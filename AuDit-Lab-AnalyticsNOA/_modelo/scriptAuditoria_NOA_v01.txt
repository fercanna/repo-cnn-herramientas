/**
 * SISTEMA DE AUDITORÍA INTERNA  |  Analytics NOA
 * ISO 9001:2015 / ISO/IEC 17025:2017
 * Basado en: scriptAuditoria_090526 v3.0
 * Adaptación para laboratorio de ensayos (Analytics NOA) – Mayo 2026
 * v01 (21/08/2026) — correcciones tras la primera corrida real:
 *   - Config: la sección CRONOGRAMA está corrida una fila respecto de lo que
 *     leía el script (Fecha de realización en B10, no B9; Hora inicio en B11,
 *     no B10; Hora fin en B12, no B11) -> el informe salía con "Fecha de
 *     auditoría" en blanco y el horario mal.
 *   - Config: los checkboxes de ámbitos están en la columna B (no C) y el
 *     nombre del ámbito en la columna A (no B), en las filas 15-19 (no
 *     14-18) -> el informe nunca detectaba ningún ámbito marcado.
 *   - Agenda: getDataRange() trae también la fila de título (fila 1) antes
 *     del encabezado real -> el informe imprimía el título como si fuera el
 *     encabezado de la tabla, y el encabezado real como si fuera una fila de
 *     datos.
 *   - Agenda: la columna Fecha (última columna) se imprimía con String(v)
 *     sobre un objeto Date de Apps Script -> volcaba el toString() crudo en
 *     vez de una fecha formateada.
 *   - Equipo auditor: si el campo queda en "NA" (placeholder de "no aplica")
 *     se imprimía literal en el informe -> ahora se filtra.
 *   - Se sacó el desdoblamiento de hallazgos por área de ensayo (Agua /
 *     Efluentes / Suelos / Agroquímicos): para Analytics NOA no tiene
 *     sentido dividir así, todos los hallazgos quedan en una sola tabla y el
 *     resumen cuantitativo pasa a ser solo por tipo (NCMa/NCme/OM/F/C), sin
 *     columnas por ámbito. Por lo mismo se sacó del menú "Duplicar cláusulas
 *     operativas por área de ensayo" (ya no aplica) y CONFIG.AMBITOS quedó
 *     con un solo valor genérico.
 */

// ============================================
// FUNCIÓN: GUARDAR HALLAZGO
// ============================================
function guardarHallazgo(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const registroSheet = ss.getSheetByName(CONFIG.SHEETS.REGISTRO);
  const lastRow = registroSheet.getLastRow();
  const numero = lastRow > 1 ? lastRow : 1;
  const estado = data.tipo === 'F' || data.tipo === 'C' ? '-' : 'Abierto';

  registroSheet.appendRow([
    numero,
    data.ambito,
    data.proceso,
    data.clausula,
    data.requisito,
    data.tipo,
    data.descripcion,
    data.evidencia,
    estado,
    '' // Trello / Acción
  ]);

  const newRow = registroSheet.getLastRow();
  aplicarFormatoHallazgo(registroSheet, newRow, data.tipo);
  renumerarHallazgos();
}

// ============================================
// FUNCIÓN: APLICAR FORMATO A HALLAZGO
// ============================================
function aplicarFormatoHallazgo(sheet, row, tipo) {
  const celdaTipo = sheet.getRange(row, 6);
  let color, fontColor;

  switch (tipo) {
    case 'NCMa': color = CONFIG.COLORES.NCMa; fontColor = '#ffffff'; break;
    case 'NCme': color = CONFIG.COLORES.NCme; fontColor = '#000000'; break;
    case 'OM':   color = CONFIG.COLORES.OM;   fontColor = '#ffffff'; break;
    case 'F':    color = CONFIG.COLORES.F;    fontColor = '#ffffff'; break;
    case 'C':    color = CONFIG.COLORES.C;    fontColor = '#000000'; break;
    default:     color = '#ffffff';           fontColor = '#000000';
  }

  celdaTipo.setBackground(color).setFontColor(fontColor).setFontWeight('bold');
}

// ============================================
// FUNCIÓN: RENUMERAR HALLAZGOS
// ============================================
function renumerarHallazgos() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const registroSheet = ss.getSheetByName(CONFIG.SHEETS.REGISTRO);
  const lastRow = registroSheet.getLastRow();
  if (lastRow <= 1) return;
  for (let i = 2; i <= lastRow; i++) {
    registroSheet.getRange(i, 1).setValue(i - 1);
  }
}

// ============================================
// FUNCIÓN: LIMPIAR REGISTRO
// ============================================
function limpiarRegistro() {
  const ui = SpreadsheetApp.getUi();
  const response = ui.alert('⚠️ Confirmar', '¿Estás seguro de que querés eliminar TODOS los hallazgos registrados?', ui.ButtonSet.YES_NO);
  if (response !== ui.Button.YES) return;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const registroSheet = ss.getSheetByName(CONFIG.SHEETS.REGISTRO);
  const lastRow = registroSheet.getLastRow();
  if (lastRow > 1) registroSheet.deleteRows(2, lastRow - 1);
  ui.alert('✅ Registro limpiado', 'Todos los hallazgos fueron eliminados.', ui.ButtonSet.OK);
}

// ============================================
// FUNCIÓN: MIGRAR HALLAZGOS LEGACY
// ============================================
function migrarHallazgosLegacy() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const legacySheet  = ss.getSheetByName(CONFIG.SHEETS.HALLAZGOS_LEGACY);
  const registroSheet = ss.getSheetByName(CONFIG.SHEETS.REGISTRO);

  if (!legacySheet) {
    SpreadsheetApp.getUi().alert('⚠️ Sin datos para migrar', 'No se encontró la pestaña "Hallazgos".', SpreadsheetApp.getUi().ButtonSet.OK);
    return;
  }

  const mapeoAmbitos = {
    'Común (Organizacional)':       'General (todos los procesos)',
    'Hormigón elaborado':           'General (todos los procesos)',
    'Producción / Entrega':         'General (todos los procesos)',
    'Análisis de Agua':             'General (todos los procesos)',
    'Análisis de Efluentes':        'General (todos los procesos)',
    'Análisis de Suelos':           'General (todos los procesos)',
    'Análisis de Agroquímicos':     'General (todos los procesos)',
  };

  const legacyData = legacySheet.getDataRange().getValues();
  let migrados = 0;

  for (let i = 1; i < legacyData.length; i++) {
    if (!legacyData[i][0]) continue;
    const ambito = mapeoAmbitos[legacyData[i][1]] || legacyData[i][1];
    registroSheet.appendRow([
      0, ambito, legacyData[i][2], legacyData[i][3], legacyData[i][4],
      legacyData[i][5], legacyData[i][6], legacyData[i][7],
      legacyData[i][8] || (legacyData[i][5] === 'F' ? '-' : 'Abierto'), ''
    ]);
    aplicarFormatoHallazgo(registroSheet, registroSheet.getLastRow(), legacyData[i][5]);
    migrados++;
  }

  renumerarHallazgos();
  SpreadsheetApp.getUi().alert('✅ Migración completada', `Se migraron ${migrados} hallazgos.\n\nRevisá el ámbito de cada uno: el mapeo automático es orientativo.`, SpreadsheetApp.getUi().ButtonSet.OK);
}

// ============================================
// FUNCIÓN: GENERAR INFORME COMPLETO
// ============================================
function generarInformeCompleto() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const configSheet    = ss.getSheetByName(CONFIG.SHEETS.CONFIG);
  const registroSheet  = ss.getSheetByName(CONFIG.SHEETS.REGISTRO);
  const agendaSheet    = ss.getSheetByName(CONFIG.SHEETS.AGENDA);

  if (!configSheet || !registroSheet) {
    SpreadsheetApp.getUi().alert('⚠️ Error', 'Faltan pestañas: Config o Registro_Hallazgos.', SpreadsheetApp.getUi().ButtonSet.OK);
    return;
  }

  // ── Leer Config ────────────────────────────────────────────────────────
  const cliente  = configSheet.getRange('B2').getValue();
  const norma    = configSheet.getRange('B3').getValue();
  const rubro    = configSheet.getRange('B4').getValue();
  const codigo   = configSheet.getRange('B6').getValue();
  // Sección CRONOGRAMA real de la plantilla: fila 9 = título "CRONOGRAMA",
  // fila 10 = Fecha de realización, fila 11 = Hora de inicio, fila 12 = Hora
  // de finalización (antes el script leía B9/B10/B11, una fila arriba de lo
  // que corresponde).
  const fechaCrono  = configSheet.getRange('B10').getValue();
  const horaInicio  = configSheet.getRange('B11').getValue();
  const horaFin     = configSheet.getRange('B12').getValue();

  // Ámbitos activos: en la plantilla real el checkbox está en la columna B
  // y el nombre del ámbito en la columna A, filas 15 a 19 (antes el script
  // buscaba el checkbox en la columna C y el nombre en la B, filas 14-18).
  const ambitos = [];
  for (let r = 15; r <= 19; r++) {
    if (configSheet.getRange(r, 2).getValue() === true) {
      ambitos.push(configSheet.getRange(r, 1).getValue());
    }
  }

  // Auditores — se descartan celdas vacías o placeholders tipo "NA"/"N/A".
  const auditores = [];
  for (let r = 28; r <= 30; r++) {
    const v = configSheet.getRange(r, 2).getValue();
    if (v && !/^n\.?\/?a\.?$/i.test(String(v).trim())) auditores.push(v);
  }

  // ── Leer Hallazgos ─────────────────────────────────────────────────────
  const hallazgosData = registroSheet.getDataRange().getValues();

  // ── Formatear fechas ───────────────────────────────────────────────────
  const fmtDate = v => v instanceof Date ? Utilities.formatDate(v, Session.getScriptTimeZone(), 'dd/MM/yyyy') : String(v || '');
  const fmtHour = v => v instanceof Date ? Utilities.formatDate(v, Session.getScriptTimeZone(), 'HH:mm') : String(v || '');

  // ── Tally de hallazgos por tipo (se usa en el resumen y en conclusiones) ─
  let totalNCMa = 0, totalNCme = 0, totalOM = 0, totalF = 0, totalC = 0;
  hallazgosData.slice(1).forEach(h => {
    if (!h[0]) return;
    if      (h[5] === 'NCMa') totalNCMa++;
    else if (h[5] === 'NCme') totalNCme++;
    else if (h[5] === 'OM')   totalOM++;
    else if (h[5] === 'F')    totalF++;
    else if (h[5] === 'C')    totalC++;
  });
  const totalHallazgos = totalNCMa + totalNCme + totalOM + totalF + totalC;

  // ── Crear documento ────────────────────────────────────────────────────
  const doc  = DocumentApp.create(`Informe Auditoría Interna ${codigo} – Analytics NOA`);
  const body = doc.getBody();
  body.clear();

  // Título
  body.appendParagraph('INFORME DE AUDITORÍA INTERNA')
    .setHeading(DocumentApp.ParagraphHeading.TITLE)
    .setAlignment(DocumentApp.HorizontalAlignment.CENTER);
  body.appendParagraph(codigo).setAlignment(DocumentApp.HorizontalAlignment.CENTER).setBold(true);
  body.appendParagraph(fmtDate(new Date())).setAlignment(DocumentApp.HorizontalAlignment.CENTER);
  body.appendParagraph(`Sistema de Gestión  |  ${norma}`)
    .setAlignment(DocumentApp.HorizontalAlignment.CENTER).setItalic(true);
  body.appendParagraph('');

  // 1. Datos de la auditoría
  body.appendParagraph('1. DATOS DE LA AUDITORÍA').setHeading(DocumentApp.ParagraphHeading.HEADING1);
  const tDatos = body.appendTable();
  const rowsDatos = [
    ['Laboratorio:', String(cliente)],
    ['Norma(s) de referencia:', String(norma)],
    ['Rubro / Sector:', String(rubro)],
    ['Código de informe:', String(codigo)],
    ['Fecha de auditoría:', fmtDate(fechaCrono)],
    ['Horario:', `${fmtHour(horaInicio)} – ${fmtHour(horaFin)}`],
    ['Alcance (ámbitos auditados):', ambitos.map(a => '• ' + a).join('\n')],
    ['Equipo auditor:', auditores.join('\n')],
  ];
  rowsDatos.forEach(([k, v]) => {
    const row = tDatos.appendTableRow();
    row.appendTableCell(k).getChild(0).asParagraph().setBold(true);
    row.appendTableCell(v);
  });
  body.appendParagraph('');

  // 1.1. Agenda
  if (agendaSheet) {
    body.appendParagraph('1.1. Agenda de Auditoría').setHeading(DocumentApp.ParagraphHeading.HEADING2);
    // getDataRange() trae también la fila de título ("AGENDA DE AUDITORÍA...")
    // antes del encabezado real ("Hora | Proceso / Área | ...") — se descarta
    // esa primera fila para que agendaData[0] sea el encabezado de verdad.
    const agendaData = agendaSheet.getDataRange().getValues().slice(1);
    if (agendaData.length > 1) {
      const tAg = body.appendTable();
      const hAg = tAg.appendTableRow();
      agendaData[0].forEach(h => hAg.appendTableCell(String(h)).getChild(0).asParagraph().setBold(true));
      const idxFecha = agendaData[0].length - 1; // última columna = Fecha
      for (let i = 1; i < agendaData.length; i++) {
        if (!agendaData[i][0]) continue;
        const row = tAg.appendTableRow();
        agendaData[i].forEach((v, c) => {
          let text;
          if (c === 0) text = fmtHour(v);
          else if (c === idxFecha) text = fmtDate(v);
          else text = String(v || '');
          row.appendTableCell(text);
        });
      }
    }
    body.appendParagraph('');
  }

  // 2. Resumen ejecutivo
  body.appendParagraph('2. RESUMEN EJECUTIVO DE HALLAZGOS').setHeading(DocumentApp.ParagraphHeading.HEADING1);
  body.appendParagraph('2.1. Resumen cuantitativo').setHeading(DocumentApp.ParagraphHeading.HEADING2);
  const tRes = body.appendTable();
  const hRes = tRes.appendTableRow();
  ['Tipo de hallazgo', 'Cantidad'].forEach(h => hRes.appendTableCell(h).getChild(0).asParagraph().setBold(true));
  const filasResumen = [
    ['No Conformidad Mayor (NCMa)', totalNCMa],
    ['No Conformidad Menor (NCme)', totalNCme],
    ['Oportunidad de Mejora (OM)', totalOM],
    ['Fortaleza (F)', totalF],
    ['Conforme (C)', totalC],
    ['TOTAL', totalHallazgos],
  ];
  filasResumen.forEach(([k, v], i) => {
    const row = tRes.appendTableRow();
    row.appendTableCell(k).getChild(0).asParagraph().setBold(i === filasResumen.length - 1);
    row.appendTableCell(String(v)).getChild(0).asParagraph().setBold(i === filasResumen.length - 1);
  });
  body.appendParagraph('');

  // 3. Hallazgos de la auditoría (sin desdoblar por ámbito/área de ensayo)
  body.appendParagraph('3. HALLAZGOS DE LA AUDITORÍA').setHeading(DocumentApp.ParagraphHeading.HEADING1);
  body.appendParagraph('Referencias: F=Fortaleza  |  OM=Oportunidad de Mejora  |  NCme=NC Menor  |  NCMa=NC Mayor  |  C=Conforme').setFontSize(9).setItalic(true);
  body.appendParagraph('');

  const hallazgosOrdenados = hallazgosData.slice(1).filter(h => h[0]);
  if (hallazgosOrdenados.length > 0) {
    const tH = body.appendTable();
    const hH = tH.appendTableRow();
    ['N°', 'Proceso', 'Cláusula', 'Tipo', 'Descripción', 'Evidencia'].forEach(h => hH.appendTableCell(h).getChild(0).asParagraph().setBold(true));
    hallazgosOrdenados.forEach(h => {
      const row = tH.appendTableRow();
      [h[0], h[2], h[3], h[5], h[6], h[7]].forEach(v => row.appendTableCell(String(v || '')));
    });
  } else {
    body.appendParagraph('No se registraron hallazgos.').setItalic(true);
  }
  body.appendParagraph('');

  // 4. Conclusiones
  body.appendParagraph('4. CONCLUSIONES Y RECOMENDACIONES').setHeading(DocumentApp.ParagraphHeading.HEADING1);

  body.appendParagraph('4.1. Conclusión general:').setBold(true);
  const conclusionGeneral = `La auditoría interna a Laboratorio Analytics NOA, realizada con referencia a los requisitos de la norma ISO 9001:2015 e ISO/IEC 17025:2017, arrojó un total de ${totalHallazgos} hallazgos: ${totalNCMa} No Conformidad(es) Mayor(es), ${totalNCme} No Conformidad(es) Menor(es), ${totalOM} Oportunidad(es) de Mejora y ${totalF} Fortaleza(s). ${totalNCMa + totalNCme === 0 ? 'No se identificaron no conformidades que comprometan el sistema de gestión ni la competencia técnica del laboratorio.' : 'Se requiere la implementación de acciones correctivas para las no conformidades identificadas dentro de los plazos acordados.'}`;
  body.appendParagraph(conclusionGeneral);
  body.appendParagraph('');

  body.appendParagraph('4.2. Recomendaciones:').setBold(true);
  const reco = totalNCMa + totalNCme === 0
    ? 'El SGC del laboratorio se encuentra en condiciones conformes con los requisitos normativos auditados. Se recomienda implementar las oportunidades de mejora identificadas para fortalecer el desempeño técnico y del sistema de gestión.'
    : 'Se recomienda priorizar la implementación de acciones correctivas para las no conformidades identificadas, definiendo responsables y plazos de implementación y verificación de eficacia. Las oportunidades de mejora deben ser evaluadas en la próxima Revisión por la Dirección.';
  body.appendParagraph(reco);
  body.appendParagraph('');

  // 5. Equipo Auditor y firmas
  body.appendParagraph('5. EQUIPO AUDITOR').setHeading(DocumentApp.ParagraphHeading.HEADING1);

// FUNCIÓN: GENERAR HALLAZGOS DESDE CHECKLIST
// ============================================
function generarHallazgosDesdeChecklistSimplificado() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const checklistSheet = ss.getSheetByName(CONFIG.SHEETS.CHECKLIST);
  const registroSheet  = ss.getSheetByName(CONFIG.SHEETS.REGISTRO);

  if (!checklistSheet || !registroSheet) {
    SpreadsheetApp.getUi().alert('⚠️ Error', 'No se encontraron las pestañas Checklist_ISO9001 o Registro_Hallazgos.', SpreadsheetApp.getUi().ButtonSet.OK);
    return;
  }

  const ui = SpreadsheetApp.getUi();
  const resp = ui.alert('🔄 Generar Hallazgos', '¿Limpiar el Registro_Hallazgos antes de generar?\n\n• SÍ: Borra existentes y genera desde cero\n• NO: Agrega al final', ui.ButtonSet.YES_NO);
  if (resp === ui.Button.YES) {
    const lastRow = registroSheet.getLastRow();
    if (lastRow > 1) registroSheet.deleteRows(2, lastRow - 1);
  }

  const checklistData = checklistSheet.getDataRange().getValues();
  const hallazgos = [];

  for (let i = 1; i < checklistData.length; i++) {
    const proceso  = checklistData[i][0];
    const clausula = checklistData[i][1];
    const requisito = checklistData[i][2];
    const ambito   = checklistData[i][5];
    const estado   = checklistData[i][6];
    const notas    = checklistData[i][7];

    if (estado && ['NCme','NCMa','OM','F'].includes(estado.toString().trim())) {
      const partes = notas ? notas.toString().split('---') : ['',''];
      const descripcion = partes[0].trim();
      const evidencia   = partes.length > 1 ? partes[1].trim() : '';
      hallazgos.push([
        0, ambito || 'General (todos los procesos)', proceso || '', clausula || '',
        requisito || '', estado, descripcion, evidencia,
        estado === 'F' ? '-' : 'Abierto', ''
      ]);
    }
  }

  if (hallazgos.length === 0) {
    ui.alert('ℹ️ Sin hallazgos', 'No se encontraron hallazgos en el Checklist.\nVerificá que la columna G (Estado) tenga valores: NCme, NCMa, OM o F.', ui.ButtonSet.OK);
    return;
  }

  const primeraFila = registroSheet.getLastRow() + 1;
  registroSheet.getRange(primeraFila, 1, hallazgos.length, 10).setValues(hallazgos);
  hallazgos.forEach((h, i) => aplicarFormatoHallazgo(registroSheet, primeraFila + i, h[5]));
  renumerarHallazgos();

  ui.alert('✅ Hallazgos generados', `Se generaron ${hallazgos.length} hallazgos desde el Checklist.\n\nRevisá la solapa Registro_Hallazgos.`, ui.ButtonSet.OK);
}
