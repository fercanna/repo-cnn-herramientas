// ============================================
// FUNCIÓN: GUARDAR HALLAZGO
// ============================================
function guardarHallazgo(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const registroSheet = ss.getSheetByName(CONFIG.SHEETS.REGISTRO);
  const lastRow = registroSheet.getLastRow();
  const numero = lastRow > 2 ? lastRow - 1 : 1;
  const estado = data.tipo === 'F' ? '-' : 'Abierto';

  registroSheet.appendRow([
    numero,
    data.ambito,
    data.proceso,
    data.clausula,
    data.requisito,
    data.tipo,
    data.descripcion,
    data.evidencia,
    estado,
    '' // Acción / Trello
  ]);

  const newRow = registroSheet.getLastRow();
  aplicarFormatoHallazgo(registroSheet, newRow, data.tipo);
  renumerarHallazgos();
}

// ============================================
// FUNCIÓN: APLICAR FORMATO A HALLAZGO (columna F = Tipo)
// ============================================
function aplicarFormatoHallazgo(sheet, row, tipo) {
  const celdaTipo = sheet.getRange(row, 6);
  let color, fontColor;

  switch (tipo) {
    case 'NCMa': color = CONFIG.COLORES.NCMa; fontColor = '#ffffff'; break;
    case 'NCme': color = CONFIG.COLORES.NCme; fontColor = '#000000'; break;
    case 'OM':   color = CONFIG.COLORES.OM;   fontColor = '#ffffff'; break;
    case 'F':    color = CONFIG.COLORES.F;    fontColor = '#ffffff'; break;
    case 'C':    color = CONFIG.COLORES.C;    fontColor = '#000000'; break;
    default:     color = '#ffffff';           fontColor = '#000000';
  }

  celdaTipo.setBackground(color).setFontColor(fontColor).setFontWeight('bold');
}

// ============================================
// FUNCIÓN: RENUMERAR HALLAZGOS (fila 1 = título, fila 2 = encabezado, datos desde fila 3)
// ============================================
function renumerarHallazgos() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const registroSheet = ss.getSheetByName(CONFIG.SHEETS.REGISTRO);
  const lastRow = registroSheet.getLastRow();
  if (lastRow <= 2) return;
  for (let i = 3; i <= lastRow; i++) {
    registroSheet.getRange(i, 1).setValue(i - 2);
  }
}

// ============================================
// FUNCIÓN: LIMPIAR REGISTRO
// ============================================
function limpiarRegistro() {
  const ui = SpreadsheetApp.getUi();
  const response = ui.alert('⚠️ Confirmar', '¿Estás seguro de que querés eliminar TODOS los hallazgos registrados?', ui.ButtonSet.YES_NO);
  if (response !== ui.Button.YES) return;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const registroSheet = ss.getSheetByName(CONFIG.SHEETS.REGISTRO);
  const lastRow = registroSheet.getLastRow();
  if (lastRow > 2) registroSheet.deleteRows(3, lastRow - 2);
  ui.alert('✅ Registro limpiado', 'Todos los hallazgos fueron eliminados.', ui.ButtonSet.OK);
}

// ============================================
// FUNCIÓN: MIGRAR HALLAZGOS LEGACY (por si queda una hoja "Hallazgos" vieja)
// ============================================
function migrarHallazgosLegacy() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const legacySheet  = ss.getSheetByName(CONFIG.SHEETS.HALLAZGOS_LEGACY);
  const registroSheet = ss.getSheetByName(CONFIG.SHEETS.REGISTRO);

  if (!legacySheet) {
    SpreadsheetApp.getUi().alert('⚠️ Sin datos para migrar', 'No se encontró la pestaña "Hallazgos". Si ya migraste a Registro_Hallazgos manualmente, no hace falta usar esta función.', SpreadsheetApp.getUi().ButtonSet.OK);
    return;
  }

  const legacyData = legacySheet.getDataRange().getValues();
  let migrados = 0;

  for (let i = 1; i < legacyData.length; i++) {
    if (!legacyData[i][0]) continue;
    // Formato legacy no tenía columna de ámbito -> se migra como General/Institucional
    // y el auditor la reclasifica manualmente si corresponde a una sección.
    registroSheet.appendRow([
      0, 'General/Institucional', legacyData[i][1], legacyData[i][2], legacyData[i][3],
      legacyData[i][4], legacyData[i][5], legacyData[i][6],
      legacyData[i][7] || (legacyData[i][4] === 'F' ? '-' : 'Abierto'), ''
    ]);
    aplicarFormatoHallazgo(registroSheet, registroSheet.getLastRow(), legacyData[i][4]);
    migrados++;
  }

  renumerarHallazgos();
  SpreadsheetApp.getUi().alert('✅ Migración completada', `Se migraron ${migrados} hallazgos como "General/Institucional".\n\nRevisá y reclasificá el ámbito de cada uno (Bacteriología, Virología, etc.) donde corresponda.`, SpreadsheetApp.getUi().ButtonSet.OK);
}

// ============================================
// FUNCIÓN: AMBITO POR DEFECTO PARA UNA FILA DEL CHECKLIST GENERAL
// (el checklist general mezcla varios procesos institucionales + Preanalítico)
// ============================================
function ambitoParaFilaGeneral(proceso) {
  const p = String(proceso || '');
  if (p.indexOf('Preanalítico') !== -1) return 'Preanalítico (común)';
  if (p.indexOf('Postanalítico') !== -1) return 'Preanalítico (común)'; // etapa común, ver nota en Ayuda
  return 'General/Institucional';
}

// ============================================
// FUNCIÓN: GENERAR HALLAZGOS DESDE CHECKLISTS (recorre las 3 hojas)
// ============================================
function generarHallazgosDesdeChecklistSimplificado() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const registroSheet = ss.getSheetByName(CONFIG.SHEETS.REGISTRO);

  if (!registroSheet) {
    SpreadsheetApp.getUi().alert('⚠️ Error', 'No se encontró la pestaña Registro_Hallazgos.', SpreadsheetApp.getUi().ButtonSet.OK);
    return;
  }

  const ui = SpreadsheetApp.getUi();
  const resp = ui.alert('🔄 Generar Hallazgos', '¿Limpiar el Registro_Hallazgos antes de generar?\n\n• SÍ: Borra existentes y genera desde cero (desde las 3 hojas de checklist)\n• NO: Agrega solo lo nuevo al final', ui.ButtonSet.YES_NO);
  if (resp === ui.Button.YES) {
    const lastRow = registroSheet.getLastRow();
    if (lastRow > 2) registroSheet.deleteRows(3, lastRow - 2);
  }

  const hallazgos = [];

  CONFIG.CHECKLISTS.forEach(cfg => {
    const sheet = ss.getSheetByName(cfg.sheet);
    if (!sheet) return; // hoja opcional (ej. si todavía no existe Checklist_8_Parasitologia)
    // Fila 1 = título fusionado, fila 2 = encabezado real -> se descartan ambas.
    const data = sheet.getDataRange().getValues().slice(2);

    data.forEach(row => {
      const proceso    = row[0];
      const clausula   = row[1];
      const requisito  = row[2];
      const estado     = row[3];
      const evidencia  = row[4];
      const observ     = row[5];
      const auditor    = row[6];

      if (!estado || ['NCme', 'NCMa', 'OM', 'F'].indexOf(String(estado).trim()) === -1) return;

      const ambito = cfg.ambitoFijo || ambitoParaFilaGeneral(proceso);
      hallazgos.push([
        0, ambito, proceso || '', clausula || '', requisito || '',
        String(estado).trim(), observ || '', evidencia || '',
        estado === 'F' ? '-' : 'Abierto', ''
      ]);
    });
  });

  if (hallazgos.length === 0) {
    ui.alert('ℹ️ Sin hallazgos', 'No se encontraron hallazgos en ninguno de los checklists.\nVerificá que la columna D (Estado) tenga valores: NCme, NCMa, OM o F.', ui.ButtonSet.OK);
    return;
  }

  const primeraFila = registroSheet.getLastRow() + 1;
  registroSheet.getRange(primeraFila, 1, hallazgos.length, 10).setValues(hallazgos);
  hallazgos.forEach((h, i) => aplicarFormatoHallazgo(registroSheet, primeraFila + i, h[5]));
  renumerarHallazgos();

  ui.alert('✅ Hallazgos generados', `Se generaron ${hallazgos.length} hallazgos desde los 3 checklists.\n\nRevisá la solapa Registro_Hallazgos.`, ui.ButtonSet.OK);
}

// ============================================
// FUNCIÓN: GENERAR INFORME COMPLETO
// ============================================
function generarInformeCompleto() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const configSheet   = ss.getSheetByName(CONFIG.SHEETS.CONFIG);
  const registroSheet = ss.getSheetByName(CONFIG.SHEETS.REGISTRO);
  const agendaSheet   = ss.getSheetByName(CONFIG.SHEETS.AGENDA);

  if (!configSheet || !registroSheet) {
    SpreadsheetApp.getUi().alert('⚠️ Error', 'Faltan pestañas: Config o Registro_Hallazgos.', SpreadsheetApp.getUi().ButtonSet.OK);
    return;
  }

  // ── Leer Config (layout real de la planilla CURF) ───────────────────────
  const cliente = configSheet.getRange('B4').getValue();
  const norma   = configSheet.getRange('B5').getValue();
  const rubro   = configSheet.getRange('B6').getValue();
  const codigo  = configSheet.getRange('B7').getValue();
  const fechaCrono = configSheet.getRange('B8').getValue();
  const horaInicio = configSheet.getRange('B9').getValue();
  const horaFin    = configSheet.getRange('B10').getValue();

  // Ámbitos/secciones activas: nombre en columna A, checkbox en columna B,
  // filas 13 a 26.
  const ambitosActivos = [];
  for (let r = 13; r <= 26; r++) {
    if (configSheet.getRange(r, 2).getValue() === true) {
      ambitosActivos.push(configSheet.getRange(r, 1).getValue());
    }
  }

  // Auditores — nombre en columna B, área en columna C, filas 29 a 33.
  const auditores = [];
  for (let r = 29; r <= 33; r++) {
    const nombre = configSheet.getRange(r, 2).getValue();
    const area   = configSheet.getRange(r, 3).getValue();
    if (nombre && !/^n\.?\/?a\.?$/i.test(String(nombre).trim())) {
      auditores.push(area ? `${nombre} (${area})` : String(nombre));
    }
  }

  // ── Leer Hallazgos ─────────────────────────────────────────────────────
  const hallazgosData = registroSheet.getDataRange().getValues().slice(2); // sin título ni encabezado

  // ── Formatear fechas ───────────────────────────────────────────────────
  const fmtDate = v => v instanceof Date ? Utilities.formatDate(v, Session.getScriptTimeZone(), 'dd/MM/yyyy') : String(v || '');
  const fmtHour = v => v instanceof Date ? Utilities.formatDate(v, Session.getScriptTimeZone(), 'HH:mm') : String(v || '');

  // ── Tally por tipo y por ámbito ──────────────────────────────────────────
  let totalNCMa = 0, totalNCme = 0, totalOM = 0, totalF = 0;
  const porAmbito = {};
  CONFIG.AMBITOS.forEach(a => porAmbito[a] = { NCMa: 0, NCme: 0, OM: 0, F: 0 });

  hallazgosData.forEach(h => {
    if (!h[0]) return;
    const ambito = h[1] || 'General/Institucional';
    const tipo = h[5];
    if (!porAmbito[ambito]) porAmbito[ambito] = { NCMa: 0, NCme: 0, OM: 0, F: 0 };
    if      (tipo === 'NCMa') { totalNCMa++; porAmbito[ambito].NCMa++; }
    else if (tipo === 'NCme') { totalNCme++; porAmbito[ambito].NCme++; }
    else if (tipo === 'OM')   { totalOM++;   porAmbito[ambito].OM++; }
    else if (tipo === 'F')    { totalF++;    porAmbito[ambito].F++; }
  });
  const totalHallazgos = totalNCMa + totalNCme + totalOM + totalF;

  // ── Crear documento ────────────────────────────────────────────────────
  const doc  = DocumentApp.create(`Informe Auditoría Interna ${codigo} – CURF`);
  const body = doc.getBody();
  body.clear();

  body.appendParagraph('INFORME DE AUDITORÍA INTERNA')
    .setHeading(DocumentApp.ParagraphHeading.TITLE)
    .setAlignment(DocumentApp.HorizontalAlignment.CENTER);
  body.appendParagraph(codigo).setAlignment(DocumentApp.HorizontalAlignment.CENTER).setBold(true);
  body.appendParagraph(fmtDate(new Date())).setAlignment(DocumentApp.HorizontalAlignment.CENTER);
  body.appendParagraph(`Sistema de Gestión  |  ${norma}`)
    .setAlignment(DocumentApp.HorizontalAlignment.CENTER).setItalic(true);
  body.appendParagraph('');

  // 1. Datos de la auditoría
  body.appendParagraph('1. DATOS DE LA AUDITORÍA').setHeading(DocumentApp.ParagraphHeading.HEADING1);
  const tDatos = body.appendTable();
  const rowsDatos = [
    ['Laboratorio:', String(cliente)],
    ['Norma de referencia:', String(norma)],
    ['Rubro / Sector:', String(rubro)],
    ['Código de informe:', String(codigo)],
    ['Fecha de auditoría:', fmtDate(fechaCrono)],
    ['Horario:', `${fmtHour(horaInicio)} – ${fmtHour(horaFin)}`],
    ['Alcance (ámbitos auditados):', ambitosActivos.map(a => '• ' + a).join('\n')],
    ['Equipo auditor:', auditores.join('\n')],
  ];
  rowsDatos.forEach(([k, v]) => {
    const row = tDatos.appendTableRow();
    row.appendTableCell(k).getChild(0).asParagraph().setBold(true);
    row.appendTableCell(v);
  });
  body.appendParagraph('');

  // 1.1. Agenda
  if (agendaSheet) {
    body.appendParagraph('1.1. Agenda de Auditoría').setHeading(DocumentApp.ParagraphHeading.HEADING2);
    const agendaData = agendaSheet.getDataRange().getValues().slice(1); // saca la fila de título
    if (agendaData.length > 1) {
      const tAg = body.appendTable();
      const hAg = tAg.appendTableRow();
      agendaData[0].forEach(h => hAg.appendTableCell(String(h)).getChild(0).asParagraph().setBold(true));
      const idxFecha = agendaData[0].length - 1; // última columna = Fecha
      for (let i = 1; i < agendaData.length; i++) {
        if (!agendaData[i][0]) continue;
        const row = tAg.appendTableRow();
        agendaData[i].forEach((v, c) => {
          let text;
          if (c === 0) text = fmtHour(v);
          else if (c === idxFecha) text = fmtDate(v);
          else text = String(v || '');
          row.appendTableCell(text);
        });
      }
    }
    body.appendParagraph('');
  }

  // 2. Resumen ejecutivo
  body.appendParagraph('2. RESUMEN EJECUTIVO DE HALLAZGOS').setHeading(DocumentApp.ParagraphHeading.HEADING1);

  body.appendParagraph('2.1. Resumen cuantitativo por tipo').setHeading(DocumentApp.ParagraphHeading.HEADING2);
  const tRes = body.appendTable();
  const hRes = tRes.appendTableRow();
  ['Tipo de hallazgo', 'Cantidad'].forEach(h => hRes.appendTableCell(h).getChild(0).asParagraph().setBold(true));
  const filasResumen = [
    ['No Conformidad Mayor (NCMa)', totalNCMa],
    ['No Conformidad Menor (NCme)', totalNCme],
    ['Oportunidad de Mejora (OM)', totalOM],
    ['Fortaleza (F)', totalF],
    ['TOTAL', totalHallazgos],
  ];
  filasResumen.forEach(([k, v], i) => {
    const row = tRes.appendTableRow();
    row.appendTableCell(k).getChild(0).asParagraph().setBold(i === filasResumen.length - 1);
    row.appendTableCell(String(v)).getChild(0).asParagraph().setBold(i === filasResumen.length - 1);
  });
  body.appendParagraph('');

  body.appendParagraph('2.2. Resumen por ámbito / sección').setHeading(DocumentApp.ParagraphHeading.HEADING2);
  const tAmb = body.appendTable();
  const hAmb = tAmb.appendTableRow();
  ['Ámbito', 'NCMa', 'NCme', 'OM', 'F', 'Total'].forEach(h => hAmb.appendTableCell(h).getChild(0).asParagraph().setBold(true));
  CONFIG.AMBITOS.forEach(a => {
    const c = porAmbito[a] || { NCMa: 0, NCme: 0, OM: 0, F: 0 };
    const totalA = c.NCMa + c.NCme + c.OM + c.F;
    const row = tAmb.appendTableRow();
    [a, c.NCMa, c.NCme, c.OM, c.F, totalA].forEach(v => row.appendTableCell(String(v)));
  });
  body.appendParagraph('');

  // 3. Hallazgos de la auditoría
  body.appendParagraph('3. HALLAZGOS DE LA AUDITORÍA').setHeading(DocumentApp.ParagraphHeading.HEADING1);
  body.appendParagraph('Referencias: F=Fortaleza  |  OM=Oportunidad de Mejora  |  NCme=NC Menor  |  NCMa=NC Mayor').setFontSize(9).setItalic(true);
  body.appendParagraph('');

  const hallazgosOrdenados = hallazgosData.filter(h => h[0]);
  if (hallazgosOrdenados.length > 0) {
    const tH = body.appendTable();
    const hH = tH.appendTableRow();
    ['N°', 'Ámbito', 'Proceso', 'Cláusula', 'Tipo', 'Descripción', 'Evidencia'].forEach(h => hH.appendTableCell(h).getChild(0).asParagraph().setBold(true));
    hallazgosOrdenados.forEach(h => {
      const row = tH.appendTableRow();
      [h[0], h[1], h[2], h[3], h[5], h[6], h[7]].forEach(v => row.appendTableCell(String(v || '')));
    });
  } else {
    body.appendParagraph('No se registraron hallazgos.').setItalic(true);
  }
  body.appendParagraph('');

  // 4. Conclusiones
  body.appendParagraph('4. CONCLUSIONES Y RECOMENDACIONES').setHeading(DocumentApp.ParagraphHeading.HEADING1);

  body.appendParagraph('4.1. Conclusión general:').setBold(true);
  const conclusionGeneral = `La auditoría interna al Laboratorio de Microbiología CURF, realizada con referencia a los requisitos de la norma ${norma}, arrojó un total de ${totalHallazgos} hallazgos: ${totalNCMa} No Conformidad(es) Mayor(es), ${totalNCme} No Conformidad(es) Menor(es), ${totalOM} Oportunidad(es) de Mejora y ${totalF} Fortaleza(s). ${totalNCMa + totalNCme === 0 ? 'No se identificaron no conformidades que comprometan el sistema de gestión ni la competencia técnica del laboratorio.' : 'Se requiere la implementación de acciones correctivas para las no conformidades identificadas dentro de los plazos acordados.'}`;
  body.appendParagraph(conclusionGeneral);
  body.appendParagraph('');

  body.appendParagraph('4.2. Recomendaciones:').setBold(true);
  const reco = totalNCMa + totalNCme === 0
    ? 'El SGC del laboratorio se encuentra en condiciones conformes con los requisitos normativos auditados. Se recomienda implementar las oportunidades de mejora identificadas para fortalecer el desempeño de cada sección (Bacteriología, Virología, Parasitología, Micología).'
    : 'Se recomienda priorizar la implementación de acciones correctivas para las no conformidades identificadas, definiendo responsables y plazos de implementación y verificación de eficacia. Las oportunidades de mejora deben ser evaluadas en la próxima Revisión por la Dirección.';
  body.appendParagraph(reco);
  body.appendParagraph('');

  // 5. Equipo Auditor y firmas
  body.appendParagraph('5. EQUIPO AUDITOR').setHeading(DocumentApp.ParagraphHeading.HEADING1);
  auditores.forEach((aud, i) => body.appendParagraph(`• ${i === 0 ? 'Auditor Líder' : 'Auditor'}: ${aud}`));
  body.appendParagraph('');
  body.appendParagraph('');

  const tFirmas = body.appendTable();
  const rFirmas = tFirmas.appendTableRow();
  rFirmas.appendTableCell('_________________________________\nFirma Auditor Líder\n\nAclaración:');
  rFirmas.appendTableCell('_________________________________\nFirma Director Técnico / Representante del Lab.\n\nAclaración:');

  // ── Guardar y abrir ────────────────────────────────────────────────────
  doc.saveAndClose();
  const url = doc.getUrl();
  const htmlOut = HtmlService.createHtmlOutput(
    '<h3 style="color:#1F4E79;">✅ Informe generado correctamente</h3>' +
    '<p><a href="' + url + '" target="_blank" style="font-size:16px;color:#2E75B6;font-weight:bold;">👉 Abrir informe en nueva pestaña</a></p>' +
    '<p style="font-size:11px;color:#666;">Revisá las secciones de Conclusiones y completá los datos que lo requieran.</p>'
  ).setWidth(460).setHeight(160);
  SpreadsheetApp.getUi().showModalDialog(htmlOut, 'Informe generado – CURF');
  Logger.log('Informe: ' + url);
}
