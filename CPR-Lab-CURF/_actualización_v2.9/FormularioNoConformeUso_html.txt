  <script>
    let insumos = [];
    let lotes   = [];
    let origenLote = null;

    window.onload = function() {
      google.script.run
        .withSuccessHandler(function(lista) {
          insumos = (lista || []).filter(function(i) { return i.stock_total > 0; });
          filtrarInsumos();
        })
        .withFailureHandler(function(e) { mostrarMensaje('Error al cargar insumos: ' + e.message); })
        .getInsumosConStockYSinStock();
    };

    function filtrarInsumos() {
      const q = document.getElementById('buscar').value.trim().toLowerCase();
      const sel = document.getElementById('insumo');
      sel.innerHTML = '';
      insumos.filter(function(i) {
        return !q || i.descripcion.toLowerCase().indexOf(q) !== -1 || i.codigo.toLowerCase().indexOf(q) !== -1;
      }).forEach(function(i) {
        const o = document.createElement('option');
        o.value = i.codigo;
        o.textContent = i.descripcion + ' (' + i.codigo + ') — stock: ' + i.stock_total;
        sel.appendChild(o);
      });
    }

    function cargarLotes() {
      const cod = document.getElementById('insumo').value;
      const selLote = document.getElementById('lote');
      selLote.innerHTML = '<option value="">Cargando lotes...</option>';
      document.getElementById('infoLote').style.display = 'none';
      if (!cod) return;
      google.script.run
        .withSuccessHandler(function(l) {
          lotes = (l || []).filter(function(x) { return x.stock_actual > 0; });
          selLote.innerHTML = '<option value="">— Elegí el lote —</option>';
          lotes.forEach(function(x, idx) {
            const o = document.createElement('option');
            o.value = x.lote;
            o.textContent = x.lote + ' — stock: ' + x.stock_actual + (x.vencimiento ? ' — vence ' + x.vencimiento : '');
            selLote.appendChild(o);
          });
          if (lotes.length === 1) { selLote.value = lotes[0].lote; mostrarInfoLote(); }
        })
        .withFailureHandler(function(e) { mostrarMensaje('Error al cargar lotes: ' + e.message); })
        .getLotesParaAjuste(cod);
    }

    function mostrarInfoLote() {
      const cod  = document.getElementById('insumo').value;
      const lote = document.getElementById('lote').value;
      const info = document.getElementById('infoLote');
      origenLote = null;
      if (!lote) { info.style.display = 'none'; return; }
      const l = lotes.filter(function(x) { return x.lote === lote; })[0];
      if (l) document.getElementById('cantidad').value = l.stock_actual;   // por defecto, todo el lote
      info.style.display = 'block';
      info.textContent = 'Buscando origen del lote...';
      google.script.run
        .withSuccessHandler(function(o) {
          origenLote = o;
          const partes = [];
          partes.push('Stock del lote: ' + (l ? l.stock_actual : '?'));
          partes.push('Orden: ' + (o.numero_orden || 'no se pudo determinar'));
          partes.push('Proveedor: ' + (o.proveedor || 'no se pudo determinar'));
          info.textContent = partes.join('  |  ');
          document.getElementById('grpProveedor').style.display = o.proveedor ? 'none' : 'block';
        })
        .withFailureHandler(function() {
          info.textContent = 'No se pudo determinar el origen del lote.';
          document.getElementById('grpProveedor').style.display = 'block';
        })
        .getOrigenLote(cod, lote);
    }

    function mostrarMensaje(txt) {
      const m = document.getElementById('message');
      m.textContent = txt; m.className = 'message error'; m.style.display = 'block';
    }

    function guardar() {
      document.getElementById('message').style.display = 'none';
      const cod   = document.getElementById('insumo').value;
      const lote  = document.getElementById('lote').value;
      const cant  = parseFloat(document.getElementById('cantidad').value);
      const motivo = document.getElementById('motivo').value.trim();
      if (!cod)  return mostrarMensaje('Elegí el insumo.');
      if (!lote) return mostrarMensaje('Elegí el lote afectado.');
      if (!(cant > 0)) return mostrarMensaje('Ingresá la cantidad afectada (mayor a 0).');
      const l = lotes.filter(function(x) { return x.lote === lote; })[0];
      if (l && cant > l.stock_actual) return mostrarMensaje('La cantidad supera el stock del lote (' + l.stock_actual + ').');
      if (!motivo) return mostrarMensaje('Describí qué se detectó (obligatorio, queda como evidencia).');

      if (!confirm('Se van a dar de baja ' + cant + ' unidad(es) del lote ' + lote + ' y se registrará el no conforme. ¿Confirmás?')) return;

      document.getElementById('formSection').style.display = 'none';
      document.getElementById('loading').style.display = 'block';
      google.script.run
        .withSuccessHandler(function(r) {
          document.getElementById('loading').style.display = 'none';
          document.getElementById('resultado').style.display = 'block';
          document.getElementById('resMensaje').textContent = r.mensaje;
          if (r.aviso) {
            const a = document.getElementById('resAlerta');
            a.textContent = r.aviso; a.style.display = 'block';
          }
        })
        .withFailureHandler(function(e) {
          document.getElementById('loading').style.display = 'none';
          document.getElementById('formSection').style.display = 'block';
          mostrarMensaje(e.message);
        })
        .registrarNoConformeEnUso({
          codigo_insumo: cod,
          lote: lote,
          cantidad: cant,
          motivo: motivo,
          proveedor_manual: document.getElementById('proveedorManual').value.trim()
        });
    }
  </script>
