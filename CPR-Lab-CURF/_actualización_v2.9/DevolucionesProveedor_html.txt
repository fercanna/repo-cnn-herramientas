  <script>
    let devolucionesGlobal = [];
    let filaEnResolucion = null;
    let devEnResolucion = null;

    window.onload = function() {
      google.script.run
        .withSuccessHandler(dibujarTabla)
        .withFailureHandler(mostrarError)
        .getTodasLasDevoluciones();
    };

    function esUsoDev(d) { return d && d.origen === 'Detectado en uso'; }

    function badgeClase(estado) {
      if (estado === 'Pendiente devolución') return 'st-pendiente';
      if (estado === 'Devuelto') return 'st-devuelto';
      if (estado === 'Reemplazado') return 'st-reemplazado';
      return 'st-credito';
    }

    function dibujarTabla(devoluciones) {
      document.getElementById('loader').style.display = 'none';
      document.getElementById('content').style.display = 'block';
      devolucionesGlobal = devoluciones;

      const container = document.getElementById('tablaContainer');
      if (!devoluciones || devoluciones.length === 0) {
        container.innerHTML = '<div class="empty-state">✅ <strong>Sin devoluciones registradas</strong><br>No hay ítems No Conforme pendientes ni resueltos.</div>';
        return;
      }

      let html = '<table><thead><tr>';
      html += '<th>Fecha</th><th>Orden / origen</th><th>Proveedor</th><th>Insumo</th><th>Cant.</th><th>Motivo</th><th>Estado</th><th></th>';
      html += '</tr></thead><tbody>';

      devoluciones.forEach(function(d, idx) {
        html += '<tr>' +
          '<td>' + d.fecha + '</td>' +
          '<td>' + d.numero_orden + (esUsoDev(d) ? '<div><span class="badge-uso">⚠️ Detectado en uso</span></div>' : '') + '</td>' +
          '<td>' + d.proveedor + '</td>' +
          '<td>' + d.descripcion + ' (' + d.codigo_insumo + ')</td>' +
          '<td>' + d.cantidad + '</td>' +
          '<td>' + (d.motivo || '<em>—</em>') + '</td>' +
          '<td><span class="status-badge ' + badgeClase(d.estado) + '">' + d.estado + '</span>' +
            (d.estado !== 'Pendiente devolución' && d.fecha_resolucion
              ? '<div style="font-size:10px;color:#888;margin-top:3px;">' + d.fecha_resolucion + '</div>' : '') +
          '</td>' +
          '<td>' + (d.estado === 'Pendiente devolución'
            ? '<button class="btn-resolver" onclick="abrirResolver(' + idx + ')">Resolver</button>'
            : '') + '</td>' +
        '</tr>';
      });

      html += '</tbody></table>';
      container.innerHTML = html;
    }

    function abrirResolver(idx) {
      filaEnResolucion = devolucionesGlobal[idx].fila;
      devEnResolucion = devolucionesGlobal[idx];
      document.getElementById('resolverTitulo').textContent =
        'Resolver: ' + devolucionesGlobal[idx].descripcion + ' — Orden ' + devolucionesGlobal[idx].numero_orden;
      document.getElementById('resolverObs').value = '';
      document.getElementById('chkNoEsperar').checked = false;
      actualizarAyudaEstado();
      document.getElementById('resolverPanel').classList.add('visible');
    }

    // v2.9.3: explica qué pasa con la línea pendiente de la OC según el estado elegido
    function actualizarAyudaEstado() {
      const est = document.getElementById('resolverEstado').value;
      const ayuda = document.getElementById('ayudaEstado');
      const chk = document.getElementById('lblNoEsperar');
      if (esUsoDev(devEnResolucion)) {
        // v2.9.5: detectado en uso → la OC original ya estaba recepcionada, no se toca
        ayuda.textContent = 'Detectado en uso: este lote ya se dio de baja del stock. Registrá cómo se resolvió el reclamo; no modifica ninguna orden de compra. Si llega un reemplazo, ingresalo con una recepción normal.';
        chk.style.display = 'none';
        document.getElementById('chkNoEsperar').checked = false;
        return;
      }
      if (est === 'Nota de crédito') {
        ayuda.textContent = 'La línea pendiente de la OC se cierra automáticamente: ya no va a aparecer para recepcionar.';
        chk.style.display = 'none';
      } else if (est === 'Devuelto') {
        ayuda.textContent = 'Por defecto la línea sigue pendiente (se espera el reemplazo). Tildá la casilla si el producto NO se va a reponer.';
        chk.style.display = 'flex';
      } else {
        ayuda.textContent = 'La línea sigue pendiente: cuando llegue el reemplazo lo recepcionás desde Ver Pendientes.';
        chk.style.display = 'none';
      }
      if (est !== 'Devuelto') document.getElementById('chkNoEsperar').checked = false;
    }

    function cerrarResolver() {
      filaEnResolucion = null;
      document.getElementById('resolverPanel').classList.remove('visible');
    }

    function confirmarResolver() {
      const nuevoEstado = document.getElementById('resolverEstado').value;
      const obs = document.getElementById('resolverObs').value;
      const noEsperar = document.getElementById('chkNoEsperar').checked;
      google.script.run
        .withSuccessHandler(function(res) {
          cerrarResolver();
          if (res && res.cierra) {
            alert(res.cerrado > 0
              ? '✅ Devolución resuelta. Se cerró la línea pendiente de la OC (' + res.cerrado + ' unidad/es).'
              : '✅ Devolución resuelta. La línea de la OC ya no tenía cantidad pendiente.');
          }
          document.getElementById('loader').style.display = 'block';
          document.getElementById('content').style.display = 'none';
          google.script.run.withSuccessHandler(dibujarTabla).withFailureHandler(mostrarError).getTodasLasDevoluciones();
        })
        .withFailureHandler(function(e) { alert('❌ ' + e.message); })
        .marcarDevolucionResuelta(filaEnResolucion, nuevoEstado, obs, noEsperar);
    }

    function mostrarError(e) {
      document.getElementById('loader').innerHTML = '<div style="color:#d93025">❌ Error: ' + e.message + '</div>';
    }
  </script>
