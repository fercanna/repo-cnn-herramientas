  <div id="loader" class="loader">
    <div style="font-size: 24px; margin-bottom: 10px;">⏳</div>
    Cargando devoluciones...
  </div>

  <div id="content" style="display:none;">
    <div id="tablaContainer"></div>
  </div>

  <div class="resolver-panel" id="resolverPanel">
    <div class="resolver-box">
      <h3 id="resolverTitulo">Resolver devolución</h3>
      <label style="font-size:12px;font-weight:bold;">Nuevo estado</label>
      <select id="resolverEstado">
        <option value="Devuelto">Devuelto</option>
        <option value="Reemplazado">Reemplazado</option>
        <option value="Nota de crédito">Nota de crédito</option>
      </select>
      <label style="font-size:12px;font-weight:bold;">Observaciones (opcional)</label>
      <textarea id="resolverObs" rows="3" placeholder="Ej: cambiado por el proveedor el 10/09, remito N° 123"></textarea>
      <div class="botones">
        <button class="btn-cancelar" onclick="cerrarResolver()">Cancelar</button>
        <button class="btn-confirmar" onclick="confirmarResolver()">Confirmar</button>
      </div>
    </div>
  </div>

  <script>
    let devolucionesGlobal = [];
    let filaEnResolucion = null;

    window.onload = function() {
      google.script.run
        .withSuccessHandler(dibujarTabla)
        .withFailureHandler(mostrarError)
        .getTodasLasDevoluciones();
    };

    function badgeClase(estado) {
      if (estado === 'Pendiente devolución') return 'st-pendiente';
      if (estado === 'Devuelto') return 'st-devuelto';
      if (estado === 'Reemplazado') return 'st-reemplazado';
      return 'st-credito';
    }

    function dibujarTabla(devoluciones) {
      document.getElementById('loader').style.display = 'none';
      document.getElementById('content').style.display = 'block';
      devolucionesGlobal = devoluciones;

      const container = document.getElementById('tablaContainer');
      if (!devoluciones || devoluciones.length === 0) {
        container.innerHTML = '<div class="empty-state">✅ <strong>Sin devoluciones registradas</strong><br>No hay ítems No Conforme pendientes ni resueltos.</div>';
        return;
      }

      let html = '<table><thead><tr>';
      html += '<th>Fecha</th><th>Orden</th><th>Proveedor</th><th>Insumo</th><th>Cant.</th><th>Motivo</th><th>Estado</th><th></th>';
      html += '</tr></thead><tbody>';

      devoluciones.forEach(function(d, idx) {
        html += '<tr>' +
          '<td>' + d.fecha + '</td>' +
          '<td>' + d.numero_orden + '</td>' +
          '<td>' + d.proveedor + '</td>' +
          '<td>' + d.descripcion + ' (' + d.codigo_insumo + ')</td>' +
          '<td>' + d.cantidad + '</td>' +
          '<td>' + (d.motivo || '<em>—</em>') + '</td>' +
          '<td><span class="status-badge ' + badgeClase(d.estado) + '">' + d.estado + '</span>' +
            (d.estado !== 'Pendiente devolución' && d.fecha_resolucion
              ? '<div style="font-size:10px;color:#888;margin-top:3px;">' + d.fecha_resolucion + '</div>' : '') +
          '</td>' +
          '<td>' + (d.estado === 'Pendiente devolución'
            ? '<button class="btn-resolver" onclick="abrirResolver(' + idx + ')">Resolver</button>'
            : '') + '</td>' +
        '</tr>';
      });

      html += '</tbody></table>';
      container.innerHTML = html;
    }

    function abrirResolver(idx) {
      filaEnResolucion = devolucionesGlobal[idx].fila;
      document.getElementById('resolverTitulo').textContent =
        'Resolver: ' + devolucionesGlobal[idx].descripcion + ' — Orden ' + devolucionesGlobal[idx].numero_orden;
      document.getElementById('resolverObs').value = '';
      document.getElementById('resolverPanel').classList.add('visible');
    }

    function cerrarResolver() {
      filaEnResolucion = null;
      document.getElementById('resolverPanel').classList.remove('visible');
    }

    function confirmarResolver() {
      const nuevoEstado = document.getElementById('resolverEstado').value;
      const obs = document.getElementById('resolverObs').value;
      google.script.run
        .withSuccessHandler(function() {
          cerrarResolver();
          document.getElementById('loader').style.display = 'block';
          document.getElementById('content').style.display = 'none';
          google.script.run.withSuccessHandler(dibujarTabla).withFailureHandler(mostrarError).getTodasLasDevoluciones();
        })
        .withFailureHandler(function(e) { alert('❌ ' + e.message); })
        .marcarDevolucionResuelta(filaEnResolucion, nuevoEstado, obs);
    }

    function mostrarError(e) {
      document.getElementById('loader').innerHTML = '<div style="color:#d93025">❌ Error: ' + e.message + '</div>';
    }
  </script>
