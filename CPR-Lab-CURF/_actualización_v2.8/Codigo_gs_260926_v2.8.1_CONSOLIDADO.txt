function getSheet(sheetName) {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(sheetName);
  if (!sheet) throw new Error('No se encontró la hoja: ' + sheetName);
  return sheet;
}

function getCurrentUser() {
  return Session.getActiveUser().getEmail();
}

function formatearFecha(fecha) {
  if (!fecha) return '';
  return Utilities.formatDate(new Date(fecha), Session.getScriptTimeZone(), 'dd/MM/yyyy');
}

function getDatosSheet(sheetName) {
  const sheet   = getSheet(sheetName);
  const data    = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  const headers = data[0];
  return data.slice(1).map(function(row) {
    const obj = {};
    headers.forEach(function(header, index) { obj[header] = row[index]; });
    return obj;
  }).filter(function(row) {
    return Object.values(row).some(function(val) { return val !== '' && val !== null; });
  });
}

function getInsumosActivos() {
  return getDatosSheet(CONFIG.SHEETS.INSUMOS).filter(function(ins) {
    return ins.activo === true || ins.activo === 'TRUE' || ins.activo === 'VERDADERO';
  });
}

function getProveedoresActivos() {
  return getDatosSheet(CONFIG.SHEETS.PROVEEDORES).filter(function(prov) {
    return prov.activo === true || prov.activo === 'TRUE' || prov.activo === 'VERDADERO';
  });
}

function getInsumosPorProveedor(codigoProveedor) {
  // 1. Insumos activos del proveedor
  const insumos = getInsumosActivos().filter(function(ins) {
    return String(ins.proveedor_preferido).trim() === String(codigoProveedor).trim();
  });

  if (insumos.length === 0) return [];

  // 2. Stock consolidado por insumo (suma todos los lotes)
  const stockSheet      = getDatosSheet(CONFIG.SHEETS.STOCK);
  const stockPorCodigo  = {};
  stockSheet.forEach(function(row) {
    const cod = String(row.codigo_insumo).trim();
    if (!stockPorCodigo[cod]) stockPorCodigo[cod] = 0;
    stockPorCodigo[cod] += Number(row.stock_disponible) || 0;
  });

  // 3. Construir resultado con estado de stock
  const resultado = insumos.map(function(ins) {
    const cod     = String(ins.codigo_insumo).trim();
    const stock   = stockPorCodigo[cod] || 0;
    const critico = Number(ins.stock_critico) || 0;
    const estado  = stock <= 0 ? 'Agotado' : stock <= critico ? 'Crítico' : 'OK';

    return {
      codigo_insumo:      cod,
      descripcion:        ins.descripcion,
      categoria:          ins.categoria || 'Sin categoría',
      unidad_medida:      ins.unidad_medida || '',
      precio_referencial: Number(ins.precio_referencial) || 0,
      stock_actual:       stock,
      stock_critico:      critico,
      estado_stock:       estado
    };
  });

  // 4. Ordenar por categoría luego por descripción
  resultado.sort(function(a, b) {
    if (a.categoria < b.categoria) return -1;
    if (a.categoria > b.categoria) return  1;
    return a.descripcion.localeCompare(b.descripcion);
  });

  return resultado;
}

function getInsumo(codigoInsumo) {
  return getDatosSheet(CONFIG.SHEETS.INSUMOS)
    .find(function(ins) { return ins.codigo_insumo === codigoInsumo; });
}

function getProveedor(codigoProveedor) {
  return getDatosSheet(CONFIG.SHEETS.PROVEEDORES)
    .find(function(prov) { return prov.codigo_proveedor === codigoProveedor; });
}

function generarNumeroOrden() {
  const sheet   = getSheet(CONFIG.SHEETS.ORDENES);
  const lastRow = sheet.getLastRow();
  const anio    = new Date().getFullYear();
  if (lastRow <= 1) return 'OC-' + anio + '-001';
  const ultima  = sheet.getRange(lastRow, 1).getValue();
  const numero  = parseInt(ultima.split('-')[2]) + 1;
  return 'OC-' + anio + '-' + numero.toString().padStart(3, '0');
}

/**
 * Crea una o más OC a partir de los datos del formulario.
 * Si los ítems pertenecen a distintos proveedores, genera una OC por proveedor.
 *
 * @param {Object} ordenData
 *   proveedor              {string}  — código del proveedor principal seleccionado
 *   items                  {Array}   — [ { codigo_insumo, descripcion, cantidad, precio_unitario, subtotal } ]
 *   fecha_entrega_estimada {string}
 *   observaciones          {string}
 *   solicitado_por         {string}
 *
 * @returns {string} Número de la primera OC creada
 */
function crearOrdenCompra(ordenData) {
  try {
    const fechaEmision  = new Date();
    const usuario       = getCurrentUser();
    const sheetOrdenes  = getSheet(CONFIG.SHEETS.ORDENES);
    const sheetDetalle  = getSheet(CONFIG.SHEETS.DETALLE_ORDENES);

    // Leer proveedor_preferido de cada insumo desde el catálogo
    const catalogoInsumos = getDatosSheet(CONFIG.SHEETS.INSUMOS);
    const mapaProveedor   = {};
    catalogoInsumos.forEach(function(ins) {
      mapaProveedor[String(ins.codigo_insumo).trim()] =
        String(ins.proveedor_preferido || '').trim();
    });

    // Agrupar items por proveedor
    // Si el ítem es del proveedor principal → usa ordenData.proveedor
    // Si es alternativo → usa su proveedor_preferido del catálogo
    const itemsPorProveedor = {};

    ordenData.items.forEach(function(item) {
      const codIns   = String(item.codigo_insumo).trim();
      const provItem = mapaProveedor[codIns] || ordenData.proveedor;
      if (!itemsPorProveedor[provItem]) itemsPorProveedor[provItem] = [];
      itemsPorProveedor[provItem].push(item);
    });

    // Crear una OC por cada proveedor
    let primeraOrden = null;

    Object.keys(itemsPorProveedor).forEach(function(codProv) {
      const itemsDelProv = itemsPorProveedor[codProv];

      const totalProv = itemsDelProv.reduce(function(sum, item) {
        return sum + (Number(item.subtotal) || 0);
      }, 0);

      const numeroOrden     = generarNumeroOrden();
      const proveedor       = getProveedor(codProv);
      const nombreProveedor = proveedor ? proveedor.razon_social : codProv;

      // Cabecera OC (10 columnas)
      // numero_orden | fecha_emision | codigo_proveedor | nombre_proveedor |
      // estado | usuario_solicita | fecha_entrega_estimada | total | observaciones | solicitado_por
      sheetOrdenes.appendRow([
        numeroOrden,
        fechaEmision,
        codProv,
        nombreProveedor,
        CONFIG.ESTADOS.ORDEN.PENDIENTE,
        usuario,
        ordenData.fecha_entrega_estimada || '',
        totalProv,
        ordenData.observaciones || '',
        ordenData.solicitado_por || ''
      ]);

      // Detalle — renumerar ítem desde 1 por OC
      itemsDelProv.forEach(function(item, idx) {
        sheetDetalle.appendRow([
          numeroOrden,
          idx + 1,
          item.codigo_insumo,
          item.descripcion,
          item.cantidad,
          item.precio_unitario,
          item.subtotal,
          0,                           // cantidad_recepcionada inicial = 0
          CONFIG.ESTADOS.ITEM.PENDIENTE,
          fechaEmision
        ]);
      });

      // Historial de precios
      registrarHistorialPrecios(
        itemsDelProv,
        numeroOrden,
        usuario,
        fechaEmision,
        nombreProveedor
      );

      if (!primeraOrden) primeraOrden = numeroOrden;
    });

    return primeraOrden;

  } catch (error) {
    throw new Error('Error al crear OC: ' + error.message);
  }
}

/**
 * Retorna las OC en estado Pendiente o Recepción Parcial,
 * con el desglose de ítems pendientes por OC.
 * Usada por FormularioRecepcion y ListaPendientes.
 */
function getOrdenesPendientes() {
  try {
    SpreadsheetApp.flush();
    const sheetOrdenes = getSheet(CONFIG.SHEETS.ORDENES);
    const sheetDetalle = getSheet(CONFIG.SHEETS.DETALLE_ORDENES);

    const lastRow = sheetOrdenes.getLastRow();
    if (lastRow <= 1) return [];

    // ✅ NUEVO (v2.7): mapa código→proveedor para saber si tienen email cargado
    // (usado por ListaPendientes para habilitar/deshabilitar Revisar/Enviar)
    const proveedoresMap = {};
    getDatosSheet(CONFIG.SHEETS.PROVEEDORES).forEach(function(p) {
      proveedoresMap[String(p.codigo_proveedor).trim()] = p;
    });

    // ✅ NUEVO (v2.8): set de órdenes con devoluciones pendientes (para el aviso
    // "↩️ Devolución pendiente" en ListaPendientes — ver Devoluciones_Proveedor)
    const ordenesConDevolucionPendiente = {};
    getDevolucionesPendientes().forEach(function(d) {
      ordenesConDevolucionPendiente[String(d.numero_orden).trim()] = true;
    });

    // Leer 10 columnas (incluye nombre_proveedor en col D y solicitado_por en col J)
    const dataOrdenes = sheetOrdenes.getRange(2, 1, lastRow - 1, 10).getValues();
    const dataDetalle = sheetDetalle.getDataRange().getValues();
    const headersDetalle = dataDetalle[0];

    const idxNumOrden    = headersDetalle.indexOf('numero_orden');
    const idxCodigo      = headersDetalle.indexOf('codigo_insumo');
    const idxDescripcion = headersDetalle.indexOf('descripcion');
    const idxCantidad    = headersDetalle.indexOf('cantidad');
    const idxRecepcionada = headersDetalle.indexOf('cantidad_recepcionada');

    return dataOrdenes
      .filter(function(row) {
        const estado = String(row[4]).trim();
        return row[0] &&
          (estado === CONFIG.ESTADOS.ORDEN.PENDIENTE ||
           estado === CONFIG.ESTADOS.ORDEN.PARCIAL);
      })
      .map(function(row) {
        const numeroOrden = row[0];

        const itemsPendientes = [];
        for (let i = 1; i < dataDetalle.length; i++) {
          if (dataDetalle[i][idxNumOrden] !== numeroOrden) continue;
          const solicitado = Number(dataDetalle[i][idxCantidad])    || 0;
          const recibido   = Number(dataDetalle[i][idxRecepcionada]) || 0;
          const pendiente  = solicitado - recibido;
          if (pendiente > 0) {
            itemsPendientes.push({
              codigo:      dataDetalle[i][idxCodigo],
              descripcion: dataDetalle[i][idxDescripcion],
              solicitado,
              recibido,
              pendiente
            });
          }
        }

        return {
          numero_orden:            numeroOrden,
          fecha_emision:           formatearFecha(row[1]),
          codigo_proveedor:        row[2],
          proveedor:               row[3],   // nombre_proveedor guardado en col D
          estado:                  String(row[4]).trim(),
          usuario_solicita:        row[5],
          fecha_entrega_estimada:  formatearFecha(row[6]),
          total_orden:             row[7],
          observaciones:           row[8],
          solicitado_por:          row[9],
          // ✅ NUEVO (v2.7): true si el proveedor tiene al menos un email cargado
          tiene_email:             getEmailsProveedor(proveedoresMap[String(row[2]).trim()]).length > 0,
          // ✅ NUEVO (v2.8): true si esta OC tiene algún ítem rechazado esperando devolución
          tiene_devolucion_pendiente: !!ordenesConDevolucionPendiente[String(numeroOrden).trim()],
          items_pendientes:        itemsPendientes
        };
      });

  } catch (e) {
    throw e;
  }
}

/**
 * Retorna el detalle completo de una OC (cabecera + ítems).
 * Usada por FormularioRecepcion para pre-cargar los datos.
 */
function getDetalleOrden(numeroOrden) {
  const ordenesData = getDatosSheet(CONFIG.SHEETS.ORDENES);
  const ordenRaw    = ordenesData.find(function(o) { return o.numero_orden === numeroOrden; });
  if (!ordenRaw) throw new Error('Orden no encontrada: ' + numeroOrden);

  const orden = Object.assign({}, ordenRaw);
  orden.fecha_emision            = formatearFecha(ordenRaw.fecha_emision);
  orden.fecha_entrega_estimada   = formatearFecha(ordenRaw.fecha_entrega_estimada);

  // Leer detalle con mapeo dinámico de columnas
  const sheetDetalle = getSheet(CONFIG.SHEETS.DETALLE_ORDENES);
  const data         = sheetDetalle.getDataRange().getValues();
  const headers      = data[0];

  const idxNumeroOrden  = headers.indexOf('numero_orden');
  const idxCodigo       = headers.indexOf('codigo_insumo');
  const idxDescripcion  = headers.indexOf('descripcion');
  const idxCantidad     = headers.indexOf('cantidad');
  const idxRecepcionada = headers.indexOf('cantidad_recepcionada');
  const idxPrecio       = headers.indexOf('precio_unitario');
  const idxEstado       = headers.indexOf('estado_item');

  const items = [];
  for (let i = 1; i < data.length; i++) {
    if (data[i][idxNumeroOrden] !== numeroOrden) continue;
    items.push({
      codigo_insumo:        data[i][idxCodigo],
      descripcion:          data[i][idxDescripcion],
      cantidad:             Number(data[i][idxCantidad])    || 0,
      cantidad_recepcionada: Number(data[i][idxRecepcionada]) || 0,
      precio_unitario:      Number(data[i][idxPrecio])      || 0,
      estado_item:          data[i][idxEstado]
    });
  }

  return { orden, items };
}

// ── Generar OC desde la hoja Stock_por_Proveedor ─────────────────────────────

function generarOrdenDesdeStock() {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Stock_por_Proveedor');
  const ui    = SpreadsheetApp.getUi();

  if (!sheet) {
    ui.alert('Primero actualizá la hoja Stock_por_Proveedor desde el menú.');
    return;
  }

  const mapaProp = PropertiesService.getScriptProperties().getProperty('MAPA_PROV_STOCK');
  if (!mapaProp) {
    ui.alert('No se encontró el mapa de proveedores. Actualizá la hoja Stock_por_Proveedor primero.');
    return;
  }
  const mapa    = JSON.parse(mapaProp);
  const allData = sheet.getDataRange().getValues();

  // Buscar filas tildadas y agrupar por proveedor
  const itemsPorProv = {};

  Object.keys(mapa).forEach(function(codProv) {
    const rango = mapa[codProv];
    for (let r = rango.filaInicio - 1; r <= rango.filaFin - 1; r++) {
      const row = allData[r];
      if (row[0] !== true) continue;                // col A = checkbox

      const cod       = String(row[1]).trim();      // col B
      const desc      = String(row[2]).trim();      // col C
      const precioRaw = String(row[6]).replace('$', '').trim();
      const precio    = parseFloat(precioRaw) || 0;
      const cant      = Number(row[9]) || 1;        // col J

      if (!cod) continue;
      if (!itemsPorProv[codProv]) itemsPorProv[codProv] = [];
      itemsPorProv[codProv].push({
        codigo_insumo:   cod,
        descripcion:     desc,
        cantidad:        cant,
        precio_unitario: precio,
        subtotal:        cant * precio
      });
    }
  });

  const proveedoresConItems = Object.keys(itemsPorProv);

  if (proveedoresConItems.length === 0) {
    ui.alert('No hay insumos seleccionados.\n\nTildá los insumos en la columna "✔ Pedir" y volvé a ejecutar.');
    return;
  }

  if (proveedoresConItems.length > 1) {
    const nombres = proveedoresConItems.map(function(c) {
      return '• ' + mapa[c].nombre + ' (' + itemsPorProv[c].length + ' items)';
    }).join('\n');
    const resp = ui.alert(
      'Se crearán ' + proveedoresConItems.length + ' órdenes separadas:',
      nombres + '\n\n¿Confirmás?',
      ui.ButtonSet.YES_NO
    );
    if (resp !== ui.Button.YES) return;
  }

  // Guardar datos para el diálogo HTML
  PropertiesService.getScriptProperties().setProperty('OC_DESDE_STOCK', JSON.stringify({
    proveedores: proveedoresConItems,
    items:       itemsPorProv,
    mapa
  }));

  const datosJSON = JSON.stringify({ proveedores: proveedoresConItems, items: itemsPorProv, mapa });
  const template  = HtmlService.createTemplateFromFile('DialogOrdenStock');
  template.datosOC = datosJSON;

  const html = template.evaluate()
    .setWidth(420).setHeight(320)
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  ui.showModalDialog(html, '🛒 Confirmar Orden de Compra');
}

/**
 * Ejecutada desde DialogOrdenStock.html al confirmar.
 * Crea las OC y limpia los checkboxes de la hoja.
 */
function confirmarOrdenDesdeStock(fechaEntrega, observaciones, solicitadoPor) {
  const datos  = JSON.parse(
    PropertiesService.getScriptProperties().getProperty('OC_DESDE_STOCK')
  );

  // Leer precio_referencial del catálogo como fallback
  const catalogoInsumos = getDatosSheet(CONFIG.SHEETS.INSUMOS);
  const mapaPrecioRef   = {};
  catalogoInsumos.forEach(function(ins) {
    mapaPrecioRef[String(ins.codigo_insumo).trim()] = Number(ins.precio_referencial) || 0;
  });

  const ordenesCreadas = [];

  datos.proveedores.forEach(function(codProv) {
    const items = datos.items[codProv];
    if (!items || items.length === 0) return;

    const itemsConIdx = items.map(function(it, i) {
      let precio = Number(it.precio_unitario) || 0;
      if (precio === 0) precio = mapaPrecioRef[String(it.codigo_insumo).trim()] || 0;
      const subtotal = it.cantidad * precio;
      return {
        item:            i + 1,
        codigo_insumo:   it.codigo_insumo,
        descripcion:     it.descripcion,
        cantidad:        it.cantidad,
        precio_unitario: precio,
        subtotal
      };
    });

    const ordenData = {
      proveedor:              codProv,
      fecha_entrega_estimada: fechaEntrega   || '',
      observaciones:          observaciones  || '',
      solicitado_por:         solicitadoPor  || '',
      items:                  itemsConIdx
    };

    const numeroOrden = crearOrdenCompra(ordenData);
    ordenesCreadas.push(numeroOrden + ' → ' + datos.mapa[codProv].nombre);
  });

  // Limpiar checkboxes
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Stock_por_Proveedor');
  if (sheet) {
    const allData = sheet.getDataRange().getValues();
    for (let r = 0; r < allData.length; r++) {
      if (allData[r][0] === true) sheet.getRange(r + 1, 1).setValue(false);
    }
  }

  PropertiesService.getScriptProperties().deleteProperty('OC_DESDE_STOCK');
  return ordenesCreadas;
}

  SpreadsheetApp.getUi().showModalDialog(html, 'Recepción de Pedido');
}

function generarNumeroRecepcion() {
  const sheet   = getSheet(CONFIG.SHEETS.RECEPCIONES);
  const lastRow = sheet.getLastRow();
  const anio    = new Date().getFullYear();
  if (lastRow <= 1) return 'REC-' + anio + '-001';
  const ultima  = sheet.getRange(lastRow, 1).getValue();
  const numero  = parseInt(ultima.split('-')[2]) + 1;
  return 'REC-' + anio + '-' + numero.toString().padStart(3, '0');
}

/**
 * Registra el ingreso de mercadería a stock.
 *
 * @param {Object} recepcionData
 *   numero_orden   {string}
 *   recibido_por   {string}
 *   observaciones  {string}
 *   items          {Array} — cada item:
 *     { codigo_insumo, cantidad_recibida, cantidad_esperada,
 *       lote, vencimiento, estado_conformidad, observaciones,
 *       precio_recepcion? }
 *
 * @returns {string} Número de recepción generado
 */
function registrarRecepcion(recepcionData) {
  try {
    const numeroRecepcion   = generarNumeroRecepcion();
    const fechaRecepcion    = new Date();
    const usuario           = getCurrentUser();
    const catalogoInsumos   = getInsumosActivos();
    const tablaConversiones = getTablaConversiones();

    const tieneDiscrepancias = recepcionData.items.some(function(item) {
      return item.estado_conformidad !== 'Conforme';
    });

    // ── Cabecera de recepción ──────────────────────────────────────
    // numero_recepcion | fecha | numero_orden | usuario | observaciones | tiene_discrepancias | recibido_por
    getSheet(CONFIG.SHEETS.RECEPCIONES).appendRow([
      numeroRecepcion,
      fechaRecepcion,
      recepcionData.numero_orden,
      usuario,
      recepcionData.observaciones || '',
      tieneDiscrepancias,
      recepcionData.recibido_por  || ''
    ]);

    // ── Acumular filas para escritura batch ────────────────────────
    const sheetDetalleRecep = getSheet(CONFIG.SHEETS.DETALLE_RECEPCIONES);
    const sheetMovimientos  = getSheet(CONFIG.SHEETS.MOVIMIENTOS);

    const filasDetalle  = [];
    const filasMovs     = [];
    const itemsConvInfo = [];
    const itemsRechazados = []; // ✅ NUEVO (v2.8): ítems "No Conforme" — no van a stock, van a Devoluciones_Proveedor

    recepcionData.items.forEach(function(item) {
      const infoInsumo      = catalogoInsumos.find(function(ins) {
        return ins.codigo_insumo === item.codigo_insumo;
      });
      const nombreInsumo    = infoInsumo ? infoInsumo.descripcion : 'No encontrado';
      const categoriaInsumo = infoInsumo ? infoInsumo.categoria   : '';

      const conv = aplicarConversion(
        item.codigo_insumo,
        Number(item.cantidad_recibida),
        tablaConversiones
      );

      const notaConversion = conv.tiene_conversion
        ? '[CONV x' + conv.factor + '] ' + item.cantidad_recibida + ' ' +
          (infoInsumo ? infoInsumo.unidad_medida || 'caja(s)' : 'caja(s)') +
          ' → ' + conv.cantidad_stock + ' ' + conv.unidad_stock
        : '';

      // Fila Detalle_Recepciones
      // numero_recepcion | item | codigo_insumo | nombre | cant_recibida | cant_esperada |
      // estado_conformidad | observaciones | lote | vencimiento | categoria | nota_conversion
      filasDetalle.push([
        numeroRecepcion,
        item.item,
        item.codigo_insumo,
        nombreInsumo,
        item.cantidad_recibida,
        item.cantidad_esperada,
        item.estado_conformidad,
        item.observaciones || '',
        item.lote          || 'SIN LOTE',
        item.vencimiento   || '',
        categoriaInsumo,
        notaConversion
      ]);

      // ✅ NUEVO (v2.8): un ítem "No Conforme" no suma a stock ni a Movimientos_Inventario
      // — queda afuera y se registra en Devoluciones_Proveedor (ver más abajo).
      if (item.estado_conformidad === 'No Conforme') {
        itemsRechazados.push({
          codigo_insumo: item.codigo_insumo,
          descripcion:   nombreInsumo,
          cantidad:      item.cantidad_recibida,
          lote:          item.lote || 'SIN LOTE',
          motivo:        item.observaciones || ''
        });
      } else {
        // Fila Movimientos_Inventario
        filasMovs.push([
          'MOV-' + Date.now() + '-' + Math.random().toString(36).substr(2, 5),
          fechaRecepcion,
          'Ingreso',
          item.codigo_insumo,
          nombreInsumo,
          conv.cantidad_stock,
          numeroRecepcion,
          usuario,
          'Rec. Orden ' + recepcionData.numero_orden +
            (notaConversion ? ' | ' + notaConversion : ''),
          item.lote        || 'SIN LOTE',
          item.vencimiento || '',
          categoriaInsumo
        ]);

        itemsConvInfo.push({
          codigo_insumo:  item.codigo_insumo,
          lote:           item.lote        || 'SIN LOTE',
          vencimiento:    item.vencimiento || '',
          cantidad_stock: conv.cantidad_stock,
          descripcion:    nombreInsumo,
          infoInsumo
        });
      }

      // Historial de precios si el formulario trae precio de recepción
      if (item.precio_recepcion && item.precio_recepcion > 0) {
        registrarHistorialPrecios(
          [{ codigo_insumo: item.codigo_insumo, descripcion: nombreInsumo,
             precio_unitario: item.precio_recepcion }],
          numeroRecepcion,
          usuario,
          fechaRecepcion,
          'Recepción ' + recepcionData.numero_orden
        );
      }
    });

    // ── Escritura batch ────────────────────────────────────────────
    const lastDetalle = sheetDetalleRecep.getLastRow();
    sheetDetalleRecep
      .getRange(lastDetalle + 1, 1, filasDetalle.length, filasDetalle[0].length)
      .setValues(filasDetalle);

    // ✅ NUEVO (v2.8): filasMovs puede venir vacío si TODOS los ítems de esta
    // recepción fueron "No Conforme" — evitar escribir un rango de 0 filas.
    if (filasMovs.length > 0) {
      const lastMovs = sheetMovimientos.getLastRow();
      sheetMovimientos
        .getRange(lastMovs + 1, 1, filasMovs.length, filasMovs[0].length)
        .setValues(filasMovs);
    }

    // ── Actualizar cantidades en la OC y su estado ──────────────────
    // ✅ v2.8: solo los ítems aceptados (No Conforme no cuenta como recepcionado —
    // la cantidad pedida sigue pendiente hasta que el proveedor reponga/acredite).
    const itemsAceptados = recepcionData.items.filter(function(it) {
      return it.estado_conformidad !== 'No Conforme';
    });
    if (itemsAceptados.length > 0) {
      actualizarCantidadesEnOrden(Object.assign({}, recepcionData, { items: itemsAceptados }));
    } else {
      // Ningún ítem aceptado en esta recepción — igual hay que recalcular el
      // estado de la orden por si otra recepción anterior ya la había avanzado.
      actualizarEstadoOrden(recepcionData.numero_orden);
    }

    // ── Stock incremental (solo los lotes afectados, ya excluye No Conforme) ──
    if (itemsConvInfo.length > 0) {
      actualizarStockIncremental(itemsConvInfo, catalogoInsumos);
    }

    // ── Devoluciones a proveedor (ítems No Conforme) ────────────────
    if (itemsRechazados.length > 0) {
      const proveedorOrden = getProveedor(
        (getDatosSheet(CONFIG.SHEETS.ORDENES).find(function(o) {
          return o.numero_orden === recepcionData.numero_orden;
        }) || {}).proveedor
      );
      const nombreProveedor = (proveedorOrden && proveedorOrden.razon_social) || recepcionData.numero_orden;
      registrarDevolucionProveedor(itemsRechazados, recepcionData.numero_orden, nombreProveedor, usuario, fechaRecepcion);
    }

    return numeroRecepcion;

  } catch (error) {
    throw new Error('No se pudo registrar la recepción: ' + error.message);
  }
}

function actualizarCantidadesEnOrden(recepcionData) {
  const sheetDetalle = getSheet(CONFIG.SHEETS.DETALLE_ORDENES);
  const data         = sheetDetalle.getDataRange().getValues();

  recepcionData.items.forEach(function(itemRec) {
    for (let i = 1; i < data.length; i++) {
      if (data[i][0] !== recepcionData.numero_orden) continue;
      if (data[i][2] !== itemRec.codigo_insumo)      continue;

      const cantPrevia     = Number(data[i][7]) || 0;
      const nuevaCant      = cantPrevia + Number(itemRec.cantidad_recibida);
      const cantSolicitada = Number(data[i][4]);
      const estado         = nuevaCant >= cantSolicitada
        ? CONFIG.ESTADOS.ITEM.COMPLETO
        : CONFIG.ESTADOS.ITEM.PARCIAL;

      // Columnas H (índice 7→col 8) e I (índice 8→col 9) en una sola escritura
      sheetDetalle.getRange(i + 1, 8, 1, 2).setValues([[nuevaCant, estado]]);
      break;
    }
  });

  actualizarEstadoOrden(recepcionData.numero_orden);
}

function actualizarEstadoOrden(numeroOrden) {
  const sheetDetalle = getSheet(CONFIG.SHEETS.DETALLE_ORDENES);
  const data         = sheetDetalle.getDataRange().getValues();
  const items        = data.slice(1).filter(function(r) { return r[0] === numeroOrden; });
  if (items.length === 0) return;

  const estados         = items.map(function(item) { return item[8]; });
  const todosCompletos  = estados.every(function(e) { return e === CONFIG.ESTADOS.ITEM.COMPLETO; });
  const todosPendientes = estados.every(function(e) { return e === CONFIG.ESTADOS.ITEM.PENDIENTE; });

  const nuevoEstado = todosCompletos  ? CONFIG.ESTADOS.ORDEN.RECEPCIONADA :
                      todosPendientes ? CONFIG.ESTADOS.ORDEN.PENDIENTE    :
                                        CONFIG.ESTADOS.ORDEN.PARCIAL;

  const sheetOrdenes  = getSheet(CONFIG.SHEETS.ORDENES);
  const dataOrdenes   = sheetOrdenes.getDataRange().getValues();

  for (let i = 1; i < dataOrdenes.length; i++) {
    if (dataOrdenes[i][0] !== numeroOrden) continue;
    if (dataOrdenes[i][4] !== nuevoEstado) {
      sheetOrdenes.getRange(i + 1, 5).setValue(nuevoEstado); // col E = estado
    }
    break;
  }
}

/**
 * Reconstrucción total de Stock_Actual a partir de Movimientos_Inventario.
 * Usar sólo cuando se necesite recalcular todo (ej. después de un ajuste masivo).
 * Para operaciones normales usar actualizarStockIncremental().
 */
function actualizarStockActual() {
  const sheetStock  = getSheet(CONFIG.SHEETS.STOCK);
  const movimientos = getDatosSheet(CONFIG.SHEETS.MOVIMIENTOS);
  const insumos     = getDatosSheet(CONFIG.SHEETS.INSUMOS);

  // 1. Calcular stock por código+lote
  const stockDetallado = {};
  movimientos.forEach(function(mov) {
    const loteId = mov.lote ? String(mov.lote).trim() : 'SIN LOTE';
    const llave  = String(mov.codigo_insumo).trim() + '|' + loteId;

    if (!stockDetallado[llave]) {
      stockDetallado[llave] = {
        cantidad:    0,
        vencimiento: mov.vencimiento || '',
        codigo:      mov.codigo_insumo,
        lote:        loteId
      };
    }

    const cant = Number(mov.cantidad) || 0;
    if (mov.tipo_movimiento === 'Ingreso' || mov.tipo_movimiento === 'Ajuste') {
      stockDetallado[llave].cantidad += cant;
    } else if (mov.tipo_movimiento === 'Egreso') {
      stockDetallado[llave].cantidad -= cant;
    }
  });

  // 2. Stock total por insumo (para calcular estado)
  const stockTotalPorInsumo = {};
  Object.keys(stockDetallado).forEach(function(llave) {
    const item   = stockDetallado[llave];
    const codigo = item.codigo;
    if (!stockTotalPorInsumo[codigo]) stockTotalPorInsumo[codigo] = 0;
    stockTotalPorInsumo[codigo] += item.cantidad;
  });

  // 3. Limpiar hoja
  if (sheetStock.getLastRow() > 1) {
    sheetStock.getRange(2, 1, sheetStock.getLastRow() - 1, sheetStock.getLastColumn())
      .clearContent();
  }

  const filasStock = [];
  const ahora      = new Date();

  Object.keys(stockDetallado).forEach(function(llave) {
    const item = stockDetallado[llave];
    const ins  = insumos.find(function(i) { return i.codigo_insumo === item.codigo; });

    if (!ins || item.cantidad <= 0) return;

    const critico    = Number(ins.stock_critico) || 0;
    const stockTotal = stockTotalPorInsumo[item.codigo] || 0;
    const estado     = stockTotal <= 0 ? 'Agotado' :
                       stockTotal <= critico ? 'Crítico' : 'OK';

    filasStock.push([
      item.codigo,           // A: codigo_insumo
      ins.descripcion,       // B: descripcion
      item.cantidad,         // C: stock_disponible (este lote)
      critico,               // D: stock_critico
      estado,                // E: estado (basado en stock TOTAL)
      ahora,                 // F: fecha_actualizacion
      item.lote,             // G: lote
      item.vencimiento,      // H: vencimiento
      ins.clasificacion || ''// I: clasificacion
    ]);
  });

  if (filasStock.length > 0) {
    sheetStock.getRange(2, 1, filasStock.length, 9).setValues(filasStock);
    // FIFO: ordenar por vencimiento ascendente
    sheetStock.getRange(2, 1, filasStock.length, 9).sort({ column: 8, ascending: true });
  }

  aplicarFormatoStock();
  actualizarStockPorProveedor();
}

/**
 * Actualización incremental: solo toca los lotes afectados por la operación.
 * Mucho más rápida que actualizarStockActual() para operaciones individuales.
 *
 * @param {Array}  itemsRecibidos  — lista de { codigo_insumo, lote, vencimiento, cantidad_stock, descripcion, infoInsumo }
 * @param {Array}  catalogoInsumos — resultado de getInsumosActivos()
 */
function actualizarStockIncremental(itemsRecibidos, catalogoInsumos) {
  const sheetStock = getSheet(CONFIG.SHEETS.STOCK);
  const lastRow    = sheetStock.getLastRow();
  const ahora      = new Date();

  const dataStock = lastRow > 1
    ? sheetStock.getRange(2, 1, lastRow - 1, 9).getValues()
    : [];

  // Índice rápido: "codigo|lote" → posición en dataStock
  const mapaStock = {};
  dataStock.forEach(function(row, idx) {
    const cod  = String(row[0]).trim();
    const lote = String(row[6]).trim();
    mapaStock[cod + '|' + lote] = idx;
  });

  // Stock total actual por código (para recalcular estado al final)
  const stockTotalPorCodigo = {};
  dataStock.forEach(function(row) {
    const cod = String(row[0]).trim();
    stockTotalPorCodigo[cod] = (stockTotalPorCodigo[cod] || 0) + (Number(row[2]) || 0);
  });

  const codigosAfectados = {};

  itemsRecibidos.forEach(function(item) {
    const cod       = String(item.codigo_insumo).trim();
    const lote      = String(item.lote).trim();
    const llave     = cod + '|' + lote;
    const cantAgregar = Number(item.cantidad_stock) || 0;

    codigosAfectados[cod] = true;

    if (mapaStock.hasOwnProperty(llave)) {
      // Lote existente: sumar cantidad
      const idx    = mapaStock[llave];
      dataStock[idx][2] = (Number(dataStock[idx][2]) || 0) + cantAgregar;
      dataStock[idx][5] = ahora;
      stockTotalPorCodigo[cod] = (stockTotalPorCodigo[cod] || 0) + cantAgregar;

    } else {
      // Lote nuevo: agregar fila
      const ins     = item.infoInsumo;
      const critico = ins ? Number(ins.stock_critico) || 0 : 0;

      const nuevaFila = [
        cod,
        item.descripcion,
        cantAgregar,
        critico,
        'OK',           // se recalcula abajo
        ahora,
        lote,
        item.vencimiento,
        ins ? ins.clasificacion || '' : ''
      ];
      dataStock.push(nuevaFila);
      mapaStock[llave] = dataStock.length - 1;
      stockTotalPorCodigo[cod] = (stockTotalPorCodigo[cod] || 0) + cantAgregar;
    }
  });

  // Recalcular estado para todos los lotes de los códigos afectados
  dataStock.forEach(function(row) {
    const cod = String(row[0]).trim();
    if (!codigosAfectados[cod]) return;

    const ins     = catalogoInsumos.find(function(i) {
      return String(i.codigo_insumo).trim() === cod;
    });
    const critico = ins ? Number(ins.stock_critico) || 0 : Number(row[3]) || 0;
    const total   = stockTotalPorCodigo[cod] || 0;

    row[4] = total <= 0 ? 'Agotado' : total <= critico ? 'Crítico' : 'OK';
  });

  // Filtrar filas con stock > 0 y reescribir
  const filasValidas = dataStock.filter(function(row) { return Number(row[2]) > 0; });

  if (sheetStock.getLastRow() > 1) {
    sheetStock.getRange(2, 1, sheetStock.getLastRow() - 1, 9).clearContent();
  }
  if (filasValidas.length > 0) {
    sheetStock.getRange(2, 1, filasValidas.length, 9).setValues(filasValidas);
    sheetStock.getRange(2, 1, filasValidas.length, 9).sort({ column: 8, ascending: true });
  }
  // Stock_por_Proveedor NO se actualiza aquí — es una vista de consulta.
}

function procesarBajaDesdeHoja() {
  const ui         = SpreadsheetApp.getUi();
  const sheetBajas = getSheet('Bajas_Consumo');

  const valorB2          = String(sheetBajas.getRange('B2').getValue()).trim();
  const codigoEnC2       = String(sheetBajas.getRange('C2').getValue()).trim();
  const loteSeleccionado = String(sheetBajas.getRange('B3').getValue()).trim();
  const cantidadBaja     = Number(sheetBajas.getRange('B4').getValue());
  const observacion      = String(sheetBajas.getRange('B5').getValue()).trim();

  // C2 siempre contiene el código resuelto (puesto por _cargarInsumoEnBajas)
  const valorBusqueda = codigoEnC2 || valorB2;

  if (!valorBusqueda || !loteSeleccionado || cantidadBaja <= 0) {
    ui.alert('⚠️ Error: Complete Insumo, Lote y una Cantidad válida.');
    return;
  }

  const stockData = getDatosSheet(CONFIG.SHEETS.STOCK);

  const itemStock = stockData.find(function(s) {
    const codS  = String(s.codigo_insumo || '').trim();
    const descS = String(s.descripcion   || '').trim();
    const loteS = String(s.lote          || '').trim();
    return (codS === valorBusqueda || descS === valorBusqueda) &&
           loteS === loteSeleccionado;
  });

  if (!itemStock || Number(itemStock.stock_disponible) < cantidadBaja) {
    const disp = itemStock ? itemStock.stock_disponible : 0;
    ui.alert('❌ Stock insuficiente en el lote: ' + loteSeleccionado + '\nDisponible: ' + disp);
    return;
  }

  const confirmar = ui.alert(
    '¿Confirmar consumo de ' + cantidadBaja + ' unidad(es) de ' + itemStock.descripcion + '?',
    ui.ButtonSet.YES_NO
  );
  if (confirmar !== ui.Button.YES) return;

  try {
    const infoInsumo = getInsumo(itemStock.codigo_insumo);
    const fechaAhora = new Date();
    const usuario    = getCurrentUser();

    // Registrar egreso en Movimientos_Inventario
    getSheet(CONFIG.SHEETS.MOVIMIENTOS).appendRow([
      'MOV-' + Date.now(),
      fechaAhora,
      'Egreso',
      itemStock.codigo_insumo,
      itemStock.descripcion,
      cantidadBaja,
      'BAJA-HOJA',
      usuario,
      'Consumo: ' + observacion,
      loteSeleccionado,
      itemStock.vencimiento || '',
      infoInsumo ? infoInsumo.categoria : ''
    ]);

    // Stock incremental (egreso)
    const sheetStock    = getSheet(CONFIG.SHEETS.STOCK);
    const dataStock     = sheetStock.getDataRange().getValues();
    const catalogoIns   = getInsumosActivos();

    const stockTotalPorCodigo = {};
    dataStock.slice(1).forEach(function(row) {
      const cod = String(row[0]).trim();
      stockTotalPorCodigo[cod] = (stockTotalPorCodigo[cod] || 0) + (Number(row[2]) || 0);
    });

    const codBaja  = String(itemStock.codigo_insumo).trim();
    const loteBaja = String(loteSeleccionado).trim();

    for (let i = 1; i < dataStock.length; i++) {
      if (String(dataStock[i][0]).trim() !== codBaja) continue;
      if (String(dataStock[i][6]).trim() !== loteBaja) continue;
      dataStock[i][2] = Math.max((Number(dataStock[i][2]) || 0) - cantidadBaja, 0);
      dataStock[i][5] = fechaAhora;
      stockTotalPorCodigo[codBaja] = (stockTotalPorCodigo[codBaja] || 0) - cantidadBaja;
      break;
    }

    const ins      = catalogoIns.find(function(i) {
      return String(i.codigo_insumo).trim() === codBaja;
    });
    const critico    = ins ? Number(ins.stock_critico) || 0 : 0;
    const totalFinal = stockTotalPorCodigo[codBaja] || 0;
    const nuevoEst   = totalFinal <= 0 ? 'Agotado' : totalFinal <= critico ? 'Crítico' : 'OK';

    for (let j = 1; j < dataStock.length; j++) {
      if (String(dataStock[j][0]).trim() === codBaja) dataStock[j][4] = nuevoEst;
    }

    const filasValidas = dataStock.slice(1).filter(function(row) { return Number(row[2]) > 0; });
    if (sheetStock.getLastRow() > 1) {
      sheetStock.getRange(2, 1, sheetStock.getLastRow() - 1, 9).clearContent();
    }
    if (filasValidas.length > 0) {
      sheetStock.getRange(2, 1, filasValidas.length, 9).setValues(filasValidas);
      sheetStock.getRange(2, 1, filasValidas.length, 9).sort({ column: 8, ascending: true });
    }

    sheetBajas.getRange('B2:B5').clearContent();
    ui.alert('✅ Consumo registrado exitosamente.\n\nStock actualizado.');

  } catch (e) {
    ui.alert('❌ Error crítico: ' + e.message);
  }
}

// ── onEdit: búsqueda inteligente en Bajas_Consumo ────────────────────────────

function onEdit(e) {
  const range = e.range;
  const sheet = range.getSheet();

  if (sheet.getName() !== 'Bajas_Consumo') return;

  // CASO 1: Se editó B2 (búsqueda de insumo)
  if (range.getA1Notation() === 'B2') {
    const valorIngresado   = String(range.getValue()).trim();
    const celdaLote        = sheet.getRange('B3');
    const celdaInfo        = sheet.getRange('C2');
    const celdaDescripcion = sheet.getRange('D2');

    range.clearDataValidations();   // ← NUEVO: limpia cualquier validación fantasma de búsquedas previas
    celdaLote.clearDataValidations();
    celdaLote.clearContent();
    sheet.getRange('C3').clearContent();
    celdaInfo.clearContent();
    celdaDescripcion.clearContent();

    if (valorIngresado === '') return;

    const stockData = getDatosSheet(CONFIG.SHEETS.STOCK);

    // Formato "COD — Descripción" (viene de dropdown múltiple)
    if (valorIngresado.indexOf(' — ') !== -1) {
      const partes    = valorIngresado.split(' — ');
      const codLimpio = partes[0].trim();
      const descLimpia = partes.slice(1).join(' — ').trim();
      range.clearDataValidations();
      range.setValue(codLimpio);
      _cargarInsumoEnBajas(sheet, stockData, codLimpio, descLimpia,
        celdaLote, celdaInfo, celdaDescripcion);
      return;
    }

    // Match exacto por código
    const matchCod = stockData.find(function(s) {
      return String(s.codigo_insumo).trim().toUpperCase() === valorIngresado.toUpperCase();
    });
    if (matchCod) {
      _cargarInsumoEnBajas(sheet, stockData,
        String(matchCod.codigo_insumo).trim(), String(matchCod.descripcion).trim(),
        celdaLote, celdaInfo, celdaDescripcion);
      return;
    }

    // Match exacto por descripción
    const valorNorm = valorIngresado.toUpperCase().replace(/\s+/g, ' ');
    const matchDesc = stockData.find(function(s) {
      return String(s.descripcion).trim().toUpperCase().replace(/\s+/g, ' ') === valorNorm;
    });
    if (matchDesc) {
      _cargarInsumoEnBajas(sheet, stockData,
        String(matchDesc.codigo_insumo).trim(), String(matchDesc.descripcion).trim(),
        celdaLote, celdaInfo, celdaDescripcion);
      return;
    }

    // Búsqueda parcial
    const codigosMatch = {};
    stockData.forEach(function(s) {
      const descNorm = String(s.descripcion).trim().toUpperCase().replace(/\s+/g, ' ');
      const codNorm  = String(s.codigo_insumo).trim().toUpperCase();
      if (descNorm.indexOf(valorNorm) !== -1 || codNorm.indexOf(valorNorm) !== -1) {
        const cod = String(s.codigo_insumo).trim();
        if (!codigosMatch[cod]) codigosMatch[cod] = String(s.descripcion).trim();
      }
    });

    const keys = Object.keys(codigosMatch);

    if (keys.length === 0) {
      celdaInfo.setValue('❌ Sin resultados para: ' + valorIngresado);
      return;
    }
    if (keys.length === 1) {
      _cargarInsumoEnBajas(sheet, stockData,
        keys[0], codigosMatch[keys[0]],
        celdaLote, celdaInfo, celdaDescripcion);
      return;
    }

    // Múltiples coincidencias → dropdown en B2
    const opciones = keys.map(function(cod) { return cod + ' — ' + codigosMatch[cod]; });
    const ruleB2   = SpreadsheetApp.newDataValidation()
      .requireValueInList(opciones, true).setAllowInvalid(true).build();
    range.setDataValidation(ruleB2);
    range.setValue('');
    celdaInfo.setValue('↓ ' + keys.length + ' coincidencias — elegí de la lista');
  }

  // CASO 2: Se editó B3 (selección de lote)
  if (range.getA1Notation() === 'B3') {
    const codigoInsumo     = String(sheet.getRange('B2').getValue()).trim();
    const loteSeleccionado = String(range.getValue()).trim();
    if (!codigoInsumo || !loteSeleccionado) return;
    const stockData = getDatosSheet(CONFIG.SHEETS.STOCK);
    mostrarInfoLote(sheet, codigoInsumo, loteSeleccionado, stockData);
  }
}

function _cargarInsumoEnBajas(sheet, stockData, codigoFinal, descFinal,
                               celdaLote, celdaInfo, celdaDescripcion) {
  // Poner el código limpio en B2 para que procesarBajaDesdeHoja lo lea bien
  sheet.getRange('B2').setValue(codigoFinal);
  celdaInfo.setValue(codigoFinal);
  celdaDescripcion.setValue(descFinal);

  const lotesDisponibles = stockData.filter(function(s) {
    return String(s.codigo_insumo).trim() === codigoFinal &&
           Number(s.stock_disponible) > 0;
  });

  if (lotesDisponibles.length === 0) {
    celdaInfo.setValue('⚠️ Sin stock: ' + codigoFinal);
    return;
  }

  // FIFO: ordenar por vencimiento
  lotesDisponibles.sort(function(a, b) {
    const fA = a.vencimiento ? new Date(a.vencimiento) : new Date(2099, 11, 31);
    const fB = b.vencimiento ? new Date(b.vencimiento) : new Date(2099, 11, 31);
    return fA - fB;
  });

  const listLotes = lotesDisponibles.map(function(s) { return String(s.lote || 'SIN LOTE'); });

  const rule = SpreadsheetApp.newDataValidation()
    .requireValueInList(listLotes, true).setAllowInvalid(false).build();
  celdaLote.setDataValidation(rule);
  celdaLote.setValue(listLotes[0]);
  mostrarInfoLote(sheet, codigoFinal, listLotes[0], lotesDisponibles);
}

function mostrarInfoLote(sheet, codigoInsumo, loteSeleccionado, stockData) {
  const itemLote = stockData.find(function(s) {
    return String(s.codigo_insumo).trim() === codigoInsumo &&
           String(s.lote || '').trim()    === loteSeleccionado;
  });
  if (!itemLote) return;
  sheet.getRange('C3').setValue(
    'Disponible: ' + (itemLote.stock_disponible || 0) +
    (itemLote.vencimiento ? ' | Vence: ' + formatearFecha(itemLote.vencimiento) : '')
  );
}

      .setRanges([rangeEstado]).build()
  ];
  sheetStock.setConditionalFormatRules(rules);

function actualizarDashboard() {
  const sheet = getSheet(CONFIG.SHEETS.DASHBOARD);
  sheet.clear();
  sheet.getCharts().forEach(function(c) { sheet.removeChart(c); });
  sheet.setTabColor('#4285f4');

  sheet.getRange('B2')
    .setValue('📊 Panel de Control — Gestión de Compras CURF Microbiología')
    .setFontSize(18).setFontWeight('bold').setFontColor('#202124');

  // ── KPIs principales ──────────────────────────────────────────
  const ordenes    = getDatosSheet(CONFIG.SHEETS.ORDENES);
  const pendientes = ordenes.filter(function(o) {
    return o.estado === CONFIG.ESTADOS.ORDEN.PENDIENTE ||
           o.estado === CONFIG.ESTADOS.ORDEN.PARCIAL;
  }).length;

  // Stock consolidado por insumo (suma de todos los lotes)
  const stockDetallado    = getDatosSheet(CONFIG.SHEETS.STOCK);
  const stockConsolidado  = {};
  stockDetallado.forEach(function(item) {
    const cod = item.codigo_insumo;
    if (!stockConsolidado[cod]) {
      stockConsolidado[cod] = {
        descripcion:  item.descripcion,
        stock_total:  0,
        nivel_critico: Number(item.stock_critico) || 0
      };
    }
    stockConsolidado[cod].stock_total += Number(item.stock_disponible) || 0;
  });

  const itemsDashboard = Object.keys(stockConsolidado).map(function(cod) {
    const item   = stockConsolidado[cod];
    const estado = item.stock_total <= 0            ? 'Agotado' :
                   item.stock_total <= item.nivel_critico ? 'Crítico' : 'OK';
    return {
      codigo:      cod,
      descripcion: item.descripcion,
      total:       item.stock_total,
      critico:     item.nivel_critico,
      estado
    };
  });

  const criticos = itemsDashboard.filter(function(i) { return i.estado === 'Crítico'; }).length;
  const agotados = itemsDashboard.filter(function(i) { return i.estado === 'Agotado'; }).length;

  crearTarjetaKPI(sheet, 4, 2, 'Órdenes Pendientes', pendientes, '#fbbc04');
  crearTarjetaKPI(sheet, 4, 5, 'Stock Crítico',      criticos,   '#ea4335');
  crearTarjetaKPI(sheet, 4, 8, 'Insumos Agotados',   agotados,   '#a61c00');

  // ── Tabla de alertas ──────────────────────────────────────────
  sheet.getRange('B8')
    .setValue('⚠️ Alertas de Reposición Prioritaria')
    .setFontSize(14).setFontWeight('bold');

  sheet.getRange('B10:F10')
    .setValues([['Código', 'Descripción', 'Stock Total', 'Nivel Crítico', 'Estado']])
    .setBackground('#4285f4').setFontColor('white').setFontWeight('bold');

  const itemsAlerta = itemsDashboard
    .filter(function(i) { return i.estado === 'Crítico' || i.estado === 'Agotado'; })
    .sort(function(a, b) { return a.total - b.total; });

  let filaLibre = 12;

  if (itemsAlerta.length > 0) {
    const filasTabla = itemsAlerta.map(function(i) {
      return [i.codigo, i.descripcion, i.total, i.critico, i.estado];
    });
    const rangoTabla = sheet.getRange(11, 2, filasTabla.length, 5);
    rangoTabla.setValues(filasTabla);
    rangoTabla.setBorder(true, true, true, true, true, true,
      '#dadce0', SpreadsheetApp.BorderStyle.SOLID);

    itemsAlerta.forEach(function(item, index) {
      sheet.getRange(11 + index, 2, 1, 5)
        .setBackground(item.estado === 'Agotado' ? '#fce8e6' : '#fef7e0');
    });
    filaLibre = 11 + filasTabla.length + 2;
  } else {
    sheet.getRange('B11')
      .setValue('✅ Todo el inventario está en niveles óptimos')
      .setFontStyle('italic').setFontColor('#137333');
    filaLibre = 14;
  }

  // ── Secciones de precios y gastos ─────────────────────────────
  const filaPostPrecios = agregarSeccionPrecios(sheet, filaLibre + 1);
  agregarSeccionGastos(sheet, filaPostPrecios + 1);

  sheet.getRange('B1')
    .setValue('Última actualización: ' + formatearFecha(new Date()) +
              ' ' + new Date().toLocaleTimeString());
}

}

/**
 * Registra variaciones de precio al crear una OC o al recepcionar.
 * Solo registra si hay precio cargado y si hubo cambio (o es primer registro).
 *
 * @param {Array}  items          — [ { codigo_insumo, descripcion, precio_unitario } ]
 * @param {string} numeroOrden
 * @param {string} usuario
 * @param {Date}   fecha
 * @param {string} nombreProveedor
 */
function registrarHistorialPrecios(items, numeroOrden, usuario, fecha, nombreProveedor) {
  const ss        = SpreadsheetApp.getActiveSpreadsheet();
  const sheetHist = ss.getSheetByName('Historial_Precios');
  const sheetIns  = getSheet(CONFIG.SHEETS.INSUMOS);

  if (!sheetHist) return; // la hoja puede no existir aún en un deploy nuevo

  const dataIns    = sheetIns.getDataRange().getValues();
  const headersIns = dataIns[0];
  const idxCodIns  = headersIns.indexOf('codigo_insumo');
  const idxPrecio  = headersIns.indexOf('precio_referencial');
  const idxDesc    = headersIns.indexOf('descripcion');

  items.forEach(function(item) {
    const precioNuevo = Number(item.precio_unitario) || 0;
    if (precioNuevo <= 0) return;

    let filaInsumo     = -1;
    let precioAnterior = 0;
    let descripcion    = item.descripcion || '';

    for (let r = 1; r < dataIns.length; r++) {
      if (String(dataIns[r][idxCodIns]).trim() !== String(item.codigo_insumo).trim()) continue;
      filaInsumo     = r + 1; // 1-indexed
      precioAnterior = Number(dataIns[r][idxPrecio]) || 0;
      descripcion    = String(dataIns[r][idxDesc]) || descripcion;
      break;
    }

    const variacionPct = precioAnterior > 0
      ? Math.round(((precioNuevo - precioAnterior) / precioAnterior) * 10000) / 100
      : 0;

    const esNuevo   = precioAnterior === 0;
    const hayCambio = precioAnterior !== precioNuevo;

    if (esNuevo || hayCambio) {
      sheetHist.appendRow([
        fecha,
        item.codigo_insumo,
        descripcion,
        nombreProveedor,
        esNuevo ? '-' : precioAnterior,
        precioNuevo,
        esNuevo ? '-' : variacionPct,
        numeroOrden,
        usuario
      ]);
    }

    // Actualizar precio_referencial en catálogo
    if (filaInsumo > 0 && (hayCambio || esNuevo)) {
      sheetIns.getRange(filaInsumo, idxPrecio + 1).setValue(precioNuevo);
    }
  });
}

/**
 * Permite al usuario consultar el historial de precio de un insumo.
 * Menú → 📈 Ver Historial Precios
 */
function mostrarHistorialPrecios() {
  const ui   = SpreadsheetApp.getUi();
  const resp = ui.prompt(
    '📈 Historial de Precios',
    'Ingresá el código del insumo (ej: REA001):',
    ui.ButtonSet.OK_CANCEL
  );
  if (resp.getSelectedButton() !== ui.Button.OK) return;

  const codigo = resp.getResponseText().trim().toUpperCase();
  if (!codigo) return;

  const ss        = SpreadsheetApp.getActiveSpreadsheet();
  const sheetHist = ss.getSheetByName('Historial_Precios');
  if (!sheetHist || sheetHist.getLastRow() < 3) {
    ui.alert('No hay historial de precios registrado todavía.');
    return;
  }

  const data     = sheetHist.getDataRange().getValues();
  const headers  = data[1]; // fila 2 = headers (fila 1 = banner)
  const idxCod   = headers.indexOf('codigo_insumo');
  const idxFecha = headers.indexOf('fecha');
  const idxPAnt  = headers.indexOf('precio_anterior');
  const idxPNuevo = headers.indexOf('precio_nuevo');
  const idxVar   = headers.indexOf('variacion_pct');
  const idxOrden = headers.indexOf('numero_orden');

  const registros = data.slice(2).filter(function(r) {
    return String(r[idxCod]).trim().toUpperCase() === codigo;
  });

  if (registros.length === 0) {
    ui.alert('No se encontraron registros para "' + codigo + '".\n\n' +
             'Verificá el código y que ya exista al menos una OC con ese insumo.');
    return;
  }

  const lineas = ['Historial para ' + codigo + ' (' + registros.length + ' registros):\n'];
  registros.slice(-10).reverse().forEach(function(r) {
    const fecha  = r[idxFecha] ? formatearFecha(r[idxFecha]) : '-';
    const pAnt   = r[idxPAnt] !== '-' ? '$' + Number(r[idxPAnt]).toFixed(2) : 'sin precio previo';
    const pNuevo = '$' + Number(r[idxPNuevo]).toFixed(2);
    const varStr = r[idxVar] !== '-'
      ? ' (' + (Number(r[idxVar]) > 0 ? '+' : '') + Number(r[idxVar]).toFixed(1) + '%)'
      : '';
    lineas.push(fecha + ' | ' + pAnt + ' → ' + pNuevo + varStr + ' | ' + r[idxOrden]);
  });

  ui.alert(lineas.join('\n'));
}

/**
 * Agrega la sección de variación de precios al Dashboard.
 * @returns {number} Primera fila libre después de la sección
 */
function agregarSeccionPrecios(sheet, filaInicio) {
  const ss        = SpreadsheetApp.getActiveSpreadsheet();
  const sheetHist = ss.getSheetByName('Historial_Precios');

  sheet.getRange(filaInicio, 2)
    .setValue('📈 Variación de Precios — vs. compra anterior')
    .setFontSize(14).setFontWeight('bold').setFontColor('#202124');

  if (!sheetHist || sheetHist.getLastRow() < 3) {
    sheet.getRange(filaInicio + 1, 2)
      .setValue('Sin registros aún. Se completará al crear órdenes de compra.')
      .setFontStyle('italic').setFontColor('#5f6368');
    return filaInicio + 3;
  }

  const data     = sheetHist.getDataRange().getValues();
  const headers  = data[1];
  const idxFecha  = headers.indexOf('fecha');
  const idxCod    = headers.indexOf('codigo_insumo');
  const idxDesc   = headers.indexOf('descripcion');
  const idxPAnt   = headers.indexOf('precio_anterior');
  const idxPNuevo = headers.indexOf('precio_nuevo');
  const idxVar    = headers.indexOf('variacion_pct');

  // KPI: insumos con cambio de precio este mes
  const mesActual  = new Date().getMonth();
  const anioActual = new Date().getFullYear();
  const codigosEsteMes = {};

  for (let r2 = 2; r2 < data.length; r2++) {
    const fila = data[r2];
    if (!fila[idxFecha]) continue;
    const f2 = new Date(fila[idxFecha]);
    if (isNaN(f2.getTime())) continue;
    if (f2.getMonth() === mesActual && f2.getFullYear() === anioActual &&
        fila[idxVar] !== '-') {
      codigosEsteMes[String(fila[idxCod]).trim()] = true;
    }
  }

  crearTarjetaKPI(sheet, filaInicio + 1, 2,
    'Insumos con cambio\nde precio (este mes)',
    Object.keys(codigosEsteMes).length, '#7248b9');

  // Tabla: último precio por insumo (solo los que tuvieron variación)
  const porInsumo = {};
  for (let r = 2; r < data.length; r++) {
    const row   = data[r];
    if (!row[idxFecha]) continue;
    const fecha = new Date(row[idxFecha]);
    if (isNaN(fecha.getTime())) continue;

    const cod    = String(row[idxCod]).trim();
    const pNuevo = Number(row[idxPNuevo]);
    const pAnt   = row[idxPAnt] !== '-' ? Number(row[idxPAnt]) : null;

    if (!porInsumo[cod]) {
      porInsumo[cod] = { descripcion: String(row[idxDesc]), pAnt, pNuevo, fecha };
    } else if (fecha > porInsumo[cod].fecha) {
      porInsumo[cod].pAnt   = pAnt;
      porInsumo[cod].pNuevo = pNuevo;
      porInsumo[cod].fecha  = fecha;
    }
  }

  const filasConsolidadas = [];
  Object.keys(porInsumo).forEach(function(cod) {
    const ins = porInsumo[cod];
    if (ins.pAnt === null || ins.pAnt === ins.pNuevo) return;
    const varPct = ins.pAnt > 0
      ? Math.round(((ins.pNuevo - ins.pAnt) / ins.pAnt) * 10000) / 100
      : 0;
    filasConsolidadas.push({
      cod, desc: ins.descripcion,
      pAnt: ins.pAnt, pNuevo: ins.pNuevo,
      varPct, absVar: Math.abs(varPct), fecha: ins.fecha
    });
  });

  filasConsolidadas.sort(function(a, b) { return b.absVar - a.absVar; });

  const filaTitulo = filaInicio + 5;
  sheet.getRange(filaTitulo, 2, 1, 6)
    .setValues([['Código', 'Descripción', 'Precio anterior', 'Precio actual', 'Var. %', 'Última compra']])
    .setBackground('#7248b9').setFontColor('white').setFontWeight('bold');

  if (filasConsolidadas.length === 0) {
    sheet.getRange(filaTitulo + 1, 2)
      .setValue('Sin variaciones de precio registradas.')
      .setFontStyle('italic').setFontColor('#5f6368');
    return filaTitulo + 3;
  }

  const limite = Math.min(filasConsolidadas.length, 15);
  const filas  = [];
  for (let i = 0; i < limite; i++) {
    const f    = filasConsolidadas[i];
    const vStr = f.varPct > 0 ? '+' + f.varPct.toFixed(1) + '%' : f.varPct.toFixed(1) + '%';
    filas.push([
      f.cod,
      f.desc.substring(0, 40),
      '$' + f.pAnt.toFixed(2),
      '$' + f.pNuevo.toFixed(2),
      vStr,
      formatearFecha(f.fecha)
    ]);
  }

  const rangoTabla = sheet.getRange(filaTitulo + 1, 2, limite, 6);
  rangoTabla.setValues(filas);
  rangoTabla.setBorder(true, true, true, true, true, true,
    '#dadce0', SpreadsheetApp.BorderStyle.SOLID);

  for (let j = 0; j < limite; j++) {
    const v     = filasConsolidadas[j].varPct;
    const color = v > 5 ? '#fce8e6' : v < -5 ? '#e6f4ea' : '#f8f9fa';
    sheet.getRange(filaTitulo + 1 + j, 2, 1, 6).setBackground(color);
  }

  return filaTitulo + limite + 2;
}

/**
 * Agrega la sección de historial de gastos mensuales al Dashboard.
 * @returns {number} Primera fila libre después de la sección
 */
function agregarSeccionGastos(sheet, filaInicio) {
  const tz      = Session.getScriptTimeZone();
  const ordenes = getDatosSheet(CONFIG.SHEETS.ORDENES);
  const detalle = getDatosSheet(CONFIG.SHEETS.DETALLE_ORDENES);
  const insumos = getDatosSheet(CONFIG.SHEETS.INSUMOS);

  // Mapas de lookup
  const mapOrdenes = {};
  ordenes.forEach(function(o) {
    if (String(o.estado || '').trim() === CONFIG.ESTADOS.ORDEN.CANCELADA) return;
    mapOrdenes[String(o.numero_orden).trim()] =
      String(o.nombre_proveedor || o.codigo_proveedor || 'Sin proveedor').trim();
  });

  const mapInsumos = {};
  insumos.forEach(function(ins) {
    mapInsumos[String(ins.codigo_insumo).trim()] =
      String(ins.categoria || 'Sin categoría').trim();
  });

  // Acumular por mes
  const porMes = {};
  detalle.forEach(function(item) {
    const numOrden = String(item.numero_orden || '').trim();
    if (!mapOrdenes[numOrden]) return;

    const fecha = new Date(item['fecha de pedido']);
    if (isNaN(fecha.getTime())) return;

    const mesKey   = Utilities.formatDate(fecha, tz, 'yyyy-MM');
    const mesLabel = Utilities.formatDate(fecha, tz, 'MM/yyyy');
    const subtotal = Number(item.subtotal) ||
      (Number(item.cantidad) * Number(item.precio_unitario)) || 0;
    const prov = mapOrdenes[numOrden];
    const cat  = mapInsumos[String(item.codigo_insumo || '').trim()] || 'Sin categoría';

    if (!porMes[mesKey]) {
      porMes[mesKey] = { label: mesLabel, total: 0, provs: {}, cats: {} };
    }
    porMes[mesKey].total       += subtotal;
    porMes[mesKey].provs[prov]  = (porMes[mesKey].provs[prov] || 0) + subtotal;
    porMes[mesKey].cats[cat]    = (porMes[mesKey].cats[cat]   || 0) + subtotal;
  });

  sheet.getRange(filaInicio, 2)
    .setValue('💰 Historial de Gastos Mensuales')
    .setFontSize(14).setFontWeight('bold').setFontColor('#202124');

  if (Object.keys(porMes).length === 0) {
    sheet.getRange(filaInicio + 1, 2)
      .setValue('Sin órdenes de compra registradas aún.')
      .setFontStyle('italic').setFontColor('#5f6368');
    return filaInicio + 4;
  }

  // KPI: gasto mes actual
  const ahora        = new Date();
  const mesActualKey = Utilities.formatDate(ahora, tz, 'yyyy-MM');
  const datosMes     = porMes[mesActualKey];
  const gastoMes     = datosMes ? datosMes.total : 0;

  crearTarjetaKPI(sheet, filaInicio + 1, 2,
    'Gasto ' + Utilities.formatDate(ahora, tz, 'MM/yyyy'),
    '$' + gastoMes.toFixed(2), '#0b8043');

  // Tabla: historial últimos 6 meses
  const meses  = Object.keys(porMes).sort().reverse().slice(0, 6);
  const filaTH = filaInicio + 5;

  sheet.getRange(filaTH, 2, 1, 4)
    .setValues([['Mes', 'Gasto Total', 'Var. vs anterior', 'Principal Proveedor']])
    .setBackground('#0b8043').setFontColor('white').setFontWeight('bold');

  const filasDatos = meses.map(function(mk, i) {
    const d      = porMes[mk];
    let varStr   = '-';
    if (i < meses.length - 1) {
      const ant = porMes[meses[i + 1]].total;
      if (ant > 0) {
        const pct = ((d.total - ant) / ant) * 100;
        varStr    = (pct >= 0 ? '+' : '') + pct.toFixed(1) + '%';
      }
    }
    const provKeys = Object.keys(d.provs);
    const prinProv = provKeys.length
      ? provKeys.sort(function(a, b) { return d.provs[b] - d.provs[a]; })[0]
      : '-';
    return [d.label, d.total, varStr, prinProv];
  });

  const rangoMeses = sheet.getRange(filaTH + 1, 2, filasDatos.length, 4);
  rangoMeses.setValues(filasDatos);
  sheet.getRange(filaTH + 1, 3, filasDatos.length, 1).setNumberFormat('$#,##0.00');
  rangoMeses.setBorder(true, true, true, true, true, true,
    '#dadce0', SpreadsheetApp.BorderStyle.SOLID);

  for (let i = 0; i < meses.length; i++) {
    const bgMes = meses[i] === mesActualKey ? '#e6f4ea' : (i % 2 === 0 ? '#f8f9fa' : 'white');
    sheet.getRange(filaTH + 1 + i, 2, 1, 4).setBackground(bgMes);
  }

  const filaPost = filaTH + 1 + meses.length;
  if (!datosMes) return filaPost + 2;

  // Sub-tablas: Proveedor y Categoría (mes actual)
  const filaSubH = filaPost + 2;

  // — Por proveedor (cols B–C) —
  sheet.getRange(filaSubH, 2)
    .setValue('Gasto por proveedor — ' + Utilities.formatDate(ahora, tz, 'MM/yyyy'))
    .setFontWeight('bold').setFontColor('#0b8043');
  sheet.getRange(filaSubH + 1, 2, 1, 2)
    .setValues([['Proveedor', 'Gasto']])
    .setBackground('#0b8043').setFontColor('white').setFontWeight('bold');

  const provsSorted = Object.keys(datosMes.provs)
    .sort(function(a, b) { return datosMes.provs[b] - datosMes.provs[a]; });
  const filasProv = provsSorted.map(function(p) { return [p, datosMes.provs[p]]; });

  if (filasProv.length) {
    const rangoProv = sheet.getRange(filaSubH + 2, 2, filasProv.length, 2);
    rangoProv.setValues(filasProv);
    sheet.getRange(filaSubH + 2, 3, filasProv.length, 1).setNumberFormat('$#,##0.00');
    rangoProv.setBorder(true, true, true, true, true, true,
      '#dadce0', SpreadsheetApp.BorderStyle.SOLID);
    for (let j = 0; j < filasProv.length; j++) {
      sheet.getRange(filaSubH + 2 + j, 2, 1, 2)
        .setBackground(j % 2 === 0 ? '#f8f9fa' : 'white');
    }
  }

  // — Por categoría (cols E–G) —
  sheet.getRange(filaSubH, 5)
    .setValue('Gasto por categoría — ' + Utilities.formatDate(ahora, tz, 'MM/yyyy'))
    .setFontWeight('bold').setFontColor('#0b8043');
  sheet.getRange(filaSubH + 1, 5, 1, 3)
    .setValues([['Categoría', 'Gasto', '% del total']])
    .setBackground('#0b8043').setFontColor('white').setFontWeight('bold');

  const catsSorted = Object.keys(datosMes.cats)
    .sort(function(a, b) { return datosMes.cats[b] - datosMes.cats[a]; });
  const filasCat = catsSorted.map(function(c) {
    const pct = gastoMes > 0
      ? ((datosMes.cats[c] / gastoMes) * 100).toFixed(1) + '%'
      : '-';
    return [c, datosMes.cats[c], pct];
  });

  if (filasCat.length) {
    const rangoCat = sheet.getRange(filaSubH + 2, 5, filasCat.length, 3);
    rangoCat.setValues(filasCat);
    sheet.getRange(filaSubH + 2, 6, filasCat.length, 1).setNumberFormat('$#,##0.00');
    rangoCat.setBorder(true, true, true, true, true, true,
      '#dadce0', SpreadsheetApp.BorderStyle.SOLID);
    for (let k = 0; k < filasCat.length; k++) {
      sheet.getRange(filaSubH + 2 + k, 5, 1, 3)
        .setBackground(k % 2 === 0 ? '#f8f9fa' : 'white');
    }
  }

  const alturaSubTablas = Math.max(filasProv.length, filasCat.length);
  const filaPostTablas  = filaSubH + 2 + alturaSubTablas + 2;
  const filaGraficos    = filaPostTablas + 1;

  // Gráfico 1: columnas — gastos por mes
  const rMeses = sheet.getRange(filaTH, 2, filasDatos.length + 1, 2);
  sheet.insertChart(sheet.newChart()
    .setChartType(Charts.ChartType.COLUMN)
    .addRange(rMeses)
    .setPosition(filaGraficos, 2, 0, 0)
    .setOption('title', 'Gastos por mes')
    .setOption('width', 520).setOption('height', 250)
    .setOption('colors', ['#0b8043'])
    .setOption('legend', { position: 'none' })
    .build());

  const filaGraf2 = filaGraficos + 16;

  // Gráfico 2: torta — por proveedor
  if (filasProv.length > 0) {
    const rProv = sheet.getRange(filaSubH + 1, 2, filasProv.length + 1, 2);
    sheet.insertChart(sheet.newChart()
      .setChartType(Charts.ChartType.PIE)
      .addRange(rProv)
      .setPosition(filaGraf2, 2, 0, 0)
      .setOption('title', 'Por proveedor — ' + Utilities.formatDate(ahora, tz, 'MM/yyyy'))
      .setOption('width', 420).setOption('height', 270)
      .setOption('pieSliceText', 'percentage')
      .build());
  }

  // Gráfico 3: torta — por categoría
  if (filasCat.length > 0) {
    const rCat = sheet.getRange(filaSubH + 1, 5, filasCat.length + 1, 2);
    sheet.insertChart(sheet.newChart()
      .setChartType(Charts.ChartType.PIE)
      .addRange(rCat)
      .setPosition(filaGraf2, 5, 0, 0)
      .setOption('title', 'Por categoría — ' + Utilities.formatDate(ahora, tz, 'MM/yyyy'))
      .setOption('width', 420).setOption('height', 270)
      .setOption('pieSliceText', 'percentage')
      .build());
  }

  return filaGraficos;
}

function actualizarStockPorProveedor() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  let sheet = ss.getSheetByName('Stock_por_Proveedor');
  if (!sheet) {
    sheet = ss.insertSheet('Stock_por_Proveedor');
  } else {
    sheet.clear();
    sheet.clearNotes();
    sheet.getRange(1, 1, sheet.getMaxRows(), sheet.getMaxColumns()).clearDataValidations();
    sheet.getImages().forEach(function(b) { b.remove(); });
  }

  sheet.setTabColor('#0f9d58');

  // Fuentes de datos
  const insumos           = getInsumosActivos();
  const proveedores       = getProveedoresActivos();
  const stockData         = getDatosSheet(CONFIG.SHEETS.STOCK);
  const movimientos       = getDatosSheet(CONFIG.SHEETS.MOVIMIENTOS);
  const tablaConversiones = getTablaConversiones();

  // Último precio por código (desde Historial_Precios + fallback en Insumos)
  const ultimoPrecio = {};
  const sheetHist    = ss.getSheetByName('Historial_Precios');
  if (sheetHist && sheetHist.getLastRow() > 2) {
    const rawHist = sheetHist.getDataRange().getValues();
    const hHdr    = rawHist[1];
    const hCod    = hHdr.indexOf('codigo_insumo');
    const hPNuevo = hHdr.indexOf('precio_nuevo');
    const hFecha  = hHdr.indexOf('fecha');
    const histList = [];
    for (let h = 2; h < rawHist.length; h++) {
      if (rawHist[h][hCod]) {
        histList.push({
          codigo: String(rawHist[h][hCod]).trim(),
          precio: Number(rawHist[h][hPNuevo]) || 0,
          fecha:  new Date(rawHist[h][hFecha])
        });
      }
    }
    histList.sort(function(a, b) { return a.fecha - b.fecha; });
    histList.forEach(function(r) {
      if (r.precio > 0) ultimoPrecio[r.codigo] = r.precio;
    });
  }
  insumos.forEach(function(ins) {
    const cod = String(ins.codigo_insumo).trim();
    if (!ultimoPrecio[cod] && Number(ins.precio_referencial) > 0) {
      ultimoPrecio[cod] = Number(ins.precio_referencial);
    }
  });

  // Stock por insumo+lote (unidades ya convertidas)
  const stockPorInsumo = {};
  stockData.forEach(function(row) {
    const cod  = String(row.codigo_insumo).trim();
    const cant = Number(row.stock_disponible) || 0;
    if (cant <= 0) return;
    if (!stockPorInsumo[cod]) stockPorInsumo[cod] = { total: 0, lotes: [] };
    stockPorInsumo[cod].total += cant;
    stockPorInsumo[cod].lotes.push({
      lote:        String(row.lote || 'SIN LOTE'),
      cantidad:    cant,
      vencimiento: row.vencimiento ? formatearFecha(row.vencimiento) : '-'
    });
  });

  // Consumo últimos 90 días
  const hace90   = new Date();
  hace90.setDate(hace90.getDate() - 90);
  const consumo90 = {};
  movimientos.forEach(function(mov) {
    if (String(mov.tipo_movimiento).trim() !== 'Egreso') return;
    const f = new Date(mov.fecha);
    if (isNaN(f.getTime()) || f < hace90) return;
    const cod = String(mov.codigo_insumo).trim();
    consumo90[cod] = (consumo90[cod] || 0) + (Number(mov.cantidad) || 0);
  });

  let fila = 1;

  // Título principal
  sheet.getRange(fila, 1, 1, 10).merge();
  sheet.getRange(fila, 1)
    .setValue('📦 STOCK POR PROVEEDOR  |  Tildá los insumos a pedir, ajustá cantidades y ejecutá Menú → 🛒 Generar OC desde Stock')
    .setBackground('#1F4E79').setFontColor('#FFFFFF')
    .setFontWeight('bold').setFontSize(10).setWrap(true);
  sheet.setRowHeight(fila, 28);
  fila++;

  // Banner de actualización
  const fechaActStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm');
  sheet.getRange(fila, 1, 1, 10).merge();
  sheet.getRange(fila, 1)
    .setValue('⚠️  Vista actualizada el ' + fechaActStr +
              '  —  Si hubo recepciones después de este horario, ejecutá Menú → 📦 Stock por Proveedor antes de generar una OC.')
    .setBackground('#FFF8E1').setFontColor('#B06000')
    .setFontSize(10).setHorizontalAlignment('left').setWrap(true);
  sheet.setRowHeight(fila, 32);
  fila += 2;

  const mapaProveedores = {};

  proveedores.forEach(function(prov) {
    const codProv    = String(prov.codigo_proveedor).trim();
    const nombreProv = prov.razon_social || codProv;

    const insumosDelProv = insumos.filter(function(ins) {
      return String(ins.proveedor_preferido).trim() === codProv;
    });
    if (insumosDelProv.length === 0) return;

    // Ordenar por consumo 90d desc, luego alfabético
    insumosDelProv.sort(function(a, b) {
      const cA = consumo90[String(a.codigo_insumo).trim()] || 0;
      const cB = consumo90[String(b.codigo_insumo).trim()] || 0;
      if (cB !== cA) return cB - cA;
      return a.descripcion.localeCompare(b.descripcion);
    });

    // Header de sección proveedor
    sheet.getRange(fila, 1, 1, 10).merge();
    sheet.getRange(fila, 1)
      .setValue('🏢  ' + nombreProv.toUpperCase() +
                '   ·   ' + insumosDelProv.length + ' insumos' +
                (prov.mail ? '   ·   ' + prov.mail : '') +
                '          👉 Tildá los que querés pedir y ejecutá el menú')
      .setBackground('#2E75B6').setFontColor('#FFFFFF')
      .setFontWeight('bold').setFontSize(11);
    sheet.setRowHeight(fila, 28);
    fila++;

    // Headers de columnas
    // A=Pedir | B=Código | C=Descripción | D=Categoría | E=Stock(u.) |
    // F=Estado | G=Últ.Precio | H=Consumo90d | I=Lote(s)/Venc | J=Cant.a pedir
    const colH = ['✔ Pedir', 'Código', 'Descripción', 'Categoría',
                   'Stock (u.)', 'Estado', 'Últ. Precio', 'Consumo 90d',
                   'Lote(s) / Vencimiento(s)', 'Cant. a pedir\n(unid. compra)'];
    sheet.getRange(fila, 1, 1, 10)
      .setValues([colH])
      .setBackground('#D6E4F0').setFontColor('#1F4E79')
      .setFontWeight('bold').setFontSize(10)
      .setHorizontalAlignment('center').setWrap(true);
    sheet.setRowHeight(fila, 30);
    fila++;

    const filaInicioInsumos = fila;

    insumosDelProv.forEach(function(ins, idx) {
      const cod      = String(ins.codigo_insumo).trim();
      const stockIns = stockPorInsumo[cod];
      const total    = stockIns ? stockIns.total : 0;
      const critico  = Number(ins.stock_critico) || 0;
      const estado   = total <= 0 ? '⚠ Agotado' : total <= critico ? '⚡ Crítico' : '✓ OK';
      const precio   = ultimoPrecio[cod] || 0;
      const precioStr = precio > 0 ? '$' + precio.toFixed(2) : '-';
      const consumo  = consumo90[cod] || 0;

      // Lotes y vencimientos
      let lotesStr = '-';
      if (stockIns && stockIns.lotes.length > 0) {
        const lotesOrd = stockIns.lotes.slice().sort(function(a, b) {
          if (a.vencimiento === '-') return 1;
          if (b.vencimiento === '-') return -1;
          return new Date(a.vencimiento) - new Date(b.vencimiento);
        });
        lotesStr = lotesOrd.map(function(l) {
          return l.lote + ' (' + l.cantidad + ')' +
                 (l.vencimiento !== '-' ? ' vence ' + l.vencimiento : '');
        }).join(' | ');
      }

      // Cantidad sugerida en unidades de COMPRA (cajas)
      // stock_critico está en unidades de stock → dividir por factor para obtener cajas
      const conv   = tablaConversiones[cod];
      const factor = (conv && conv.factor > 1) ? conv.factor : 1;
      const unidadesStockFaltantes = critico * 2 - total;
      const cantSugerida = Math.max(Math.ceil(unidadesStockFaltantes / factor), 1);

      // Col A: checkbox
      sheet.getRange(fila, 1).insertCheckboxes();

      // Cols B–J: datos
      sheet.getRange(fila, 2, 1, 9).setValues([[
        cod, ins.descripcion, ins.categoria || '',
        total, estado, precioStr, consumo,
        lotesStr, cantSugerida
      ]]);

      // Color de fondo según estado
      const bgBase   = idx % 2 === 0 ? '#FFFFFF' : '#F8F9FA';
      const bgEstado = estado.indexOf('Agotado') > -1 ? '#FCE8E6' :
                       estado.indexOf('Crítico') > -1 ? '#FEF7E0' : bgBase;
      sheet.getRange(fila, 1, 1, 10).setBackground(bgEstado).setFontSize(10);

      // Alineaciones
      sheet.getRange(fila, 2).setHorizontalAlignment('center');
      sheet.getRange(fila, 5).setHorizontalAlignment('center');
      sheet.getRange(fila, 6).setHorizontalAlignment('center');
      sheet.getRange(fila, 7).setHorizontalAlignment('right');
      sheet.getRange(fila, 8).setHorizontalAlignment('center');
      sheet.getRange(fila, 10).setHorizontalAlignment('center').setFontWeight('bold');

      // Top 3 en negrita (más consumidos)
      if (idx < 3 && consumo > 0) sheet.getRange(fila, 3).setFontWeight('bold');

      fila++;
    });

    mapaProveedores[codProv] = {
      filaInicio: filaInicioInsumos,
      filaFin:    fila - 1,
      nombre:     nombreProv,
      mail:       prov.mail || ''
    };

    // Fila instructiva
    sheet.getRange(fila, 1, 1, 10).merge();
    sheet.getRange(fila, 1)
      .setValue('▶  Menú → 🛒 Gestión Compras → 🛒 Generar OC desde Stock  (con filas tildadas arriba)')
      .setBackground('#E8F5E9').setFontColor('#137333')
      .setFontStyle('italic').setFontSize(10)
      .setHorizontalAlignment('center');
    sheet.setRowHeight(fila, 24);
    fila += 2;
  });

  // Guardar mapa en ScriptProperties para generarOrdenDesdeStock()
  PropertiesService.getScriptProperties()
    .setProperty('MAPA_PROV_STOCK', JSON.stringify(mapaProveedores));

  // Anchos de columna
  sheet.setColumnWidth(1,  65);   // checkbox
  sheet.setColumnWidth(2,  90);   // codigo
  sheet.setColumnWidth(3,  260);  // descripcion
  sheet.setColumnWidth(4,  120);  // categoria
  sheet.setColumnWidth(5,  75);   // stock
  sheet.setColumnWidth(6,  85);   // estado
  sheet.setColumnWidth(7,  90);   // precio
  sheet.setColumnWidth(8,  85);   // consumo
  sheet.setColumnWidth(9,  230);  // lotes
  sheet.setColumnWidth(10, 90);   // cant pedir
}

// ── Ajuste masivo de precios desde hoja Proveedores ──────────────────────────

function prepararColumnasPorcentaje() {
  const ui          = SpreadsheetApp.getUi();
  const sheetProv   = getSheet(CONFIG.SHEETS.PROVEEDORES);
  const sheetIns    = getSheet(CONFIG.SHEETS.INSUMOS);

  const dataIns    = sheetIns.getDataRange().getValues();
  const headersIns = dataIns[0];
  const idxCat     = headersIns.indexOf('categoria');
  const idxActivo  = headersIns.indexOf('activo');

  if (idxCat === -1) {
    ui.alert('No se encontró la columna "categoria" en la hoja Insumos.');
    return;
  }

  const categorias = [];
  for (let r = 1; r < dataIns.length; r++) {
    const activo = dataIns[r][idxActivo];
    if (activo !== true && activo !== 'TRUE' && activo !== 'VERDADERO') continue;
    const cat = String(dataIns[r][idxCat]).trim();
    if (cat && categorias.indexOf(cat) === -1) categorias.push(cat);
  }
  categorias.sort();

  if (categorias.length === 0) {
    ui.alert('No se encontraron categorías en Insumos.');
    return;
  }

  const headersProv = sheetProv.getRange(1, 1, 1, sheetProv.getLastColumn()).getValues()[0];
  const prefix      = 'ajuste_%_';
  let colAgregadas  = 0;

  categorias.forEach(function(cat) {
    const colName = prefix + cat;
    if (headersProv.indexOf(colName) !== -1) return;
    const nuevaCol = sheetProv.getLastColumn() + 1;
    sheetProv.getRange(1, nuevaCol).setValue(colName)
      .setBackground('#FFF8E1').setFontWeight('bold')
      .setFontColor('#B06000').setFontSize(10);
    sheetProv.setColumnWidth(nuevaCol, 130);
    headersProv.push(colName);
    colAgregadas++;
  });

  const instrCol = sheetProv.getLastColumn() + 1;
  sheetProv.getRange(1, instrCol)
    .setValue('← Ingresá el % de aumento en cada columna (ej: 10 = +10%). Luego ejecutá Menú → 💲 Aplicar ajuste de precios.')
    .setFontStyle('italic').setFontColor('#888888').setFontSize(10);
  sheetProv.setColumnWidth(instrCol, 400);

  SpreadsheetApp.flush();

  if (colAgregadas > 0) {
    ui.alert('✅ Se agregaron ' + colAgregadas + ' columna(s) para: ' + categorias.join(', ') + '.\n\nIngresá los porcentajes y ejecutá "💲 Aplicar ajuste de precios".');
  } else {
    ui.alert('Las columnas ya estaban creadas para todas las categorías.\n\nIngresá los porcentajes y ejecutá "💲 Aplicar ajuste de precios".');
  }
}

function aplicarAjustePrecios() {
  const ui         = SpreadsheetApp.getUi();
  const ss         = SpreadsheetApp.getActiveSpreadsheet();
  const sheetProv  = ss.getSheetByName(CONFIG.SHEETS.PROVEEDORES);
  const sheetIns   = ss.getSheetByName(CONFIG.SHEETS.INSUMOS);
  const sheetHist  = ss.getSheetByName(CONFIG.SHEETS.HIST_PRECIOS);

  if (!sheetProv || !sheetIns) {
    ui.alert('No se encontraron las hojas necesarias.');
    return;
  }

  const dataProv    = sheetProv.getDataRange().getValues();
  const headersProv = dataProv[0];
  const idxCodProv  = headersProv.indexOf('codigo_proveedor');
  const idxNombre   = headersProv.indexOf('razon_social');
  const idxActProv  = headersProv.indexOf('activo');
  const prefix      = 'ajuste_%_';

  const colsAjuste = [];
  headersProv.forEach(function(h, i) {
    if (String(h).indexOf(prefix) === 0) {
      colsAjuste.push({ col: i, categoria: String(h).replace(prefix, '').trim() });
    }
  });

  if (colsAjuste.length === 0) {
    ui.alert('No se encontraron columnas de ajuste.\n\nEjecutá primero "📋 Preparar columnas % en Proveedores".');
    return;
  }

  const mapaAjustes = {};
  const mapaNombres = {};
  for (let r = 1; r < dataProv.length; r++) {
    const codProv = String(dataProv[r][idxCodProv]).trim();
    if (!codProv) continue;
    const activo = dataProv[r][idxActProv];
    if (activo !== true && activo !== 'TRUE' && activo !== 'VERDADERO') continue;
    mapaNombres[codProv] = String(dataProv[r][idxNombre]).trim();
    mapaAjustes[codProv] = {};
    colsAjuste.forEach(function(ca) {
      const pct = parseFloat(dataProv[r][ca.col]);
      if (!isNaN(pct) && pct !== 0) mapaAjustes[codProv][ca.categoria] = pct;
    });
  }

  const hayAjustes = Object.keys(mapaAjustes).some(function(cod) {
    return Object.keys(mapaAjustes[cod]).length > 0;
  });
  if (!hayAjustes) {
    ui.alert('No hay porcentajes cargados en la hoja Proveedores.');
    return;
  }

  const resumen = [];
  Object.keys(mapaAjustes).forEach(function(cod) {
    const cats = mapaAjustes[cod];
    if (Object.keys(cats).length === 0) return;
    resumen.push('• ' + mapaNombres[cod] + ':');
    Object.keys(cats).forEach(function(cat) {
      resumen.push('    - ' + cat + ': ' + (cats[cat] > 0 ? '+' : '') + cats[cat] + '%');
    });
  });

  const resp = ui.alert(
    '💲 Confirmar ajuste de precios',
    'Se aplicarán:\n\n' + resumen.join('\n') +
    '\n\n⚠️ Actualiza precio_referencial en Insumos y registra en Historial_Precios.\n¿Confirmás?',
    ui.ButtonSet.YES_NO
  );
  if (resp !== ui.Button.YES) return;

  const dataIns    = sheetIns.getDataRange().getValues();
  const headersIns = dataIns[0];
  const idxCodIns  = headersIns.indexOf('codigo_insumo');
  const idxDesc    = headersIns.indexOf('descripcion');
  const idxCatIns  = headersIns.indexOf('categoria');
  const idxPrecio  = headersIns.indexOf('precio_referencial');
  const idxProvPref = headersIns.indexOf('proveedor_preferido');
  const idxActIns  = headersIns.indexOf('activo');

  const fecha   = new Date();
  const usuario = getCurrentUser();
  let contActualizados = 0;
  let contSinPrecio    = 0;

  for (let ri = 1; ri < dataIns.length; ri++) {
    const actIns = dataIns[ri][idxActIns];
    if (actIns !== true && actIns !== 'TRUE' && actIns !== 'VERDADERO') continue;

    const codIns   = String(dataIns[ri][idxCodIns]).trim();
    const desc     = String(dataIns[ri][idxDesc]).trim();
    const cat      = String(dataIns[ri][idxCatIns]).trim();
    const provPref = String(dataIns[ri][idxProvPref]).trim();
    const precioAct = Number(dataIns[ri][idxPrecio]) || 0;

    const ajustesProv = mapaAjustes[provPref];
    if (!ajustesProv) continue;
    const pct = ajustesProv[cat];
    if (pct === undefined) continue;
    if (precioAct === 0) { contSinPrecio++; continue; }

    const precioNuevo = Math.round(precioAct * (1 + pct / 100) * 100) / 100;
    sheetIns.getRange(ri + 1, idxPrecio + 1).setValue(precioNuevo);

    if (sheetHist) {
      const varPct = Math.round(((precioNuevo - precioAct) / precioAct) * 10000) / 100;
      sheetHist.appendRow([
        fecha, codIns, desc,
        mapaNombres[provPref] || provPref,
        precioAct, precioNuevo,
        (varPct > 0 ? '+' : '') + varPct + '%',
        'AJUSTE_MASIVO_' + (pct > 0 ? '+' : '') + pct + '%',
        usuario
      ]);
    }
    contActualizados++;
  }

  // Limpiar porcentajes ingresados (evitar re-aplicación accidental)
  for (let rp = 1; rp < dataProv.length; rp++) {
    colsAjuste.forEach(function(ca) {
      sheetProv.getRange(rp + 1, ca.col + 1).clearContent();
    });
  }

  SpreadsheetApp.flush();
  ui.alert(
    '✅ Ajuste aplicado.\n\n' +
    '• Insumos actualizados: ' + contActualizados + '\n' +
    (contSinPrecio > 0 ? '• Sin precio base (no actualizados): ' + contSinPrecio + '\n' : '') +
    '\nLos porcentajes fueron borrados de Proveedores para evitar re-aplicación.\n\n' +
    'Ejecutá "Actualizar Dashboard" para ver los cambios.'
  );
}

/**
 * Retorna todos los insumos activos con su estado de conversión actual.
 * Usada por FormularioConversiones.html para poblar la tabla.
 */
function getInsumosConConversiones() {
  const insumos      = getInsumosActivos();
  const conversiones = _getTablaConversionesCompleta();

  return insumos.map(function(ins) {
    const cod  = String(ins.codigo_insumo).trim();
    const conv = conversiones[cod];
    return {
      codigo:           cod,
      descripcion:      ins.descripcion,
      categoria:        ins.categoria    || '',
      unidad_medida:    ins.unidad_medida || '',
      tiene_conversion: !!conv && conv.activo,
      factor:           conv ? conv.factor         : '',
      unidad_destino:   conv ? conv.unidad_destino : '',
      unidad_compra:    conv ? conv.unidad_compra  : '',
      activo:           conv ? conv.activo         : false,
      observaciones:    conv ? conv.observaciones  : ''
    };
  }).sort(function(a, b) {
    // Primero los que tienen conversión activa
    if (a.tiene_conversion && !b.tiene_conversion) return -1;
    if (!a.tiene_conversion && b.tiene_conversion) return  1;
    return a.descripcion.localeCompare(b.descripcion);
  });
}

/**
 * Lee Config_Conversiones completo (activos e inactivos).
 * Devuelve mapa { codigo → { fila, factor, unidad_destino, unidad_compra, activo, observaciones } }.
 * Función interna — no llamar desde HTML.
 */
function _getTablaConversionesCompleta() {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Config_Conversiones');
  if (!sheet || sheet.getLastRow() < 3) return {};

  const data    = sheet.getDataRange().getValues();
  const headers = data[1]; // fila 1 = banner, fila 2 = headers

  const idxCod    = headers.indexOf('codigo_insumo');
  const idxFactor = headers.indexOf('factor_conversion');
  const idxUnidad = headers.indexOf('unidad_destino');
  const idxActivo = headers.indexOf('activo');
  const idxCompra = headers.indexOf('unidad_de_compra');
  const idxObs    = headers.indexOf('observaciones');
  // ✅ NUEVO (v2.7): columna de control del aviso preventivo de conversión (punto 2.3)
  const idxConfirmada = headers.indexOf('confirmado_primera_recepcion');

  const tabla = {};
  for (let i = 2; i < data.length; i++) {
    const cod = String(data[i][idxCod] || '').trim();
    if (!cod) continue;
    tabla[cod] = {
      fila:           i + 1, // 1-indexed para getRange
      factor:         Number(data[i][idxFactor]) || 0,
      unidad_destino: String(data[i][idxUnidad] || '').trim(),
      unidad_compra:  String(data[i][idxCompra] || '').trim(),
      activo:         data[i][idxActivo] === true ||
                      String(data[i][idxActivo]).toUpperCase() === 'TRUE',
      observaciones:  String(data[i][idxObs] || ''),
      confirmada:     idxConfirmada >= 0 &&
                      (data[i][idxConfirmada] === true ||
                       String(data[i][idxConfirmada]).toUpperCase() === 'TRUE')
    };
  }
  return tabla;
}

/**
 * Devuelve (creándola si hace falta) la columna "confirmado_primera_recepcion"
 * de Config_Conversiones — 1-indexed, lista para usar con sheet.getRange().
 * No pisa filas existentes: si la hoja es vieja y no tiene esta columna,
 * la agrega al final sin tocar el resto.
 */
function _asegurarColumnaConfirmada(sheet) {
  const headers = sheet.getRange(2, 1, 1, sheet.getLastColumn()).getValues()[0];
  let idx = headers.indexOf('confirmado_primera_recepcion');
  if (idx >= 0) return idx + 1;

  const col = sheet.getLastColumn() + 1;
  sheet.getRange(2, col).setValue('confirmado_primera_recepcion');
  return col;
}

/**
 * Lee solo las conversiones ACTIVAS con factor > 1.
 * Resultado: { codigo → { factor, unidad_destino } }
 * Usada en registrarRecepcion() y actualizarStockPorProveedor().
 */
function getTablaConversiones() {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Config_Conversiones');
  if (!sheet) return {};

  const data    = sheet.getDataRange().getValues();
  if (data.length < 3) return {};

  const headers  = data[1]; // fila 2 = headers
  const idxCodigo = headers.indexOf('codigo_insumo');
  const idxFactor = headers.indexOf('factor_conversion');
  const idxUnidad = headers.indexOf('unidad_destino');
  const idxActivo = headers.indexOf('activo');

  const tabla = {};
  for (let i = 2; i < data.length; i++) {
    const cod    = String(data[i][idxCodigo] || '').trim();
    const activo = data[i][idxActivo];
    if (!cod) continue;
    if (activo !== true && String(activo).toUpperCase() !== 'TRUE') continue;
    const factor = Number(data[i][idxFactor]);
    if (!factor || factor <= 1) continue; // factor 1 = sin conversión real

    tabla[cod] = {
      factor,
      unidad_destino: String(data[i][idxUnidad] || 'Determinación').trim()
    };
  }
  return tabla;
}

/**
 * Aplica la conversión caja → unidad a una cantidad recibida.
 * Si no hay conversión activa para el código, devuelve la cantidad sin cambios.
 *
 * @param {string} codigoInsumo
 * @param {number} cantidadRecibida   — en unidades de compra (cajas)
 * @param {Object} tablaConversiones  — resultado de getTablaConversiones()
 * @returns {{ cantidad_stock, unidad_stock, tiene_conversion, factor }}
 */
function aplicarConversion(codigoInsumo, cantidadRecibida, tablaConversiones) {
  const conv = tablaConversiones[codigoInsumo];
  if (!conv) {
    return {
      cantidad_stock:  cantidadRecibida,
      unidad_stock:    null,
      tiene_conversion: false,
      factor:          1
    };
  }
  return {
    cantidad_stock:  cantidadRecibida * conv.factor,
    unidad_stock:    conv.unidad_destino,
    tiene_conversion: true,
    factor:          conv.factor
  };
}

/**
 * Guarda o actualiza la conversión de un insumo desde el formulario.
 * Si ya existe fila en Config_Conversiones → la actualiza.
 * Si no existe → agrega fila nueva al final.
 *
 * @param {Object} convData — { codigo, descripcion, factor, unidad_destino, unidad_compra, activo, observaciones }
 */
function guardarConversionInsumo(convData) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName('Config_Conversiones');

    if (!sheet) {
      crearHojaConversiones();
      sheet = ss.getSheetByName('Config_Conversiones');
    }

    const tabla    = _getTablaConversionesCompleta();
    const existing = tabla[String(convData.codigo).trim()];

    const factorNuevo  = Number(convData.factor) || 1;
    const unidadNueva   = convData.unidad_destino || 'Unidad';

    // ✅ NUEVO (v2.7): si es una conversión nueva, o si cambió el factor/unidad de
    // destino de una existente, se resetea el aviso de "primera recepción" (punto 2.3)
    // para que el equipo vuelva a confirmar la equivalencia la próxima vez que reciban
    // este insumo. Si no cambió nada relevante, se conserva el estado de confirmación.
    const esCambioDeEquivalencia = !existing ||
      existing.factor !== factorNuevo ||
      existing.unidad_destino !== unidadNueva;
    const confirmada = esCambioDeEquivalencia ? false : existing.confirmada;

    const rowData = [
      convData.codigo,
      convData.descripcion,
      convData.unidad_compra  || 'CAJA',
      factorNuevo,
      unidadNueva,
      convData.activo === true || convData.activo === 'true',
      convData.observaciones  || '',
      confirmada
    ];

    const colConfirmada = _asegurarColumnaConfirmada(sheet);

    if (existing) {
      sheet.getRange(existing.fila, 1, 1, 7).setValues([rowData.slice(0, 7)]);
      sheet.getRange(existing.fila, colConfirmada).setValue(confirmada);
    } else {
      const filaNueva = sheet.getLastRow() + 1;
      sheet.getRange(filaNueva, 1, 1, 7).setValues([rowData.slice(0, 7)]);
      sheet.getRange(filaNueva, colConfirmada).setValue(confirmada);
    }

    SpreadsheetApp.flush();
    return { ok: true, mensaje: 'Conversión guardada para ' + convData.codigo + '.' };
  } catch (e) {
    return { ok: false, mensaje: 'Error al guardar: ' + e.message };
  }
}

/**
 * Desactiva (sin borrar) la conversión de un insumo.
 * Permite reactivarla luego cambiando activo → TRUE.
 */
function desactivarConversionInsumo(codigoInsumo) {
  try {
    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName('Config_Conversiones');
    if (!sheet) return { ok: false, mensaje: 'No existe la hoja Config_Conversiones.' };

    const tabla    = _getTablaConversionesCompleta();
    const existing = tabla[String(codigoInsumo).trim()];
    if (!existing) return { ok: false, mensaje: 'No se encontró conversión para ' + codigoInsumo + '.' };

    const headers  = sheet.getDataRange().getValues()[1];
    const idxActivo = headers.indexOf('activo') + 1;
    sheet.getRange(existing.fila, idxActivo).setValue(false);

    SpreadsheetApp.flush();
    return { ok: true, mensaje: 'Conversión desactivada para ' + codigoInsumo + '.' };
  } catch (e) {
    return { ok: false, mensaje: 'Error al desactivar: ' + e.message };
  }
}

/**
 * Muestra un resumen de las conversiones activas.
 * Útil para verificar la configuración sin abrir la hoja.
 */
function verResumenConversiones() {
  const tabla   = getTablaConversiones();
  const codigos = Object.keys(tabla);

  if (codigos.length === 0) {
    SpreadsheetApp.getUi().alert(
      'No hay conversiones activas.\n' +
      'Verificá que la hoja "Config_Conversiones" exista y tenga filas con activo = TRUE.'
    );
    return;
  }

  const catalogoInsumos = getInsumosActivos();
  const lineas = ['Conversiones activas (' + codigos.length + '):\n'];

  codigos.sort().forEach(function(cod) {
    const ins  = catalogoInsumos.find(function(i) { return i.codigo_insumo === cod; });
    const desc = ins ? ins.descripcion.substring(0, 40) : '(no encontrado en catálogo)';
    lineas.push('• ' + cod + ': ×' + tabla[cod].factor + ' ' +
                tabla[cod].unidad_destino + ' — ' + desc);
  });

  SpreadsheetApp.getUi().alert(lineas.join('\n'));
}

/**
 * Devuelve todos los insumos activos con su stock total actual.
 * Usada por el buscador del formulario de ajuste.
 */
function getInsumosConStockYSinStock() {
  const insumos  = getInsumosActivos();
  const stockData = getDatosSheet(CONFIG.SHEETS.STOCK);

  const stockPorCodigo = {};
  stockData.forEach(function(row) {
    const cod = String(row.codigo_insumo).trim();
    stockPorCodigo[cod] = (stockPorCodigo[cod] || 0) + (Number(row.stock_disponible) || 0);
  });

  return insumos.map(function(ins) {
    const cod = String(ins.codigo_insumo).trim();
    return {
      codigo:      cod,
      descripcion: ins.descripcion,
      categoria:   ins.categoria || '',
      stock_total: stockPorCodigo[cod] || 0
    };
  }).sort(function(a, b) { return a.descripcion.localeCompare(b.descripcion); });
}

/**
 * Devuelve los lotes con stock disponible de un insumo.
 * Si no tiene stock, devuelve un lote ficticio para permitir ajuste positivo desde cero.
 */
function getLotesParaAjuste(codigoInsumo) {
  const stockData = getDatosSheet(CONFIG.SHEETS.STOCK);

  const lotes = stockData
    .filter(function(row) { return String(row.codigo_insumo).trim() === codigoInsumo; })
    .map(function(row) {
      return {
        lote:         String(row.lote || 'SIN LOTE'),
        stock_actual: Number(row.stock_disponible) || 0,
        vencimiento:  row.vencimiento ? formatearFecha(row.vencimiento) : ''
      };
    });

  if (lotes.length === 0) {
    lotes.push({ lote: 'SIN LOTE', stock_actual: 0, vencimiento: '' });
  }

  return lotes;
}

/**
 * Registra un ajuste de stock sobre un lote específico.
 *
 * @param {Object} ajusteData
 *   codigo_insumo  {string}
 *   descripcion    {string}
 *   lote           {string}
 *   vencimiento    {string}
 *   stock_actual   {number}  — cantidad actual en el lote
 *   stock_correcto {number}  — cantidad que DEBERÍA haber
 *   motivo         {string}
 */
function registrarAjusteStock(ajusteData) {
  try {
    const delta = Number(ajusteData.stock_correcto) - Number(ajusteData.stock_actual);

    if (delta === 0) {
      return { ok: true, mensaje: 'Sin cambios: el stock ya es correcto.' };
    }

    const usuario        = getCurrentUser();
    const ahora          = new Date();
    const catalogoInsumos = getInsumosActivos();
    const infoInsumo     = catalogoInsumos.find(function(ins) {
      return ins.codigo_insumo === ajusteData.codigo_insumo;
    });
    const categoria = infoInsumo ? (infoInsumo.categoria || '') : '';

    // Registrar en Movimientos_Inventario
    // delta positivo = suma, delta negativo = resta (actualizarStockActual lo maneja igual)
    getSheet(CONFIG.SHEETS.MOVIMIENTOS).appendRow([
      'MOV-' + Date.now(),
      ahora,
      'Ajuste',
      ajusteData.codigo_insumo,
      ajusteData.descripcion,
      delta,
      'AJUSTE-MANUAL',
      usuario,
      'Corrección manual | ' + (ajusteData.motivo || 'Sin motivo') +
        ' | Anterior: ' + ajusteData.stock_actual +
        ' → Correcto: ' + ajusteData.stock_correcto,
      ajusteData.lote,
      ajusteData.vencimiento || '',
      categoria
    ]);

    // Actualizar Stock_Actual sin reconstrucción total
    _actualizarStockDesdeAjuste(ajusteData, delta, catalogoInsumos, ahora);

    const signo = delta > 0 ? '+' : '';
    return {
      ok:     true,
      mensaje: 'Ajuste aplicado (' + signo + delta + ' u.) en lote "' +
               ajusteData.lote + '" de ' + ajusteData.descripcion + '.'
    };
  } catch (e) {
    return { ok: false, mensaje: 'Error al ajustar: ' + e.message };
  }
}

/**
 * Actualiza Stock_Actual de forma incremental tras un ajuste.
 * No reconstruye toda la hoja, solo toca la fila del lote afectado.
 */
function _actualizarStockDesdeAjuste(ajusteData, delta, catalogoInsumos, ahora) {
  const sheetStock = getSheet(CONFIG.SHEETS.STOCK);
  const lastRow    = sheetStock.getLastRow();
  const cod        = String(ajusteData.codigo_insumo).trim();
  const lote       = String(ajusteData.lote).trim();

  const dataStock = lastRow > 1
    ? sheetStock.getRange(2, 1, lastRow - 1, 9).getValues()
    : [];

  // Totales actuales por código
  const stockTotalPorCodigo = {};
  dataStock.forEach(function(row) {
    const c = String(row[0]).trim();
    stockTotalPorCodigo[c] = (stockTotalPorCodigo[c] || 0) + (Number(row[2]) || 0);
  });

  // Buscar y actualizar la fila del lote
  let filaEncontrada = -1;
  for (let i = 0; i < dataStock.length; i++) {
    if (String(dataStock[i][0]).trim() !== cod) continue;
    if (String(dataStock[i][6]).trim() !== lote) continue;
    const nuevoStock  = Math.max((Number(dataStock[i][2]) || 0) + delta, 0);
    dataStock[i][2]   = nuevoStock;
    dataStock[i][5]   = ahora;
    stockTotalPorCodigo[cod] = (stockTotalPorCodigo[cod] || 0) + delta;
    filaEncontrada = i;
    break;
  }

  // Si el lote no existía y el ajuste es positivo → crear fila nueva
  if (filaEncontrada === -1 && delta > 0) {
    const ins     = catalogoInsumos.find(function(x) {
      return String(x.codigo_insumo).trim() === cod;
    });
    const critico = ins ? Number(ins.stock_critico) || 0 : 0;
    dataStock.push([
      cod,
      ins ? ins.descripcion : ajusteData.descripcion,
      delta,
      critico,
      'OK',
      ahora,
      lote,
      ajusteData.vencimiento || '',
      ins ? (ins.clasificacion || '') : ''
    ]);
    stockTotalPorCodigo[cod] = (stockTotalPorCodigo[cod] || 0) + delta;
  }

  // Recalcular estado para todas las filas del código afectado
  const ins2     = catalogoInsumos.find(function(x) {
    return String(x.codigo_insumo).trim() === cod;
  });
  const critico2   = ins2 ? Number(ins2.stock_critico) || 0 : 0;
  const totalFinal = stockTotalPorCodigo[cod] || 0;
  const nuevoEst   = totalFinal <= 0 ? 'Agotado' : totalFinal <= critico2 ? 'Crítico' : 'OK';

  for (let j = 0; j < dataStock.length; j++) {
    if (String(dataStock[j][0]).trim() === cod) dataStock[j][4] = nuevoEst;
  }

  // Filtrar filas con stock > 0 y reescribir
  const filasValidas = dataStock.filter(function(row) { return Number(row[2]) > 0; });

  if (sheetStock.getLastRow() > 1) {
    sheetStock.getRange(2, 1, sheetStock.getLastRow() - 1, 9).clearContent();
  }
  if (filasValidas.length > 0) {
    sheetStock.getRange(2, 1, filasValidas.length, 9).setValues(filasValidas);
    sheetStock.getRange(2, 1, filasValidas.length, 9).sort({ column: 8, ascending: true });
  }
}

/**
 * Importa stock inicial desde el payload JSON enviado por el formulario.
 *
 * Cada ítem del payload:
 *   { codigo, cantidad, lote, vencimiento, obs }
 *
 * Valida contra catálogo, aplica conversiones, escribe en batch en
 * Movimientos_Inventario y Stock_Actual, y devuelve un resumen.
 *
 * @param {Object} payload — { filas: [...], importId: string }
 * @returns {Object} — { importadas, saltadas, errores, tiempo_ms, detalle_errores }
 */
function importarStockInicial(payload) {
  const inicio          = Date.now();
  const filas           = payload.filas     || [];
  const importId        = payload.importId  || ('IMPORT-' + Utilities.formatDate(
    new Date(), Session.getScriptTimeZone(), 'yyyyMMdd-HHmmss'
  ));

  const catalogoInsumos   = getInsumosActivos();
  const tablaConversiones = getTablaConversiones();

  // Índice rápido del catálogo
  const mapaCatalogo = {};
  catalogoInsumos.forEach(function(ins) {
    mapaCatalogo[String(ins.codigo_insumo).trim()] = ins;
  });

  const filasMovimientos = [];
  const filasStock       = [];
  const errores          = [];
  let   importadas       = 0;
  let   saltadas         = 0;

  filas.forEach(function(fila, idx) {
    const cod      = String(fila.codigo || '').trim();
    const cantidad = Number(fila.cantidad) || 0;

    // Validaciones básicas
    if (!cod) {
      errores.push('Fila ' + (idx + 2) + ': código vacío.');
      saltadas++;
      return;
    }
    if (!mapaCatalogo[cod]) {
      errores.push('Fila ' + (idx + 2) + ': código "' + cod + '" no existe en el catálogo.');
      saltadas++;
      return;
    }
    if (cantidad <= 0) {
      errores.push('Fila ' + (idx + 2) + ': cantidad inválida para "' + cod + '".');
      saltadas++;
      return;
    }

    const ins       = mapaCatalogo[cod];
    const lote      = String(fila.lote || 'SIN LOTE').trim();
    const venc      = fila.vencimiento || '';
    const obs       = fila.obs || '';

    // Aplicar conversión si corresponde
    const conv = aplicarConversion(cod, cantidad, tablaConversiones);
    const notaConversion = conv.tiene_conversion
      ? '[CONV x' + conv.factor + '] ' + cantidad + ' caja(s) → ' +
        conv.cantidad_stock + ' ' + conv.unidad_stock
      : '';

    // Fila Movimientos_Inventario
    filasMovimientos.push([
      importId + '-' + (idx + 1).toString().padStart(4, '0'),
      new Date(),
      'Ajuste inicial',
      cod,
      ins.descripcion,
      conv.cantidad_stock,
      importId,
      getCurrentUser(),
      'Importación inicial' + (obs ? ' | ' + obs : '') +
        (notaConversion ? ' | ' + notaConversion : ''),
      lote,
      venc,
      ins.categoria || ''
    ]);

    // Fila Stock_Actual
    // (se ordenará por vencimiento al final)
    filasStock.push([
      cod,
      ins.descripcion,
      conv.cantidad_stock,
      Number(ins.stock_critico) || 0,
      'OK',              // estado — se recalcula al final
      new Date(),
      lote,
      venc,
      ins.clasificacion || ''
    ]);

    importadas++;
  });

  if (importadas === 0) {
    return {
      importadas: 0,
      saltadas,
      errores,
      tiempo_ms: Date.now() - inicio,
      detalle_errores: errores
    };
  }

  // ── Escritura batch en Movimientos_Inventario ──────────────────
  const sheetMov  = getSheet(CONFIG.SHEETS.MOVIMIENTOS);
  const lastMov   = sheetMov.getLastRow();
  sheetMov.getRange(lastMov + 1, 1, filasMovimientos.length, filasMovimientos[0].length)
    .setValues(filasMovimientos);

  // ── Escritura en Stock_Actual (suma a lotes existentes) ────────
  _integrarStockImportado(filasStock, catalogoInsumos);

  return {
    importadas,
    saltadas,
    errores,
    tiempo_ms:       Date.now() - inicio,
    detalle_errores: errores
  };
}

/**
 * Integra las filas importadas en Stock_Actual:
 * - Si el lote ya existe → suma la cantidad
 * - Si es lote nuevo → agrega fila
 * - Recalcula estado (OK/Crítico/Agotado) para todos los códigos afectados
 * - Ordena por vencimiento (FIFO)
 */
function _integrarStockImportado(filasNuevas, catalogoInsumos) {
  const sheetStock = getSheet(CONFIG.SHEETS.STOCK);
  const lastRow    = sheetStock.getLastRow();

  const dataStock = lastRow > 1
    ? sheetStock.getRange(2, 1, lastRow - 1, 9).getValues()
    : [];

  // Índice: "codigo|lote" → posición en dataStock
  const mapaStock = {};
  dataStock.forEach(function(row, idx) {
    const key = String(row[0]).trim() + '|' + String(row[6]).trim();
    mapaStock[key] = idx;
  });

  // Stock total por código (para recalcular estado al final)
  const stockTotal = {};
  dataStock.forEach(function(row) {
    const cod = String(row[0]).trim();
    stockTotal[cod] = (stockTotal[cod] || 0) + (Number(row[2]) || 0);
  });

  const codigosAfectados = {};

  filasNuevas.forEach(function(fila) {
    const cod   = String(fila[0]).trim();
    const lote  = String(fila[6]).trim();
    const cant  = Number(fila[2]) || 0;
    const key   = cod + '|' + lote;

    codigosAfectados[cod] = true;

    if (mapaStock.hasOwnProperty(key)) {
      // Lote existente: sumar
      const idx = mapaStock[key];
      dataStock[idx][2] = (Number(dataStock[idx][2]) || 0) + cant;
      stockTotal[cod]   = (stockTotal[cod] || 0) + cant;
    } else {
      // Lote nuevo: agregar
      dataStock.push(fila);
      mapaStock[key]  = dataStock.length - 1;
      stockTotal[cod] = (stockTotal[cod] || 0) + cant;
    }
  });

  // Recalcular estado para todos los códigos afectados
  dataStock.forEach(function(row) {
    const cod = String(row[0]).trim();
    if (!codigosAfectados[cod]) return;
    const ins     = catalogoInsumos.find(function(i) {
      return String(i.codigo_insumo).trim() === cod;
    });
    const critico = ins ? Number(ins.stock_critico) || 0 : Number(row[3]) || 0;
    const total   = stockTotal[cod] || 0;
    row[4] = total <= 0 ? 'Agotado' : total <= critico ? 'Crítico' : 'OK';
  });

  // Filtrar > 0, reescribir y ordenar FIFO
  const filasValidas = dataStock.filter(function(row) { return Number(row[2]) > 0; });

  if (sheetStock.getLastRow() > 1) {
    sheetStock.getRange(2, 1, sheetStock.getLastRow() - 1, 9).clearContent();
  }
  if (filasValidas.length > 0) {
    sheetStock.getRange(2, 1, filasValidas.length, 9).setValues(filasValidas);
    sheetStock.getRange(2, 1, filasValidas.length, 9).sort({ column: 8, ascending: true });
  }
}

/**
 * Crea la hoja Historial_Precios con sus headers y formato condicional.
 * Solo ejecutar UNA VEZ en un spreadsheet nuevo.
 */
function crearHojaHistorialPrecios() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  if (ss.getSheetByName('Historial_Precios')) {
    SpreadsheetApp.getUi().alert('La hoja Historial_Precios ya existe.');
    return;
  }

  const sheet = ss.insertSheet('Historial_Precios');

  sheet.getRange('A1:I1').merge();
  sheet.getRange('A1')
    .setValue('📈 HISTORIAL DE VARIACIÓN DE PRECIOS  |  Se registra automáticamente al crear OC y al recepcionar  |  Solo lectura')
    .setFontWeight('bold').setBackground('#1F4E79').setFontColor('#FFFFFF')
    .setFontSize(10).setWrap(true);
  sheet.setRowHeight(1, 32);

  const headers = [
    'fecha', 'codigo_insumo', 'descripcion', 'proveedor',
    'precio_anterior', 'precio_nuevo', 'variacion_pct', 'numero_orden', 'usuario'
  ];
  sheet.getRange(2, 1, 1, headers.length)
    .setValues([headers]).setFontWeight('bold')
    .setBackground('#2E75B6').setFontColor('#FFFFFF').setHorizontalAlignment('center');

  const anchos = [100, 110, 300, 200, 120, 120, 110, 130, 200];
  anchos.forEach(function(w, i) { sheet.setColumnWidth(i + 1, w); });
  sheet.setFrozenRows(2);

  // Formato condicional: rojo si sube >5%, verde si baja >5%
  const rangoVar = sheet.getRange('G3:G1000');
  const rules    = sheet.getConditionalFormatRules();
  rules.push(
    SpreadsheetApp.newConditionalFormatRule()
      .whenNumberGreaterThan(5)
      .setBackground('#FCE8E6').setFontColor('#A61C00')
      .setRanges([rangoVar]).build(),
    SpreadsheetApp.newConditionalFormatRule()
      .whenNumberLessThan(-5)
      .setBackground('#E6F4EA').setFontColor('#137333')
      .setRanges([rangoVar]).build()
  );
  sheet.setConditionalFormatRules(rules);

  SpreadsheetApp.getUi().alert(
    '✅ Hoja "Historial_Precios" creada.\n\n' +
    'Se completará automáticamente al crear órdenes de compra y al recepcionar.'
  );
}

/**
 * Retorna las OC editables: solo las que están en Pendiente
 * y no tienen ninguna recepción registrada (ni parcial).
 * Usada por FormularioEditarOC y FormularioCancelarOC.
 */
function getOCsEditables() {
  const sheetOrdenes  = getSheet(CONFIG.SHEETS.ORDENES);
  const sheetRecep    = getSheet(CONFIG.SHEETS.RECEPCIONES);
  const lastRow       = sheetOrdenes.getLastRow();
  if (lastRow <= 1) return [];

  // Números de órdenes que ya tienen al menos una recepción (no anulada)
  const recepData    = getDatosSheet(CONFIG.SHEETS.RECEPCIONES);
  const ordenesConRecepcion = {};
  recepData.forEach(function(rec) {
    const obs = String(rec.observaciones || '').toUpperCase();
    if (obs.indexOf('ANULADA') !== -1) return; // ignorar recepciones anuladas
    ordenesConRecepcion[String(rec.numero_orden).trim()] = true;
  });

  const dataOrdenes = sheetOrdenes.getRange(2, 1, lastRow - 1, 10).getValues();

  return dataOrdenes
    .filter(function(row) {
      const estado      = String(row[4]).trim();
      const numOrden    = String(row[0]).trim();
      return row[0] &&
             estado === CONFIG.ESTADOS.ORDEN.PENDIENTE &&
             !ordenesConRecepcion[numOrden];
    })
    .map(function(row) {
      return {
        numero_orden:           String(row[0]).trim(),
        fecha_emision:          formatearFecha(row[1]),
        proveedor:              String(row[3]).trim(),   // nombre guardado en col D
        codigo_proveedor:       String(row[2]).trim(),
        fecha_entrega_estimada: formatearFecha(row[6]),
        total_orden:            row[7],
        observaciones:          String(row[8] || '').trim(),
        solicitado_por:         String(row[9] || '').trim()
      };
    });
}

/**
 * Retorna las recepciones activas (no anuladas) para el formulario de cancelación.
 * Incluye nombre del proveedor y ítems recibidos para mostrar al usuario.
 */
function getRecepcionesActivas() {
  const recepData   = getDatosSheet(CONFIG.SHEETS.RECEPCIONES);
  const detalleData = getDatosSheet(CONFIG.SHEETS.DETALLE_RECEPCIONES);
  const ordenesData = getDatosSheet(CONFIG.SHEETS.ORDENES);

  // Mapa número_orden → nombre_proveedor
  const mapaProveedores = {};
  ordenesData.forEach(function(o) {
    mapaProveedores[String(o.numero_orden).trim()] =
      String(o.nombre_proveedor || o.codigo_proveedor || '').trim();
  });

  return recepData
    .filter(function(rec) {
      const obs = String(rec.observaciones || '').toUpperCase();
      return obs.indexOf('ANULADA') === -1; // excluir ya anuladas
    })
    .map(function(rec) {
      const numRec   = String(rec.numero_recepcion).trim();
      const numOrden = String(rec.numero_orden).trim();

      // Items de esta recepción
      const items = detalleData
        .filter(function(d) { return String(d.numero_recepcion).trim() === numRec; })
        .map(function(d) {
          return {
            codigo_insumo:     String(d.codigo_insumo || '').trim(),
            descripcion:       String(d.descripcion   || '').trim(),
            cantidad_recibida: Number(d.cantidad_recibida) || 0,
            lote:              String(d.lote           || 'SIN LOTE').trim(),
            vencimiento:       d.vencimiento ? formatearFecha(d.vencimiento) : '',
            nota_conversion:   String(d.nota_conversion || '').trim()
          };
        });

      return {
        numero_recepcion: numRec,
        fecha:            formatearFecha(rec.fecha),
        numero_orden:     numOrden,
        proveedor:        mapaProveedores[numOrden] || '—',
        recibido_por:     String(rec.recibido_por || '').trim(),
        items
      };
    })
    .sort(function(a, b) {
      // Más recientes primero
      return b.numero_recepcion.localeCompare(a.numero_recepcion);
    });
}

/**
 * Retorna los egresos de consumo activos (no anulados) de los últimos 180 días.
 * Solo tipo "Egreso" con referencia "BAJA-HOJA" (los generados desde Bajas_Consumo).
 * Usada por FormularioCancelarBaja.
 */
function getMovimientosEgresoActivos() {
  const sheetMov = getSheet(CONFIG.SHEETS.MOVIMIENTOS);
  const data     = sheetMov.getDataRange().getValues();
  if (data.length <= 1) return [];

  const headers        = data[0];
  const idxId          = headers.indexOf('id_movimiento');
  const idxFecha       = headers.indexOf('fecha');
  const idxTipo        = headers.indexOf('tipo_movimiento');
  const idxCodigo      = headers.indexOf('codigo_insumo');
  const idxNombre      = headers.indexOf('nombre_insumo');
  const idxCantidad    = headers.indexOf('cantidad');
  const idxReferencia  = headers.indexOf('referencia');
  const idxUsuario     = headers.indexOf('usuario');
  const idxObs        = headers.indexOf('observaciones');
  const idxLote        = headers.indexOf('lote');
  const idxVenc        = headers.indexOf('vencimiento');

  const hace180 = new Date();
  hace180.setDate(hace180.getDate() - 180);

  const resultados = [];

  for (let i = 1; i < data.length; i++) {
    const row  = data[i];
    const tipo = String(row[idxTipo] || '').trim();
    const obs  = String(row[idxObs]  || '').toUpperCase();
    const ref  = String(row[idxReferencia] || '').trim();

    if (tipo !== 'Egreso')                 continue;
    if (obs.indexOf('ANULADO') !== -1)     continue; // ya anulado

    const fecha = new Date(row[idxFecha]);
    if (isNaN(fecha.getTime()) || fecha < hace180) continue;

    resultados.push({
      id_movimiento: String(row[idxId]      || '').trim(),
      fila:          i + 1,                 // 1-indexed, para poder ubicarlo rápido
      fecha:         formatearFecha(row[idxFecha]),
      codigo_insumo: String(row[idxCodigo]  || '').trim(),
      descripcion:   String(row[idxNombre]  || '').trim(),
      cantidad:      Number(row[idxCantidad]) || 0,
      referencia:    ref,
      usuario:       String(row[idxUsuario] || '').trim(),
      observaciones: String(row[idxObs]     || '').trim(),
      lote:          String(row[idxLote]    || 'SIN LOTE').trim(),
      vencimiento:   row[idxVenc] ? formatearFecha(row[idxVenc]) : ''
    });
  }

  // Más recientes primero
  return resultados.sort(function(a, b) {
    return b.id_movimiento.localeCompare(a.id_movimiento);
  });
}

/**
 * Edita una OC existente en estado Pendiente (sin recepciones).
 *
 * @param {Object} editData
 *   numero_orden            {string}
 *   fecha_entrega_estimada  {string}
 *   observaciones           {string}
 *   solicitado_por          {string}
 *   items  {Array} — [ { codigo_insumo, descripcion, cantidad, precio_unitario } ]
 *
 * @returns {{ ok: boolean, mensaje: string }}
 */
function editarOrdenCompra(editData) {
  try {
    const numeroOrden = String(editData.numero_orden).trim();

    // ── 1. Validar que la OC sea editable ─────────────────────────
    const ocs = getOCsEditables();
    const ocBase = ocs.find(function(o) { return o.numero_orden === numeroOrden; });
    if (!ocBase) {
      return {
        ok: false,
        mensaje: 'La OC "' + numeroOrden + '" no es editable. ' +
                 'Solo se pueden editar OC en estado Pendiente sin recepciones registradas.'
      };
    }

    // ── 2. Recalcular total ────────────────────────────────────────
    const items = editData.items || [];
    if (items.length === 0) {
      return { ok: false, mensaje: 'Debe incluir al menos un ítem en la OC.' };
    }

    const nuevoTotal = items.reduce(function(sum, item) {
      return sum + (Number(item.cantidad) * Number(item.precio_unitario));
    }, 0);

    // ── 3. Actualizar cabecera en Ordenes_Compra ──────────────────
    const sheetOrdenes = getSheet(CONFIG.SHEETS.ORDENES);
    const dataOrdenes  = sheetOrdenes.getDataRange().getValues();

    let filaOC = -1;
    for (let i = 1; i < dataOrdenes.length; i++) {
      if (String(dataOrdenes[i][0]).trim() === numeroOrden) {
        filaOC = i + 1; // 1-indexed
        break;
      }
    }
    if (filaOC === -1) {
      return { ok: false, mensaje: 'No se encontró la OC "' + numeroOrden + '" en la hoja.' };
    }

    // Actualizar: fecha_entrega (col G=7), total (col H=8), observaciones (col I=9), solicitado_por (col J=10)
    sheetOrdenes.getRange(filaOC, 7, 1, 4).setValues([[
      editData.fecha_entrega_estimada || '',
      nuevoTotal,
      editData.observaciones || '',
      editData.solicitado_por || ''
    ]]);

    // ── 4. Reemplazar ítems en Detalle_Ordenes ────────────────────
    const sheetDetalle = getSheet(CONFIG.SHEETS.DETALLE_ORDENES);
    const dataDetalle  = sheetDetalle.getDataRange().getValues();

    // Encontrar y borrar filas existentes de esta OC
    // Recorremos desde el final para no desplazar índices al borrar
    for (let i = dataDetalle.length - 1; i >= 1; i--) {
      if (String(dataDetalle[i][0]).trim() === numeroOrden) {
        sheetDetalle.deleteRow(i + 1);
      }
    }

    // Escribir los nuevos ítems en batch
    const fechaAhora  = new Date();
    const filasDetalle = items.map(function(item, idx) {
      const subtotal = Number(item.cantidad) * Number(item.precio_unitario);
      return [
        numeroOrden,
        idx + 1,
        item.codigo_insumo,
        item.descripcion,
        Number(item.cantidad),
        Number(item.precio_unitario),
        subtotal,
        0,                               // cantidad_recepcionada = 0
        CONFIG.ESTADOS.ITEM.PENDIENTE,
        fechaAhora
      ];
    });

    const lastDetalle = sheetDetalle.getLastRow();
    sheetDetalle
      .getRange(lastDetalle + 1, 1, filasDetalle.length, filasDetalle[0].length)
      .setValues(filasDetalle);

    // ── 5. Actualizar historial de precios si cambiaron precios ───
    const usuario       = getCurrentUser();
    const proveedor     = getProveedor(ocBase.codigo_proveedor);
    const nombreProv    = proveedor ? proveedor.razon_social : ocBase.codigo_proveedor;
    registrarHistorialPrecios(
      items.map(function(item) {
        return {
          codigo_insumo:  item.codigo_insumo,
          descripcion:    item.descripcion,
          precio_unitario: Number(item.precio_unitario)
        };
      }),
      numeroOrden + '-EDIT',
      usuario,
      fechaAhora,
      nombreProv
    );

    return {
      ok:     true,
      mensaje: 'OC "' + numeroOrden + '" actualizada correctamente.\n' +
               '• ' + items.length + ' ítem(s) | Nuevo total: $' + nuevoTotal.toFixed(2)
    };

  } catch (e) {
    return { ok: false, mensaje: 'Error al editar la OC: ' + e.message };
  }
}

/**
 * Marca una OC como Cancelada.
 * Solo opera sobre OC en estado Pendiente sin recepciones.
 *

 * @param {string} motivo        — se agrega a observaciones

    numeroOrden = String(numeroOrden).trim();

    const ocs    = getOCsEditables();
    const ocBase = ocs.find(function(o) { return o.numero_orden === numeroOrden; });
    if (!ocBase) {
      return {
        ok: false,
        mensaje: 'La OC "' + numeroOrden + '" no puede cancelarse. ' +

/**
 * Revierte una recepción completa:
 *   1. Lee los ítems recibidos desde Detalle_Recepciones
 *   2. Lee los movimientos de Ingreso asociados desde Movimientos_Inventario
 *   3. Crea movimientos de "Anulación" (tipo Ajuste, cantidad negativa) para trazabilidad
 *   4. Actualiza Stock_Actual (resta las cantidades ingresadas)
 *   5. Revierte las cantidades en Detalle_Ordenes
 *   6. Recalcula el estado de la OC
 *   7. Marca la recepción como ANULADA en Recepciones
 *
 * @param {string} numeroRecepcion
 * @param {string} motivo
 *
 * @returns {{ ok: boolean, mensaje: string }}
 */
function cancelarRecepcion(numeroRecepcion, motivo) {
  try {
    numeroRecepcion = String(numeroRecepcion).trim();
    const usuario   = getCurrentUser();
    const ahora     = new Date();

    // ── 1. Leer recepción y validar que no esté ya anulada ────────
    const sheetRecep = getSheet(CONFIG.SHEETS.RECEPCIONES);
    const dataRecep  = sheetRecep.getDataRange().getValues();
    const headersRecep = dataRecep[0];

    const idxRecNum = headersRecep.indexOf('numero_recepcion');
    const idxRecObs = headersRecep.indexOf('observaciones');
    const idxRecOrden = headersRecep.indexOf('numero_orden');

    let filaRecep    = -1;
    let numeroOrden  = '';
    let obsActualRec = '';

    for (let i = 1; i < dataRecep.length; i++) {
      if (String(dataRecep[i][idxRecNum]).trim() !== numeroRecepcion) continue;
      const obs = String(dataRecep[i][idxRecObs] || '').toUpperCase();
      if (obs.indexOf('ANULADA') !== -1) {
        return { ok: false, mensaje: 'La recepción "' + numeroRecepcion + '" ya está anulada.' };
      }
      filaRecep    = i + 1;
      numeroOrden  = String(dataRecep[i][idxRecOrden]).trim();
      obsActualRec = String(dataRecep[i][idxRecObs] || '').trim();
      break;
    }

    if (filaRecep === -1) {
      return { ok: false, mensaje: 'No se encontró la recepción "' + numeroRecepcion + '".' };
    }

    // ── 2. Leer ítems de Detalle_Recepciones ─────────────────────
    const detalleRecepData = getDatosSheet(CONFIG.SHEETS.DETALLE_RECEPCIONES);
    const itemsRecibidos   = detalleRecepData.filter(function(d) {
      return String(d.numero_recepcion).trim() === numeroRecepcion;
    });

    if (itemsRecibidos.length === 0) {
      return { ok: false, mensaje: 'No se encontraron ítems para la recepción "' + numeroRecepcion + '".' };
    }

    // ── 3. Leer movimientos de Ingreso de esta recepción ──────────
    const sheetMov    = getSheet(CONFIG.SHEETS.MOVIMIENTOS);
    const dataMov     = sheetMov.getDataRange().getValues();
    const headersMov  = dataMov[0];

    const idxMovId   = headersMov.indexOf('id_movimiento');
    const idxMovTipo = headersMov.indexOf('tipo_movimiento');
    const idxMovRef  = headersMov.indexOf('referencia');
    const idxMovCod  = headersMov.indexOf('codigo_insumo');
    const idxMovNom  = headersMov.indexOf('nombre_insumo');
    const idxMovCant = headersMov.indexOf('cantidad');
    const idxMovLote = headersMov.indexOf('lote');
    const idxMovVenc = headersMov.indexOf('vencimiento');
    const idxMovCat  = headersMov.indexOf('categoria');

    const movIngresosRecep = [];
    for (let i = 1; i < dataMov.length; i++) {
      if (String(dataMov[i][idxMovRef]).trim()  !== numeroRecepcion) continue;
      if (String(dataMov[i][idxMovTipo]).trim() !== 'Ingreso')       continue;
      movIngresosRecep.push({
        fila:          i + 1,
        id_movimiento: String(dataMov[i][idxMovId]   || '').trim(),
        codigo_insumo: String(dataMov[i][idxMovCod]  || '').trim(),
        nombre_insumo: String(dataMov[i][idxMovNom]  || '').trim(),
        cantidad:      Number(dataMov[i][idxMovCant]) || 0,
        lote:          String(dataMov[i][idxMovLote]  || 'SIN LOTE').trim(),
        vencimiento:   dataMov[i][idxMovVenc] || '',
        categoria:     String(dataMov[i][idxMovCat]  || '').trim()
      });
    }

    // ── 4. Crear movimientos de Anulación (Ajuste negativo) ───────
    const filasAnulacion = movIngresosRecep.map(function(mov) {
      return [
        'ANUL-' + Date.now() + '-' + Math.random().toString(36).substr(2, 4),
        ahora,
        'Ajuste',                        // tipo que maneja actualizarStockActual
        mov.codigo_insumo,
        mov.nombre_insumo,
        -mov.cantidad,                   // negativo = resta stock
        'ANULACION-' + numeroRecepcion,
        usuario,
        'Anulación recepción ' + numeroRecepcion + ' | ' + (motivo || 'Sin motivo'),
        mov.lote,
        mov.vencimiento,
        mov.categoria
      ];
    });

    if (filasAnulacion.length > 0) {
      const lastMov = sheetMov.getLastRow();
      sheetMov.getRange(lastMov + 1, 1, filasAnulacion.length, filasAnulacion[0].length)
        .setValues(filasAnulacion);
    }

    // ── 5. Actualizar Stock_Actual (restar cantidades ingresadas) ─
    _revertirStockDeRecepcion(movIngresosRecep, ahora);

    // ── 6. Revertir cantidades en Detalle_Ordenes ─────────────────
    _revertirCantidadesEnOrden(numeroOrden, itemsRecibidos);

    // ── 7. Recalcular estado de la OC ─────────────────────────────
    actualizarEstadoOrden(numeroOrden);

    // ── 8. Marcar recepción como ANULADA ──────────────────────────
    const nuevaObsRec = (obsActualRec ? obsActualRec + ' | ' : '') +
                        'ANULADA: ' + (motivo || 'Sin motivo') +
                        ' (' + formatearFecha(ahora) + ' — ' + usuario + ')';
    sheetRecep.getRange(filaRecep, idxRecObs + 1).setValue(nuevaObsRec);

    return {
      ok:     true,
      mensaje: 'Recepción "' + numeroRecepcion + '" anulada exitosamente.\n' +
               '• ' + movIngresosRecep.length + ' ítem(s) revertido(s) del stock.\n' +
               '• Estado de OC "' + numeroOrden + '" recalculado.'
    };

  } catch (e) {
    return { ok: false, mensaje: 'Error al cancelar la recepción: ' + e.message };
  }
}

/**
 * Revierte un egreso de consumo (Baja).
 *   1. Busca el movimiento por ID en Movimientos_Inventario
 *   2. Valida que sea Egreso y no esté anulado
 *   3. Crea un movimiento de "Ajuste" positivo (contrapartida)
 *   4. Actualiza Stock_Actual (suma de vuelta la cantidad consumida)
 *   5. Marca el movimiento original como ANULADO en Observaciones
 *
 * @param {string} idMovimiento
 * @param {string} motivo
 *
 * @returns {{ ok: boolean, mensaje: string }}
 */
function cancelarBaja(idMovimiento, motivo) {
  try {
    idMovimiento = String(idMovimiento).trim();
    const usuario = getCurrentUser();
    const ahora   = new Date();

    // ── 1. Encontrar el movimiento ─────────────────────────────────
    const sheetMov   = getSheet(CONFIG.SHEETS.MOVIMIENTOS);
    const dataMov    = sheetMov.getDataRange().getValues();
    const headersMov = dataMov[0];

    const idxId   = headersMov.indexOf('id_movimiento');
    const idxTipo = headersMov.indexOf('tipo_movimiento');
    const idxCod  = headersMov.indexOf('codigo_insumo');
    const idxNom  = headersMov.indexOf('nombre_insumo');
    const idxCant = headersMov.indexOf('cantidad');
    const idxRef  = headersMov.indexOf('referencia');
    const idxObs  = headersMov.indexOf('observaciones');
    const idxLote = headersMov.indexOf('lote');
    const idxVenc = headersMov.indexOf('vencimiento');
    const idxCat  = headersMov.indexOf('categoria');

    let filaMovimiento = -1;
    let movData        = null;

    for (let i = 1; i < dataMov.length; i++) {
      if (String(dataMov[i][idxId]).trim() !== idMovimiento) continue;

      const tipo = String(dataMov[i][idxTipo] || '').trim();
      const obs  = String(dataMov[i][idxObs]  || '').toUpperCase();

      if (tipo !== 'Egreso') {
        return { ok: false, mensaje: 'El movimiento "' + idMovimiento + '" no es un Egreso.' };
      }
      if (obs.indexOf('ANULADO') !== -1) {
        return { ok: false, mensaje: 'El movimiento "' + idMovimiento + '" ya está anulado.' };
      }

      filaMovimiento = i + 1;
      movData = {
        codigo_insumo: String(dataMov[i][idxCod]  || '').trim(),
        nombre_insumo: String(dataMov[i][idxNom]  || '').trim(),
        cantidad:      Number(dataMov[i][idxCant]) || 0,
        referencia:    String(dataMov[i][idxRef]   || '').trim(),
        lote:          String(dataMov[i][idxLote]  || 'SIN LOTE').trim(),
        vencimiento:   dataMov[i][idxVenc] || '',
        categoria:     String(dataMov[i][idxCat]   || '').trim(),
        obsOriginal:   String(dataMov[i][idxObs]   || '').trim()
      };
      break;
    }

    if (!movData) {
      return { ok: false, mensaje: 'No se encontró el movimiento "' + idMovimiento + '".' };
    }

    // ── 2. Crear movimiento de contrapartida (Ajuste positivo) ────
    sheetMov.appendRow([
      'RECUP-' + Date.now() + '-' + Math.random().toString(36).substr(2, 4),
      ahora,
      'Ajuste',
      movData.codigo_insumo,
      movData.nombre_insumo,
      movData.cantidad,                 // positivo = suma stock
      'ANULACION-' + idMovimiento,
      usuario,
      'Anulación baja ' + idMovimiento + ' | ' + (motivo || 'Sin motivo'),
      movData.lote,
      movData.vencimiento,
      movData.categoria
    ]);

    // ── 3. Marcar el egreso original como ANULADO ──────────────────
    const nuevaObs = movData.obsOriginal +
                     ' | ANULADO: ' + (motivo || 'Sin motivo') +
                     ' (' + formatearFecha(ahora) + ' — ' + usuario + ')';
    sheetMov.getRange(filaMovimiento, idxObs + 1).setValue(nuevaObs);

    // ── 4. Devolver el stock al lote correspondiente ───────────────
    _sumarStockLote(
      movData.codigo_insumo,
      movData.lote,
      movData.vencimiento,
      movData.cantidad,
      movData.nombre_insumo,
      ahora
    );

    return {
      ok:     true,
      mensaje: 'Baja "' + idMovimiento + '" anulada exitosamente.\n' +
               '• Se restituyeron ' + movData.cantidad + ' u. de "' + movData.nombre_insumo +
               '" al lote ' + movData.lote + '.'
    };

  } catch (e) {
    return { ok: false, mensaje: 'Error al cancelar la baja: ' + e.message };
  }
}

/**
 * Revierte el stock de una recepción:
 * resta cada cantidad de Ingreso del lote correspondiente en Stock_Actual.
 */
function _revertirStockDeRecepcion(movIngresos, ahora) {
  const sheetStock = getSheet(CONFIG.SHEETS.STOCK);
  const lastRow    = sheetStock.getLastRow();
  if (lastRow <= 1) return;

  const dataStock  = sheetStock.getRange(2, 1, lastRow - 1, 9).getValues();
  const catalogoInsumos = getInsumosActivos();

  // Stock total actual por código
  const stockTotal = {};
  dataStock.forEach(function(row) {
    const cod = String(row[0]).trim();
    stockTotal[cod] = (stockTotal[cod] || 0) + (Number(row[2]) || 0);
  });

  const codigosAfectados = {};

  movIngresos.forEach(function(mov) {
    const cod  = mov.codigo_insumo;
    const lote = mov.lote;
    codigosAfectados[cod] = true;

    for (let i = 0; i < dataStock.length; i++) {
      if (String(dataStock[i][0]).trim() !== cod)  continue;
      if (String(dataStock[i][6]).trim() !== lote) continue;

      const nuevoStock = Math.max((Number(dataStock[i][2]) || 0) - mov.cantidad, 0);
      dataStock[i][2]  = nuevoStock;
      dataStock[i][5]  = ahora;
      stockTotal[cod]  = (stockTotal[cod] || 0) - mov.cantidad;
      break;
    }
  });

  // Recalcular estado para los códigos afectados
  dataStock.forEach(function(row) {
    const cod = String(row[0]).trim();
    if (!codigosAfectados[cod]) return;
    const ins     = catalogoInsumos.find(function(i) {
      return String(i.codigo_insumo).trim() === cod;
    });
    const critico = ins ? Number(ins.stock_critico) || 0 : Number(row[3]) || 0;
    const total   = stockTotal[cod] || 0;
    row[4] = total <= 0 ? 'Agotado' : total <= critico ? 'Crítico' : 'OK';
  });

  // Filtrar y reescribir
  const filasValidas = dataStock.filter(function(row) { return Number(row[2]) > 0; });
  sheetStock.getRange(2, 1, lastRow - 1, 9).clearContent();
  if (filasValidas.length > 0) {
    sheetStock.getRange(2, 1, filasValidas.length, 9).setValues(filasValidas);
    sheetStock.getRange(2, 1, filasValidas.length, 9).sort({ column: 8, ascending: true });
  }
}

/**
 * Revierte las cantidades recepcionadas en Detalle_Ordenes.
 * Resta lo recibido en esta recepción de la columna cantidad_recepcionada,
 * y vuelve el estado_item a Pendiente si queda en 0.
 */
function _revertirCantidadesEnOrden(numeroOrden, itemsRecibidos) {
  const sheetDetalle = getSheet(CONFIG.SHEETS.DETALLE_ORDENES);
  const data         = sheetDetalle.getDataRange().getValues();

  // Mapa codigo → cantidad recibida en esta recepción
  const mapaCantidades = {};
  itemsRecibidos.forEach(function(item) {
    const cod = String(item.codigo_insumo || '').trim();
    mapaCantidades[cod] = (mapaCantidades[cod] || 0) + (Number(item.cantidad_recibida) || 0);
  });

  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() !== numeroOrden) continue;

    const cod = String(data[i][2]).trim();
    if (!mapaCantidades[cod]) continue;

    const cantRecibidaActual = Number(data[i][7]) || 0;
    const cantSolicitada     = Number(data[i][4]) || 0;
    const nueva              = Math.max(cantRecibidaActual - mapaCantidades[cod], 0);
    const nuevoEstado        = nueva <= 0
      ? CONFIG.ESTADOS.ITEM.PENDIENTE
      : nueva < cantSolicitada
        ? CONFIG.ESTADOS.ITEM.PARCIAL
        : CONFIG.ESTADOS.ITEM.COMPLETO;

    sheetDetalle.getRange(i + 1, 8, 1, 2).setValues([[nueva, nuevoEstado]]);
  }
}

/**
 * Suma una cantidad a un lote específico en Stock_Actual.
 * Si el lote no existe, crea una fila nueva.
 * Usada por cancelarBaja() para restituir stock consumido.
 */
function _sumarStockLote(codigo, lote, vencimiento, cantidad, descripcion, ahora) {
  const sheetStock    = getSheet(CONFIG.SHEETS.STOCK);
  const lastRow       = sheetStock.getLastRow();
  const catalogoInsumos = getInsumosActivos();

  const dataStock = lastRow > 1
    ? sheetStock.getRange(2, 1, lastRow - 1, 9).getValues()
    : [];

  const stockTotal = {};
  dataStock.forEach(function(row) {
    const cod = String(row[0]).trim();
    stockTotal[cod] = (stockTotal[cod] || 0) + (Number(row[2]) || 0);
  });

  let encontrado = false;
  for (let i = 0; i < dataStock.length; i++) {
    if (String(dataStock[i][0]).trim() !== codigo) continue;
    if (String(dataStock[i][6]).trim() !== lote)   continue;
    dataStock[i][2] = (Number(dataStock[i][2]) || 0) + cantidad;
    dataStock[i][5] = ahora;
    stockTotal[codigo] = (stockTotal[codigo] || 0) + cantidad;
    encontrado = true;
    break;
  }

  if (!encontrado) {
    const ins     = catalogoInsumos.find(function(i) {
      return String(i.codigo_insumo).trim() === codigo;
    });
    const critico = ins ? Number(ins.stock_critico) || 0 : 0;
    dataStock.push([
      codigo, descripcion, cantidad, critico, 'OK',
      ahora, lote, vencimiento,
      ins ? (ins.clasificacion || '') : ''
    ]);
    stockTotal[codigo] = (stockTotal[codigo] || 0) + cantidad;
  }

  // Recalcular estado
  const ins2     = catalogoInsumos.find(function(i) {
    return String(i.codigo_insumo).trim() === codigo;
  });
  const critico2   = ins2 ? Number(ins2.stock_critico) || 0 : 0;
  const total      = stockTotal[codigo] || 0;
  const nuevoEst   = total <= 0 ? 'Agotado' : total <= critico2 ? 'Crítico' : 'OK';

  for (let j = 0; j < dataStock.length; j++) {
    if (String(dataStock[j][0]).trim() === codigo) dataStock[j][4] = nuevoEst;
  }

  const filasValidas = dataStock.filter(function(row) { return Number(row[2]) > 0; });
  if (sheetStock.getLastRow() > 1) {
    sheetStock.getRange(2, 1, sheetStock.getLastRow() - 1, 9).clearContent();
  }
  if (filasValidas.length > 0) {
    sheetStock.getRange(2, 1, filasValidas.length, 9).setValues(filasValidas);
    sheetStock.getRange(2, 1, filasValidas.length, 9).sort({ column: 8, ascending: true });
  }
}

/**
 * Dado un array de códigos de insumo (los que aparecen en una recepción a punto
 * de guardarse), devuelve solo los que tienen una conversión activa (factor > 1)
 * todavía NO confirmada por el equipo — o porque es la primera vez que se
 * configuró, o porque cambió el factor/unidad después de haberse confirmado antes.
 * Usada por FormularioRecepcion.html antes de guardar.
 */
function getConversionesPendientesConfirmar(codigosInsumo) {
  const tabla = _getTablaConversionesCompleta();
  const codigosUnicos = Array.from(new Set((codigosInsumo || []).map(function(c) {
    return String(c).trim();
  })));

  return codigosUnicos
    .map(function(cod) { return tabla[cod] ? Object.assign({ codigo: cod }, tabla[cod]) : null; })
    .filter(function(conv) {
      return conv && conv.activo && conv.factor > 1 && !conv.confirmada;
    });
}

/**
 * Marca como confirmadas (por el equipo, al momento de recepcionar) las
 * conversiones de los códigos indicados, para no volver a preguntar por ellas
 * hasta que el factor/unidad vuelva a cambiar. Se llama justo antes de
 * registrarRecepcion(), una vez que el usuario confirmó cada equivalencia.
 */
function marcarConversionesConfirmadas(codigosInsumo) {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Config_Conversiones');
  if (!sheet) return { ok: false, mensaje: 'No existe la hoja Config_Conversiones.' };

  const tabla = _getTablaConversionesCompleta();
  const colConfirmada = _asegurarColumnaConfirmada(sheet);

  (codigosInsumo || []).forEach(function(cod) {
    const conv = tabla[String(cod).trim()];
    if (conv) sheet.getRange(conv.fila, colConfirmada).setValue(true);
  });

  SpreadsheetApp.flush();
  return { ok: true };
}

/**
 * Devuelve los emails cargados para un proveedor como array, aceptando
 * múltiples direcciones separadas por coma, punto y coma o salto de línea.
 * Por ahora solo se usa emails[0] en revisarPedidoOC/enviarPedidoOC.
 */
function getEmailsProveedor(proveedor) {
  if (!proveedor || !proveedor.email) return [];
  return String(proveedor.email)
    .split(/[,;\n]+/)
    .map(function(e) { return e.trim(); })
    .filter(function(e) { return e.length > 0; });
}

function tieneEmailProveedor(codigoProveedor) {

}

function construirHtmlOrdenPdf(numeroOrden) {
  const detalle   = getDetalleOrden(numeroOrden);
  const orden     = detalle.orden;      // orden.proveedor = CÓDIGO del proveedor (ver Ordenes_Compra)
  const items     = detalle.items;
  const proveedor = getProveedor(orden.proveedor) || {};
  const nombreProveedor = proveedor.razon_social || orden.nombre_proveedor || orden.proveedor;

  const filas = items.map(function(it) {
    const subtotal = it.cantidad * it.precio_unitario;
    return '<tr>' +
      '<td>' + it.codigo_insumo + '</td>' +
      '<td>' + it.descripcion + '</td>' +
      '<td style="text-align:right">' + it.cantidad + '</td>' +
      '<td style="text-align:right">' + (it.precio_unitario ? '$' + it.precio_unitario.toFixed(2) : '-') + '</td>' +
      '<td style="text-align:right">' + (it.precio_unitario ? '$' + subtotal.toFixed(2) : '-') + '</td>' +
    '</tr>';
  }).join('');

  return '<html><head><meta charset="utf-8"><style>' +
    'body{font-family:Arial, sans-serif; font-size:12px; color:#222; padding:20px;}' +
    'h1{font-size:18px; margin-bottom:2px;} .sub{color:#666; margin-top:0; margin-bottom:16px;}' +
    'table{width:100%; border-collapse:collapse; margin-top:10px;}' +
    'th,td{border:1px solid #ccc; padding:6px 8px; font-size:11px;}' +
    'th{background:#f2f2f2; text-align:left;}' +
    '.total{font-weight:bold; text-align:right; margin-top:10px; font-size:13px;}' +
    '</style></head><body>' +
    '<h1>Orden de Compra ' + numeroOrden + '</h1>' +
    '<div class="sub">CURF — Laboratorio de Microbiología</div>' +
    '<p><strong>Proveedor:</strong> ' + nombreProveedor + '<br>' +
    '<strong>Fecha de emisión:</strong> ' + orden.fecha_emision + '<br>' +
    '<strong>Entrega estimada:</strong> ' + (orden.fecha_entrega_estimada || '-') + '</p>' +
    '<table><thead><tr><th>Código</th><th>Descripción</th><th>Cantidad</th><th>Precio unit.</th><th>Subtotal</th></tr></thead>' +
    '<tbody>' + filas + '</tbody></table>' +
    '<div class="total">Total: $' + (Number(orden.total_orden) || 0).toFixed(2) + '</div>' +
    (orden.observaciones ? '<p><strong>Observaciones:</strong> ' + orden.observaciones + '</p>' : '') +
  '</body></html>';
}

function generarPdfOrden(numeroOrden) {
  const html = construirHtmlOrdenPdf(numeroOrden);
  return HtmlService.createHtmlOutput(html)
    .getBlob()
    .getAs('application/pdf')
    .setName('OC_' + numeroOrden + '.pdf');
}

/** Devuelve el PDF en base64 para mostrarlo en un <iframe> sin descargar nada. */
function previsualizarPdfOrden(numeroOrden) {
  const blob = generarPdfOrden(numeroOrden);
  return Utilities.base64Encode(blob.getBytes());
}

/** Abre un diálogo modal con la vista previa del PDF de una orden puntual. */
function mostrarPreviaPdfOrden(numeroOrden) {
  try {
    const base64 = previsualizarPdfOrden(numeroOrden);
    // ✅ v2.8.1: Chrome bloquea la navegación directa de un <iframe> a una URL
    // "data:application/pdf;..." (la trata como una descarga potencialmente
    // insegura y la corta con "This page has been blocked by Chrome"). La forma
    // que sí funciona es armar el PDF como Blob del lado del cliente (con
    // atob + Uint8Array) y apuntar el iframe a un blob: URL generado con
    // URL.createObjectURL — eso Chrome sí lo permite.
    const html = HtmlService.createHtmlOutput(
      '<!DOCTYPE html><html><head><meta charset="utf-8"></head>' +
      '<body style="margin:0;padding:0;font-family:Arial,sans-serif;">' +
      '<div id="msg" style="padding:20px;color:#666;">⏳ Cargando PDF…</div>' +
      '<iframe id="pdfFrame" style="width:100%;height:100vh;border:none;display:none;"></iframe>' +
      '<script>' +
      '(function() {' +
      '  var base64 = "' + base64 + '";' +
      '  try {' +
      '    var byteChars = atob(base64);' +
      '    var byteNumbers = new Array(byteChars.length);' +
      '    for (var i = 0; i < byteChars.length; i++) { byteNumbers[i] = byteChars.charCodeAt(i); }' +
      '    var byteArray = new Uint8Array(byteNumbers);' +
      '    var blob = new Blob([byteArray], { type: "application/pdf" });' +
      '    var url = URL.createObjectURL(blob);' +
      '    var frame = document.getElementById("pdfFrame");' +
      '    frame.onload = function() { document.getElementById("msg").style.display = "none"; frame.style.display = "block"; };' +
      '    frame.src = url;' +
      '  } catch (e) {' +
      '    document.getElementById("msg").textContent = "❌ Error al mostrar el PDF: " + e.message;' +
      '  }' +
      '})();' +
      '</script>' +
      '</body></html>'
    ).setWidth(780).setHeight(680);
    SpreadsheetApp.getUi().showModalDialog(html, '📄 Vista previa — ' + numeroOrden);
  } catch (e) {
    SpreadsheetApp.getUi().alert('Error al generar la vista previa: ' + e.message);
  }
}

/**
 * Valida que el proveedor tenga email y arma los datos comunes a
 * revisarPedidoOC/enviarPedidoOC (evita repetir la validación en ambas).
 */
function prepararEnvioOrden(numeroOrden) {
  const detalle = getDetalleOrden(numeroOrden);
  const orden = detalle.orden;
  const proveedor = getProveedor(orden.proveedor);
  const emails = getEmailsProveedor(proveedor);

  if (emails.length === 0) {
    const nombre = (proveedor && proveedor.razon_social) || orden.nombre_proveedor || orden.proveedor;
    throw new Error('El proveedor "' + nombre + '" no tiene email cargado. Cargalo en la hoja Proveedores antes de continuar.');
  }

  const nombreProveedor = (proveedor && proveedor.razon_social) || orden.nombre_proveedor || orden.proveedor;
  // Por ahora se usa solo el primer email cargado (ver nota al inicio del archivo).
  const destinatario = emails[0];
  const pdf = generarPdfOrden(numeroOrden);
  const asunto = 'Orden de Compra ' + numeroOrden + ' — CURF Laboratorio';
  const cuerpo = 'Estimados,\n\nAdjuntamos la orden de compra ' + numeroOrden + ' correspondiente a ' +
    nombreProveedor + '.\n\nAnte cualquier consulta, quedamos a disposición.\n\nSaludos,\n' +
    'CURF — Laboratorio de Microbiología';

  return { orden, nombreProveedor, destinatario, pdf, asunto, cuerpo };
}

/** Crea un borrador de Gmail con el PDF adjunto, dirigido al proveedor. No lo envía. */
function revisarPedidoOC(numeroOrden) {
  const datos = prepararEnvioOrden(numeroOrden);
  GmailApp.createDraft(datos.destinatario, datos.asunto, datos.cuerpo, { attachments: [datos.pdf] });
  registrarLogEnvio(numeroOrden, datos.nombreProveedor, CONFIG_ENVIO.ACCION.BORRADOR, datos.destinatario);
  return { ok: true, destinatario: datos.destinatario };
}

/** Envía el mail directo al proveedor con el PDF adjunto. Sin vuelta atrás. */
function enviarPedidoOC(numeroOrden) {
  const datos = prepararEnvioOrden(numeroOrden);
  GmailApp.sendEmail(datos.destinatario, datos.asunto, datos.cuerpo, { attachments: [datos.pdf] });
  registrarLogEnvio(numeroOrden, datos.nombreProveedor, CONFIG_ENVIO.ACCION.ENVIADO, datos.destinatario);
  return { ok: true, destinatario: datos.destinatario };
}

/** Devuelve todas las devoluciones con estado = Pendiente devolución. */
function getDevolucionesPendientes() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.SHEETS.DEVOLUCIONES);
  if (!sheet || sheet.getLastRow() < 2) return [];

  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const idxEstado = headers.indexOf('estado');

  return data.slice(1)
    .map(function(row, i) {
      const obj = {};
      headers.forEach(function(h, j) { obj[h] = row[j]; });
      obj.fila = i + 2; // 1-indexed, +1 por encabezado
      return obj;
    })
    .filter(function(d) { return d.estado === CONFIG_DEVOLUCIONES.ESTADOS.PENDIENTE; });
}

/** Devuelve todas las devoluciones (pendientes y resueltas) para el diálogo. */
function getTodasLasDevoluciones() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.SHEETS.DEVOLUCIONES);
  if (!sheet || sheet.getLastRow() < 2) return [];

  const data = sheet.getDataRange().getValues();
  const headers = data[0];

  return data.slice(1).map(function(row, i) {
    const obj = {};
    headers.forEach(function(h, j) { obj[h] = row[j]; });
    obj.fila = i + 2;
    obj.fecha = formatearFecha(obj.fecha);
    obj.fecha_resolucion = obj.fecha_resolucion ? formatearFecha(obj.fecha_resolucion) : '';
    return obj;
  }).sort(function(a, b) {
    // Pendientes primero, después por fecha descendente
    if (a.estado === CONFIG_DEVOLUCIONES.ESTADOS.PENDIENTE && b.estado !== CONFIG_DEVOLUCIONES.ESTADOS.PENDIENTE) return -1;
    if (a.estado !== CONFIG_DEVOLUCIONES.ESTADOS.PENDIENTE && b.estado === CONFIG_DEVOLUCIONES.ESTADOS.PENDIENTE) return 1;
    return b.fila - a.fila;
  });
}

/** Marca una devolución como resuelta (Devuelto / Reemplazado / Nota de crédito). */
function marcarDevolucionResuelta(fila, nuevoEstado, observaciones) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.SHEETS.DEVOLUCIONES);
  if (!sheet) throw new Error('No existe la hoja Devoluciones_Proveedor.');

  const estadosValidos = [
    CONFIG_DEVOLUCIONES.ESTADOS.DEVUELTO,
    CONFIG_DEVOLUCIONES.ESTADOS.REEMPLAZADO,
    CONFIG_DEVOLUCIONES.ESTADOS.NOTA_CREDITO
  ];
  if (estadosValidos.indexOf(nuevoEstado) === -1) {
    throw new Error('Estado inválido: ' + nuevoEstado);
  }

  // Columnas: J=estado(10), K=fecha_resolucion(11), L=observaciones_resolucion(12)
  sheet.getRange(fila, 10, 1, 3).setValues([[nuevoEstado, new Date(), observaciones || '']]);
  return { ok: true };
}
