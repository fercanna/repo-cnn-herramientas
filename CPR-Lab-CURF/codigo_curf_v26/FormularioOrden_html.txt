  <script>
    let insumosProveedor    = [];
    let todosLosInsumos     = [];
    let todosLosProveedores = [];
    let itemsAlt            = [];
    let contadorAlt         = 0;

    window.onload = function() {
      google.script.run
        .withSuccessHandler(function(data) {
          todosLosProveedores = data;
          const selectPpal = document.getElementById('proveedor');
          const selectAlt  = document.getElementById('proveedorAlt');
          data.forEach(prov => {
            const makeOpt = () => {
              const opt = document.createElement('option');
              opt.value = prov.codigo_proveedor;
              opt.textContent = prov.razon_social;
              return opt;
            };
            selectPpal.appendChild(makeOpt());
            selectAlt.appendChild(makeOpt());
          });
        })
        .getProveedoresActivos();

      google.script.run
        .withSuccessHandler(function(data) {
          todosLosInsumos = data;
          const dl = document.getElementById('datalistInsumos');
          data.forEach(ins => {
            const opt = document.createElement('option');
            opt.value = ins.descripcion;
            opt.dataset.codigo = ins.codigo_insumo;
            opt.dataset.precio = ins.precio_referencial || 0;
            dl.appendChild(opt);
          });
        })
        .getInsumosActivos();
    };

    function cargarInsumos() {
      const codProv = document.getElementById('proveedor').value;
      const container = document.getElementById('insumosContainer');
      const estado = document.getElementById('estadoCarga');

      container.innerHTML = '';
      insumosProveedor = [];
      document.getElementById('resumenSeleccion').style.display = 'none';
      document.getElementById('totalOrden').textContent = 'Total: $0.00';
      document.getElementById('btnToggleAlt').style.display = 'none';
      document.getElementById('seccionAlternativo').style.display = 'none';
      resetAlt();
      actualizarSelectAlt(codProv);

      if (!codProv) {
        estado.style.display = 'block';
        estado.textContent = '← Seleccioná un proveedor para ver sus insumos';
        return;
      }

      estado.style.display = 'block';
      estado.textContent = '⏳ Cargando insumos del proveedor...';

      google.script.run
        .withSuccessHandler(function(data) {
          insumosProveedor = data;
          estado.style.display = 'none';
          if (!data || data.length === 0) {
            estado.style.display = 'block';
            estado.textContent = 'Este proveedor no tiene insumos asignados en el catálogo.';
            return;
          }
          renderizarInsumos(data);
          document.getElementById('btnToggleAlt').style.display = 'block';
        })
        .withFailureHandler(function(err) {
          estado.style.display = 'block';
          estado.textContent = '❌ Error al cargar insumos: ' + err.message;
        })
        .getInsumosPorProveedor(codProv);
    }

    function actualizarSelectAlt(codPrincipal) {
      const selectAlt = document.getElementById('proveedorAlt');
      selectAlt.innerHTML = '<option value="">-- Seleccionar --</option>';
      todosLosProveedores.forEach(prov => {
        if (prov.codigo_proveedor === codPrincipal) return;
        const opt = document.createElement('option');
        opt.value = prov.codigo_proveedor;
        opt.textContent = prov.razon_social;
        selectAlt.appendChild(opt);
      });
    }

    function toggleAlternativo() {
      const sec = document.getElementById('seccionAlternativo');
      const btn = document.getElementById('btnToggleAlt');
      if (sec.style.display === 'none') {
        sec.style.display = 'block';
        btn.textContent = '✕ Cancelar proveedor alternativo';
        btn.style.borderColor = '#ea4335';
        btn.style.color = '#ea4335';
      } else {
        sec.style.display = 'none';
        btn.textContent = '➕ Agregar insumos de otro proveedor';
        btn.style.borderColor = '#4285f4';
        btn.style.color = '#4285f4';
        resetAlt();
        calcularTotal();
      }
    }

    function resetAlt() {
      itemsAlt = [];
      contadorAlt = 0;
      document.getElementById('itemsAltContainer').innerHTML = '';
      document.getElementById('altHeaders').style.display = 'none';
      document.getElementById('altBuscarInsumo').value = '';
      document.getElementById('altCodigoInsumo').value = '';
      document.getElementById('altCantidad').value = '1';
      document.getElementById('altPrecio').value = '';
      document.getElementById('proveedorAlt').value = '';
    }

    function autocompletarPrecioAlt() {
      const nombre = document.getElementById('altBuscarInsumo').value;
      const ins = todosLosInsumos.find(i => i.descripcion === nombre);
      if (ins) {
        document.getElementById('altCodigoInsumo').value = ins.codigo_insumo;
        if (ins.precio_referencial > 0) {
          document.getElementById('altPrecio').value = ins.precio_referencial.toFixed(2);
        }
      } else {
        document.getElementById('altCodigoInsumo').value = '';
      }
    }

    function agregarItemAlt() {
      const provAlt  = document.getElementById('proveedorAlt').value;
      const nombre   = document.getElementById('altBuscarInsumo').value.trim();
      const codigo   = document.getElementById('altCodigoInsumo').value.trim();
      const cant     = parseFloat(document.getElementById('altCantidad').value) || 0;
      const precio   = parseFloat(document.getElementById('altPrecio').value)   || 0;

      if (!provAlt)  return alert('Seleccioná el proveedor alternativo.');
      if (!codigo)   return alert('Seleccioná un insumo del listado.');
      if (cant <= 0) return alert('La cantidad debe ser mayor a 0.');

      const yaExiste = itemsAlt.find(i => i.codigo_insumo === codigo && i.proveedor === provAlt);
      if (yaExiste) return alert('Ese insumo ya fue agregado para este proveedor.');

      const subtotal = cant * precio;
      contadorAlt++;
      const id = contadorAlt;
      itemsAlt.push({ id, proveedor: provAlt, codigo_insumo: codigo, descripcion: nombre, cantidad: cant, precio_unitario: precio, subtotal });

      document.getElementById('altHeaders').style.display = 'grid';

      const row = document.createElement('div');
      row.className = 'item-alt';
      row.id = `itemAlt_${id}`;
      row.innerHTML = `
        <div>
          <div class="item-alt-nombre">${nombre}</div>
          <div class="item-alt-codigo">${codigo} · <em>${getNombreProveedor(provAlt)}</em></div>
        </div>
        <input type="number" min="1" value="${cant}" onchange="actualizarItemAlt(${id}, this)" style="text-align:center;">
        <input type="number" min="0" step="0.01" value="${precio.toFixed(2)}" onchange="actualizarPrecioAlt(${id}, this)" style="text-align:right;">
        <div style="text-align:right; font-weight:bold; color:#1F4E79;" id="subAlt_${id}">$${subtotal.toFixed(2)}</div>
        <button type="button" class="btn-quitar-alt" onclick="quitarItemAlt(${id})">✕</button>
      `;
      document.getElementById('itemsAltContainer').appendChild(row);

      document.getElementById('altBuscarInsumo').value = '';
      document.getElementById('altCodigoInsumo').value = '';
      document.getElementById('altCantidad').value = '1';
      document.getElementById('altPrecio').value = '';
      document.getElementById('altBuscarInsumo').focus();
      calcularTotal();
    }

    function getNombreProveedor(cod) {
      const p = todosLosProveedores.find(p => p.codigo_proveedor === cod);
      return p ? p.razon_social : cod;
    }

    function actualizarItemAlt(id, input) {
      const item = itemsAlt.find(i => i.id === id);
      if (!item) return;
      item.cantidad = parseFloat(input.value) || 0;
      item.subtotal = item.cantidad * item.precio_unitario;
      document.getElementById(`subAlt_${id}`).textContent = '$' + item.subtotal.toFixed(2);
      calcularTotal();
    }

    function actualizarPrecioAlt(id, input) {
      const item = itemsAlt.find(i => i.id === id);
      if (!item) return;
      item.precio_unitario = parseFloat(input.value) || 0;
      item.subtotal = item.cantidad * item.precio_unitario;
      document.getElementById(`subAlt_${id}`).textContent = '$' + item.subtotal.toFixed(2);
      calcularTotal();
    }

    function quitarItemAlt(id) {
      itemsAlt = itemsAlt.filter(i => i.id !== id);
      document.getElementById(`itemAlt_${id}`).remove();
      if (itemsAlt.length === 0) document.getElementById('altHeaders').style.display = 'none';
      calcularTotal();
    }

    function renderizarInsumos(insumos) {
      const container = document.getElementById('insumosContainer');
      container.innerHTML = '';

      const grupos = {};
      insumos.forEach(ins => {
        const cat = ins.categoria || 'Sin categoría';
        if (!grupos[cat]) grupos[cat] = [];
        grupos[cat].push(ins);
      });

      const hdrs = document.createElement('div');
      hdrs.className = 'col-headers';
      hdrs.innerHTML = `<span></span><span>Insumo</span><span style="text-align:center">Stock</span><span style="text-align:center">Cantidad</span><span style="text-align:right">Precio unit.</span>`;
      container.appendChild(hdrs);

      Object.keys(grupos).sort().forEach(cat => {
        const catDiv = document.createElement('div');
        catDiv.className = 'categoria-header';
        catDiv.textContent = `📂 ${cat} (${grupos[cat].length})`;
        container.appendChild(catDiv);

        grupos[cat].forEach(ins => {
          const row = document.createElement('div');
          row.className = 'insumo-row' +
            (ins.estado_stock === 'Crítico' ? ' critico' : '') +
            (ins.estado_stock === 'Agotado' ? ' agotado' : '');
          row.dataset.codigo = ins.codigo_insumo;

          const badgeClass = ins.estado_stock === 'OK' ? 'badge-ok' :
                             ins.estado_stock === 'Crítico' ? 'badge-critico' : 'badge-agotado';
          const badgeText  = ins.estado_stock === 'OK' ? `✓ ${ins.stock_actual}` :
                             ins.estado_stock === 'Crítico' ? `⚠ ${ins.stock_actual}` : `✗ 0`;
          const precio = ins.precio_referencial > 0 ? ins.precio_referencial.toFixed(2) : '0.00';

          row.innerHTML = `
            <input type="checkbox" onchange="toggleInsumo(this)">
            <div>
              <div class="insumo-nombre">${ins.descripcion}</div>
              <div class="insumo-codigo">${ins.codigo_insumo} · ${ins.unidad_medida}</div>
            </div>
            <div style="text-align:center"><span class="badge ${badgeClass}">${badgeText}</span></div>
            <input type="number" class="item-cantidad" min="1" value="1" disabled onchange="calcularTotal()">
            <input type="number" class="item-precio" min="0" step="0.01" value="${precio}" disabled onchange="calcularTotal()">
          `;
          container.appendChild(row);
        });
      });
    }

    function toggleInsumo(checkbox) {
      const row = checkbox.closest('.insumo-row');
      const cantInput   = row.querySelector('.item-cantidad');
      const precioInput = row.querySelector('.item-precio');
      if (checkbox.checked) {
        row.classList.add('seleccionado');
        cantInput.disabled = false;
        precioInput.disabled = false;
        cantInput.focus();
      } else {
        row.classList.remove('seleccionado');
        cantInput.disabled = true;
        precioInput.disabled = true;
      }
      actualizarResumen();
      calcularTotal();
    }

    function actualizarResumen() {
      const sel = document.querySelectorAll('#insumosContainer .insumo-row.seleccionado');
      const div = document.getElementById('resumenSeleccion');
      if (sel.length === 0) { div.style.display = 'none'; }
      else {
        div.style.display = 'block';
        div.textContent = `✔ ${sel.length} insumo${sel.length > 1 ? 's' : ''} seleccionado${sel.length > 1 ? 's' : ''}`;
      }
    }

    function calcularTotal() {
      let total = 0;
      document.querySelectorAll('#insumosContainer .insumo-row.seleccionado').forEach(row => {
        const cant   = parseFloat(row.querySelector('.item-cantidad').value)  || 0;
        const precio = parseFloat(row.querySelector('.item-precio').value) || 0;
        total += cant * precio;
      });
      itemsAlt.forEach(i => { total += i.subtotal; });
      document.getElementById('totalOrden').textContent = `Total: $${total.toFixed(2)}`;
    }

    document.getElementById('ordenForm').addEventListener('submit', function(e) {
      e.preventDefault();

      const proveedorPpal = document.getElementById('proveedor').value;
      if (!proveedorPpal) return mostrarMensaje('Debe seleccionar un proveedor.', 'error');

      const solicitadoPor = document.getElementById('solicitadoPor').value;
      if (!solicitadoPor) return mostrarMensaje('Debe seleccionar quién solicita la orden.', 'error');

      const fechaEntrega  = document.getElementById('fechaEntrega').value;
      const observaciones = document.getElementById('observaciones').value;

      const itemsPpal = [];
      let idxPpal = 1;
      document.querySelectorAll('#insumosContainer .insumo-row.seleccionado').forEach(row => {
        const cod    = row.dataset.codigo;
        const ins    = insumosProveedor.find(i => i.codigo_insumo === cod);
        const cant   = parseFloat(row.querySelector('.item-cantidad').value) || 0;
        const precio = parseFloat(row.querySelector('.item-precio').value)   || 0;
        if (!ins || cant <= 0) return;
        itemsPpal.push({ item: idxPpal++, codigo_insumo: cod, descripcion: ins.descripcion, cantidad: cant, precio_unitario: precio, subtotal: cant * precio });
      });

      const itemsAltPorProv = {};
      itemsAlt.forEach(i => {
        if (!itemsAltPorProv[i.proveedor]) itemsAltPorProv[i.proveedor] = [];
        itemsAltPorProv[i.proveedor].push(i);
      });

      if (itemsPpal.length === 0 && itemsAlt.length === 0) {
        return mostrarMensaje('Debe seleccionar al menos un insumo.', 'error');
      }

      const colaOrdenes = [];
      if (itemsPpal.length > 0) {
        colaOrdenes.push({ proveedor: proveedorPpal, items: itemsPpal });
      }
      Object.keys(itemsAltPorProv).forEach(codProv => {
        const loteItems = itemsAltPorProv[codProv].map((i, idx) => ({
          item: idx + 1,
          codigo_insumo: i.codigo_insumo,
          descripcion: i.descripcion,
          cantidad: i.cantidad,
          precio_unitario: i.precio_unitario,
          subtotal: i.subtotal
        }));
        colaOrdenes.push({ proveedor: codProv, items: loteItems });
      });

      document.getElementById('ordenForm').style.display = 'none';
      document.getElementById('loading').style.display   = 'block';

      const ordenesCreadas = [];

      function crearSiguiente(posicion) {
        if (posicion >= colaOrdenes.length) {
          document.getElementById('loading').style.display = 'none';
          const msg = ordenesCreadas.length === 1
            ? `✅ Orden ${ordenesCreadas[0]} creada exitosamente`
            : `✅ ${ordenesCreadas.length} órdenes creadas: ${ordenesCreadas.join(', ')}`;
          mostrarMensaje(msg, 'success');
          setTimeout(() => google.script.host.close(), 2500);
          return;
        }

        const lote = colaOrdenes[posicion];
        const ordenData = {
          proveedor:              lote.proveedor,
          fecha_entrega_estimada: fechaEntrega,
          observaciones:          observaciones,
          solicitado_por:         solicitadoPor,
          items:                  lote.items
        };

        google.script.run
          .withSuccessHandler(function(numeroOrden) {
            ordenesCreadas.push(numeroOrden);
            crearSiguiente(posicion + 1);
          })
          .withFailureHandler(function(err) {
            document.getElementById('loading').style.display = 'none';
            document.getElementById('ordenForm').style.display = 'block';
            const yaCreadas = ordenesCreadas.length > 0
              ? ` (ya se crearon: ${ordenesCreadas.join(', ')})`
              : '';
            mostrarMensaje(`❌ Error al crear orden para proveedor ${lote.proveedor}: ${err.message}${yaCreadas}`, 'error');
          })
          .crearOrdenCompra(ordenData);
      }

      crearSiguiente(0);
    });

    function mostrarMensaje(texto, tipo) {
      const msgDiv = document.getElementById('message');
      msgDiv.textContent = texto;
      msgDiv.className = `message ${tipo}`;
      msgDiv.style.display = 'block';
      msgDiv.scrollIntoView({ behavior: 'smooth' });
    }
  </script>
