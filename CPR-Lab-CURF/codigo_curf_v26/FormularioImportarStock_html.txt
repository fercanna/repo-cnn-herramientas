<script>
  let parsedRows = [];
  let validRows  = [];
  let errorRows  = [];

  // ── Drag & drop ──────────────────────────────────────────────
  const dropZone = document.getElementById('dropZone');
  const fileInput = document.getElementById('fileInput');

  dropZone.addEventListener('dragover', e => { e.preventDefault(); dropZone.classList.add('dragover'); });
  dropZone.addEventListener('dragleave', () => dropZone.classList.remove('dragover'));
  dropZone.addEventListener('drop', e => {
    e.preventDefault();
    dropZone.classList.remove('dragover');
    const file = e.dataTransfer.files[0];
    if (file) processFile(file);
  });
  fileInput.addEventListener('change', e => {
    if (e.target.files[0]) processFile(e.target.files[0]);
  });

  // ── Parse Excel ───────────────────────────────────────────────
  function processFile(file) {
    if (!file.name.match(/\.xlsx?$/i)) {
      alert('Por favor seleccioná un archivo Excel (.xlsx)');
      return;
    }

    const reader = new FileReader();
    reader.onload = e => {
      const data = new Uint8Array(e.target.result);
      const wb   = XLSX.read(data, { type: 'array', cellDates: true });
      const ws   = wb.Sheets[wb.SheetNames[0]];
      const json = XLSX.utils.sheet_to_json(ws, { header: 1, defval: '' });

      parseRows(json);
      renderPreview();
      setStep(2);
    };
    reader.readAsArrayBuffer(file);
  }

  function parseRows(raw) {
    parsedRows = [];
    validRows  = [];
    errorRows  = [];

    // Buscar fila de encabezado (primera que tenga "codigo" en col A)
    let dataStart = 1;
    for (let i = 0; i < Math.min(raw.length, 5); i++) {
      const cell = String(raw[i][0] || '').toLowerCase();
      if (cell.includes('codigo') || cell.includes('código')) {
        dataStart = i + 1;
        break;
      }
    }

    for (let i = dataStart; i < raw.length; i++) {
      const row = raw[i];
      const codigo  = String(row[0] || '').trim();
      const desc    = String(row[1] || '').trim();
      const cantRaw = row[2];
      const lote    = String(row[3] || 'SIN LOTE').trim() || 'SIN LOTE';
      const vencRaw = row[4];
      const obs     = String(row[5] || '').trim();

      // Saltar filas vacías
      if (!codigo && !desc && !cantRaw) continue;

      const errors = [];

      if (!codigo) errors.push('Código vacío');

      const cantidad = parseFloat(cantRaw);
      if (isNaN(cantidad) || cantidad <= 0) errors.push('Cantidad inválida');

      // Vencimiento: puede ser Date (xlsx parseó), string, o vacío
      let vencimiento = '';
      if (vencRaw) {
        if (vencRaw instanceof Date) {
          vencimiento = formatDate(vencRaw);
        } else {
          const s = String(vencRaw).trim();
          // Aceptar DD/MM/AAAA o DD/MM/AA
          if (/^\d{1,2}\/\d{1,2}\/\d{2,4}$/.test(s)) {
            vencimiento = s;
          } else {
            errors.push('Formato de vencimiento inválido (usar DD/MM/AAAA)');
          }
        }
      }
      // Vencimiento vacío es válido (insumos sin fecha)

      const entry = { rowNum: i + 1, codigo, desc, cantidad: isNaN(cantidad) ? 0 : cantidad, lote, vencimiento, obs, errors };
      parsedRows.push(entry);

      if (errors.length === 0) validRows.push(entry);
      else errorRows.push(entry);
    }
  }

  function formatDate(d) {
    const dd = String(d.getDate()).padStart(2,'0');
    const mm = String(d.getMonth()+1).padStart(2,'0');
    const yy = d.getFullYear();
    return `${dd}/${mm}/${yy}`;
  }

  // ── Render preview ────────────────────────────────────────────
  function renderPreview() {
    // Stats
    const sb = document.getElementById('statsBar');
    sb.innerHTML = `
      <div class="stat total"><div class="num">${parsedRows.length}</div><div class="lbl">Total filas</div></div>
      <div class="stat ok">  <div class="num">${validRows.length}</div><div class="lbl">Válidas ✅</div></div>
      <div class="stat err"> <div class="num">${errorRows.length}</div><div class="lbl">Con errores ❌</div></div>
    `;

    // Table
    const tbody = document.getElementById('previewBody');
    tbody.innerHTML = '';
    parsedRows.forEach(r => {
      const isOk = r.errors.length === 0;
      const badge = isOk
        ? '<span class="badge badge-ok">✅ OK</span>'
        : `<span class="badge badge-err">❌ ${r.errors[0]}</span>`;
      tbody.innerHTML += `
        <tr>
          <td>${r.rowNum}</td>
          <td><strong>${r.codigo || '—'}</strong></td>
          <td>${r.desc || '—'}</td>
          <td>${r.cantidad || '—'}</td>
          <td>${r.lote}</td>
          <td>${r.vencimiento || 'Sin fecha'}</td>
          <td>${r.obs || '—'}</td>
          <td>${badge}</td>
        </tr>`;
    });

    // Errors detail
    if (errorRows.length > 0) {
      document.getElementById('errorsDetail').style.display = 'block';
      const el = document.getElementById('errorList');
      el.innerHTML = errorRows.map(r =>
        `<div class="error-item">Fila ${r.rowNum}: <strong>${r.codigo || 'sin código'}</strong> — ${r.errors.join(', ')}</div>`
      ).join('');
    }

    // Disable import if no valid rows
    document.getElementById('btnImport').disabled = validRows.length === 0;

    document.getElementById('cardUpload').style.display   = 'none';
    document.getElementById('cardPreview').style.display  = 'block';
  }

  // ── Confirm & send ────────────────────────────────────────────
  function confirmarImportacion() {
    if (validRows.length === 0) return;

    const msg = errorRows.length > 0
      ? `Se importarán ${validRows.length} filas válidas. Las ${errorRows.length} filas con errores serán saltadas. ¿Confirmás?`
      : `Se importarán ${validRows.length} filas. ¿Confirmás?`;

    if (!confirm(msg)) return;

    setStep(3);
    document.getElementById('cardPreview').style.display = 'none';
    document.getElementById('cardResult').style.display  = 'block';
    animateProgress();

    // Serializar para GAS (solo filas válidas)
    const payload = validRows.map(r => ({
      codigo:      r.codigo,
      cantidad:    r.cantidad,
      lote:        r.lote,
      vencimiento: r.vencimiento,
      obs:         r.obs
    }));

    google.script.run
      .withSuccessHandler(onImportSuccess)
      .withFailureHandler(onImportError)
      .importarStockInicial(payload);
  }

  // Fake progress animation while waiting
  let progressInterval;
  function animateProgress() {
    let pct = 0;
    const bar = document.getElementById('progressBar');
    const lbl = document.getElementById('progressLabel');
    progressInterval = setInterval(() => {
      if (pct < 85) {
        pct += Math.random() * 8;
        bar.style.width = Math.min(pct, 85) + '%';
        const msgs = ['Validando contra catálogo...','Escribiendo en Movimientos_Inventario...','Actualizando Stock_Actual...','Finalizando...'];
        lbl.textContent = msgs[Math.floor(pct / 25)] || msgs[3];
      }
    }, 400);
  }

  function onImportSuccess(result) {
    clearInterval(progressInterval);
    document.getElementById('progressBar').style.width = '100%';
    document.getElementById('progressLabel').textContent = '¡Completado!';

    const box = document.getElementById('resultBox');
    box.style.display = 'block';

    if (result.ok) {
      box.className = 'result-box success';
      let html = `<strong>✅ Importación completada</strong><br><br>
        📦 Filas importadas: <strong>${result.importadas}</strong><br>
        ❌ Saltadas (código no en catálogo): <strong>${result.saltadas}</strong><br>
        ⏱️ Tiempo: <strong>${result.tiempo}s</strong>`;

      if (result.errores && result.errores.length > 0) {
        html += `<br><br><strong>Códigos no encontrados en catálogo:</strong>
          <div class="error-list">
            ${result.errores.map(e => `<div class="error-item">• ${e}</div>`).join('')}
          </div>`;
      }
      html += `<br><br>✅ <em>Stock_Actual fue actualizado automáticamente.</em>`;
      box.innerHTML = html;
      setStep(3, true);
    } else {
      box.className = 'result-box error';
      box.innerHTML = `<strong>❌ Error durante la importación</strong><br>${result.mensaje}`;
    }
  }

  function onImportError(err) {
    clearInterval(progressInterval);
    const box = document.getElementById('resultBox');
    box.style.display = 'block';
    box.className = 'result-box error';
    box.innerHTML = `<strong>❌ Error de comunicación</strong><br>${err.message}`;
  }

  // ── Helpers ───────────────────────────────────────────────────
  function setStep(n, done) {
    for (let i = 1; i <= 3; i++) {
      const el = document.getElementById('step' + i);
      el.className = 'step' + (i < n || (i === n && done) ? ' done' : i === n ? ' active' : '');
    }
  }

  function resetForm() {
    parsedRows = []; validRows = []; errorRows = [];
    document.getElementById('fileInput').value = '';
    document.getElementById('cardUpload').style.display  = 'block';
    document.getElementById('cardPreview').style.display = 'none';
    document.getElementById('cardResult').style.display  = 'none';
    document.getElementById('previewBody').innerHTML = '';
    document.getElementById('errorList').innerHTML   = '';
    document.getElementById('errorsDetail').style.display = 'none';
    setStep(1);
  }
</script>
