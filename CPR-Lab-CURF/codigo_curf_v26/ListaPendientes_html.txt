  <script>
    let ordenesGlobal = [];
    const hoy = new Date();
    hoy.setHours(0,0,0,0);

    window.onload = function() {
      google.script.run
        .withSuccessHandler(dibujarTabla)
        .withFailureHandler(mostrarError)
        .getOrdenesPendientes();
    };

    // ✅ NUEVO: Analiza si una fecha de entrega está vencida o próxima
    function estadoFecha(fechaStr) {
      if (!fechaStr || fechaStr === '-') return 'normal';
      // fechaStr viene como dd/MM/yyyy
      const partes = fechaStr.split('/');
      if (partes.length !== 3) return 'normal';
      const fecha = new Date(partes[2], partes[1]-1, partes[0]);
      const diffDias = Math.floor((fecha - hoy) / (1000*60*60*24));
      if (diffDias < 0) return 'vencida';
      if (diffDias <= 3) return 'proxima';
      return 'normal';
    }

    function dibujarTabla(ordenes) {
      document.getElementById('loader').style.display = 'none';
      const container = document.getElementById('tablaContainer');
      ordenesGlobal = ordenes;

      if (!ordenes || ordenes.length === 0) {
        container.innerHTML = '<div class="empty-state">✅ <strong>Todo al día</strong><br>No hay órdenes pendientes de recepción.</div>';
        return;
      }

      document.getElementById('toolbar').style.display = 'flex';
      document.getElementById('filtros').style.display = 'flex';

      // ✅ Barra de alertas de demora
      const vencidas = ordenes.filter(o => estadoFecha(o.fecha_entrega_estimada) === 'vencida');
      if (vencidas.length > 0) {
        const bar = document.getElementById('alertaBar');
        bar.style.display = 'block';
        bar.innerHTML = `🔴 <strong>${vencidas.length} orden(es) con fecha de entrega vencida</strong>: ${vencidas.map(o => o.numero_orden).join(', ')} — Considerá hacer seguimiento con el proveedor.`;
      }

      renderTabla(ordenes);
    }

    function renderTabla(ordenes) {
      const container = document.getElementById('tablaContainer');
      
      // ✅ NUEVO: columna Entrega Estimada
      let html = '<table id="tablaDatos"><thead><tr>';
      html += '<th>Orden</th><th>Proveedor</th><th>Emitida</th><th>Entrega Est.</th><th>Estado</th><th>Progreso</th><th>Items Pendientes</th>';
      html += '</tr></thead><tbody>';
      
      ordenes.forEach((orden, idx) => {
        const cssClass = orden.estado === 'Recepción Parcial' ? 'status-parcial' : 'status-pendiente';
        const total = orden.total_orden ? '$' + parseFloat(orden.total_orden).toFixed(2) : '-';
        const cantPendientes = orden.items_pendientes ? orden.items_pendientes.length : 0;
        
        // ✅ Badge de progreso
        let progresoBadge = '';
        if (orden.estado === 'Recepción Parcial') {
          const totalItems = (orden.items_pendientes || []).length; // pendientes restantes
          progresoBadge = `<span class="progreso-badge progreso-parcial">⚠️ parcial</span>`;
        } else {
          progresoBadge = `<span class="progreso-badge progreso-sin">nuevo</span>`;
        }

        // ✅ Fecha de entrega con color
        const efEstado = estadoFecha(orden.fecha_entrega_estimada);
        const fechaEntrega = orden.fecha_entrega_estimada || '—';
        const fechaClass = 'fecha-' + efEstado;

        html += `<tr data-estado="${orden.estado}" data-fecha-estado="${efEstado}">
          <td><strong>${orden.numero_orden}</strong></td>
          <td>${orden.proveedor}</td>
          <td>${orden.fecha_emision}</td>
          <td><span class="${fechaClass}">${fechaEntrega}</span></td>
          <td><span class="status-badge ${cssClass}">${orden.estado}</span></td>
          <td>${progresoBadge}</td>
          <td>
            <button class="btn-expand" onclick="toggleItems(${idx})">
              ${cantPendientes} producto(s) ▼
            </button>
            <div class="items-detalle" id="items-${idx}">
              ${dibujarItemsPendientes(orden.items_pendientes)}
            </div>
          </td>
        </tr>`;
      });

      html += '</tbody></table>';
      container.innerHTML = html;
      actualizarContador(ordenes.length);
    }

    function dibujarItemsPendientes(items) {
      if (!items || items.length === 0) return '<em>Sin items pendientes</em>';
      
      let html = '<div style="font-weight: bold; margin-bottom: 5px; color: #202124;">Detalle pendiente:</div>';
      items.forEach(item => {
        const pct = item.solicitado > 0 ? Math.round((item.recibido / item.solicitado) * 100) : 0;
        html += `<div class="item-pendiente">
          <strong>${item.descripcion}</strong> (${item.codigo})<br>
          <span style="color: #5f6368;">
            Pedido: ${item.solicitado} | Recibido: <span style="color:#1e7e34;font-weight:bold;">${item.recibido}</span> (${pct}%) | 
            <strong style="color: #d93025;">Pendiente: ${item.pendiente}</strong>
          </span>
        </div>`;
      });
      return html;
    }

    // ✅ NUEVO: Filtros aplicados sin recargar del servidor
    function aplicarFiltros() {
      const filtroEstado = document.getElementById('filtroEstado').value;
      const filtroAlerta = document.getElementById('filtroAlerta').value;
      
      let filtradas = ordenesGlobal;
      
      if (filtroEstado) {
        filtradas = filtradas.filter(o => o.estado === filtroEstado);
      }
      if (filtroAlerta) {
        filtradas = filtradas.filter(o => estadoFecha(o.fecha_entrega_estimada) === filtroAlerta);
      }
      
      renderTabla(filtradas);
    }

    function actualizarContador(n) {
      document.getElementById('contadorFiltro').textContent = n + ' orden(es) mostradas';
    }

    function toggleItems(idx) {
      const div = document.getElementById(`items-${idx}`);
      div.classList.toggle('visible');
    }

    function copiarTabla() {
      document.querySelectorAll('.items-detalle').forEach(div => div.classList.add('visible'));
      const tabla = document.getElementById('tablaDatos');
      if (!tabla) return;
      const range = document.createRange();
      range.selectNode(tabla);
      window.getSelection().removeAllRanges();
      window.getSelection().addRange(range);
      try {
        document.execCommand('copy');
        window.getSelection().removeAllRanges();
        mostrarNotificacion();
      } catch (err) {
        alert('No se pudo copiar automáticamente. Seleccioná la tabla y presioná Ctrl+C');
      }
    }

    function mostrarNotificacion() {
      const msg = document.getElementById('mensajeCopia');
      msg.style.opacity = '1';
      setTimeout(() => { msg.style.opacity = '0'; }, 2000);
    }

    function mostrarError(e) {
      document.getElementById('loader').innerHTML = '<div style="color:#d93025">❌ Error: ' + e.message + '</div>';
    }
