  <script>
    let recepciones = [];

    window.onload = function() {
      google.script.run
        .withSuccessHandler(function(data) {
          recepciones = data;
          const sel = document.getElementById('recSelect');
          if (!data || data.length === 0) {
            sel.innerHTML = '<option value="">— Sin recepciones activas —</option>';
            return;
          }
          sel.innerHTML = '<option value="">— Seleccioná una recepción —</option>' +
            data.map(function(rec) {
              return '<option value="' + esc(rec.numero_recepcion) + '">' +
                esc(rec.numero_recepcion) + ' — ' + esc(rec.proveedor) +
                ' (' + esc(rec.fecha) + ')' +
                '</option>';
            }).join('');
        })
        .withFailureHandler(function(e) { mostrarMensaje('Error al cargar recepciones: ' + e.message, 'error'); })
        .getRecepcionesActivas();
    };

    function onRecepcionChange() {
      const num = document.getElementById('recSelect').value;
      const rec = recepciones.find(function(r) { return r.numero_recepcion === num; });

      const detalleEl = document.getElementById('detalleRecepcion');
      const btnAnular = document.getElementById('btnAnular');

      if (!rec) {
        detalleEl.style.display = 'none';
        btnAnular.disabled = true;
        return;
      }

      document.getElementById('dOrden').textContent = rec.numero_orden;
      document.getElementById('dProv').textContent  = rec.proveedor;
      document.getElementById('dFecha').textContent = rec.fecha;
      document.getElementById('dRec').textContent   = rec.recibido_por || '—';

      const itemsEl = document.getElementById('itemsRecepcion');
      if (rec.items && rec.items.length > 0) {
        itemsEl.innerHTML = rec.items.map(function(it) {
          return '<div class="items-list-row">' +
            '<div><div>' + esc(it.descripcion) + '</div><div class="cod">' + esc(it.codigo_insumo) + '</div></div>' +
            '<div class="cant" style="text-align:center">' + it.cantidad_recibida + '</div>' +
            '<div class="lote">' + esc(it.lote) + '</div>' +
            '<div class="lote">' + (it.vencimiento || '—') + '</div>' +
            '</div>';
        }).join('');
      } else {
        itemsEl.innerHTML = '<div style="padding:12px; color:#888;">Sin ítems</div>';
      }

      detalleEl.style.display = 'block';
      btnAnular.disabled = false;
    }

    function confirmarAnulacion() {
      const num    = document.getElementById('recSelect').value;
      const motivo = document.getElementById('motivo').value.trim();

      if (!num)    return mostrarMensaje('Seleccioná una recepción.', 'error');
      if (!motivo) return mostrarMensaje('El motivo es obligatorio.', 'error');

      const rec = recepciones.find(function(r) { return r.numero_recepcion === num; });
      const confirmMsg = '¿Anular recepción ' + num + '?\n\nSe revertirán ' +
        (rec ? rec.items.length : '?') + ' ítem(s) del stock.\nEsta acción queda registrada en los movimientos.';
      if (!confirm(confirmMsg)) return;

      document.getElementById('formContent').style.display = 'none';
      document.getElementById('loading').style.display     = 'block';

      google.script.run
        .withSuccessHandler(function(result) {
          document.getElementById('loading').style.display = 'none';
          if (result.ok) {
            mostrarMensaje('✅ ' + result.mensaje, 'success');
            setTimeout(function() { google.script.host.close(); }, 3000);
          } else {
            document.getElementById('formContent').style.display = 'block';
            mostrarMensaje('❌ ' + result.mensaje, 'error');
          }
        })
        .withFailureHandler(function(e) {
          document.getElementById('loading').style.display     = 'none';
          document.getElementById('formContent').style.display = 'block';
          mostrarMensaje('❌ Error: ' + e.message, 'error');
        })
        .cancelarRecepcion(num, motivo);
    }

    function mostrarMensaje(texto, tipo) {
      const el = document.getElementById('message');
      el.textContent = texto;
      el.className = 'message ' + tipo;
      el.style.display = 'block';
    }

    function esc(str) {
      return String(str || '').replace(/&/g,'&amp;').replace(/"/g,'&quot;')
        .replace(/</g,'&lt;').replace(/>/g,'&gt;');
    }
