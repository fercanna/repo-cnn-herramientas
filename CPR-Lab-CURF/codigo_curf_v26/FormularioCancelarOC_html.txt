      <button class="btn btn-danger" id="btnCancelar" onclick="confirmarCancelacion()" disabled>
        🗑️ Cancelar esta OC

  <script>
    let ocData = [];

    window.onload = function() {
      google.script.run
        .withSuccessHandler(function(ocs) {
          ocData = ocs;
          const sel = document.getElementById('ocSelect');
          if (!ocs || ocs.length === 0) {
            sel.innerHTML = '<option value="">— Sin OC pendientes —</option>';
            return;
          }
          sel.innerHTML = '<option value="">— Seleccioná una OC —</option>' +
            ocs.map(function(oc) {
              return '<option value="' + esc(oc.numero_orden) + '">' +
                esc(oc.numero_orden) + ' — ' + esc(oc.proveedor) +
                ' ($' + parseFloat(oc.total_orden || 0).toFixed(2) + ')' +
                '</option>';
            }).join('');
        })
        .withFailureHandler(function(e) {
          mostrarMensaje('Error al cargar OC: ' + e.message, 'error');
        })
        .getOCsEditables();
    };

    function onOCChange() {
      const num = document.getElementById('ocSelect').value;
      const oc  = ocData.find(function(o) { return o.numero_orden === num; });

      const detalle    = document.getElementById('detalleOC');
      const advert     = document.getElementById('advertencia');
      const btnCancelar = document.getElementById('btnCancelar');

      if (!oc) {
        detalle.style.display   = 'none';
        advert.style.display    = 'none';
        btnCancelar.disabled    = true;
        return;
      }

      document.getElementById('dProv').textContent  = oc.proveedor;
      document.getElementById('dFecha').textContent = oc.fecha_emision;
      document.getElementById('dTotal').textContent = '$' + parseFloat(oc.total_orden || 0).toFixed(2);
      document.getElementById('dSol').textContent   = oc.solicitado_por || '—';

      detalle.style.display   = 'block';
      advert.style.display    = 'block';
      btnCancelar.disabled    = false;
    }

    function confirmarCancelacion() {
      const num    = document.getElementById('ocSelect').value;
      const motivo = document.getElementById('motivo').value.trim();

      if (!num)    return mostrarMensaje('Seleccioná una OC.', 'error');
      if (!motivo) return mostrarMensaje('El motivo es obligatorio.', 'error');

      const oc = ocData.find(function(o) { return o.numero_orden === num; });
      const confirmMsg = '¿Confirmar cancelación de ' + num + ' (' + (oc ? oc.proveedor : '') + ')?\n\nEsta acción NO puede deshacerse.';
      if (!confirm(confirmMsg)) return;

      document.getElementById('formContent').style.display = 'none';
      document.getElementById('loading').style.display     = 'block';

      google.script.run
        .withSuccessHandler(function(result) {
          document.getElementById('loading').style.display = 'none';
          if (result.ok) {
            mostrarMensaje('✅ ' + result.mensaje, 'success');
            setTimeout(function() { google.script.host.close(); }, 2500);
          } else {
            document.getElementById('formContent').style.display = 'block';
            mostrarMensaje('❌ ' + result.mensaje, 'error');
          }
        })
        .withFailureHandler(function(e) {
          document.getElementById('loading').style.display     = 'none';
          document.getElementById('formContent').style.display = 'block';
          mostrarMensaje('❌ Error: ' + e.message, 'error');
        })
        .cancelarOrdenCompra(num, motivo);
    }

    function mostrarMensaje(texto, tipo) {
      const el = document.getElementById('message');
      el.textContent = texto;
      el.className = 'message ' + tipo;
      el.style.display = 'block';
    }

    function esc(str) {
      return String(str || '').replace(/&/g,'&amp;').replace(/"/g,'&quot;')
        .replace(/</g,'&lt;').replace(/>/g,'&gt;');
    }
  </script>
