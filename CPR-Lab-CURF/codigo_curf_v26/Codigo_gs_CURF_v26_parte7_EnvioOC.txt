/**
 * CPR-CURF — Sistema de Gestión de Compras e Inventario
 * Parte 7 — Envío de Órdenes de Compra por PDF (v2.7)
 *
 * Agrega, a cada orden de ListaPendientes:
 *   - Ver PDF        → previsualiza el PDF de esa orden puntual (sin descargar nada)
 *   - Revisar pedido → crea un borrador de Gmail con el PDF adjunto, al mail del proveedor
 *   - Enviar mail    → envía el mail directo al proveedor (con confirmación previa en el cliente)
 *
 * Requiere que Codigo_gs_CURF_v26_parte1_OC.txt (CONFIG, getSheet, getDatosSheet,
 * getCurrentUser, getDetalleOrden, getProveedor) esté cargado en el mismo proyecto.
 *
 * Nota: si un proveedor tiene más de un email cargado (separados por coma, punto y
 * coma o salto de línea), por ahora se usa solo el primero — a la espera de acordar
 * con el cliente si conviene enviar a todos como destinatarios. Ver getEmailsProveedor().
 */

const CONFIG_ENVIO = {
  LOG_SHEET: 'Log_Envios_OC',
  ACCION: {
    BORRADOR: 'Borrador',
    ENVIADO:  'Enviado'
  }
};

// ========================================
// EMAILS DEL PROVEEDOR
// ========================================

/**
 * Devuelve los emails cargados para un proveedor como array, aceptando
 * múltiples direcciones separadas por coma, punto y coma o salto de línea.
 * Por ahora solo se usa emails[0] en revisarPedidoOC/enviarPedidoOC.
 */
function getEmailsProveedor(proveedor) {
  if (!proveedor || !proveedor.email) return [];
  return String(proveedor.email)
    .split(/[,;\n]+/)
    .map(function(e) { return e.trim(); })
    .filter(function(e) { return e.length > 0; });
}

/** Usada por ListaPendientes para deshabilitar Revisar/Enviar si falta el email. */
function tieneEmailProveedor(codigoProveedor) {
  return getEmailsProveedor(getProveedor(codigoProveedor)).length > 0;
}

// ========================================
// GENERACIÓN DEL PDF
// ========================================

function construirHtmlOrdenPdf(numeroOrden) {
  const detalle   = getDetalleOrden(numeroOrden);
  const orden     = detalle.orden;      // orden.proveedor = CÓDIGO del proveedor (ver Ordenes_Compra)
  const items     = detalle.items;
  const proveedor = getProveedor(orden.proveedor) || {};
  const nombreProveedor = proveedor.razon_social || orden.nombre_proveedor || orden.proveedor;

  const filas = items.map(function(it) {
    const subtotal = it.cantidad * it.precio_unitario;
    return '<tr>' +
      '<td>' + it.codigo_insumo + '</td>' +
      '<td>' + it.descripcion + '</td>' +
      '<td style="text-align:right">' + it.cantidad + '</td>' +
      '<td style="text-align:right">' + (it.precio_unitario ? '$' + it.precio_unitario.toFixed(2) : '-') + '</td>' +
      '<td style="text-align:right">' + (it.precio_unitario ? '$' + subtotal.toFixed(2) : '-') + '</td>' +
    '</tr>';
  }).join('');

  return '<html><head><meta charset="utf-8"><style>' +
    'body{font-family:Arial, sans-serif; font-size:12px; color:#222; padding:20px;}' +
    'h1{font-size:18px; margin-bottom:2px;} .sub{color:#666; margin-top:0; margin-bottom:16px;}' +
    'table{width:100%; border-collapse:collapse; margin-top:10px;}' +
    'th,td{border:1px solid #ccc; padding:6px 8px; font-size:11px;}' +
    'th{background:#f2f2f2; text-align:left;}' +
    '.total{font-weight:bold; text-align:right; margin-top:10px; font-size:13px;}' +
    '</style></head><body>' +
    '<h1>Orden de Compra ' + numeroOrden + '</h1>' +
    '<div class="sub">CURF — Laboratorio de Microbiología</div>' +
    '<p><strong>Proveedor:</strong> ' + nombreProveedor + '<br>' +
    '<strong>Fecha de emisión:</strong> ' + orden.fecha_emision + '<br>' +
    '<strong>Entrega estimada:</strong> ' + (orden.fecha_entrega_estimada || '-') + '</p>' +
    '<table><thead><tr><th>Código</th><th>Descripción</th><th>Cantidad</th><th>Precio unit.</th><th>Subtotal</th></tr></thead>' +
    '<tbody>' + filas + '</tbody></table>' +
    '<div class="total">Total: $' + (Number(orden.total_orden) || 0).toFixed(2) + '</div>' +
    (orden.observaciones ? '<p><strong>Observaciones:</strong> ' + orden.observaciones + '</p>' : '') +
  '</body></html>';
}

function generarPdfOrden(numeroOrden) {
  const html = construirHtmlOrdenPdf(numeroOrden);
  return HtmlService.createHtmlOutput(html)
    .getBlob()
    .getAs('application/pdf')
    .setName('OC_' + numeroOrden + '.pdf');
}

/** Devuelve el PDF en base64 para mostrarlo en un <iframe> sin descargar nada. */
function previsualizarPdfOrden(numeroOrden) {
  const blob = generarPdfOrden(numeroOrden);
  return Utilities.base64Encode(blob.getBytes());
}

/** Abre un diálogo modal con la vista previa del PDF de una orden puntual. */
function mostrarPreviaPdfOrden(numeroOrden) {
  try {
    const base64 = previsualizarPdfOrden(numeroOrden);
    // ✅ v2.8.1: Chrome bloquea la navegación directa de un <iframe> a una URL
    // "data:application/pdf;..." (la trata como una descarga potencialmente
    // insegura y la corta con "This page has been blocked by Chrome"). La forma
    // que sí funciona es armar el PDF como Blob del lado del cliente (con
    // atob + Uint8Array) y apuntar el iframe a un blob: URL generado con
    // URL.createObjectURL — eso Chrome sí lo permite.
    const html = HtmlService.createHtmlOutput(
      '<!DOCTYPE html><html><head><meta charset="utf-8"></head>' +
      '<body style="margin:0;padding:0;font-family:Arial,sans-serif;">' +
      '<div id="msg" style="padding:20px;color:#666;">⏳ Cargando PDF…</div>' +
      '<iframe id="pdfFrame" style="width:100%;height:100vh;border:none;display:none;"></iframe>' +
      '<script>' +
      '(function() {' +
      '  var base64 = "' + base64 + '";' +
      '  try {' +
      '    var byteChars = atob(base64);' +
      '    var byteNumbers = new Array(byteChars.length);' +
      '    for (var i = 0; i < byteChars.length; i++) { byteNumbers[i] = byteChars.charCodeAt(i); }' +
      '    var byteArray = new Uint8Array(byteNumbers);' +
      '    var blob = new Blob([byteArray], { type: "application/pdf" });' +
      '    var url = URL.createObjectURL(blob);' +
      '    var frame = document.getElementById("pdfFrame");' +
      '    frame.onload = function() { document.getElementById("msg").style.display = "none"; frame.style.display = "block"; };' +
      '    frame.src = url;' +
      '  } catch (e) {' +
      '    document.getElementById("msg").textContent = "❌ Error al mostrar el PDF: " + e.message;' +
      '  }' +
      '})();' +
      '</script>' +
      '</body></html>'
    ).setWidth(780).setHeight(680);
    SpreadsheetApp.getUi().showModalDialog(html, '📄 Vista previa — ' + numeroOrden);
  } catch (e) {
    SpreadsheetApp.getUi().alert('Error al generar la vista previa: ' + e.message);
  }
}

// ========================================
// TRAZABILIDAD (Log_Envios_OC)
// ========================================

function getOrCrearHojaLogEnvios() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(CONFIG_ENVIO.LOG_SHEET);
  if (!sheet) {
    sheet = ss.insertSheet(CONFIG_ENVIO.LOG_SHEET);
    sheet.appendRow(['fecha_hora', 'numero_orden', 'proveedor', 'accion', 'usuario', 'destinatario']);
    sheet.getRange(1, 1, 1, 6).setFontWeight('bold');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function registrarLogEnvio(numeroOrden, nombreProveedor, accion, destinatario) {
  const sheet = getOrCrearHojaLogEnvios();
  sheet.appendRow([new Date(), numeroOrden, nombreProveedor, accion, getCurrentUser(), destinatario]);
}

// REVISAR / ENVIAR
// ========================================

/**
 * Valida que el proveedor tenga email y arma los datos comunes a
 * revisarPedidoOC/enviarPedidoOC (evita repetir la validación en ambas).
 */
function prepararEnvioOrden(numeroOrden) {
  const detalle = getDetalleOrden(numeroOrden);
  const orden = detalle.orden;
  const proveedor = getProveedor(orden.proveedor);
  const emails = getEmailsProveedor(proveedor);

  if (emails.length === 0) {
    const nombre = (proveedor && proveedor.razon_social) || orden.nombre_proveedor || orden.proveedor;
    throw new Error('El proveedor "' + nombre + '" no tiene email cargado. Cargalo en la hoja Proveedores antes de continuar.');
  }

  const nombreProveedor = (proveedor && proveedor.razon_social) || orden.nombre_proveedor || orden.proveedor;
  // Por ahora se usa solo el primer email cargado (ver nota al inicio del archivo).
  const destinatario = emails[0];
  const pdf = generarPdfOrden(numeroOrden);
  const asunto = 'Orden de Compra ' + numeroOrden + ' — CURF Laboratorio';
  const cuerpo = 'Estimados,\n\nAdjuntamos la orden de compra ' + numeroOrden + ' correspondiente a ' +
    nombreProveedor + '.\n\nAnte cualquier consulta, quedamos a disposición.\n\nSaludos,\n' +
    'CURF — Laboratorio de Microbiología';

  return { orden, nombreProveedor, destinatario, pdf, asunto, cuerpo };
}

/** Crea un borrador de Gmail con el PDF adjunto, dirigido al proveedor. No lo envía. */
function revisarPedidoOC(numeroOrden) {
  const datos = prepararEnvioOrden(numeroOrden);
  GmailApp.createDraft(datos.destinatario, datos.asunto, datos.cuerpo, { attachments: [datos.pdf] });
  registrarLogEnvio(numeroOrden, datos.nombreProveedor, CONFIG_ENVIO.ACCION.BORRADOR, datos.destinatario);
  return { ok: true, destinatario: datos.destinatario };
}

/** Envía el mail directo al proveedor con el PDF adjunto. Sin vuelta atrás. */
function enviarPedidoOC(numeroOrden) {
  const datos = prepararEnvioOrden(numeroOrden);
  GmailApp.sendEmail(datos.destinatario, datos.asunto, datos.cuerpo, { attachments: [datos.pdf] });
  registrarLogEnvio(numeroOrden, datos.nombreProveedor, CONFIG_ENVIO.ACCION.ENVIADO, datos.destinatario);
  return { ok: true, destinatario: datos.destinatario };
}
