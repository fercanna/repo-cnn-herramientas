  <script>
    let ocSeleccionada = null;
    let itemsEditados  = [];

    // ── Carga inicial de OCs editables ────────────────────────────
    window.onload = function() {
      google.script.run
        .withSuccessHandler(renderizarListaOC)
        .withFailureHandler(function(e) { mostrarMensaje('Error al cargar OC: ' + e.message, 'error'); })
        .getOCsEditables();
    };

    function renderizarListaOC(ocs) {
      const lista = document.getElementById('listaOC');
      if (!ocs || ocs.length === 0) {
        lista.innerHTML = '<div class="empty-state">✅ No hay OC en estado Pendiente editables.<br><small>Las OC con recepciones parciales no pueden editarse.</small></div>';
        return;
      }
      lista.innerHTML = ocs.map(oc => `
        <div class="oc-item" onclick="seleccionarOC('${esc(oc.numero_orden)}', this)">
          <div>
            <div class="oc-num">${esc(oc.numero_orden)}</div>
            <div class="oc-meta">Emitida: ${esc(oc.fecha_emision)} | Entrega est.: ${esc(oc.fecha_entrega_estimada || '—')}</div>
          </div>
          <div class="oc-prov">${esc(oc.proveedor)}<br><small>$${parseFloat(oc.total_orden || 0).toFixed(2)}</small></div>
        </div>
      `).join('');
    }

    function seleccionarOC(numeroOrden, el) {
      document.querySelectorAll('.oc-item').forEach(i => i.classList.remove('selected'));
      el.classList.add('selected');

      google.script.run
        .withSuccessHandler(function(data) {
          ocSeleccionada = data.orden;
          itemsEditados  = data.items.map(function(item, idx) {
            return {
              idx,
              codigo_insumo:  item.codigo_insumo,
              descripcion:    item.descripcion,
              cantidad:       parseFloat(item.cantidad) || 0,
              precio_unitario: parseFloat(item.precio_unitario) || 0
            };
          });
          cargarFormularioEdicion(data.orden);
        })
        .withFailureHandler(function(e) { mostrarMensaje('Error al cargar detalle: ' + e.message, 'error'); })
        .getDetalleOCEditable(numeroOrden);
    }

    function cargarFormularioEdicion(orden) {
      document.getElementById('panelSelector').style.display = 'none';
      document.getElementById('panelEdicion').style.display  = 'block';

      document.getElementById('cabeceraOC').innerHTML =
        '<strong>' + esc(orden.numero_orden) + '</strong> — ' +
        esc(orden.proveedor) + ' | Estado: <strong style="color:#1967d2">Pendiente</strong>';

      document.getElementById('fechaEntrega').value  = orden.fecha_entrega_estimada || '';
      document.getElementById('observaciones').value  = orden.observaciones || '';
      document.getElementById('solicitadoPor').value  = orden.solicitado_por || '';

      renderizarItems();
    }

    function renderizarItems() {
      const container = document.getElementById('itemsContainer');
      container.innerHTML = '';

      if (itemsEditados.length === 0) {
        container.innerHTML = '<div style="padding:16px; color:#888; text-align:center;">Sin ítems</div>';
        return;
      }

      itemsEditados.forEach(function(item, i) {
        const sub = (item.cantidad * item.precio_unitario).toFixed(2);
        const div = document.createElement('div');
        div.className = 'item-row-edit';
        div.id = 'item_row_' + i;
        div.innerHTML = `
          <div>
            <div class="item-nombre">${esc(item.descripcion)}</div>
            <div class="item-codigo">${esc(item.codigo_insumo)}</div>
          </div>
          <input type="number" min="1" step="1" value="${item.cantidad}"
            onchange="actualizarItem(${i}, 'cantidad', this.value)">
          <input type="number" min="0" step="0.01" value="${item.precio_unitario.toFixed(2)}"
            onchange="actualizarItem(${i}, 'precio', this.value)">
          <div class="subtotal-cell" id="sub_${i}">$${sub}</div>
          <button class="btn-quitar" onclick="quitarItem(${i})" title="Quitar ítem">✕</button>
        `;
        container.appendChild(div);
      });

      actualizarTotal();
    }

    function actualizarItem(idx, campo, valor) {
      if (campo === 'cantidad') itemsEditados[idx].cantidad = parseFloat(valor) || 0;
      if (campo === 'precio')   itemsEditados[idx].precio_unitario = parseFloat(valor) || 0;
      const sub = (itemsEditados[idx].cantidad * itemsEditados[idx].precio_unitario).toFixed(2);
      document.getElementById('sub_' + idx).textContent = '$' + sub;
      actualizarTotal();
    }

    function quitarItem(idx) {
      if (itemsEditados.length <= 1) {
        alert('La orden debe tener al menos un ítem.');
        return;
      }
      itemsEditados.splice(idx, 1);
      renderizarItems();
    }

    function actualizarTotal() {
      const total = itemsEditados.reduce(function(sum, it) {
        return sum + it.cantidad * it.precio_unitario;
      }, 0);
      document.getElementById('totalEdit').textContent = 'Total: $' + total.toFixed(2);
    }

    function guardarEdicion() {
      if (!ocSeleccionada) return;

      if (itemsEditados.length === 0) {
        mostrarMensaje('La orden debe tener al menos un ítem.', 'error');
        return;
      }

      const invalidos = itemsEditados.filter(function(it) { return it.cantidad <= 0; });
      if (invalidos.length > 0) {
        mostrarMensaje('Todos los ítems deben tener cantidad mayor a 0.', 'error');
        return;
      }

      const payload = {
        numero_orden:           ocSeleccionada.numero_orden,
        fecha_entrega_estimada: document.getElementById('fechaEntrega').value,
        observaciones:          document.getElementById('observaciones').value,
        solicitado_por:         document.getElementById('solicitadoPor').value,
        items:                  itemsEditados.map(function(it) {
          return {
            codigo_insumo:   it.codigo_insumo,
            descripcion:     it.descripcion,
            cantidad:        it.cantidad,
            precio_unitario: it.precio_unitario
          };
        })
      };

      document.getElementById('panelEdicion').style.display = 'none';
      document.getElementById('loading').style.display = 'block';

      google.script.run
        .withSuccessHandler(function(result) {
          document.getElementById('loading').style.display = 'none';
          if (result.ok) {
            mostrarMensaje('✅ ' + result.mensaje, 'success');
            setTimeout(function() { google.script.host.close(); }, 2500);
          } else {
            document.getElementById('panelEdicion').style.display = 'block';
            mostrarMensaje('❌ ' + result.mensaje, 'error');
          }
        })
        .withFailureHandler(function(e) {
          document.getElementById('loading').style.display = 'none';
          document.getElementById('panelEdicion').style.display = 'block';
          mostrarMensaje('❌ Error: ' + e.message, 'error');
        })
        .editarOrdenCompra(payload);
    }

    function volverSelector() {
      ocSeleccionada = null;
      itemsEditados  = [];
      document.getElementById('panelEdicion').style.display  = 'none';
      document.getElementById('panelSelector').style.display = 'block';
      ocultarMensaje();
    }

    function mostrarMensaje(texto, tipo) {
      const el = document.getElementById('message');
      el.textContent = texto;
      el.className = 'message ' + tipo;
      el.style.display = 'block';
      el.scrollIntoView({ behavior: 'smooth' });
    }

    function ocultarMensaje() {
      document.getElementById('message').style.display = 'none';
    }

    function esc(str) {
      return String(str || '')
        .replace(/&/g,'&amp;').replace(/"/g,'&quot;')
        .replace(/</g,'&lt;').replace(/>/g,'&gt;');
    }
