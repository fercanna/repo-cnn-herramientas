//   - Cancelar Recepción  → cualquier recepción no anulada
//   - Cancelar Baja       → cualquier egreso tipo "Egreso" no anulado

//   para mantener el trazado de auditoría completo.

/**
 * Retorna las OC editables: solo las que están en Pendiente
 * y no tienen ninguna recepción registrada (ni parcial).
 * Usada por FormularioEditarOC y FormularioCancelarOC.
 */
function getOCsEditables() {
  const sheetOrdenes  = getSheet(CONFIG.SHEETS.ORDENES);
  const sheetRecep    = getSheet(CONFIG.SHEETS.RECEPCIONES);
  const lastRow       = sheetOrdenes.getLastRow();
  if (lastRow <= 1) return [];

  // Números de órdenes que ya tienen al menos una recepción (no anulada)
  const recepData    = getDatosSheet(CONFIG.SHEETS.RECEPCIONES);
  const ordenesConRecepcion = {};
  recepData.forEach(function(rec) {
    const obs = String(rec.observaciones || '').toUpperCase();
    if (obs.indexOf('ANULADA') !== -1) return; // ignorar recepciones anuladas
    ordenesConRecepcion[String(rec.numero_orden).trim()] = true;
  });

  const dataOrdenes = sheetOrdenes.getRange(2, 1, lastRow - 1, 10).getValues();

  return dataOrdenes
    .filter(function(row) {
      const estado      = String(row[4]).trim();
      const numOrden    = String(row[0]).trim();
      return row[0] &&
             estado === CONFIG.ESTADOS.ORDEN.PENDIENTE &&
             !ordenesConRecepcion[numOrden];
    })
    .map(function(row) {
      return {
        numero_orden:           String(row[0]).trim(),
        fecha_emision:          formatearFecha(row[1]),
        proveedor:              String(row[3]).trim(),   // nombre guardado en col D
        codigo_proveedor:       String(row[2]).trim(),
        fecha_entrega_estimada: formatearFecha(row[6]),
        total_orden:            row[7],
        observaciones:          String(row[8] || '').trim(),
        solicitado_por:         String(row[9] || '').trim()
      };
    });
}

/**
 * Retorna las recepciones activas (no anuladas) para el formulario de cancelación.
 * Incluye nombre del proveedor y ítems recibidos para mostrar al usuario.
 */
function getRecepcionesActivas() {
  const recepData   = getDatosSheet(CONFIG.SHEETS.RECEPCIONES);
  const detalleData = getDatosSheet(CONFIG.SHEETS.DETALLE_RECEPCIONES);
  const ordenesData = getDatosSheet(CONFIG.SHEETS.ORDENES);

  // Mapa número_orden → nombre_proveedor
  const mapaProveedores = {};
  ordenesData.forEach(function(o) {
    mapaProveedores[String(o.numero_orden).trim()] =
      String(o.nombre_proveedor || o.codigo_proveedor || '').trim();
  });

  return recepData
    .filter(function(rec) {
      const obs = String(rec.observaciones || '').toUpperCase();
      return obs.indexOf('ANULADA') === -1; // excluir ya anuladas
    })
    .map(function(rec) {
      const numRec   = String(rec.numero_recepcion).trim();
      const numOrden = String(rec.numero_orden).trim();

      // Items de esta recepción
      const items = detalleData
        .filter(function(d) { return String(d.numero_recepcion).trim() === numRec; })
        .map(function(d) {
          return {
            codigo_insumo:     String(d.codigo_insumo || '').trim(),
            descripcion:       String(d.descripcion   || '').trim(),
            cantidad_recibida: Number(d.cantidad_recibida) || 0,
            lote:              String(d.lote           || 'SIN LOTE').trim(),
            vencimiento:       d.vencimiento ? formatearFecha(d.vencimiento) : '',
            nota_conversion:   String(d.nota_conversion || '').trim()
          };
        });

      return {
        numero_recepcion: numRec,
        fecha:            formatearFecha(rec.fecha),
        numero_orden:     numOrden,
        proveedor:        mapaProveedores[numOrden] || '—',
        recibido_por:     String(rec.recibido_por || '').trim(),
        items
      };
    })
    .sort(function(a, b) {
      // Más recientes primero
      return b.numero_recepcion.localeCompare(a.numero_recepcion);
    });
}

/**
 * Retorna los egresos de consumo activos (no anulados) de los últimos 180 días.
 * Solo tipo "Egreso" con referencia "BAJA-HOJA" (los generados desde Bajas_Consumo).
 * Usada por FormularioCancelarBaja.
 */
function getMovimientosEgresoActivos() {
  const sheetMov = getSheet(CONFIG.SHEETS.MOVIMIENTOS);
  const data     = sheetMov.getDataRange().getValues();
  if (data.length <= 1) return [];

  const headers        = data[0];
  const idxId          = headers.indexOf('id_movimiento');
  const idxFecha       = headers.indexOf('fecha');
  const idxTipo        = headers.indexOf('tipo_movimiento');
  const idxCodigo      = headers.indexOf('codigo_insumo');
  const idxNombre      = headers.indexOf('nombre_insumo');
  const idxCantidad    = headers.indexOf('cantidad');
  const idxReferencia  = headers.indexOf('referencia');
  const idxUsuario     = headers.indexOf('usuario');
  const idxObs        = headers.indexOf('observaciones');
  const idxLote        = headers.indexOf('lote');
  const idxVenc        = headers.indexOf('vencimiento');

  const hace180 = new Date();
  hace180.setDate(hace180.getDate() - 180);

  const resultados = [];

  for (let i = 1; i < data.length; i++) {
    const row  = data[i];
    const tipo = String(row[idxTipo] || '').trim();
    const obs  = String(row[idxObs]  || '').toUpperCase();
    const ref  = String(row[idxReferencia] || '').trim();

    if (tipo !== 'Egreso')                 continue;
    if (obs.indexOf('ANULADO') !== -1)     continue; // ya anulado

    const fecha = new Date(row[idxFecha]);
    if (isNaN(fecha.getTime()) || fecha < hace180) continue;

    resultados.push({
      id_movimiento: String(row[idxId]      || '').trim(),
      fila:          i + 1,                 // 1-indexed, para poder ubicarlo rápido
      fecha:         formatearFecha(row[idxFecha]),
      codigo_insumo: String(row[idxCodigo]  || '').trim(),
      descripcion:   String(row[idxNombre]  || '').trim(),
      cantidad:      Number(row[idxCantidad]) || 0,
      referencia:    ref,
      usuario:       String(row[idxUsuario] || '').trim(),
      observaciones: String(row[idxObs]     || '').trim(),
      lote:          String(row[idxLote]    || 'SIN LOTE').trim(),
      vencimiento:   row[idxVenc] ? formatearFecha(row[idxVenc]) : ''
    });
  }

  // Más recientes primero
  return resultados.sort(function(a, b) {
    return b.id_movimiento.localeCompare(a.id_movimiento);
  });
}

/**
 * Edita una OC existente en estado Pendiente (sin recepciones).
 *
 * @param {Object} editData
 *   numero_orden            {string}
 *   fecha_entrega_estimada  {string}
 *   observaciones           {string}
 *   solicitado_por          {string}
 *   items  {Array} — [ { codigo_insumo, descripcion, cantidad, precio_unitario } ]
 *
 * @returns {{ ok: boolean, mensaje: string }}
 */
function editarOrdenCompra(editData) {
  try {
    const numeroOrden = String(editData.numero_orden).trim();

    // ── 1. Validar que la OC sea editable ─────────────────────────
    const ocs = getOCsEditables();
    const ocBase = ocs.find(function(o) { return o.numero_orden === numeroOrden; });
    if (!ocBase) {
      return {
        ok: false,
        mensaje: 'La OC "' + numeroOrden + '" no es editable. ' +
                 'Solo se pueden editar OC en estado Pendiente sin recepciones registradas.'
      };
    }

    // ── 2. Recalcular total ────────────────────────────────────────
    const items = editData.items || [];
    if (items.length === 0) {
      return { ok: false, mensaje: 'Debe incluir al menos un ítem en la OC.' };
    }

    const nuevoTotal = items.reduce(function(sum, item) {
      return sum + (Number(item.cantidad) * Number(item.precio_unitario));
    }, 0);

    // ── 3. Actualizar cabecera en Ordenes_Compra ──────────────────
    const sheetOrdenes = getSheet(CONFIG.SHEETS.ORDENES);
    const dataOrdenes  = sheetOrdenes.getDataRange().getValues();

    let filaOC = -1;
    for (let i = 1; i < dataOrdenes.length; i++) {
      if (String(dataOrdenes[i][0]).trim() === numeroOrden) {
        filaOC = i + 1; // 1-indexed
        break;
      }
    }
    if (filaOC === -1) {
      return { ok: false, mensaje: 'No se encontró la OC "' + numeroOrden + '" en la hoja.' };
    }

    // Actualizar: fecha_entrega (col G=7), total (col H=8), observaciones (col I=9), solicitado_por (col J=10)
    sheetOrdenes.getRange(filaOC, 7, 1, 4).setValues([[
      editData.fecha_entrega_estimada || '',
      nuevoTotal,
      editData.observaciones || '',
      editData.solicitado_por || ''
    ]]);

    // ── 4. Reemplazar ítems en Detalle_Ordenes ────────────────────
    const sheetDetalle = getSheet(CONFIG.SHEETS.DETALLE_ORDENES);
    const dataDetalle  = sheetDetalle.getDataRange().getValues();

    // Encontrar y borrar filas existentes de esta OC
    // Recorremos desde el final para no desplazar índices al borrar
    for (let i = dataDetalle.length - 1; i >= 1; i--) {
      if (String(dataDetalle[i][0]).trim() === numeroOrden) {
        sheetDetalle.deleteRow(i + 1);
      }
    }

    // Escribir los nuevos ítems en batch
    const fechaAhora  = new Date();
    const filasDetalle = items.map(function(item, idx) {
      const subtotal = Number(item.cantidad) * Number(item.precio_unitario);
      return [
        numeroOrden,
        idx + 1,
        item.codigo_insumo,
        item.descripcion,
        Number(item.cantidad),
        Number(item.precio_unitario),
        subtotal,
        0,                               // cantidad_recepcionada = 0
        CONFIG.ESTADOS.ITEM.PENDIENTE,
        fechaAhora
      ];
    });

    const lastDetalle = sheetDetalle.getLastRow();
    sheetDetalle
      .getRange(lastDetalle + 1, 1, filasDetalle.length, filasDetalle[0].length)
      .setValues(filasDetalle);

    // ── 5. Actualizar historial de precios si cambiaron precios ───
    const usuario       = getCurrentUser();
    const proveedor     = getProveedor(ocBase.codigo_proveedor);
    const nombreProv    = proveedor ? proveedor.razon_social : ocBase.codigo_proveedor;
    registrarHistorialPrecios(
      items.map(function(item) {
        return {
          codigo_insumo:  item.codigo_insumo,
          descripcion:    item.descripcion,
          precio_unitario: Number(item.precio_unitario)
        };
      }),
      numeroOrden + '-EDIT',
      usuario,
      fechaAhora,
      nombreProv
    );

    return {
      ok:     true,
      mensaje: 'OC "' + numeroOrden + '" actualizada correctamente.\n' +
               '• ' + items.length + ' ítem(s) | Nuevo total: $' + nuevoTotal.toFixed(2)
    };

  } catch (e) {
    return { ok: false, mensaje: 'Error al editar la OC: ' + e.message };
  }
}

/**
 * Marca una OC como Cancelada.
 * Solo opera sobre OC en estado Pendiente sin recepciones.
 *
 * @param {string} numeroOrden
 * @param {string} motivo        — se agrega a observaciones
 *
 * @returns {{ ok: boolean, mensaje: string }}
 */
function cancelarOrdenCompra(numeroOrden, motivo) {
  try {
    numeroOrden = String(numeroOrden).trim();

    // Validar editabilidad (mismas condiciones que editar)
    const ocs    = getOCsEditables();
    const ocBase = ocs.find(function(o) { return o.numero_orden === numeroOrden; });
    if (!ocBase) {
      return {
        ok: false,
        mensaje: 'La OC "' + numeroOrden + '" no puede cancelarse. ' +
                 'Solo se pueden cancelar OC en Pendiente sin recepciones.'
      };
    }

    const sheetOrdenes = getSheet(CONFIG.SHEETS.ORDENES);
    const dataOrdenes  = sheetOrdenes.getDataRange().getValues();

    for (let i = 1; i < dataOrdenes.length; i++) {
      if (String(dataOrdenes[i][0]).trim() !== numeroOrden) continue;

      const obsActual  = String(dataOrdenes[i][8] || '').trim();
      const nuevaObs   = (obsActual ? obsActual + ' | ' : '') +
                         'CANCELADA: ' + (motivo || 'Sin motivo especificado') +
                         ' (' + formatearFecha(new Date()) + ' — ' + getCurrentUser() + ')';

      // Col E=5 (estado), Col I=9 (observaciones)
      sheetOrdenes.getRange(i + 1, 5).setValue(CONFIG.ESTADOS.ORDEN.CANCELADA);
      sheetOrdenes.getRange(i + 1, 9).setValue(nuevaObs);
      break;
    }

    return {
      ok:     true,
      mensaje: 'OC "' + numeroOrden + '" cancelada exitosamente.\n' +
               'El motivo quedó registrado en Observaciones.'
    };

  } catch (e) {
    return { ok: false, mensaje: 'Error al cancelar la OC: ' + e.message };
  }
}

/**
 * Revierte una recepción completa:
 *   1. Lee los ítems recibidos desde Detalle_Recepciones
 *   2. Lee los movimientos de Ingreso asociados desde Movimientos_Inventario
 *   3. Crea movimientos de "Anulación" (tipo Ajuste, cantidad negativa) para trazabilidad
 *   4. Actualiza Stock_Actual (resta las cantidades ingresadas)
 *   5. Revierte las cantidades en Detalle_Ordenes
 *   6. Recalcula el estado de la OC
 *   7. Marca la recepción como ANULADA en Recepciones
 *
 * @param {string} numeroRecepcion
 * @param {string} motivo
 *
 * @returns {{ ok: boolean, mensaje: string }}
 */
function cancelarRecepcion(numeroRecepcion, motivo) {
  try {
    numeroRecepcion = String(numeroRecepcion).trim();
    const usuario   = getCurrentUser();
    const ahora     = new Date();

    // ── 1. Leer recepción y validar que no esté ya anulada ────────
    const sheetRecep = getSheet(CONFIG.SHEETS.RECEPCIONES);
    const dataRecep  = sheetRecep.getDataRange().getValues();
    const headersRecep = dataRecep[0];

    const idxRecNum = headersRecep.indexOf('numero_recepcion');
    const idxRecObs = headersRecep.indexOf('observaciones');
    const idxRecOrden = headersRecep.indexOf('numero_orden');

    let filaRecep    = -1;
    let numeroOrden  = '';
    let obsActualRec = '';

    for (let i = 1; i < dataRecep.length; i++) {
      if (String(dataRecep[i][idxRecNum]).trim() !== numeroRecepcion) continue;
      const obs = String(dataRecep[i][idxRecObs] || '').toUpperCase();
      if (obs.indexOf('ANULADA') !== -1) {
        return { ok: false, mensaje: 'La recepción "' + numeroRecepcion + '" ya está anulada.' };
      }
      filaRecep    = i + 1;
      numeroOrden  = String(dataRecep[i][idxRecOrden]).trim();
      obsActualRec = String(dataRecep[i][idxRecObs] || '').trim();
      break;
    }

    if (filaRecep === -1) {
      return { ok: false, mensaje: 'No se encontró la recepción "' + numeroRecepcion + '".' };
    }

    // ── 2. Leer ítems de Detalle_Recepciones ─────────────────────
    const detalleRecepData = getDatosSheet(CONFIG.SHEETS.DETALLE_RECEPCIONES);
    const itemsRecibidos   = detalleRecepData.filter(function(d) {
      return String(d.numero_recepcion).trim() === numeroRecepcion;
    });

    if (itemsRecibidos.length === 0) {
      return { ok: false, mensaje: 'No se encontraron ítems para la recepción "' + numeroRecepcion + '".' };
    }

    // ── 3. Leer movimientos de Ingreso de esta recepción ──────────
    const sheetMov    = getSheet(CONFIG.SHEETS.MOVIMIENTOS);
    const dataMov     = sheetMov.getDataRange().getValues();
    const headersMov  = dataMov[0];

    const idxMovId   = headersMov.indexOf('id_movimiento');
    const idxMovTipo = headersMov.indexOf('tipo_movimiento');
    const idxMovRef  = headersMov.indexOf('referencia');
    const idxMovCod  = headersMov.indexOf('codigo_insumo');
    const idxMovNom  = headersMov.indexOf('nombre_insumo');
    const idxMovCant = headersMov.indexOf('cantidad');
    const idxMovLote = headersMov.indexOf('lote');
    const idxMovVenc = headersMov.indexOf('vencimiento');
    const idxMovCat  = headersMov.indexOf('categoria');

    const movIngresosRecep = [];
    for (let i = 1; i < dataMov.length; i++) {
      if (String(dataMov[i][idxMovRef]).trim()  !== numeroRecepcion) continue;
      if (String(dataMov[i][idxMovTipo]).trim() !== 'Ingreso')       continue;
      movIngresosRecep.push({
        fila:          i + 1,
        id_movimiento: String(dataMov[i][idxMovId]   || '').trim(),
        codigo_insumo: String(dataMov[i][idxMovCod]  || '').trim(),
        nombre_insumo: String(dataMov[i][idxMovNom]  || '').trim(),
        cantidad:      Number(dataMov[i][idxMovCant]) || 0,
        lote:          String(dataMov[i][idxMovLote]  || 'SIN LOTE').trim(),
        vencimiento:   dataMov[i][idxMovVenc] || '',
        categoria:     String(dataMov[i][idxMovCat]  || '').trim()
      });
    }

    // ── 4. Crear movimientos de Anulación (Ajuste negativo) ───────
    const filasAnulacion = movIngresosRecep.map(function(mov) {
      return [
        'ANUL-' + Date.now() + '-' + Math.random().toString(36).substr(2, 4),
        ahora,
        'Ajuste',                        // tipo que maneja actualizarStockActual
        mov.codigo_insumo,
        mov.nombre_insumo,
        -mov.cantidad,                   // negativo = resta stock
        'ANULACION-' + numeroRecepcion,
        usuario,
        'Anulación recepción ' + numeroRecepcion + ' | ' + (motivo || 'Sin motivo'),
        mov.lote,
        mov.vencimiento,
        mov.categoria
      ];
    });

    if (filasAnulacion.length > 0) {
      const lastMov = sheetMov.getLastRow();
      sheetMov.getRange(lastMov + 1, 1, filasAnulacion.length, filasAnulacion[0].length)
        .setValues(filasAnulacion);
    }

    // ── 5. Actualizar Stock_Actual (restar cantidades ingresadas) ─
    _revertirStockDeRecepcion(movIngresosRecep, ahora);

    // ── 6. Revertir cantidades en Detalle_Ordenes ─────────────────
    _revertirCantidadesEnOrden(numeroOrden, itemsRecibidos);

    // ── 7. Recalcular estado de la OC ─────────────────────────────
    actualizarEstadoOrden(numeroOrden);

    // ── 8. Marcar recepción como ANULADA ──────────────────────────
    const nuevaObsRec = (obsActualRec ? obsActualRec + ' | ' : '') +
                        'ANULADA: ' + (motivo || 'Sin motivo') +
                        ' (' + formatearFecha(ahora) + ' — ' + usuario + ')';
    sheetRecep.getRange(filaRecep, idxRecObs + 1).setValue(nuevaObsRec);

    return {
      ok:     true,
      mensaje: 'Recepción "' + numeroRecepcion + '" anulada exitosamente.\n' +
               '• ' + movIngresosRecep.length + ' ítem(s) revertido(s) del stock.\n' +
               '• Estado de OC "' + numeroOrden + '" recalculado.'
    };

  } catch (e) {
    return { ok: false, mensaje: 'Error al cancelar la recepción: ' + e.message };
  }
}

/**
 * Revierte un egreso de consumo (Baja).
 *   1. Busca el movimiento por ID en Movimientos_Inventario
 *   2. Valida que sea Egreso y no esté anulado
 *   3. Crea un movimiento de "Ajuste" positivo (contrapartida)
 *   4. Actualiza Stock_Actual (suma de vuelta la cantidad consumida)
 *   5. Marca el movimiento original como ANULADO en Observaciones
 *
 * @param {string} idMovimiento
 * @param {string} motivo
 *
 * @returns {{ ok: boolean, mensaje: string }}
 */
function cancelarBaja(idMovimiento, motivo) {
  try {
    idMovimiento = String(idMovimiento).trim();
    const usuario = getCurrentUser();
    const ahora   = new Date();

    // ── 1. Encontrar el movimiento ─────────────────────────────────
    const sheetMov   = getSheet(CONFIG.SHEETS.MOVIMIENTOS);
    const dataMov    = sheetMov.getDataRange().getValues();
    const headersMov = dataMov[0];

    const idxId   = headersMov.indexOf('id_movimiento');
    const idxTipo = headersMov.indexOf('tipo_movimiento');
    const idxCod  = headersMov.indexOf('codigo_insumo');
    const idxNom  = headersMov.indexOf('nombre_insumo');
    const idxCant = headersMov.indexOf('cantidad');
    const idxRef  = headersMov.indexOf('referencia');
    const idxObs  = headersMov.indexOf('observaciones');
    const idxLote = headersMov.indexOf('lote');
    const idxVenc = headersMov.indexOf('vencimiento');
    const idxCat  = headersMov.indexOf('categoria');

    let filaMovimiento = -1;
    let movData        = null;

    for (let i = 1; i < dataMov.length; i++) {
      if (String(dataMov[i][idxId]).trim() !== idMovimiento) continue;

      const tipo = String(dataMov[i][idxTipo] || '').trim();
      const obs  = String(dataMov[i][idxObs]  || '').toUpperCase();

      if (tipo !== 'Egreso') {
        return { ok: false, mensaje: 'El movimiento "' + idMovimiento + '" no es un Egreso.' };
      }
      if (obs.indexOf('ANULADO') !== -1) {
        return { ok: false, mensaje: 'El movimiento "' + idMovimiento + '" ya está anulado.' };
      }

      filaMovimiento = i + 1;
      movData = {
        codigo_insumo: String(dataMov[i][idxCod]  || '').trim(),
        nombre_insumo: String(dataMov[i][idxNom]  || '').trim(),
        cantidad:      Number(dataMov[i][idxCant]) || 0,
        referencia:    String(dataMov[i][idxRef]   || '').trim(),
        lote:          String(dataMov[i][idxLote]  || 'SIN LOTE').trim(),
        vencimiento:   dataMov[i][idxVenc] || '',
        categoria:     String(dataMov[i][idxCat]   || '').trim(),
        obsOriginal:   String(dataMov[i][idxObs]   || '').trim()
      };
      break;
    }

    if (!movData) {
      return { ok: false, mensaje: 'No se encontró el movimiento "' + idMovimiento + '".' };
    }

    // ── 2. Crear movimiento de contrapartida (Ajuste positivo) ────
    sheetMov.appendRow([
      'RECUP-' + Date.now() + '-' + Math.random().toString(36).substr(2, 4),
      ahora,
      'Ajuste',
      movData.codigo_insumo,
      movData.nombre_insumo,
      movData.cantidad,                 // positivo = suma stock
      'ANULACION-' + idMovimiento,
      usuario,
      'Anulación baja ' + idMovimiento + ' | ' + (motivo || 'Sin motivo'),
      movData.lote,
      movData.vencimiento,
      movData.categoria
    ]);

    // ── 3. Marcar el egreso original como ANULADO ──────────────────
    const nuevaObs = movData.obsOriginal +
                     ' | ANULADO: ' + (motivo || 'Sin motivo') +
                     ' (' + formatearFecha(ahora) + ' — ' + usuario + ')';
    sheetMov.getRange(filaMovimiento, idxObs + 1).setValue(nuevaObs);

    // ── 4. Devolver el stock al lote correspondiente ───────────────
    _sumarStockLote(
      movData.codigo_insumo,
      movData.lote,
      movData.vencimiento,
      movData.cantidad,
      movData.nombre_insumo,
      ahora
    );

    return {
      ok:     true,
      mensaje: 'Baja "' + idMovimiento + '" anulada exitosamente.\n' +
               '• Se restituyeron ' + movData.cantidad + ' u. de "' + movData.nombre_insumo +
               '" al lote ' + movData.lote + '.'
    };

  } catch (e) {
    return { ok: false, mensaje: 'Error al cancelar la baja: ' + e.message };
  }
}

/**
 * Revierte el stock de una recepción:
 * resta cada cantidad de Ingreso del lote correspondiente en Stock_Actual.
 */
function _revertirStockDeRecepcion(movIngresos, ahora) {
  const sheetStock = getSheet(CONFIG.SHEETS.STOCK);
  const lastRow    = sheetStock.getLastRow();
  if (lastRow <= 1) return;

  const dataStock  = sheetStock.getRange(2, 1, lastRow - 1, 9).getValues();
  const catalogoInsumos = getInsumosActivos();

  // Stock total actual por código
  const stockTotal = {};
  dataStock.forEach(function(row) {
    const cod = String(row[0]).trim();
    stockTotal[cod] = (stockTotal[cod] || 0) + (Number(row[2]) || 0);
  });

  const codigosAfectados = {};

  movIngresos.forEach(function(mov) {
    const cod  = mov.codigo_insumo;
    const lote = mov.lote;
    codigosAfectados[cod] = true;

    for (let i = 0; i < dataStock.length; i++) {
      if (String(dataStock[i][0]).trim() !== cod)  continue;
      if (String(dataStock[i][6]).trim() !== lote) continue;

      const nuevoStock = Math.max((Number(dataStock[i][2]) || 0) - mov.cantidad, 0);
      dataStock[i][2]  = nuevoStock;
      dataStock[i][5]  = ahora;
      stockTotal[cod]  = (stockTotal[cod] || 0) - mov.cantidad;
      break;
    }
  });

  // Recalcular estado para los códigos afectados
  dataStock.forEach(function(row) {
    const cod = String(row[0]).trim();
    if (!codigosAfectados[cod]) return;
    const ins     = catalogoInsumos.find(function(i) {
      return String(i.codigo_insumo).trim() === cod;
    });
    const critico = ins ? Number(ins.stock_critico) || 0 : Number(row[3]) || 0;
    const total   = stockTotal[cod] || 0;
    row[4] = total <= 0 ? 'Agotado' : total <= critico ? 'Crítico' : 'OK';
  });

  // Filtrar y reescribir
  const filasValidas = dataStock.filter(function(row) { return Number(row[2]) > 0; });
  sheetStock.getRange(2, 1, lastRow - 1, 9).clearContent();
  if (filasValidas.length > 0) {
    sheetStock.getRange(2, 1, filasValidas.length, 9).setValues(filasValidas);
    sheetStock.getRange(2, 1, filasValidas.length, 9).sort({ column: 8, ascending: true });
  }
}

/**
 * Revierte las cantidades recepcionadas en Detalle_Ordenes.
 * Resta lo recibido en esta recepción de la columna cantidad_recepcionada,
 * y vuelve el estado_item a Pendiente si queda en 0.
 */
function _revertirCantidadesEnOrden(numeroOrden, itemsRecibidos) {
  const sheetDetalle = getSheet(CONFIG.SHEETS.DETALLE_ORDENES);
  const data         = sheetDetalle.getDataRange().getValues();

  // Mapa codigo → cantidad recibida en esta recepción
  const mapaCantidades = {};
  itemsRecibidos.forEach(function(item) {
    const cod = String(item.codigo_insumo || '').trim();
    mapaCantidades[cod] = (mapaCantidades[cod] || 0) + (Number(item.cantidad_recibida) || 0);
  });

  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() !== numeroOrden) continue;

    const cod = String(data[i][2]).trim();
    if (!mapaCantidades[cod]) continue;

    const cantRecibidaActual = Number(data[i][7]) || 0;
    const cantSolicitada     = Number(data[i][4]) || 0;
    const nueva              = Math.max(cantRecibidaActual - mapaCantidades[cod], 0);
    const nuevoEstado        = nueva <= 0
      ? CONFIG.ESTADOS.ITEM.PENDIENTE
      : nueva < cantSolicitada
        ? CONFIG.ESTADOS.ITEM.PARCIAL
        : CONFIG.ESTADOS.ITEM.COMPLETO;

    sheetDetalle.getRange(i + 1, 8, 1, 2).setValues([[nueva, nuevoEstado]]);
  }
}

/**
 * Suma una cantidad a un lote específico en Stock_Actual.
 * Si el lote no existe, crea una fila nueva.
 * Usada por cancelarBaja() para restituir stock consumido.
 */
function _sumarStockLote(codigo, lote, vencimiento, cantidad, descripcion, ahora) {
  const sheetStock    = getSheet(CONFIG.SHEETS.STOCK);
  const lastRow       = sheetStock.getLastRow();
  const catalogoInsumos = getInsumosActivos();

  const dataStock = lastRow > 1
    ? sheetStock.getRange(2, 1, lastRow - 1, 9).getValues()
    : [];

  const stockTotal = {};
  dataStock.forEach(function(row) {
    const cod = String(row[0]).trim();
    stockTotal[cod] = (stockTotal[cod] || 0) + (Number(row[2]) || 0);
  });

  let encontrado = false;
  for (let i = 0; i < dataStock.length; i++) {
    if (String(dataStock[i][0]).trim() !== codigo) continue;
    if (String(dataStock[i][6]).trim() !== lote)   continue;
    dataStock[i][2] = (Number(dataStock[i][2]) || 0) + cantidad;
    dataStock[i][5] = ahora;
    stockTotal[codigo] = (stockTotal[codigo] || 0) + cantidad;
    encontrado = true;
    break;
  }

  if (!encontrado) {
    const ins     = catalogoInsumos.find(function(i) {
      return String(i.codigo_insumo).trim() === codigo;
    });
    const critico = ins ? Number(ins.stock_critico) || 0 : 0;
    dataStock.push([
      codigo, descripcion, cantidad, critico, 'OK',
      ahora, lote, vencimiento,
      ins ? (ins.clasificacion || '') : ''
    ]);
    stockTotal[codigo] = (stockTotal[codigo] || 0) + cantidad;
  }

  // Recalcular estado
  const ins2     = catalogoInsumos.find(function(i) {
    return String(i.codigo_insumo).trim() === codigo;
  });
  const critico2   = ins2 ? Number(ins2.stock_critico) || 0 : 0;
  const total      = stockTotal[codigo] || 0;
  const nuevoEst   = total <= 0 ? 'Agotado' : total <= critico2 ? 'Crítico' : 'OK';

  for (let j = 0; j < dataStock.length; j++) {
    if (String(dataStock[j][0]).trim() === codigo) dataStock[j][4] = nuevoEst;
  }

  const filasValidas = dataStock.filter(function(row) { return Number(row[2]) > 0; });
  if (sheetStock.getLastRow() > 1) {
    sheetStock.getRange(2, 1, sheetStock.getLastRow() - 1, 9).clearContent();
  }
  if (filasValidas.length > 0) {
    sheetStock.getRange(2, 1, filasValidas.length, 9).setValues(filasValidas);
    sheetStock.getRange(2, 1, filasValidas.length, 9).sort({ column: 8, ascending: true });
  }
}
