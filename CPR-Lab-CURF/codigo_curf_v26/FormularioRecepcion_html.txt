  <script>
    let itemsOrden  = [];
    let ordenActual = null;

    window.onload = function() { cargarOrdenesManual(); };

    function cargarOrdenesManual() {
      google.script.run
        .withSuccessHandler(function(ordenes) {
          const select = document.getElementById('ordenSelect');
          select.innerHTML = '<option value="">-- Seleccione una orden --</option>';
          ordenes.forEach(o => {
            const opt = document.createElement('option');
            opt.value = o.numero_orden;
            const esParcial = o.estado === 'Recepción Parcial';
            opt.textContent = o.numero_orden + ' — ' + o.proveedor + (esParcial ? ' [PARCIAL]' : '');
            if (esParcial) opt.style.color = '#856404';
            select.appendChild(opt);
          });
        })
        .getOrdenesPendientes();
    }

    function cargarDetalleOrden() {
      const num = document.getElementById('ordenSelect').value;
      if (!num) return;
      google.script.run
        .withSuccessHandler(mostrarDetalle)
        .getDetalleOrden(num);
    }

    function mostrarDetalle(data) {
      ordenActual = data.orden;
      itemsOrden  = data.items;

      const esParcial = ordenActual.estado === 'Recepción Parcial';

      document.getElementById('ordenDetalle').innerHTML =
        '<strong>Orden:</strong> ' + ordenActual.numero_orden +
        ' | <strong>Proveedor:</strong> ' + ordenActual.proveedor +
        ' | <strong>Estado:</strong> <span style="color:' + (esParcial ? '#856404' : '#1967d2') + ';font-weight:bold;">' + ordenActual.estado + '</span>';

      const banner       = document.getElementById('historialBanner');
      const historialEl  = document.getElementById('historialItems');

      if (esParcial) {
        let html = '';
        itemsOrden.forEach(item => {
          const solicitado = parseFloat(item.cantidad) || 0;
          const yaRecibido = parseFloat(item.cantidad_recepcionada) || 0;
          const pendiente  = solicitado - yaRecibido;
          if (yaRecibido > 0) {
            html += `<div class="historial-item">
              <span>${item.descripcion}</span>
              <span style="color:#1e7e34;font-weight:bold;">${yaRecibido} recib.</span>
              <span style="color:#d93025;font-weight:bold;">${pendiente} pend.</span>
            </div>`;
          }
        });
        historialEl.innerHTML = html || '<em>Sin historial previo</em>';
        banner.style.display = 'block';
      } else {
        banner.style.display = 'none';
      }

      const container = document.getElementById('itemsContainer');
      container.innerHTML = '<h3>Items a Recepcionar</h3>';

      itemsOrden.forEach((item, i) => {
        const solicitado = parseFloat(item.cantidad) || 0;
        const yaRecibido = parseFloat(item.cantidad_recepcionada) || 0;
        const pendiente  = solicitado - yaRecibido;

        let badgeHtml = '';
        if (yaRecibido > 0) {
          badgeHtml = `<span class="progreso-badge progreso-parcial">⚠️ ${yaRecibido}/${solicitado} ya recibidos</span>`;
        } else {
          badgeHtml = `<span class="progreso-badge progreso-nuevo">✨ Sin recepciones previas</span>`;
        }

        const div = document.createElement('div');
        div.className = 'item-row';
        div.innerHTML = `
  <div class="item-header">${item.descripcion} (${item.codigo_insumo}) ${badgeHtml}</div>
  <div class="item-grid">
    <div><label>Pedido</label><input type="number" value="${solicitado}" disabled></div>
    <div><label>Pendiente</label><input type="number" value="${pendiente}" disabled style="font-weight:bold; color:#d93025;"></div>
    <div><label>A recibir *</label><input type="number" class="cant-rec" value="${pendiente}" data-idx="${i}" onchange="validarCantidad(this, ${pendiente})"></div>
    <div><label>Lote *</label><input type="text" class="lote-rec" placeholder="Lote/Partida" data-idx="${i}"></div>
    <div><label>Vence</label><input type="date" class="vence-rec" data-idx="${i}"></div>
    <div>
      <label>Conformidad *</label>
      <select class="conf-rec" data-idx="${i}">
        <option value="Conforme">Conforme</option>
        <option value="Conforme Parcial">Parcial</option>
        <option value="No Conforme">No Conforme</option>
      </select>
    </div>
    <div>
      <label>Precio unit. <span style="font-weight:normal;color:#888;font-size:11px;">(si difiere)</span></label>
      <input type="number" class="precio-rec" step="0.01" min="0"
             placeholder="Opcional" data-idx="${i}"
             style="border-color: #fbbc04;">
    </div>
    <div><label>Observaciones</label><input type="text" class="obs-rec" placeholder="Temperatura, condición..." data-idx="${i}"></div>
  </div>`;
        container.appendChild(div);
      });

      document.getElementById('recepcionForm').style.display = 'block';
    }

    function validarCantidad(input, max) {
      if (parseFloat(input.value) > max) {
        alert('La cantidad ingresada (' + input.value + ') es mayor a la pendiente (' + max + ').');
        input.value = max;
      }
    }

    function guardarRecepcion() {
      const cants  = document.querySelectorAll('.cant-rec');
      const lotes  = document.querySelectorAll('.lote-rec');
      const vence  = document.querySelectorAll('.vence-rec');
      const confs  = document.querySelectorAll('.conf-rec');
      const obss   = document.querySelectorAll('.obs-rec');
      const precios = document.querySelectorAll('.precio-rec');

      const itemsEnviar = [];
      let errorLote = false;

      cants.forEach((input, i) => {
        const cant = parseFloat(input.value) || 0;
        if (cant > 0) {
          const loteVal = lotes[i].value.trim();
          if (!loteVal) errorLote = true;
          itemsEnviar.push({
            item:              i + 1,
            codigo_insumo:     itemsOrden[i].codigo_insumo,
            cantidad_recibida: cant,
            cantidad_esperada: itemsOrden[i].cantidad - (itemsOrden[i].cantidad_recepcionada || 0),
            estado_conformidad: confs[i].value,
            lote:              loteVal,
            vencimiento:       vence[i].value,
            observaciones:     obss[i].value,
            precio_recepcion:  parseFloat(precios[i].value) || 0
          });
        }
      });

      if (itemsEnviar.length === 0) return alert('Debe ingresar al menos una cantidad.');
      if (errorLote) return alert('Por trazabilidad ISO 9001, debe ingresar el LOTE de cada insumo recibido.');

      const recibidoPor = document.getElementById('recibidoPor').value;
      if (!recibidoPor) return alert('Debe seleccionar quién recibe los insumos.');

      document.getElementById('recepcionForm').style.display = 'none';
      document.getElementById('loading').style.display = 'block';

      google.script.run
        .withSuccessHandler(function(resultado) {
          document.getElementById('loading').style.display = 'none';

          let recId, estadoOrden;
          if (typeof resultado === 'object' && resultado !== null) {
            recId      = resultado.id || resultado;
            estadoOrden = resultado.estado_orden || 'Recepcionada';
          } else {
            recId = resultado;
            const todoRecibido = itemsEnviar.every((it, i) => {
              const pedido = itemsOrden[i] ? (itemsOrden[i].cantidad - (itemsOrden[i].cantidad_recepcionada || 0)) : 0;
              return it.cantidad_recibida >= pedido;
            });
            estadoOrden = todoRecibido ? 'Recepcionada ✅' : 'Recepción Parcial ⚠️';
          }

          document.getElementById('res-id').textContent    = recId;
          document.getElementById('res-orden').textContent = ordenActual.numero_orden;

          const esCompleta = estadoOrden.includes('Recepcionada') && !estadoOrden.includes('Parcial');
          const estadoEl   = document.getElementById('res-estado');
          estadoEl.textContent = estadoOrden;
          estadoEl.className   = 'ri-valor ' + (esCompleta ? 'estado-completa' : 'estado-parcial');

          document.getElementById('res-mensaje').textContent = esCompleta
            ? 'La orden quedó completamente recepcionada y se actualizó el stock.'
            : 'Quedan ítems pendientes. La orden permanecerá activa para recepciones futuras.';

          document.getElementById('resultadoPanel').style.display = 'block';
        })
        .withFailureHandler(function(err) {
          alert('Error: ' + err.message);
          document.getElementById('loading').style.display = 'none';
          document.getElementById('recepcionForm').style.display = 'block';
        })
        .registrarRecepcion({
          numero_orden:  ordenActual.numero_orden,
          observaciones: document.getElementById('observacionesRecepcion').value,
          recibido_por:  recibidoPor,
          items:         itemsEnviar
        });
    }
  </script>
