  <script>
    let movimientos    = [];
    let movFiltrados   = [];
    let movSeleccionado = null;

    window.onload = function() {
      google.script.run
        .withSuccessHandler(function(data) {
          movimientos  = data;
          movFiltrados = data;
          renderizarSelect(data);
        })
        .withFailureHandler(function(e) { mostrarMensaje('Error al cargar movimientos: ' + e.message, 'error'); })
        .getMovimientosEgresoActivos();
    };

    function renderizarSelect(data) {
      const sel = document.getElementById('movSelect');
      if (!data || data.length === 0) {
        sel.innerHTML = '<option value="">— Sin egresos activos en los últimos 180 días —</option>';
        return;
      }
      sel.innerHTML = data.map(function(m) {
        const label = m.fecha + ' | ' + m.descripcion + ' (' + m.codigo_insumo + ')' +
                      ' — ' + m.cantidad + ' u. | Lote: ' + m.lote;
        return '<option value="' + esc(m.id_movimiento) + '">' + esc(label) + '</option>';
      }).join('');
    }

    function filtrarMovimientos() {
      const busq = document.getElementById('busqueda').value.trim().toUpperCase();
      if (busq.length < 2) {
        movFiltrados = movimientos;
      } else {
        movFiltrados = movimientos.filter(function(m) {
          return m.descripcion.toUpperCase().indexOf(busq) > -1 ||
                 m.codigo_insumo.toUpperCase().indexOf(busq) > -1 ||
                 m.referencia.toUpperCase().indexOf(busq) > -1 ||
                 m.lote.toUpperCase().indexOf(busq) > -1;
        });
      }
      renderizarSelect(movFiltrados);
      document.getElementById('detalleMov').style.display = 'none';
      document.getElementById('btnRestituir').disabled = true;
      movSeleccionado = null;
    }

    function onMovimientoChange() {
      const id = document.getElementById('movSelect').value;
      movSeleccionado = movFiltrados.find(function(m) { return m.id_movimiento === id; });

      const detalleEl    = document.getElementById('detalleMov');
      const btnRestituir = document.getElementById('btnRestituir');

      if (!movSeleccionado) {
        detalleEl.style.display = 'none';
        btnRestituir.disabled = true;
        return;
      }

      document.getElementById('dInsumo').textContent  = movSeleccionado.descripcion + ' (' + movSeleccionado.codigo_insumo + ')';
      document.getElementById('dCantidad').textContent = movSeleccionado.cantidad + ' u.';
      document.getElementById('dLote').textContent    = movSeleccionado.lote + (movSeleccionado.vencimiento ? ' · Vence: ' + movSeleccionado.vencimiento : '');
      document.getElementById('dFecha').textContent   = movSeleccionado.fecha;
      document.getElementById('dUsuario').textContent = movSeleccionado.usuario || '—';

      if (movSeleccionado.referencia) {
        document.getElementById('dRef').textContent       = movSeleccionado.referencia;
        document.getElementById('filaRef').style.display  = 'flex';
      } else {
        document.getElementById('filaRef').style.display  = 'none';
      }

      document.getElementById('cantAdvert').textContent  = movSeleccionado.cantidad;
      document.getElementById('insAdvert').textContent   = movSeleccionado.descripcion;
      document.getElementById('loteAdvert').textContent  = movSeleccionado.lote;

      detalleEl.style.display = 'block';
      btnRestituir.disabled = false;
    }

    function confirmarAnulacion() {
      if (!movSeleccionado) return mostrarMensaje('Seleccioná un movimiento.', 'error');

      const motivo = document.getElementById('motivo').value.trim();
      if (!motivo) return mostrarMensaje('El motivo es obligatorio.', 'error');

      const confirmMsg = '¿Anular egreso de ' + movSeleccionado.cantidad + ' u. de "' +
        movSeleccionado.descripcion + '"?\n\nLas unidades volverán al stock del lote ' +
        movSeleccionado.lote + '.';
      if (!confirm(confirmMsg)) return;

      document.getElementById('formContent').style.display = 'none';
      document.getElementById('loading').style.display     = 'block';

      google.script.run
        .withSuccessHandler(function(result) {
          document.getElementById('loading').style.display = 'none';
          if (result.ok) {
            mostrarMensaje('✅ ' + result.mensaje, 'success');
            setTimeout(function() { google.script.host.close(); }, 3000);
          } else {
            document.getElementById('formContent').style.display = 'block';
            mostrarMensaje('❌ ' + result.mensaje, 'error');
          }
        })
        .withFailureHandler(function(e) {
          document.getElementById('loading').style.display     = 'none';
          document.getElementById('formContent').style.display = 'block';
          mostrarMensaje('❌ Error: ' + e.message, 'error');
        })
        .cancelarBaja(movSeleccionado.id_movimiento, motivo);
    }

    function mostrarMensaje(texto, tipo) {
      const el = document.getElementById('message');
      el.textContent = texto;
      el.className = 'message ' + tipo;
      el.style.display = 'block';
    }

    function esc(str) {
      return String(str || '').replace(/&/g,'&amp;').replace(/"/g,'&quot;')
        .replace(/</g,'&lt;').replace(/>/g,'&gt;');
    }
