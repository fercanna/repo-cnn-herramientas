  <script>
    const _datos = <?= datosOC ?>;

    window.onload = function() {
      const d = new Date();
      d.setDate(d.getDate() + 7);
      document.getElementById('fechaEntrega').value = d.toISOString().split('T')[0];

      let html = '';
      _datos.proveedores.forEach(function(codProv) {
        const items  = _datos.items[codProv];
        const nombre = _datos.mapa[codProv].nombre;
        html += '<strong>' + nombre + '</strong> (' + items.length + ' item' + (items.length > 1 ? 's' : '') + '):<br>';
        items.forEach(function(it) {
          const precio = it.precio_unitario || 0;
          const sub    = (it.cantidad * precio).toFixed(2);
          html += '&nbsp;&nbsp;• ' + it.descripcion +
                  ' — ' + it.cantidad + ' u. × $' + precio.toFixed(2) +
                  ' = <strong>$' + sub + '</strong><br>';
        });
        html += '<br>';
      });
      document.getElementById('resumen').innerHTML = html;
    };

    function confirmar() {
      const solicitadoPor = document.getElementById('solicitadoPor').value;
      const fecha         = document.getElementById('fechaEntrega').value;
      const obs           = document.getElementById('observaciones').value;

      if (!solicitadoPor) {
        alert('Por favor seleccioná quién solicita la orden.');
        return;
      }

      document.getElementById('formSection').style.display = 'none';
      document.getElementById('loading').style.display = 'block';

      google.script.run
        .withSuccessHandler(function(ordenesCreadas) {
          document.getElementById('loading').style.display = 'none';
          const div = document.getElementById('success');
          div.style.display = 'block';
          const plural = ordenesCreadas.length > 1;
          div.innerHTML = '✅ <strong>Orden' + (plural ? 'es' : '') +
                          ' generada' + (plural ? 's' : '') + ':</strong><br>' +
                          ordenesCreadas.map(function(o) { return '• ' + o; }).join('<br>');
          setTimeout(function() { google.script.host.close(); }, 2500);
        })
        .withFailureHandler(function(err) {
          document.getElementById('loading').style.display = 'none';
          document.getElementById('formSection').style.display = 'block';
          alert('Error: ' + err.message);
        })
        .confirmarOrdenDesdeStock(fecha, obs, solicitadoPor);
    }
  </script>
