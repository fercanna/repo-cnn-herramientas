      <select id="loteSelect" onchange="onLoteChange()">
        <option value="">— Seleccioná un lote —</option>
      </select>
      <div class="lote-info" id="loteInfo">
        <div class="fila">
          <span class="etiqueta">Stock actual en este lote:</span>
          <span class="valor" id="stockActualLabel">0</span>
        </div>
        <div class="venc" id="vencLabel"></div>

      <input type="number" id="stockCorrecto" min="0" step="1"
        placeholder="Ingresá el stock real..."
        oninput="calcularDelta()">

    <button class="btn btn-secondary" onclick="google.script.host.close()">Cerrar</button>

<script>
  var insumos    = [];
  var loteActual = null;
  var itemActual = null;

  // Carga inicial de insumos
  google.script.run
    .withSuccessHandler(function(data) { insumos = data; })
    .withFailureHandler(function(e) { mostrarError('Error al cargar insumos: ' + e.message); })
    .getInsumosConStockYSinStock();

  // ── Buscador ──────────────────────────────────────────────────
  var inputBusq = document.getElementById('busqueda');
  var dropdown  = document.getElementById('dropdown');

  inputBusq.addEventListener('input', function() {
    var val = this.value.trim().toUpperCase();
    if (val.length < 2) { dropdown.style.display = 'none'; return; }

    var filtrados = insumos.filter(function(ins) {
      return ins.codigo.toUpperCase().indexOf(val) > -1 ||
             ins.descripcion.toUpperCase().indexOf(val) > -1;
    }).slice(0, 14);

    if (!filtrados.length) { dropdown.style.display = 'none'; return; }

    dropdown.innerHTML = filtrados.map(function(ins) {
      var stockStr = ins.stock_total > 0
        ? ins.stock_total + ' u.'
        : '<span style="color:#ea4335">Sin stock</span>';
      return '<div class="dropdown-item" onclick="seleccionarInsumo(\'' +
             esc(ins.codigo) + '\')">' +
        '<div class="di-left">' +
          '<div class="desc">' + esc(ins.descripcion) + '</div>' +
          '<div class="cod">'  + esc(ins.codigo) + ' · ' + esc(ins.categoria) + '</div>' +
        '</div>' +
        '<div class="di-right">' + stockStr + '</div>' +
      '</div>';
    }).join('');
    dropdown.style.display = 'block';
  });

  inputBusq.addEventListener('blur', function() {
    setTimeout(function() { dropdown.style.display = 'none'; }, 180);
  });

  // ── Selección de insumo ──────────────────────────────────────
  function seleccionarInsumo(codigo) {
    var ins = insumos.find(function(x) { return x.codigo === codigo; });
    if (!ins) return;
    itemActual = ins;

    inputBusq.value            = '';
    dropdown.style.display     = 'none';

    document.getElementById('cardNombre').textContent = ins.descripcion;
    document.getElementById('cardSub').textContent    =
      ins.codigo + (ins.categoria ? ' · ' + ins.categoria : '');
    document.getElementById('itemCard').style.display    = 'flex';
    document.getElementById('panelAjuste').style.display = 'block';
    document.getElementById('btnAjustar').style.display  = 'inline-block';

    loteActual = null;
    document.getElementById('loteSelect').innerHTML = '<option value="">Cargando lotes…</option>';
    document.getElementById('loteInfo').style.display = 'none';
    document.getElementById('deltaBox').style.display  = 'none';
    document.getElementById('stockCorrecto').value     = '';
    ocultarMensajes();

    google.script.run
      .withSuccessHandler(cargarLotes)
      .withFailureHandler(function(e) { mostrarError('Error al cargar lotes: ' + e.message); })
      .getLotesParaAjuste(codigo);
  }

  function cargarLotes(lotes) {
    var sel = document.getElementById('loteSelect');
    sel.innerHTML = '<option value="">— Seleccioná un lote —</option>' +
      lotes.map(function(l) {
        var label = l.lote;
        if (l.vencimiento) label += '  (vence ' + l.vencimiento + ')';
        label += '  —  ' + l.stock_actual + ' u.';
        return '<option value="' + esc(l.lote) + '"' +
          ' data-stock="' + l.stock_actual + '"' +
          ' data-venc="'  + esc(l.vencimiento || '') + '">' +
          label + '</option>';
      }).join('');
  }

  function onLoteChange() {
    var sel = document.getElementById('loteSelect');
    var opt = sel.options[sel.selectedIndex];
    var inf = document.getElementById('loteInfo');

    if (!sel.value) {
      inf.style.display = 'none';
      loteActual = null;
      document.getElementById('deltaBox').style.display = 'none';
      return;
    }

    var stock = Number(opt.getAttribute('data-stock'));
    var venc  = opt.getAttribute('data-venc') || '';
    loteActual = { lote: sel.value, stock: stock, vencimiento: venc };

    document.getElementById('stockActualLabel').textContent = stock;
    document.getElementById('vencLabel').textContent =
      venc ? 'Vencimiento: ' + venc : '';
    inf.style.display = 'block';

    // Pre-completar con el valor actual para que el usuario solo corrija
    document.getElementById('stockCorrecto').value = stock;
    calcularDelta();
  }

  function calcularDelta() {
    var box = document.getElementById('deltaBox');
    if (!loteActual) { box.style.display = 'none'; return; }

    var val = document.getElementById('stockCorrecto').value;
    if (val === '' || isNaN(Number(val))) { box.style.display = 'none'; return; }

    var correcto = Number(val);
    var delta    = correcto - loteActual.stock;

    box.style.display = 'block';
    box.className = 'delta ' +
      (delta > 0 ? 'delta-pos' : delta < 0 ? 'delta-neg' : 'delta-cero');

    if (delta === 0) {
      box.textContent = '✓ Sin cambios (el stock ya tiene ese valor)';
    } else if (delta > 0) {
      box.textContent = '⬆ Se sumarán +' + delta + ' unidades al lote';
    } else {
      box.textContent = '⬇ Se restarán ' + Math.abs(delta) + ' unidades del lote';
    }
  }

  // ── Limpiar selección ────────────────────────────────────────
  function limpiarSeleccion() {
    itemActual = null; loteActual = null;
    document.getElementById('itemCard').style.display    = 'none';
    document.getElementById('panelAjuste').style.display = 'none';
    document.getElementById('btnAjustar').style.display  = 'none';
    document.getElementById('deltaBox').style.display    = 'none';
    document.getElementById('loteInfo').style.display    = 'none';
    ocultarMensajes();
    inputBusq.focus();
  }

  // ── Enviar ajuste ────────────────────────────────────────────
  function aplicarAjuste() {
    if (!itemActual)  { mostrarError('Seleccioná un insumo primero.'); return; }
    if (!loteActual)  { mostrarError('Seleccioná un lote.'); return; }

    var correcto = Number(document.getElementById('stockCorrecto').value);
    if (isNaN(correcto) || correcto < 0) {
      mostrarError('Ingresá una cantidad válida (número entero ≥ 0).');
      return;
    }

    var motivo = document.getElementById('motivo').value.trim();
    if (!motivo) { mostrarError('El motivo es obligatorio.'); return; }

    if (correcto === loteActual.stock) {
      mostrarError('El stock ya tiene ese valor. No hay nada que ajustar.');
      return;
    }

    ocultarMensajes();
    document.getElementById('loading').style.display = 'block';
    document.getElementById('btnAjustar').disabled   = true;

    var payload = {
      codigo_insumo:  itemActual.codigo,
      descripcion:    itemActual.descripcion,
      lote:           loteActual.lote,
      vencimiento:    loteActual.vencimiento,
      stock_actual:   loteActual.stock,
      stock_correcto: correcto,
      motivo:         motivo
    };

    google.script.run
      .withSuccessHandler(function(result) {
        document.getElementById('loading').style.display = 'none';
        document.getElementById('btnAjustar').disabled   = false;

        if (result.ok) {
          var el = document.getElementById('msgOk');
          el.textContent    = '✅ ' + result.mensaje;
          el.style.display  = 'block';
          // Actualizar estado local para que el delta se recalcule bien
          loteActual.stock  = correcto;
          document.getElementById('stockActualLabel').textContent = correcto;
          calcularDelta();
          // Limpiar motivo para el próximo ajuste
          document.getElementById('motivo').value = '';
        } else {
          mostrarError(result.mensaje);
        }
      })
      .withFailureHandler(function(e) {
        document.getElementById('loading').style.display = 'none';
        document.getElementById('btnAjustar').disabled   = false;
        mostrarError('Error inesperado: ' + e.message);
      })
      .registrarAjusteStock(payload);
  }

  // ── Helpers ──────────────────────────────────────────────────
  function mostrarError(msg) {
    var el = document.getElementById('msgErr');
    el.textContent   = '❌ ' + msg;
    el.style.display = 'block';
  }
  function ocultarMensajes() {
    document.getElementById('msgOk').style.display  = 'none';
    document.getElementById('msgErr').style.display = 'none';
  }
  function esc(str) {
    return String(str || '')
      .replace(/&/g,'&amp;').replace(/"/g,'&quot;')
      .replace(/</g,'&lt;').replace(/>/g,'&gt;');
  }
</script>
