<script>
  var insumos      = [];
  var filtroActivo = 'todos';
  var itemSeleccionado = null;

  // ── Carga inicial ────────────────────────────────────────────
  google.script.run
    .withSuccessHandler(function(data) {
      insumos = data;
      renderizarLista();
    })
    .withFailureHandler(function(e) {
      document.getElementById('listaItems').innerHTML =
        '<div style="padding:14px;color:#a61c00;">Error al cargar insumos: ' + esc(e.message) + '</div>';
    })
    .getInsumosConConversiones();

  // ── Filtros ──────────────────────────────────────────────────
  function setFiltro(f) {
    filtroActivo = f;
    ['chipTodos','chipConv','chipSin'].forEach(function(id) {
      document.getElementById(id).classList.remove('activo');
    });
    var chipMap = { todos: 'chipTodos', con: 'chipConv', sin: 'chipSin' };
    document.getElementById(chipMap[f]).classList.add('activo');
    renderizarLista();
  }

  function filtrarLista() { renderizarLista(); }

  function renderizarLista() {
    var busq = document.getElementById('searchInput').value.trim().toUpperCase();
    var lista = document.getElementById('listaItems');

    var filtrados = insumos.filter(function(ins) {
      // Filtro por búsqueda
      if (busq.length >= 2) {
        if (ins.codigo.toUpperCase().indexOf(busq) === -1 &&
            ins.descripcion.toUpperCase().indexOf(busq) === -1) return false;
      }
      // Filtro por estado
      if (filtroActivo === 'con' && !ins.tiene_conversion) return false;
      if (filtroActivo === 'sin' &&  ins.tiene_conversion) return false;
      return true;
    });

    if (!filtrados.length) {
      lista.innerHTML = '<div style="padding:16px;color:#9aa0a6;text-align:center;">Sin resultados.</div>';
      document.getElementById('listaFooter').textContent = '0 insumos';
      return;
    }

    lista.innerHTML = filtrados.map(function(ins) {
      var badge = ins.tiene_conversion
        ? '<span class="badge badge-on">x' + ins.factor + '</span>'
        : '<span class="badge badge-off">sin conv.</span>';
      var sel = (itemSeleccionado && itemSeleccionado.codigo === ins.codigo) ? ' selected' : '';
      return '<div class="lista-item' + sel + '" onclick="seleccionarItem(\'' + esc(ins.codigo) + '\')">' +
        '<div>' +
          '<div class="li-nombre">' + esc(ins.descripcion) + '</div>' +
          '<div class="li-codigo">' + esc(ins.codigo) +
            (ins.categoria ? ' · ' + esc(ins.categoria) : '') + '</div>' +
        '</div>' +
        badge +
      '</div>';
    }).join('');

    document.getElementById('listaFooter').textContent =
      filtrados.length + ' insumo' + (filtrados.length !== 1 ? 's' : '');
  }

  // ── Selección de insumo ──────────────────────────────────────
  function seleccionarItem(codigo) {
    var ins = insumos.find(function(x) { return x.codigo === codigo; });
    if (!ins) return;
    itemSeleccionado = ins;

    // Resaltar en la lista
    document.querySelectorAll('.lista-item').forEach(function(el) {
      el.classList.remove('selected');
    });
    event.currentTarget.classList.add('selected');

    // Poblar formulario
    document.getElementById('placeholder').style.display    = 'none';
    var fc = document.getElementById('formContent');
    fc.style.display = 'flex';

    document.getElementById('fTitulo').textContent = ins.descripcion;
    document.getElementById('fSub').textContent    =
      ins.codigo + (ins.categoria ? ' · ' + ins.categoria : '') +
      (ins.unidad_medida ? ' · Unidad base: ' + ins.unidad_medida : '');

    document.getElementById('chkActivo').checked        = ins.activo || ins.tiene_conversion;
    document.getElementById('fUnidadCompra').value      = ins.unidad_compra  || '';
    document.getElementById('fFactor').value            = ins.factor         || '';
    document.getElementById('fUnidadDestino').value     = ins.unidad_destino || '';
    document.getElementById('fObservaciones').value     = ins.observaciones  || '';

    document.getElementById('btnDesactivar').style.display =
      ins.tiene_conversion ? 'block' : 'none';

    ocultarMensajes();
    actualizarPreview();
  }

  // ── Preview ──────────────────────────────────────────────────
  function actualizarPreview() {
    var pv = document.getElementById('preview');
    var activo  = document.getElementById('chkActivo').checked;
    var factor  = Number(document.getElementById('fFactor').value);
    var uCompra = document.getElementById('fUnidadCompra').value.trim() || 'caja';
    var uStock  = document.getElementById('fUnidadDestino').value.trim() || 'unidad';

    if (!activo) {
      pv.style.display = 'block';
      pv.textContent = 'ℹ️ Con la conversión desactivada, al recepcionar 1 ' + uCompra +
        ' se registrará 1 ' + uStock + ' en stock.';
      return;
    }

    if (factor >= 2) {
      pv.style.display = 'block';
      pv.textContent = '✅ Al recepcionar 1 ' + uCompra + ' se registrarán ' +
        factor + ' ' + uStock + '(s) en stock.';
    } else {
      pv.style.display = 'none';
    }
  }

  // ── Guardar ──────────────────────────────────────────────────
  function guardarConversion() {
    if (!itemSeleccionado) return;

    var factor = Number(document.getElementById('fFactor').value);
    if (document.getElementById('chkActivo').checked) {
      if (!factor || factor < 1) { mostrarError('Ingresá un factor válido (número ≥ 1).'); return; }
      if (!document.getElementById('fUnidadDestino').value.trim()) {
        mostrarError('Ingresá la unidad de stock (ej: Determinación, Tubo, Unidad).');
        return;
      }
    }

    ocultarMensajes();
    document.getElementById('loading').style.display = 'block';

    var payload = {
      codigo:         itemSeleccionado.codigo,
      descripcion:    itemSeleccionado.descripcion,
      factor:         factor || 1,
      unidad_compra:  document.getElementById('fUnidadCompra').value.trim(),
      unidad_destino: document.getElementById('fUnidadDestino').value.trim(),
      activo:         document.getElementById('chkActivo').checked,
      observaciones:  document.getElementById('fObservaciones').value.trim()
    };

    google.script.run
      .withSuccessHandler(function(result) {
        document.getElementById('loading').style.display = 'none';
        if (result.ok) {
          document.getElementById('msgOk').textContent   = '✅ ' + result.mensaje;
          document.getElementById('msgOk').style.display = 'block';
          // Actualizar estado local
          itemSeleccionado.tiene_conversion = payload.activo && payload.factor >= 2;
          itemSeleccionado.factor           = payload.factor;
          itemSeleccionado.unidad_destino   = payload.unidad_destino;
          itemSeleccionado.unidad_compra    = payload.unidad_compra;
          itemSeleccionado.activo           = payload.activo;
          itemSeleccionado.observaciones    = payload.observaciones;
          // Actualizar el insumo en el array global
          var idx = insumos.findIndex(function(x) { return x.codigo === itemSeleccionado.codigo; });
          if (idx > -1) insumos[idx] = JSON.parse(JSON.stringify(itemSeleccionado));
          document.getElementById('btnDesactivar').style.display =
            itemSeleccionado.tiene_conversion ? 'block' : 'none';
          renderizarLista();
        } else {
          mostrarError(result.mensaje);
        }
      })
      .withFailureHandler(function(e) {
        document.getElementById('loading').style.display = 'none';
        mostrarError('Error: ' + e.message);
      })
      .guardarConversionInsumo(payload);
  }

  // ── Desactivar ───────────────────────────────────────────────
  function desactivarConversion() {
    if (!itemSeleccionado || !itemSeleccionado.tiene_conversion) return;
    if (!confirm('¿Desactivar la conversión de "' + itemSeleccionado.descripcion + '"?\n\n' +
        'Los movimientos anteriores no se modifican. Podés reactivarla en cualquier momento.')) return;

    ocultarMensajes();
    document.getElementById('loading').style.display = 'block';

    google.script.run
      .withSuccessHandler(function(result) {
        document.getElementById('loading').style.display = 'none';
        if (result.ok) {
          document.getElementById('msgOk').textContent   = '✅ ' + result.mensaje;
          document.getElementById('msgOk').style.display = 'block';
          itemSeleccionado.tiene_conversion = false;
          itemSeleccionado.activo           = false;
          document.getElementById('chkActivo').checked   = false;
          document.getElementById('btnDesactivar').style.display = 'none';
          var idx = insumos.findIndex(function(x) { return x.codigo === itemSeleccionado.codigo; });
          if (idx > -1) { insumos[idx].tiene_conversion = false; insumos[idx].activo = false; }
          actualizarPreview();
          renderizarLista();
        } else {
          mostrarError(result.mensaje);
        }
      })
      .withFailureHandler(function(e) {
        document.getElementById('loading').style.display = 'none';
        mostrarError('Error: ' + e.message);
      })
      .desactivarConversionInsumo(itemSeleccionado.codigo);
  }

  // ── Helpers ──────────────────────────────────────────────────
  function mostrarError(msg) {
    var el = document.getElementById('msgErr');
    el.textContent   = '❌ ' + msg;
    el.style.display = 'block';
  }
  function ocultarMensajes() {
    document.getElementById('msgOk').style.display  = 'none';
    document.getElementById('msgErr').style.display = 'none';
  }
  function esc(str) {
    return String(str || '')
      .replace(/&/g,'&amp;').replace(/"/g,'&quot;')
      .replace(/</g,'&lt;').replace(/>/g,'&gt;');
  }
