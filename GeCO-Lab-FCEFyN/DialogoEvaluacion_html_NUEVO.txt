<script>
// ── Estado global ─────────────────────────────────────────────────
var personas       = [];      // todas las personas activas
var puestosMap     = {};      // puesto → [personas]

var modoSeleccion  = 'individual';  // 'individual' | 'puesto'

// Selección individual
var personaSel     = null;

// Selección por puesto
var puestoSel      = '';      // nombre del puesto seleccionado
var personasSelMulti = [];    // array de objetos persona seleccionadas

var compPendientes = [];
var compTodas      = [];
var compSel        = null;
var modoSel        = null;
var docsDrive      = [];
var pasoActual     = 1;

// ── Init ──────────────────────────────────────────────────────────
window.onload = function() { cargarPersonas(); };

// ── Cargar personas y construir mapa por puesto ───────────────────
function cargarPersonas() {
  google.script.run
    .withSuccessHandler(function(data) {
      personas = data || [];

      // Construir mapa puesto → personas
      puestosMap = {};
      personas.forEach(function(p) {
        var puesto = p.puesto || '(Sin puesto)';
        if (!puestosMap[puesto]) puestosMap[puesto] = [];
        puestosMap[puesto].push(p);
      });

      // Poblar select individual
      var sel = document.getElementById('selectPersona');
      personas.forEach(function(p) {
        var o = document.createElement('option');
        o.value = p.idPersona;
        o.text  = p.nombre + (p.puesto ? ' — ' + p.puesto : '') + (p.area ? ' (' + p.area + ')' : '');
        sel.appendChild(o);
      });

      // Poblar select de puestos
      var selP = document.getElementById('selectPuesto');
      Object.keys(puestosMap).sort().forEach(function(puesto) {
        var o = document.createElement('option');
        o.value = puesto;
        o.text  = puesto + ' (' + puestosMap[puesto].length + ' personas)';
        selP.appendChild(o);
      });
    })
    .withFailureHandler(function(err) {
      mostrarAlerta('alertaPaso1', 'error', 'Error cargando personas: ' + err.message);
    })
    .getPersonas();
}

// ── Toggle modo de selección ──────────────────────────────────────
function cambiarModoSel(modo) {
  modoSeleccion = modo;
  personaSel = null;
  personasSelMulti = [];
  puestoSel = '';

  document.getElementById('btnModoIndividual').classList.toggle('activo',  modo === 'individual');
  document.getElementById('btnModoPuesto').classList.toggle('activo',      modo === 'puesto');
  document.getElementById('btnModoTransversal').classList.toggle('activo', modo === 'transversal');

  document.getElementById('bloqueIndividual').style.display  = modo === 'individual'  ? 'block' : 'none';
  document.getElementById('bloquePuesto').style.display      = modo === 'puesto'      ? 'block' : 'none';
  document.getElementById('bloqueTransversal').style.display = modo === 'transversal' ? 'block' : 'none';

  // Resetear selecciones
  document.getElementById('selectPersona').value = '';
  document.getElementById('personaCard').classList.remove('visible');
  document.getElementById('selectPuesto').value = '';
  document.getElementById('puestoInfo').classList.remove('visible');
  document.getElementById('personasMulti').classList.remove('visible');
  document.getElementById('selCount').classList.remove('visible');
  document.getElementById('selCountTransversal').classList.remove('visible');

  // Cargar lista transversal al activar
  if (modo === 'transversal') renderListaTransversal('');

  document.getElementById('btnSiguiente').disabled = true;
  ocultarAlerta('alertaPaso1');
}

// ── TRANSVERSAL — lista de todas las personas ────────────────────
function renderListaTransversal(filtro) {
  var lista = document.getElementById('listaTransversal');
  var filtroNorm = filtro.toLowerCase();
  var visibles = personas.filter(function(p) {
    return !filtroNorm ||
      p.nombre.toLowerCase().includes(filtroNorm) ||
      (p.puesto || '').toLowerCase().includes(filtroNorm);
  });

  lista.innerHTML = visibles.map(function(p) {
    var estaSeleccionada = personasSelMulti.some(function(s) { return s.idPersona === p.idPersona; });
    return '<label class="persona-check-item' + (estaSeleccionada ? ' seleccionada' : '') +
      '" id="pct-' + esc(p.idPersona) + '">' +
      '<input type="checkbox" value="' + esc(p.idPersona) + '"' +
      (estaSeleccionada ? ' checked' : '') +
      ' onchange="onCheckTransversal(this)">' +
      '<div class="persona-check-info">' +
        '<div class="persona-check-nombre">' + esc(p.nombre) + '</div>' +
        '<div class="persona-check-sub">' + esc(p.puesto || '') + (p.area ? ' · ' + esc(p.area) : '') + '</div>' +
      '</div></label>';
  }).join('');
}

function filtrarTransversal() {
  var filtro = document.getElementById('buscarTransversal').value;
  renderListaTransversal(filtro);
}

function onCheckTransversal(checkbox) {
  var id = checkbox.value;
  var persona = personas.find(function(p) { return p.idPersona === id; });
  var item = document.getElementById('pct-' + id);

  if (checkbox.checked) {
    if (persona && !personasSelMulti.find(function(p) { return p.idPersona === id; })) {
      personasSelMulti.push(persona);
    }
    if (item) item.classList.add('seleccionada');
  } else {
    personasSelMulti = personasSelMulti.filter(function(p) { return p.idPersona !== id; });
    if (item) item.classList.remove('seleccionada');
  }

  var count = personasSelMulti.length;
  var selCount = document.getElementById('selCountTransversal');
  if (count > 0) {
    selCount.textContent = count + ' persona(s) seleccionada(s)';
    selCount.classList.add('visible');

  var info     = document.getElementById('puestoInfo');

        '<div class="persona-check-nombre">' + esc(p.nombre) + '</div>' +
        '<div class="persona-check-sub">' + esc(p.area || '') + (p.mail ? ' · ' + esc(p.mail) : '') + '</div>' +

  }).join('');

  selCount.classList.remove('visible');

function onCheckPersona(checkbox) {
  var id = checkbox.value;
  var persona = personas.find(function(p) { return p.idPersona === id; });
  var item = document.getElementById('pci-' + id);

  if (checkbox.checked) {
    if (persona && !personasSelMulti.find(function(p) { return p.idPersona === id; })) {
      personasSelMulti.push(persona);
    }
    if (item) item.classList.add('seleccionada');
  } else {
    personasSelMulti = personasSelMulti.filter(function(p) { return p.idPersona !== id; });
    if (item) item.classList.remove('seleccionada');
  }

  var count = personasSelMulti.length;
  var selCount = document.getElementById('selCount');
  if (count > 0) {
    selCount.textContent = count + ' persona(s) seleccionada(s)';
    selCount.classList.add('visible');
  } else {
    selCount.classList.remove('visible');
  }

  document.getElementById('btnSiguiente').disabled = (count === 0);
  ocultarAlerta('alertaPaso1');
}

// ── PASO 2 — Competencias ─────────────────────────────────────────
function cargarCompetencias() {
  var puesto = getPuestoActivo();
  if (!puesto) return;

  // Tomar una persona de referencia para verificar pendientes
  var personaRef = modoSeleccion === 'individual'
    ? personaSel
    : (personasSelMulti[0] || null);

  document.getElementById('emptyPendientes').textContent = 'Cargando…';
  document.getElementById('listaPendientes').innerHTML = '<div class="empty-hint" id="emptyPendientes">Cargando…</div>';
  document.getElementById('listaTodas').innerHTML = '';
  document.getElementById('listaTodas').classList.remove('visible');
  document.getElementById('btnToggleTodas').textContent = '▸ Ver todas las competencias del puesto';
  compSel = null;

  var cargados = 0;
  function check() { cargados++; if (cargados === 2) document.getElementById('btnSiguiente').disabled = !compSel; }

  if (personaRef && modoSeleccion !== 'transversal') {
    google.script.run
      .withSuccessHandler(function(data) {
        compPendientes = data || [];
        renderCompLista('listaPendientes', compPendientes, true);
        document.getElementById('cntPendientes').textContent = compPendientes.length;
        if (compPendientes.length === 0) {
          document.getElementById('listaPendientes').innerHTML =
            '<div class="empty-hint">Sin competencias vencidas o pendientes.</div>';
        }
        check();
      })
      .withFailureHandler(function(err) {
        document.getElementById('listaPendientes').innerHTML =
          '<div class="empty-hint" style="color:#c5221f">Error: ' + esc(err.message) + '</div>';
        check();
      })
      .getCompetenciasPendientesPorPersona(personaRef.idPersona, puesto);
  } else {
    // Transversal o sin referencia: no hay pendientes por persona
    document.getElementById('listaPendientes').innerHTML =
      modoSeleccion === 'transversal'
        ? '<div class="empty-hint">Seleccioná la evaluación transversal a aplicar.</div>'
        : '<div class="empty-hint">Sin competencias vencidas.</div>';
    document.getElementById('cntPendientes').textContent = '—';
    check();
  }

  google.script.run
    .withSuccessHandler(function(data) {
      compTodas = data || [];
      renderCompLista('listaTodas', compTodas, false);
      if (compTodas.length === 0) {
        document.getElementById('listaTodas').innerHTML =
          '<div class="empty-hint">No hay competencias definidas para "' + esc(puesto) + '".</div>';
      }
      check();
    })
    .withFailureHandler(function() { check(); })
    .getCompetencias(puesto);
}

function renderCompLista(contenedorId, lista, esPendiente) {
  var cont = document.getElementById(contenedorId);
  if (!lista || lista.length === 0) return;

  cont.innerHTML = lista.map(function(c) {
    var claseTipo = '', badgePendiente = '';
    if (esPendiente) {
      if (c.diasVencida !== null && c.diasVencida !== undefined) {
        claseTipo = 'vencido';
        badgePendiente = '<span class="badge badge-vencido">Vencida ' + c.diasVencida + 'd</span>';
      } else {
        claseTipo = 'nunca-eval';
        badgePendiente = '<span class="badge badge-nunca">Nunca evaluada</span>';
      }
    }

    var badgeCrit = 'badge-' + (c.criticidad || 'media').toLowerCase();
    var badgeNat  = (c.naturaleza || '').toLowerCase().includes('tácit') ? 'badge-tacito' : 'badge-explicito';
    var badgeDoc  = c.driveId
      ? '<span class="badge badge-doc">📄 ' + esc(c.driveNombre || c.dominio) + '</span>'
      : '<span class="badge badge-sin-doc">Sin doc</span>';
    var nivelEsp  = c.nivelEsperado
      ? '<span style="font-size:11px;color:#6b7280">Nivel esp.: ' + esc(c.nivelEsperado) + '</span>'
      : '';
    var ultEval   = c.ultimaEvaluacion
      ? '<span style="font-size:11px;color:#6b7280">Última: ' + esc(c.ultimaEvaluacion) + '</span>'
      : '';

    var idComp = c.idCompetencia || (c.dominio + '|' + (c.puesto || ''));

    return '<div class="comp-item ' + claseTipo + '" id="ci-' + esc(idComp) + '"' +
           ' onclick=\'seleccionarComp(' + esc(JSON.stringify(c)) + ')\'>' +
      '<div class="comp-nombre">' + esc(c.dominio) + '</div>' +
      '<div class="comp-meta">' +
        '<span class="badge ' + badgeCrit + '">' + esc(c.criticidad) + '</span>' +
        '<span class="badge ' + badgeNat  + '">' + esc(c.naturaleza) + '</span>' +
        badgeDoc + badgePendiente + nivelEsp + ultEval +
      '</div>' +
      '<div class="comp-seleccionada-icon">✓</div>' +
    '</div>';
  }).join('');
}

  if (el) el.classList.add('seleccionado');

// ── PASO 3 — Configuración ────────────────────────────────────────
function prepararPaso3() {
  var personasSel = getPersonasSeleccionadas();
  var resPersonas;
  if (modoSeleccion === 'individual') {
    resPersonas = personaSel ? personaSel.nombre + (personaSel.puesto ? ' — ' + personaSel.puesto : '') : '—';
  } else if (modoSeleccion === 'puesto') {
    resPersonas = personasSel.length + ' personas (puesto: ' + puestoSel + ')';
  } else {
    resPersonas = personasSel.length + ' personas (evaluación transversal)';
  }

  document.getElementById('resPersonas').textContent    = resPersonas;
  document.getElementById('resCompetencia').textContent = compSel.dominio +
    (compSel.criticidad ? ' (' + compSel.criticidad + ')' : '') +
    (modoSeleccion === 'transversal' ? ' — transversal' : '');
  document.getElementById('resDocumento').textContent   = compSel.driveNombre || compSel.driveId || '— Sin documento vinculado';

  var grupoDoc = document.getElementById('grupoDocManual');
  if (!compSel.driveId) {
    grupoDoc.style.display = 'flex';
    cargarDocsDrive();
  } else {
    grupoDoc.style.display = 'none';
  }

  // Sugerir modo según naturaleza (transversal sigue la misma lógica)
  var naturaleza = (compSel.naturaleza || '').toLowerCase();
  if (naturaleza.includes('tácit')) {
    seleccionarModo('presencial');
    document.getElementById('modoSugerido').classList.add('visible');
    document.getElementById('modoSugerido').textContent = 'Sugerido: Presencial/Rubrica (competencia tacita)';
  } else {
    seleccionarModo('autonomo');
    document.getElementById('modoSugerido').classList.add('visible');
    document.getElementById('modoSugerido').textContent = 'Sugerido: Autonomo (competencia explicita)';
  }

  document.getElementById('btnGenerar').disabled = false;
}

function cargarDocsDrive() {
  var sel = document.getElementById('selectDocDrive');
  while (sel.options.length > 1) sel.remove(1);
  google.script.run
    .withSuccessHandler(function(docs) {
      docsDrive = docs || [];
      docs.forEach(function(d) {
        var o = document.createElement('option');
        o.value = d.id; o.text = d.nombre; sel.appendChild(o);
      });
    })
    .getDocumentosDrive();
}

// ── Navegación ────────────────────────────────────────────────────
function siguiente() {
  if (pasoActual === 1) {
    if (modoSeleccion === 'individual' && !personaSel) {
      mostrarAlerta('alertaPaso1', 'error', 'Seleccioná una persona primero.'); return;
    }
    if ((modoSeleccion === 'puesto' || modoSeleccion === 'transversal') && personasSelMulti.length === 0) {
      mostrarAlerta('alertaPaso1', 'error', 'Seleccioná al menos una persona.'); return;
    }
    irPaso(2);
    // Para todos los modos se cargan competencias del puesto activo
    // (transversal → puesto 'Todos', los demás → puesto de la persona/selector)
    document.getElementById('bloquePaso2Comp').style.display = 'block';
    cargarCompetencias();
  } else if (pasoActual === 2) {
    if (!compSel) { mostrarAlerta('alertaPaso2', 'error', 'Seleccioná una competencia.'); return; }
    irPaso(3);
    prepararPaso3();
  }
}

function irPaso(n) {
  if (n < 1 || n > 3) return;
  document.getElementById('paso' + pasoActual).classList.remove('visible');
  document.getElementById('paso' + n).classList.add('visible');
  pasoActual = n;

  [1, 2, 3].forEach(function(i) {
    var tab = document.getElementById('tab' + i);
    tab.classList.remove('activo', 'completado');
    if (i < pasoActual)       tab.classList.add('completado');
    else if (i === pasoActual) tab.classList.add('activo');
  });

  document.getElementById('btnVolver').style.display = pasoActual > 1 ? 'inline-block' : 'none';

  if (pasoActual === 3) {
    document.getElementById('btnSiguiente').style.display = 'none';
    document.getElementById('btnGenerar').style.display   = 'inline-block';
  } else {
    document.getElementById('btnSiguiente').style.display = 'inline-block';
    document.getElementById('btnGenerar').style.display   = 'none';
    document.getElementById('btnSiguiente').disabled = (pasoActual === 2 && !compSel);
  }
}

// ── Generar ───────────────────────────────────────────────────────
function generar() {
  var personasSel = getPersonasSeleccionadas();
  if (personasSel.length === 0 || !compSel || !modoSel) {
    mostrarAlerta('alertaPaso3', 'error', 'Faltan datos obligatorios.'); return;
  }

  var cantidad = parseInt(document.getElementById('sliderCantidad').value) || 5;

  var driveId     = compSel.driveId || '';
  var driveNombre = compSel.driveNombre || compSel.dominio;

  if (!driveId) {
    var selDoc2 = document.getElementById('selectDocDrive');
    if (selDoc2 && selDoc2.value) {
      driveId = selDoc2.value;
      driveNombre = selDoc2.options[selDoc2.selectedIndex].text;
    }
  }

  if (modoSel === 'autonomo' && !driveId) {
    mostrarAlerta('alertaPaso3', 'error',
      'Para el modo autónomo es necesario un documento Drive. Vinculá uno en la hoja Puestos_Competencias o cambiá al modo presencial.');
    return;
  }

  var params = {
    personas: personasSel.map(function(p) {
      return { idPersona: p.idPersona, nombre: p.nombre, mail: p.mail, puesto: p.puesto, area: p.area };
    }),
    competencia: {
      idCompetencia:   compSel.idCompetencia   || '',
      dominio:         compSel.dominio,
      categoria:       compSel.categoria        || '',
      naturaleza:      compSel.naturaleza        || '',
      criticidad:      compSel.criticidad        || '',
      driveId:         driveId,
      driveNombre:     driveNombre,
      frecuenciaMeses: compSel.frecuenciaMeses  || 12,
      nivelEsperado:   compSel.nivelEsperado    || ''
    },
    cantidad:    cantidad,
    modo:        modoSel,
    transversal: modoSeleccion === 'transversal'
  };

  // Bloquear UI
  document.getElementById('btnGenerar').disabled  = true;
  document.getElementById('btnVolver').disabled   = true;
  spin('spinGen', true);

  var nPersonas = personasSel.length;
  var msgEspera = modoSel === 'autonomo'
    ? '⏳ Generando preguntas con IA y creando el formulario… puede tardar unos segundos.'
    : '⏳ Registrando evaluación presencial' + (nPersonas > 1 ? ' para ' + nPersonas + ' personas' : '') + '…';
  mostrarAlerta('alertaPaso3', 'info', msgEspera);

  google.script.run
    .withSuccessHandler(function(res) {
      spin('spinGen', false);
      if (res.ok) {
        var msg = modoSel === 'autonomo'
          ? '✅ Evaluación generada para ' + nPersonas + ' persona(s). Borrador de Gmail creado.'
          : '✅ Evaluación presencial registrada para ' + nPersonas + ' persona(s).';
        mostrarAlerta('alertaPaso3', 'success', msg);
        setTimeout(function() { google.script.host.close(); }, 2200);
      } else {
        document.getElementById('btnGenerar').disabled = false;
        document.getElementById('btnVolver').disabled  = false;
        mostrarAlerta('alertaPaso3', 'error', '❌ ' + (res.error || 'Error desconocido.'));
      }
    })
    .withFailureHandler(function(err) {
      spin('spinGen', false);
      document.getElementById('btnGenerar').disabled = false;
      document.getElementById('btnVolver').disabled  = false;
      mostrarAlerta('alertaPaso3', 'error', '❌ Error: ' + err.message);
    })
    .generarEvaluacion(params);
}

// ── Utilidades ────────────────────────────────────────────────────
function esc(s) {
  return String(s || '').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}
function spin(id, on) { var el = document.getElementById(id); if (el) el.classList.toggle('visible', on); }
function mostrarAlerta(id, tipo, msg) {
  var el = document.getElementById(id); if (!el) return;
  el.className = 'alert ' + tipo; el.textContent = msg;
}
function ocultarAlerta(id) {
  var el = document.getElementById(id); if (el) { el.className = 'alert'; el.textContent = ''; }
}
