<script>
  // ── Estado ───────────────────────────────────────────────────
  var personaSel    = null;
  var competenciaSel = null;
  var idEvaluacion  = null;
  var nivelSel      = -1;
  var rubricaData   = [];

  var tagClases  = ['tag-0', 'tag-1', 'tag-2', 'tag-3'];
  var selClases  = ['sel-0', 'sel-1', 'sel-2', 'sel-3'];
  var iconosEscala = {
    Excelente: '🏆', Bueno: '✅', Regular: '⚠️', Insuficiente: '❌'
  };

  // ── Inicialización ────────────────────────────────────────────
  window.onload = function() {
    google.script.run
      .withSuccessHandler(renderPersonas)
      .withFailureHandler(function(e) { mostrarAlerta('Error cargando personas: ' + e.message, 'error'); })
      .getPersonasActivas();
  };

  // ── Paso 1: personas ──────────────────────────────────────────
  function renderPersonas(lista) {
    document.getElementById('spinnerPersonas').classList.remove('visible');
    if (!lista || lista.length === 0) {
      mostrarAlerta('No hay personas activas en el directorio. Verificá la hoja Personas.', 'error');
      return;
    }
    var grid = document.getElementById('personaGrid');
    lista.forEach(function(p) {
      var iniciales = p.nombre.split(' ').map(function(w) { return w[0] || ''; })
                      .slice(0, 2).join('').toUpperCase();
      var card = document.createElement('div');
      card.className = 'persona-card';
      card.innerHTML =
        '<div class="avatar">' + iniciales + '</div>' +
        '<div>' +
          '<div class="persona-nombre">' + p.nombre + '</div>' +
          '<div class="persona-puesto">' + (p.puesto || 'Sin puesto asignado') + '</div>' +
        '</div>';
      card.onclick = function() { seleccionarPersona(p); };
      grid.appendChild(card);
    });
    grid.style.display = 'grid';
  }

  function seleccionarPersona(p) {
    personaSel = p;
    ocultarAlerta();
    actualizarSteps(2);

    document.getElementById('step1').style.display = 'none';
    document.getElementById('step2').style.display = 'block';

    document.getElementById('personaSelBadge').innerHTML =
      '<strong>' + p.nombre + '</strong> · ' + (p.puesto || 'Sin puesto') +
      (p.area ? ' · ' + p.area : '');

    document.getElementById('spinnerComps').classList.add('visible');
    document.getElementById('compList').style.display = 'none';

    google.script.run
      .withSuccessHandler(renderCompetencias)
      .withFailureHandler(function(e) {
        document.getElementById('spinnerComps').classList.remove('visible');
        mostrarAlerta('Error cargando competencias: ' + e.message, 'error');
      })
      .getCompetenciasTacitasPendientes(p.id, p.puesto);
  }

  // ── Paso 2: competencias ──────────────────────────────────────
  function renderCompetencias(lista) {
    document.getElementById('spinnerComps').classList.remove('visible');

    if (!lista || lista.length === 0) {
      document.getElementById('sinComps').style.display = 'block';
      return;
    }

    var ul = document.getElementById('compList');
    lista.forEach(function(c) {
      var item = document.createElement('div');
      item.className = 'comp-item';

      var badgeVenc = '';
      if (c.vencida) {
        var txt = (c.diasVencida === null) ? 'Sin evaluar' : c.diasVencida + ' días vencida';
        badgeVenc = '<span class="badge badge-red">' + txt + '</span>';
      } else {
        badgeVenc = '<span class="badge badge-gray">Vigente</span>';
      }

      var badgeCrit = c.criticidad === 'Alta'
        ? '<span class="badge badge-amber">' + c.criticidad + '</span>' : '';
      var badgeNat  = '<span class="badge badge-blue">' + c.naturaleza + '</span>';

      item.innerHTML =
        '<div class="comp-row1">' +
          '<span class="comp-nombre">' + c.dominio + '</span>' +
          badgeVenc + badgeCrit + badgeNat +
        '</div>' +
        '<div class="comp-meta">' + c.categoria +
          (c.ultimaEval ? ' · Última evaluación: ' + c.ultimaEval : '') +
        '</div>';
      item.onclick = function() { seleccionarCompetencia(c); };
      ul.appendChild(item);
    });
    ul.style.display = 'flex';
  }

  function seleccionarCompetencia(c) {
    competenciaSel = c;
    nivelSel       = -1;
    ocultarAlerta();
    actualizarSteps(3);

    document.getElementById('step2').style.display = 'none';
    document.getElementById('step3').style.display = 'block';
    document.getElementById('spinnerRubrica').classList.add('visible');
    document.getElementById('rubricaContent').style.display = 'none';

    google.script.run
      .withSuccessHandler(onIniciarOk)
      .withFailureHandler(function(e) {
        document.getElementById('spinnerRubrica').classList.remove('visible');
        mostrarAlerta('Error al generar la rúbrica: ' + e.message, 'error');
      })
      .iniciarEvaluacionPresencial({ idPersona: personaSel.id, idCompetencia: c.id });
  }

  // ── Paso 3: rúbrica ───────────────────────────────────────────
  function onIniciarOk(res) {
    document.getElementById('spinnerRubrica').classList.remove('visible');

    if (!res.ok) {
      mostrarAlerta(res.error || 'Error desconocido al iniciar la evaluación.', 'error');
      return;
    }

    idEvaluacion = res.idEvaluacion;
    rubricaData  = res.rubrica;

    document.getElementById('rubricaContexto').innerHTML =
      '<strong>' + res.persona.nombre + '</strong> · ' +
      res.competencia.dominio +
      ' <span style="color:#5f6368;font-size:11px;">· ' + res.competencia.criticidad + ' · ' + res.competencia.naturaleza + '</span>';

    var grid = document.getElementById('nivelesGrid');
    grid.innerHTML = '';
    res.rubrica.forEach(function(nivel, i) {
      var card = document.createElement('div');
      card.className = 'nivel-card';
      card.id = 'nv' + i;
      card.innerHTML =
        '<span class="nivel-tag ' + tagClases[i] + '">' + nivel.nivel + '</span>' +
        '<p class="nivel-desc">' + nivel.descriptor + '</p>';
      card.onclick = function() { seleccionarNivel(i); };
      grid.appendChild(card);
    });

    document.getElementById('obsText').value = '';
    document.getElementById('btnRegistrar').disabled = true;
    document.getElementById('rubricaContent').style.display = 'block';
  }

  function seleccionarNivel(i) {
    nivelSel = i;
    [0, 1, 2, 3].forEach(function(j) {
      var el = document.getElementById('nv' + j);
      if (el) el.className = 'nivel-card' + (j === i ? ' ' + selClases[i] : '');
    });
    validarBoton();
  }

  document.addEventListener('input', function(e) {
    if (e.target && e.target.id === 'obsText') validarBoton();
  });

  function validarBoton() {
    var obs = document.getElementById('obsText').value.trim();
    document.getElementById('btnRegistrar').disabled = (nivelSel === -1 || !obs);
  }

  // ── Registro ──────────────────────────────────────────────────
  function registrar() {
    var obs = document.getElementById('obsText').value.trim();
    if (nivelSel === -1 || !obs) return;

    var nivel = rubricaData[nivelSel].nivel;

    document.getElementById('btnRegistrar').disabled = true;
    document.getElementById('spinnerReg').innerHTML =
      '<div class="spinner" style="width:18px;height:18px;border-width:2px;"></div>';

    google.script.run
      .withSuccessHandler(onRegistroOk)
      .withFailureHandler(function(e) {
        document.getElementById('spinnerReg').innerHTML = '';
        document.getElementById('btnRegistrar').disabled = false;
        mostrarAlerta('Error al registrar: ' + e.message, 'error');
      })
      .registrarEvaluacionRubrica({
        idEvaluacion:  idEvaluacion,
        nivel:         nivel,
        observaciones: obs
      });
  }

  function onRegistroOk(res) {
    document.getElementById('spinnerReg').innerHTML = '';
    if (!res.ok) {
      document.getElementById('btnRegistrar').disabled = false;
      mostrarAlerta(res.error || 'Error desconocido al registrar.', 'error');
      return;
    }

    document.getElementById('step3').style.display = 'none';
    document.getElementById('stepsBar').style.display = 'none';
    ocultarAlerta();

    var icono = iconosEscala[res.escala] || '✅';
    document.getElementById('resultadoIcon').textContent  = icono;
    document.getElementById('resultadoEscala').textContent = res.escala;
    document.getElementById('resultadoPct').textContent   = res.puntaje + '%';

    if (res.generaAccion) {
      var diasObj = res.escala === 'Insuficiente' ? 30 : 60;
      document.getElementById('avisoAccion').innerHTML =
        '📋 <strong>Plan de acción creado</strong> — objetivo en ' + diasObj +
        ' días. Revisalo en la hoja Plan_Acciones.';
      document.getElementById('avisoAccion').style.display = 'block';
    }

    var nivel    = rubricaData[nivelSel].nivel;
    var obs      = document.getElementById('obsText').value.trim();
    var filas = [
      ['Evaluado',        personaSel.nombre],
      ['Puesto',          personaSel.puesto],
      ['Competencia',     competenciaSel.dominio],
      ['Nivel asignado',  nivel + ' (' + res.puntaje + '%)'],
      ['Escala GeCO',     res.escala],
      ['Evidencia',       obs],
      ['ID Evaluación',   idEvaluacion]
    ];

    var table = document.getElementById('detalleTable');
    table.innerHTML = filas.map(function(f) {
      return '<tr><td>' + f[0] + '</td><td><strong>' + f[1] + '</strong></td></tr>';
    }).join('');

    document.getElementById('step4').style.display = 'block';
  }

  // ── Navegación ────────────────────────────────────────────────
  function volverAlPaso(n) {
    ocultarAlerta();
    if (n === 1) {
      document.getElementById('step2').style.display = 'none';
      document.getElementById('step1').style.display = 'block';
      actualizarSteps(1);
    } else if (n === 2) {
      nivelSel = -1;
      document.getElementById('step3').style.display = 'none';
      document.getElementById('step2').style.display = 'block';
      actualizarSteps(2);
    }
  }

  function nuevaEvaluacion() {
    personaSel    = null;
    competenciaSel = null;
    idEvaluacion  = null;
    nivelSel      = -1;
    rubricaData   = [];

    document.getElementById('step4').style.display = 'none';
    document.getElementById('stepsBar').style.display = 'flex';
    document.getElementById('nivelesGrid').innerHTML = '';
    document.getElementById('compList').innerHTML    = '';
    document.getElementById('personaGrid').innerHTML = '';
    document.getElementById('personaGrid').style.display = 'none';
    document.getElementById('spinnerPersonas').classList.add('visible');
    actualizarSteps(1);
    document.getElementById('step1').style.display = 'block';

    google.script.run
      .withSuccessHandler(renderPersonas)
      .withFailureHandler(function(e) { mostrarAlerta('Error: ' + e.message, 'error'); })
      .getPersonasActivas();
  }

  // ── Helpers ───────────────────────────────────────────────────
  function actualizarSteps(paso) {
    [1, 2, 3].forEach(function(n) {
      var el = document.getElementById('si' + n);
      el.className = 'step-item' +
        (n < paso ? ' done' : n === paso ? ' active' : '');
    });
  }

  function mostrarAlerta(msg, tipo) {
    var el = document.getElementById('alertMsg');
    el.textContent = msg;
    el.className = 'alert visible alert-' + tipo;
  }

  function ocultarAlerta() {
    document.getElementById('alertMsg').className = 'alert';
  }
