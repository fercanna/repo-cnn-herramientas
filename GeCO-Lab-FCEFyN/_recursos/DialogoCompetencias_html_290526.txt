<script>
// ── Estado global ───────────────────────────────────────────────────
var competencias = [];   // array completo desde servidor
var competenciasFiltradas = [];
var modoEdicion = null;  // null = nueva, string idCompetencia = edición
var documentosDrive = []; // cache de archivos Drive

// ── Init ────────────────────────────────────────────────────────────
window.onload = function() {
  cargarCompetencias();
  cargarDocumentosDrive();
};

// ── Carga de datos ──────────────────────────────────────────────────
function cargarCompetencias() {
  spin('spinMain', true);
  google.script.run
    .withSuccessHandler(function(data) {
      spin('spinMain', false);
      competencias = data || [];
      llenarFiltroPuesto();
      aplicarFiltros();
    })
    .withFailureHandler(function(err) {
      spin('spinMain', false);
      mostrarToast('Error al cargar: ' + err.message, 'error');
    })
    .getCompetencias();
}

function cargarDocumentosDrive() {
  spin('spinDocs', true);
  google.script.run
    .withSuccessHandler(function(docs) {
      spin('spinDocs', false);
      documentosDrive = docs || [];
      poblarSelectDrive(docs);
    })
    .withFailureHandler(function() {
      spin('spinDocs', false);
    })
    .buscarDocumentosDrive();
}

function poblarSelectDrive(docs) {
  var sel = document.getElementById('fDriveDoc');
  // conservar la primera opción vacía
  while (sel.options.length > 1) sel.remove(1);
  docs.forEach(function(d) {
    var o = document.createElement('option');
    o.value = d.id;
    o.text  = d.nombre;
    sel.appendChild(o);
  });
}

// ── Filtros ─────────────────────────────────────────────────────────
function llenarFiltroPuesto() {
  var puestos = {};
  competencias.forEach(function(c) { puestos[c.puesto] = true; });
  var sel = document.getElementById('filtroPuesto');
  while (sel.options.length > 1) sel.remove(1);
  Object.keys(puestos).sort().forEach(function(p) {
    var o = document.createElement('option');
    o.value = p; o.text = p;
    sel.appendChild(o);
  });

  // datalist para el formulario
  var dl = document.getElementById('listaPuestos');
  dl.innerHTML = '';
  Object.keys(puestos).sort().forEach(function(p) {
    var o = document.createElement('option'); o.value = p; dl.appendChild(o);
  });
}

function aplicarFiltros() {
  var puesto     = document.getElementById('filtroPuesto').value.toLowerCase();
  var categoria  = document.getElementById('filtroCategoria').value.toLowerCase();
  var criticidad = document.getElementById('filtroCriticidad').value.toLowerCase();
  var naturaleza = document.getElementById('filtroNaturaleza').value.toLowerCase();
  var busq       = document.getElementById('buscador').value.toLowerCase();

  competenciasFiltradas = competencias.filter(function(c) {
    if (puesto     && c.puesto.toLowerCase()     !== puesto)     return false;
    if (categoria  && c.categoria.toLowerCase()  !== categoria)  return false;
    if (criticidad && c.criticidad.toLowerCase() !== criticidad) return false;
    if (naturaleza && c.naturaleza.toLowerCase() !== naturaleza) return false;
    if (busq && !(
      c.puesto.toLowerCase().includes(busq)   ||
      c.dominio.toLowerCase().includes(busq)  ||
      c.driveNombre.toLowerCase().includes(busq)
    )) return false;
    return true;
  });

  renderTabla();
  actualizarStats();
}

// ── Render tabla ─────────────────────────────────────────────────────
function renderTabla() {
  var tbody = document.getElementById('tbodyComp');

  if (competenciasFiltradas.length === 0) {
    tbody.innerHTML = '<tr><td colspan="10" class="empty">No se encontraron competencias con esos filtros.</td></tr>';
    document.getElementById('btnBaja').disabled = true;
    document.getElementById('chkTodos').checked = false;
    return;
  }

  tbody.innerHTML = competenciasFiltradas.map(function(c) {
    var badgeCrit = 'badge-' + c.criticidad.toLowerCase();
    var badgeNat  = c.naturaleza === 'Tácito' ? 'badge-tacito' : 'badge-explicito';
    var docBadge  = c.driveId
      ? '<span class="badge badge-con-doc">✓ ' + esc(c.driveNombre) + '</span>'
      : '<span class="badge badge-sin-doc">Sin doc</span>';
    var iconoActivo = c.activo
      ? '<span class="ico-activo" title="Activa">●</span>'
      : '<span class="ico-inactivo" title="Inactiva">○</span>';
    var freq = c.frecuenciaMeses === 0 ? 'Sin venc.' : c.frecuenciaMeses + ' m';

    return '<tr id="row-' + esc(c.idCompetencia) + '">' +
      '<td style="text-align:center"><input type="checkbox" class="chk-fila" value="' + esc(c.idCompetencia) + '" onchange="onCheckFila()"></td>' +
      '<td>' + esc(c.puesto) + '</td>' +
      '<td>' + esc(c.dominio) + '</td>' +
      '<td>' + esc(c.categoria) + '</td>' +
      '<td><span class="badge ' + badgeNat + '">' + esc(c.naturaleza) + '</span></td>' +
      '<td><span class="badge ' + badgeCrit + '">' + esc(c.criticidad) + '</span></td>' +
      '<td style="text-align:center">' + freq + '</td>' +
      '<td>' + docBadge + '</td>' +
      '<td style="text-align:center">' + iconoActivo + '</td>' +
      '<td><button class="btn btn-gray" style="padding:3px 8px;font-size:11px" onclick="editarCompetencia(\'' + esc(c.idCompetencia) + '\')">Editar</button></td>' +
    '</tr>';
  }).join('');
}

function actualizarStats() {
  var data = competenciasFiltradas;
  document.getElementById('statTotal').textContent   = data.length;
  document.getElementById('statAlta').textContent    = data.filter(function(c){ return c.criticidad === 'Alta'; }).length;
  document.getElementById('statTacito').textContent  = data.filter(function(c){ return c.naturaleza === 'Tácito'; }).length;
  document.getElementById('statSinDoc').textContent  = data.filter(function(c){ return !c.driveId; }).length;
}

// ── Checkboxes ───────────────────────────────────────────────────────
function toggleTodos(chk) {
  document.querySelectorAll('.chk-fila').forEach(function(c) { c.checked = chk.checked; });
  onCheckFila();
}

function onCheckFila() {
  var sel = document.querySelectorAll('.chk-fila:checked');
  document.getElementById('btnBaja').disabled = sel.length === 0;
  document.getElementById('chkTodos').checked = sel.length > 0 && sel.length === document.querySelectorAll('.chk-fila').length;
}

// ── Panel formulario ─────────────────────────────────────────────────
function mostrarFormNueva() {
  modoEdicion = null;
  document.getElementById('panelTitulo').textContent = 'Nueva competencia';
  limpiarFormPanel();
  document.getElementById('panelOverlay').classList.add('visible');
}

function editarCompetencia(idCompetencia) {
  var comp = competencias.find(function(c) { return c.idCompetencia === idCompetencia; });
  if (!comp) return;
  modoEdicion = idCompetencia;
  document.getElementById('panelTitulo').textContent = 'Editar competencia';
  cargarEnFormPanel(comp);
  document.getElementById('panelOverlay').classList.add('visible');
}

function cerrarPanel() {
  document.getElementById('panelOverlay').classList.remove('visible');
}

function limpiarFormPanel() {
  document.getElementById('fPuesto').value    = '';
  document.getElementById('fDominio').value   = '';
  document.getElementById('fCategoria').value = 'Técnico';
  document.getElementById('fNaturaleza').value = 'Explícito';
  document.getElementById('fCriticidad').value = 'Alta';
  document.getElementById('fFrecuencia').value = '12';
  document.getElementById('fDriveDoc').value   = '';
  document.getElementById('fDriveNombre').value = '';
}

function cargarEnFormPanel(comp) {
  document.getElementById('fPuesto').value    = comp.puesto;
  document.getElementById('fDominio').value   = comp.dominio;
  document.getElementById('fCategoria').value = comp.categoria;
  document.getElementById('fNaturaleza').value = comp.naturaleza;
  document.getElementById('fCriticidad').value = comp.criticidad;
  document.getElementById('fFrecuencia').value = String(comp.frecuenciaMeses);
  document.getElementById('fDriveDoc').value   = comp.driveId || '';
  document.getElementById('fDriveNombre').value = comp.driveNombre || '';
}

function onDriveDocChange() {
  var sel  = document.getElementById('fDriveDoc');
  var id   = sel.value;
  var nombre = id ? (sel.options[sel.selectedIndex].text) : '';
  document.getElementById('fDriveNombre').value = nombre;
}

function guardarCompetencia() {
  var puesto  = document.getElementById('fPuesto').value.trim();
  var dominio = document.getElementById('fDominio').value.trim();

  if (!puesto || !dominio) {
    mostrarToast('Puesto y Dominio son obligatorios.', 'error');
    return;
  }

  var datos = {
    puesto:          puesto,
    dominio:         dominio,
    categoria:       document.getElementById('fCategoria').value,
    naturaleza:      document.getElementById('fNaturaleza').value,
    criticidad:      document.getElementById('fCriticidad').value,
    frecuenciaMeses: parseInt(document.getElementById('fFrecuencia').value) || 12,
    driveId:         document.getElementById('fDriveDoc').value,
    driveNombre:     document.getElementById('fDriveNombre').value.trim() || dominio
  };

  spin('spinPanel', true);
  document.getElementById('btnGuardar').disabled = true;

  if (modoEdicion) {
    // Edición
    google.script.run
      .withSuccessHandler(function(res) {
        spin('spinPanel', false);
        document.getElementById('btnGuardar').disabled = false;
        if (res.ok) {
          mostrarToast('Competencia actualizada correctamente.', 'ok');
          cerrarPanel();
          cargarCompetencias();
        } else {
          mostrarToast('Error: ' + res.error, 'error');
        }
      })
      .withFailureHandler(function(err) {
        spin('spinPanel', false);
        document.getElementById('btnGuardar').disabled = false;
        mostrarToast('Error: ' + err.message, 'error');
      })
      .modificarCompetencia(modoEdicion, datos);
  } else {
    // Alta nueva
    google.script.run
      .withSuccessHandler(function(res) {
        spin('spinPanel', false);
        document.getElementById('btnGuardar').disabled = false;
        if (res.ok) {
          mostrarToast('Competencia agregada: ' + res.idCompetencia, 'ok');
          cerrarPanel();
          cargarCompetencias();
        } else {
          mostrarToast('Error: ' + res.error, 'error');
        }
      })
      .withFailureHandler(function(err) {
        spin('spinPanel', false);
        document.getElementById('btnGuardar').disabled = false;
        mostrarToast('Error: ' + err.message, 'error');
      })
      .agregarCompetencia(datos);
  }
}

// ── Desactivar seleccionadas ─────────────────────────────────────────
function desactivarSeleccionadas() {
  var ids = Array.from(document.querySelectorAll('.chk-fila:checked')).map(function(c) { return c.value; });
  if (ids.length === 0) return;
  if (!confirm('¿Desactivar ' + ids.length + ' competencia(s) seleccionada(s)? Se pueden reactivar manualmente en la hoja.')) return;

  spin('spinMain', true);
  var promesas = ids.length;
  var errores  = 0;

  ids.forEach(function(id) {
    google.script.run
      .withSuccessHandler(function(res) {
        if (!res.ok) errores++;
        promesas--;
        if (promesas === 0) {
          spin('spinMain', false);
          if (errores > 0) mostrarToast('Se produjeron ' + errores + ' error(es).', 'error');
          else mostrarToast(ids.length + ' competencia(s) desactivada(s).', 'ok');
          cargarCompetencias();
        }
      })
      .withFailureHandler(function() {
        errores++; promesas--;
        if (promesas === 0) {
          spin('spinMain', false);
          mostrarToast('Se produjeron errores.', 'error');
          cargarCompetencias();
        }
      })
      .desactivarCompetencia(id);
  });
}

// ── Importar CSV ─────────────────────────────────────────────────────
function toggleCSV() {
  var panel = document.getElementById('csvPanel');
  panel.classList.toggle('visible');
  document.getElementById('csvResult').style.display = 'none';
  document.getElementById('csvError').style.display  = 'none';
}

function leerArchivoCSV(event) {
  var file = event.target.files[0];
  if (!file) return;
  var reader = new FileReader();
  reader.onload = function(e) {
    document.getElementById('csvTexto').value = e.target.result;
  };
  reader.readAsText(file, 'UTF-8');
}

function importarCSV() {
  var texto = document.getElementById('csvTexto').value.trim();
  if (!texto) { mostrarToast('Pegá o cargá un CSV primero.', 'error'); return; }

  var resEl = document.getElementById('csvResult');
  var errEl = document.getElementById('csvError');
  resEl.style.display = 'none';
  errEl.style.display = 'none';

  spin('spinCSV', true);

  google.script.run
    .withSuccessHandler(function(res) {
      spin('spinCSV', false);
      if (!res.ok) {
        errEl.textContent  = 'Error: ' + res.error;
        errEl.style.display = 'inline-block';
        return;
      }
      var msg = '✓ ' + res.agregadas + ' agregadas';
      if (res.saltadas > 0) msg += ' · ' + res.saltadas + ' saltadas';
      resEl.textContent  = msg;
      resEl.style.display = 'inline-block';
      if (res.errores && res.errores.length > 0) {
        errEl.textContent  = res.errores.join(' | ');
        errEl.style.display = 'inline-block';
      }
      cargarCompetencias();
    })
    .withFailureHandler(function(err) {
      spin('spinCSV', false);
      errEl.textContent  = 'Error: ' + err.message;
      errEl.style.display = 'inline-block';
    })
    .importarCompetenciasDesdeCSV(texto);
}

function esc(s) {
  return String(s || '').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}

function spin(id, on) {
  var el = document.getElementById(id);
  if (el) el.classList.toggle('visible', on);
}

function mostrarToast(msg, tipo) {
  var t = document.getElementById('toast');
  t.textContent = msg;
  t.className   = 'toast ' + tipo + ' show';
  setTimeout(function() { t.classList.remove('show'); }, 3500);
}
