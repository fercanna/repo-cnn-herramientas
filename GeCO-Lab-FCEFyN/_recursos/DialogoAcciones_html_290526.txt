<script>
// ── Estado global ─────────────────────────────────────────────────
var planes = [];         // todos los planes cargados del servidor
var planesFiltrados = [];
var planActual = null;   // plan abierto en el panel

// ── Init ──────────────────────────────────────────────────────────
window.onload = function() {
  cargarPlanes();
};

// ── Carga ─────────────────────────────────────────────────────────
function cargarPlanes() {
  spin('spinMain', true);
  // Cargar todos los planes y el resumen KPI en paralelo
  var cargados = 0;
  var kpiData  = null;
  var planData = null;

  function check() {
    cargados++;
    if (cargados < 2) return;
    spin('spinMain', false);
    planes = planData || [];
    aplicarFiltros();
    if (kpiData) mostrarKPI(kpiData);
  }

  google.script.run
    .withSuccessHandler(function(data) { planData = data; check(); })
    .withFailureHandler(function(err)  { spin('spinMain', false); mostrarToast('Error: ' + err.message, 'error'); })
    .getPlanesActivos();   // carga activos con diasRestantes calculados

  google.script.run
    .withSuccessHandler(function(data) { kpiData = data; check(); })
    .withFailureHandler(function()     { check(); })
    .getResumenPlanesAccion();
}

function mostrarKPI(data) {
  document.getElementById('kpiVencidos').textContent   = data.vencidos      || 0;
  document.getElementById('kpiPendientes').textContent = data.pendientes     || 0;
  document.getElementById('kpiEnCurso').textContent    = data.enCurso        || 0;
  document.getElementById('kpiCerrados').textContent   = data.cerrados       || 0;
  document.getElementById('kpiPorVencer').textContent  = data.porVencer7dias || 0;
}

// ── Filtros ───────────────────────────────────────────────────────
function aplicarFiltros() {
  var estado    = document.getElementById('filtroEstado').value;
  var tipo      = document.getElementById('filtroTipo').value.toLowerCase();
  var busq      = document.getElementById('buscador').value.toLowerCase();
  var soloVenc  = document.getElementById('soloVencidos').checked;

  planesFiltrados = planes.filter(function(p) {
    if (estado === 'activos' && p.estado === 'Cerrado') return false;
    if (estado !== '' && estado !== 'activos' && p.estado !== estado) return false;
    if (tipo   && p.tipoAccion.toLowerCase() !== tipo) return false;
    if (busq   && !(p.evaluado.toLowerCase().includes(busq) || p.dominio.toLowerCase().includes(busq))) return false;
    if (soloVenc && (p.diasRestantes === null || p.diasRestantes >= 0)) return false;
    return true;
  });

  renderTabla();
}

// ── Render ────────────────────────────────────────────────────────
function renderTabla() {
  var tbody = document.getElementById('tbodyPlanes');
  if (planesFiltrados.length === 0) {
    tbody.innerHTML = '<tr><td colspan="10" class="empty">No hay planes que coincidan con los filtros.</td></tr>';
    return;
  }

  tbody.innerHTML = planesFiltrados.map(function(p) {
    var badgeEstado = 'badge estado-' + p.estado.toLowerCase().replace(' ', '-');
    var badgeEscala = 'badge escala-' + p.escala.toLowerCase();
    var diasEl = renderDias(p);
    var reevalEl = p.idReevaluacion
      ? '<span style="font-size:11px;color:#1a56db" title="' + esc(p.resultadoFinal) + '">✓ Vinculada</span>'
      : '<span style="color:#d1d5db;font-size:11px">—</span>';

    return '<tr>' +
      '<td style="font-size:11px;font-family:monospace">' + esc(p.idAccion) + '</td>' +
      '<td>' + esc(p.evaluado) + '</td>' +
      '<td class="text-truncate" title="' + esc(p.dominio) + '">' + esc(p.dominio) + '</td>' +
      '<td><span class="' + badgeEscala + '">' + esc(p.escala) + '</span></td>' +
      '<td>' + esc(p.tipoAccion || '—') + '</td>' +
      '<td style="font-size:11px">' + esc(p.responsable || '—') + '</td>' +
      '<td>' + diasEl + '</td>' +
      '<td><span class="' + badgeEstado + '">' + esc(p.estado) + '</span></td>' +
      '<td style="text-align:center">' + reevalEl + '</td>' +
      '<td><button class="btn btn-blue" style="padding:3px 8px;font-size:11px" onclick="abrirPanel(\'' + esc(p.idAccion) + '\')">Ver</button></td>' +
    '</tr>';
  }).join('');
}

function renderDias(p) {
  if (p.estado === 'Cerrado') {
    return '<span class="dias-ok" style="font-size:11px">Cerrado ' + (p.fechaCierre || '') + '</span>';
  }
  if (p.diasRestantes === null) return '<span class="dias-ninguno">—</span>';
  if (p.diasRestantes < 0) {
    return '<span class="dias-vencido">⚠ Vencido ' + Math.abs(p.diasRestantes) + 'd</span>';
  }
  if (p.diasRestantes <= 7) {
    return '<span class="dias-urgente">⏰ ' + p.diasRestantes + 'd</span>';
  }
  return '<span class="dias-ok">' + p.fechaObjetivo + '</span>';
}

// ── Panel ─────────────────────────────────────────────────────────
function abrirPanel(idAccion) {
  planActual = planes.find(function(p) { return p.idAccion === idAccion; });
  if (!planActual) return;

  document.getElementById('panelTitulo').textContent = planActual.idAccion;

  // Info de solo lectura
  document.getElementById('infoTexto').innerHTML =
    '<b>Evaluado:</b> ' + esc(planActual.evaluado) + '<br>' +
    '<b>Dominio:</b> '  + esc(planActual.dominio)  + '<br>' +
    '<b>Puntaje:</b> '  + planActual.puntajePct + '% — <b>' + esc(planActual.escala) + '</b><br>' +
    '<b>Evaluación:</b> ' + esc(planActual.idEvaluacion) + '<br>' +
    '<b>Generado:</b> ' + esc(planActual.fechaGeneracion);

  // Campos editables
  document.getElementById('pEstado').value      = planActual.estado;
  document.getElementById('pTipo').value        = planActual.tipoAccion    || 'Capacitación';
  document.getElementById('pDescripcion').value = planActual.descripcion   || '';
  document.getElementById('pResponsable').value = planActual.responsable   || '';
  document.getElementById('pNotas').value       = planActual.notas         || '';
  document.getElementById('pIdReeval').value    = planActual.idReevaluacion || '';
  document.getElementById('pResultado').value   = planActual.resultadoFinal || '';

  // Fecha objetivo: convertir dd/MM/yyyy → yyyy-MM-dd para input[type=date]
  var fo = planActual.fechaObjetivo ? parseFecha(planActual.fechaObjetivo) : '';
  document.getElementById('pFechaObj').value = fo;

  onEstadoChange();
  document.getElementById('panelOverlay').classList.add('visible');
}

function cerrarPanel() {
  document.getElementById('panelOverlay').classList.remove('visible');
  planActual = null;
}

function onEstadoChange() {
  var estado = document.getElementById('pEstado').value;
  document.getElementById('cierreRapido').classList.toggle('visible', estado === 'Cerrado');
}

function guardarCambios() {
  if (!planActual) return;

  var cambios = {
    estado:      document.getElementById('pEstado').value,
    tipoAccion:  document.getElementById('pTipo').value,
    descripcion: document.getElementById('pDescripcion').value.trim(),
    responsable: document.getElementById('pResponsable').value.trim(),
    notas:       document.getElementById('pNotas').value.trim()
  };

  // Fecha objetivo: yyyy-MM-dd → Date object (se pasa como string ISO)
  var foStr = document.getElementById('pFechaObj').value;
  if (foStr) cambios.fechaObjetivo = new Date(foStr + 'T12:00:00');

  spin('spinPanel', true);
  document.getElementById('btnGuardar').disabled = true;

  google.script.run
    .withSuccessHandler(function(res) {
      spin('spinPanel', false);
      document.getElementById('btnGuardar').disabled = false;
      if (res.ok) {
        mostrarToast('Plan actualizado correctamente.', 'ok');
        cerrarPanel();
        cargarPlanes();
      } else {
        mostrarToast('Error: ' + res.error, 'error');
      }
    })
    .withFailureHandler(function(err) {
      spin('spinPanel', false);
      document.getElementById('btnGuardar').disabled = false;
      mostrarToast('Error: ' + err.message, 'error');
    })
    .actualizarPlanAccion(planActual.idAccion, cambios);
}

// ── Utilidades ────────────────────────────────────────────────────
function parseFecha(ddMMyyy) {
  // "24/01/2026" → "2026-01-24"
  var partes = ddMMyyy.split('/');
  if (partes.length !== 3) return '';
  return partes[2] + '-' + partes[1] + '-' + partes[0];
}

function esc(s) {
  return String(s || '').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}

function spin(id, on) {
  var el = document.getElementById(id);
  if (el) el.classList.toggle('visible', on);
}

function mostrarToast(msg, tipo) {
  var t = document.getElementById('toast');
  t.textContent = msg;
  t.className   = 'toast ' + tipo + ' show';
  setTimeout(function() { t.classList.remove('show'); }, 3500);
}
