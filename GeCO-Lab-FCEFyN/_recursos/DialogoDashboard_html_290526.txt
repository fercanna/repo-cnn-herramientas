// ── Init ──────────────────────────────────────────────────────────
window.onload = function() { cargar(); };

function recargar() { cargar(); }

function cargar() {
  document.getElementById('loader').classList.remove('hidden');
  google.script.run
    .withSuccessHandler(function(datos) {
      document.getElementById('loader').classList.add('hidden');
      renderDashboard(datos);
    })
    .withFailureHandler(function(err) {
      document.getElementById('loader').classList.add('hidden');
      document.getElementById('dashBody').innerHTML =
        '<div style="padding:20px;color:#f05252;font-weight:700">Error al cargar: ' + esc(err.message) + '</div>';
    })
    .getDatosDashboard();
}

// ── Render principal ──────────────────────────────────────────────
function renderDashboard(d) {
  document.getElementById('timestamp').textContent = 'Actualizado: ' + new Date().toLocaleTimeString('es-AR');

  var html = '';

  // KPI fila 1
  html += renderKPIs(d.kpis);

  // Alertas
  html += renderAlertas(d.alertas);

  // Estado personas + Distribución (lado a lado si caben)
  html += '<div style="display:grid;grid-template-columns:1fr 1fr;gap:16px">';
  html += renderDistribucion(d.distribucionEscalas, d.resumenPlanes);
  html += renderTendencia(d.tendenciaMensual);
  html += '</div>';

  html += renderPersonas(d.estadoPersonas);
  html += renderRecientes(d.ultimasEvaluaciones);

  document.getElementById('dashBody').innerHTML = html;
}

// ── KPIs ──────────────────────────────────────────────────────────
function renderKPIs(k) {
  if (!k) return '';
  var html = '<div class="kpi-row">';
  html += kpiCard('Total eval.', k.totalEvaluaciones, 'c-blue', 'históricas');
  html += kpiCard('Completadas', k.completadas, 'c-green', k.tasaCompletado + '% completado');
  html += kpiCard('Pendientes', k.pendientes, 'c-gray', k.enCurso + ' en curso');
  html += kpiCard('Planes activos', k.planesActivos, 'c-orange', k.planesVencidos + ' vencidos');
  html += kpiCard('Cobertura', k.personasEvaluadas + '/' + k.personasTotal, 'c-blue', 'personas con al menos 1 eval.');
  html += '</div>';
  return html;
}

function kpiCard(label, valor, colorCls, sub) {
  return '<div class="kpi-card">' +
    '<div class="kpi-label">' + esc(label) + '</div>' +
    '<div class="kpi-valor ' + colorCls + '">' + esc(String(valor)) + '</div>' +
    (sub ? '<div class="kpi-sub">' + esc(sub) + '</div>' : '') +
    '</div>';
}

// ── Alertas ───────────────────────────────────────────────────────
function renderAlertas(alertas) {
  var html = '<div class="seccion sec-alertas">' +
    '<div class="sec-header">⚠️ Alertas operacionales' +
    (alertas && alertas.length ? ' <span style="background:rgba(0,0,0,.12);border-radius:10px;padding:1px 7px;font-size:11px">' + alertas.length + '</span>' : '') +
    '</div><div class="sec-body">';

  if (!alertas || alertas.length === 0) {
    html += '<div class="sin-alertas">✅ Sin alertas activas — todo en orden</div>';
  } else {
    html += '<div class="alerta-lista">';
    alertas.forEach(function(a) {
      html += '<div class="alerta-item ' + esc(a.nivel) + '">' +
        '<div class="alerta-titulo">' + esc(a.titulo) + '</div>' +
        (a.detalle ? '<div class="alerta-detalle">' + esc(a.detalle) + '</div>' : '') +
        '</div>';
    });
    html += '</div>';
  }

  html += '</div></div>';
  return html;
}

// ── Distribución ──────────────────────────────────────────────────
function renderDistribucion(dist, planes) {
  var html = '<div class="seccion sec-dist">' +
    '<div class="sec-header">📊 Resultados</div>' +
    '<div class="sec-body">';

  if (!dist || dist.total === 0) {
    html += '<div class="empty">Sin evaluaciones completadas aún.</div>';
  } else {
    html += '<div class="dist-row">';
    [
      { key: 'Excelente',    cls: 'd-excelente' },
      { key: 'Bueno',        cls: 'd-bueno' },
      { key: 'Regular',      cls: 'd-regular' },
      { key: 'Insuficiente', cls: 'd-insuficiente' }
    ].forEach(function(e) {
      var n   = dist[e.key] || 0;
      var pct = dist.total > 0 ? Math.round((n / dist.total) * 100) : 0;
      html += '<div class="dist-card ' + e.cls + '">' +
        '<div class="dist-num">' + n + '</div>' +
        '<div class="dist-pct">' + pct + '%</div>' +
        '<div class="dist-label">' + e.key + '</div>' +
        '</div>';
    });
    html += '</div>';

    // Mini resumen planes
    if (planes) {
      html += '<div style="margin-top:12px;font-size:11px;color:#64748b;display:flex;gap:14px;flex-wrap:wrap">' +
        '<span>Planes pendientes: <strong style="color:#d97706">' + (planes.pendientes || 0) + '</strong></span>' +
        '<span>En curso: <strong style="color:#1a56db">' + (planes.enCurso || 0) + '</strong></span>' +
        '<span>Cerrados: <strong style="color:#0e9f6e">' + (planes.cerrados || 0) + '</strong></span>' +
        '</div>';
    }
  }
  html += '</div></div>';
  return html;
}

// ── Tendencia ─────────────────────────────────────────────────────
function renderTendencia(tend) {
  var html = '<div class="seccion sec-tendencia">' +
    '<div class="sec-header">📈 Tendencia mensual</div>' +
    '<div class="sec-body tendencia-chart">';

  if (!tend || tend.length === 0) {
    html += '<div class="empty">Sin datos suficientes.</div>';
  } else {
    var maxVal = 1;
    tend.forEach(function(t) { if (t.completadas > maxVal) maxVal = t.completadas; });

    html += '<div class="tend-bars">';
    tend.forEach(function(t) {
      var h = maxVal > 0 ? Math.round((t.completadas / maxVal) * 70) : 0;
      var ha = t.conAccion && maxVal > 0 ? Math.round((t.conAccion / maxVal) * 70) : 0;
      html += '<div class="tend-col">' +
        '<div class="tend-num">' + t.completadas + '</div>' +
        '<div class="tend-bar-wrap">' +
          '<div class="tend-bar completadas" style="height:' + h + 'px"></div>' +
        '</div>' +
        '<div class="tend-label">' + esc(t.label) + '</div>' +
        '</div>';
    });
    html += '</div>';

    html += '<div class="tend-leyenda">' +
      '<div class="tend-leyenda-item"><div class="tend-punto" style="background:#1a56db"></div> Completadas</div>' +
      '</div>';
  }
  html += '</div></div>';
  return html;
}

// ── Estado personas ───────────────────────────────────────────────
function renderPersonas(personas) {
  var html = '<div class="seccion sec-personas">' +
    '<div class="sec-header">👥 Estado de competencias por persona</div>' +
    '<div class="sec-body">';

  if (!personas || personas.length === 0) {
    html += '<div class="empty">No hay datos de personas o competencias configuradas.</div>';
  } else {
    html += '<table class="tabla-personas">' +
      '<thead><tr>' +
        '<th>Persona</th><th>Puesto</th>' +
        '<th style="text-align:center">Total</th>' +
        '<th style="text-align:center">Al día</th>' +
        '<th style="text-align:center">Vencidas</th>' +
        '<th style="text-align:center">Sin eval.</th>' +
        '<th>Cobertura</th>' +
      '</tr></thead><tbody>';

    personas.forEach(function(p) {
      var pct = p.total > 0 ? Math.round((p.alDia / p.total) * 100) : 0;
      var barCls = pct >= 80 ? 'verde' : pct >= 50 ? 'ambar' : 'rojo';
      var pctColor = pct >= 80 ? '#0e9f6e' : pct >= 50 ? '#d97706' : '#f05252';

      html += '<tr>' +
        '<td><strong>' + esc(p.nombre) + '</strong></td>' +
        '<td style="color:#64748b">' + esc(p.puesto) + '</td>' +
        '<td style="text-align:center">' + p.total + '</td>' +
        '<td style="text-align:center;color:#0e9f6e;font-weight:700">' + p.alDia + '</td>' +
        '<td style="text-align:center">' + (p.vencidas > 0 ? '<span class="badge-vencida">' + p.vencidas + '</span>' : '<span style="color:#94a3b8">—</span>') + '</td>' +
        '<td style="text-align:center">' + (p.pendientes > 0 ? '<span class="badge-nunca">' + p.pendientes + '</span>' : '<span style="color:#94a3b8">—</span>') + '</td>' +
        '<td><div class="bar-cobertura">' +
          '<div class="bar-track"><div class="bar-fill ' + barCls + '" style="width:' + pct + '%"></div></div>' +
          '<div class="bar-pct" style="color:' + pctColor + '">' + pct + '%</div>' +
        '</div></td>' +
      '</tr>';
    });
    html += '</tbody></table>';
  }
  html += '</div></div>';
  return html;
}

// ── Últimas evaluaciones ──────────────────────────────────────────
function renderRecientes(evals) {
  var html = '<div class="seccion sec-recientes">' +
    '<div class="sec-header">📋 Últimas evaluaciones</div>' +
    '<div class="sec-body">';

  if (!evals || evals.length === 0) {
    html += '<div class="empty">Sin evaluaciones registradas.</div>';
  } else {
    html += '<table class="tabla-eval"><thead><tr>' +
      '<th>Fecha</th><th>Evaluado</th><th>Dominio</th>' +
      '<th>Modo</th><th>Puntaje</th><th>Escala</th><th>Estado</th><th>Plan</th>' +
      '</tr></thead><tbody>';

    evals.forEach(function(e) {
      var escCls = e.escala ? 'esc-' + e.escala : 'esc-pendiente';
      var esc_label = e.escala || '—';
      var estadoCls = e.estado === 'Completado' ? 'estado-comp' : 'estado-pend';
      var domCorto  = e.dominio.length > 32 ? e.dominio.substring(0, 32) + '…' : e.dominio;

      html += '<tr>' +
        '<td style="font-size:11px;color:#64748b">' + esc(e.fecha) + '</td>' +
        '<td><strong>' + esc(e.evaluado) + '</strong></td>' +
        '<td title="' + esc(e.dominio) + '" style="color:#64748b">' + esc(domCorto) + '</td>' +
        '<td style="font-size:11px">' + esc(e.modo) + '</td>' +
        '<td style="text-align:center;font-weight:700">' + (e.puntaje !== null ? e.puntaje + '%' : '—') + '</td>' +
        '<td><span class="esc-badge ' + escCls + '">' + esc(esc_label) + '</span></td>' +
        '<td class="' + estadoCls + '">' + esc(e.estado) + '</td>' +
        '<td>' + (e.generaAccion ? '<span class="accion-si">⚡ Sí</span>' : '<span style="color:#94a3b8;font-size:11px">—</span>') + '</td>' +
        '</tr>';
    });
    html += '</tbody></table>';
  }
  html += '</div></div>';
  return html;
}

// ── Utils ─────────────────────────────────────────────────────────
function esc(s) {
  return String(s || '').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}
