/**
 * Devuelve personas activas desde la hoja Personas.
 * Columnas nuevas: A=ID_Persona, B=Nombre, C=Mail, D=Puesto, E=Área
 * @returns {Array} [{ idPersona, nombre, mail, puesto, area }]
 */
function getPersonas() {
  const ss   = SpreadsheetApp.getActiveSpreadsheet();
  const hoja = ss.getSheetByName('Personas');
  if (!hoja || hoja.getLastRow() < 2) return [];

  const datos    = hoja.getDataRange().getValues();
  const personas = [];

  for (let i = 1; i < datos.length; i++) {
    // A=ID_Persona(0), B=Nombre(1), C=Mail(2), D=Puesto(3), E=Área(4), F=Activo(5)
    const activo = datos[i][5];
    if (activo === true || activo === 'TRUE' || activo === 'Sí') {
      personas.push({
        idPersona: String(datos[i][0] || '').trim(),
        nombre:    String(datos[i][1] || '').trim(),
        mail:      String(datos[i][2] || '').trim(),
        puesto:    String(datos[i][3] || '').trim(),
        area:      String(datos[i][4] || '').trim()
      });
    }
  }
  return personas;
}

/**
 * Genera la evaluación completa.
 * A partir de Fase 4 acepta el nuevo formato de params del DialogoEvaluacion v2.
 * Mantiene compatibilidad descendente con el formato anterior (Fase 1–3).
 *
 *   1. Lee el documento desde Drive
 *   2. Llama a Claude para generar preguntas
 *   3. Crea Google Form (modo autónomo) o registra sin Form (modo presencial)
 *   4. Guarda claves en Claves_Evaluacion (con ID_Evaluacion)
 *   5. Registra en Registro_Competencias incluyendo idCompetencia, dominio, categoría, criticidad
 *   6. Instala trigger onFormSubmit
 *   7. Crea borrador de Gmail
 *
 * @param {Object} params  — FORMATO NUEVO (Fase 4):
 *   - persona    {Object}  { idPersona, nombre, mail, puesto, area }
 *   - competencia {Object} { idCompetencia, dominio, categoria, naturaleza,
 *                            criticidad, driveId, driveNombre, frecuenciaMeses }
 *   - cantidad   {number}  preguntas (3–20)
 *   - modo       {string}  'autonomo' | 'presencial'
 *
 * @returns {Object} { ok, formUrl?, formEditUrl?, idEvaluacion?, error? }
 */
function generarEvaluacion(params) {
  try {
    // ── Normalizar params: Fase 4 (nuevo) vs Fase 1–3 (legado) ──────
    var persona;
    var comp;
    var docs;

    if (params.persona) {
      // ── NUEVO formato (Fase 4) ───────────────────────────────────
      persona = params.persona;
      comp    = params.competencia || {};

      // Construir array docs desde la competencia.
      // Drive_ID puede ser un ID simple o varios separados por "|"
      // (ej: "abc123|def456"), igual que Drive_Nombre ("Manual ICP|PO-digestión").
      if (comp.driveId) {
        var ids     = comp.driveId.split('|').map(function(s){ return s.trim(); }).filter(Boolean);
        var nombres = (comp.driveNombre || comp.dominio).split('|').map(function(s){ return s.trim(); });
        docs = ids.map(function(id, i) {
          return { id: id, nombre: nombres[i] || comp.dominio };
        });
      } else {
        docs = [];
      }

    } else {
      // ── LEGADO (Fase 1–3): params.personas[] + params.docs[] ─────
      const primeraPersona = (params.personas || [])[0] || {};
      persona = {
        idPersona: primeraPersona.idPersona || '',
        nombre:    primeraPersona.nombre    || '',
        mail:      primeraPersona.mail      || '',
        puesto:    primeraPersona.puesto    || primeraPersona.area || '',
        area:      primeraPersona.area      || ''
      };
      comp = {
        idCompetencia: '',
        dominio:       (params.docs || []).map(function(d){ return d.nombre; }).join(', '),
        categoria:     '',
        naturaleza:    '',
        criticidad:    '',
        driveId:       '',
        driveNombre:   ''
      };
      docs = params.docs || [];
    }

    const cantidad = Number(params.cantidad) || 5;
    const modo     = (params.modo || 'autonomo').toLowerCase();

    // ── Validaciones ─────────────────────────────────────────────
    if (!persona.nombre) return { ok: false, error: 'No se recibió una persona.' };
    if (modo === 'autonomo' && docs.length === 0) {
      return { ok: false, error: 'Modo autónomo requiere al menos un documento Drive.' };
    }

    const nombresDocumentos = docs.map(function(d){ return d.nombre; }).join(', ')
                           || comp.dominio || '—';
    const puesto = persona.puesto || persona.area || 'Personal';

    // ── Generar ID_Evaluacion ────────────────────────────────────
    const ahora        = new Date();
    const fechaStamp   = Utilities.formatDate(ahora, Session.getScriptTimeZone(), 'yyyyMMdd-HHmmss');
    const primerNombre = persona.nombre.split(' ')[0].toUpperCase();
    const idEvaluacion = 'EVAL-' + fechaStamp + '-' + primerNombre;

    // ── Modo presencial sin documento: registrar directamente ────
    if (modo === 'presencial' && docs.length === 0) {
      return _procesarModoPresencial({
        idEvaluacion:      idEvaluacion,
        personas:          [persona],
        docs:              [],
        todasLasPreguntas: [],
        nombresDocumentos: nombresDocumentos,
        competencia:       comp,
        puesto:            puesto,
        ahora:             ahora
      });
    }

    // ── Generar preguntas con Claude (para autónomo y presencial con doc) ──
    var todasLasPreguntas = [];

    for (var i = 0; i < docs.length; i++) {
      var doc = docs[i];
      try {
        var texto       = leerTextoDeArchivo(doc.id);
        var cantidadDoc = docs.length === 1
          ? cantidad
          : Math.max(2, Math.round(cantidad / docs.length));
        var preguntas   = generarPreguntasDesdeTexto(texto, puesto, doc.nombre, cantidadDoc);
        todasLasPreguntas = todasLasPreguntas.concat(preguntas);
      } catch (docErr) {
        Logger.log('[Codigo] Error procesando "' + doc.nombre + '": ' + docErr.message);
        return { ok: false, error: 'Error al procesar "' + doc.nombre + '": ' + docErr.message };
      }
    }

    if (todasLasPreguntas.length === 0) {
      return { ok: false, error: 'No se pudieron generar preguntas. Revisá el documento.' };
    }

    // ── Modo presencial con documento ────────────────────────────
    if (modo === 'presencial') {
      return _procesarModoPresencial({
        idEvaluacion:      idEvaluacion,
        personas:          [persona],
        docs:              docs,
        todasLasPreguntas: todasLasPreguntas,
        nombresDocumentos: nombresDocumentos,
        competencia:       comp,
        puesto:            puesto,
        ahora:             ahora
      });
    }

    // ── Modo autónomo: crear Google Form ─────────────────────────
    const tituloForm = docs.length === 1 ? docs[0].nombre : nombresDocumentos;
    const form       = FormApp.create(tituloForm);
    form.setDescription('Evaluación GeCO Lab — ' + persona.nombre);
    form.setCollectEmail(false);
    form.setShowLinkToRespondAgain(false);

    if (docs.length > 1) {
      var docActual = null;
      todasLasPreguntas.forEach(function(p) {
        var docPregunta = p.documento || '';
        if (docPregunta && docPregunta !== docActual) {
          form.addSectionHeaderItem().setTitle(docPregunta);
          docActual = docPregunta;
        }
        _agregarPreguntaAlForm(form, p);
      });
    } else {
      todasLasPreguntas.forEach(function(p) { _agregarPreguntaAlForm(form, p); });
    }

    const formUrl     = form.getPublishedUrl();
    const formEditUrl = form.getEditUrl();
    const formId      = form.getId();

    // ── Mover Form a carpeta Drive ────────────────────────────────
    try {
      const folder = DriveApp.getFolderById(getDriveFolderId());
      DriveApp.getFileById(formId).moveTo(folder);
    } catch (moveErr) {
      Logger.log('[Codigo] No se pudo mover el Form: ' + moveErr.message);
    }

    // ── ① Instalar trigger ───────────────────────────────────────
    instalarTriggerForm(formId);

    // ── ② Guardar claves en Claves_Evaluacion ────────────────────
    const ss       = SpreadsheetApp.getActiveSpreadsheet();
    const fechaStr = Utilities.formatDate(ahora, Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm');
    var hojaClaves = ss.getSheetByName('Claves_Evaluacion');
    if (!hojaClaves) {
      hojaClaves = ss.insertSheet('Claves_Evaluacion');
      hojaClaves.appendRow(['FormID','ID_Evaluacion','N°_Pregunta','Enunciado',
                            'Resp_Correcta','Justificación','Doc_Fuente','Fecha','Modo']);
      hojaClaves.getRange(1, 1, 1, 9)
        .setBackground('#1a73e8').setFontColor('#ffffff').setFontWeight('bold');
    }
    todasLasPreguntas.forEach(function(p, idx) {
      hojaClaves.appendRow([
        formId,
        idEvaluacion,
        idx + 1,
        p.enunciado,
        p.correcta,
        p.justificacion || '',
        p.documento || nombresDocumentos,
        fechaStr,
        'autonomo'
      ]);
    });

    // ── ③ Registrar en Registro_Competencias ─────────────────────
    // Fase 4: idCompetencia, dominio, categoria, criticidad propagados desde la competencia
    registrarEvaluacionInicial({
      persona: {
        nombre:    persona.nombre,
        mail:      persona.mail,
        area:      persona.area,
        puesto:    puesto,
        idPersona: persona.idPersona || ''
      },
      competencia: {
        idCompetencia: comp.idCompetencia || '',
        dominio:       comp.dominio       || nombresDocumentos,
        categoria:     comp.categoria     || '',
        criticidad:    comp.criticidad    || ''
      },
      evaluacion: {
        idEvaluacion: idEvaluacion,
        documento:    nombresDocumentos,
        driveId:      docs.map(function(d){ return d.id; }).join('|'),
        modo:         'Autónomo',
        nroPreguntas: todasLasPreguntas.length,
        evaluador:    Session.getActiveUser().getEmail(),
        formUrl:      formUrl,
        formEditUrl:  formEditUrl
      }
    });

    // ── ④ Crear borrador de Gmail ─────────────────────────────────
    _crearBorradorGmail(persona.mail, persona.nombre, nombresDocumentos,
                        todasLasPreguntas.length, formUrl, 'autonomo');

    Logger.log('[Codigo] Evaluación generada — ID: ' + idEvaluacion +
               ' | Form: ' + formId + ' | Comp: ' + (comp.idCompetencia || 'sin ID'));

    return { ok: true, formUrl: formUrl, formEditUrl: formEditUrl, idEvaluacion: idEvaluacion };

  } catch (e) {
    Logger.log('[Codigo] generarEvaluacion() ERROR: ' + e.message + '\n' + e.stack);
    return { ok: false, error: e.message };
  }
}

/**
 * Procesa una evaluación en modo presencial:
 *   - No crea Google Form
 *   - Registra en Registro_Competencias con Estado = Pendiente
 *   - Crea borrador de Gmail informando al evaluado
 *   - Abre directamente el diálogo de corrección presencial
 *
 * @param {Object} ctx — contexto interno desde generarEvaluacion()
 */
function _procesarModoPresencial(ctx) {
  const ss      = SpreadsheetApp.getActiveSpreadsheet();
  const comp    = ctx.competencia || {};

  const mailsEvaluados   = ctx.personas.map(function(p){ return p.mail;   }).join(', ');
  const nombresEvaluados = ctx.personas.map(function(p){ return p.nombre; }).join(', ');

  // Registrar en Registro_Competencias (sin Form URL)
  // Fase 4: propaga idCompetencia, dominio, categoria, criticidad
  ctx.personas.forEach(function(persona) {
    registrarEvaluacionInicial({
      persona: {
        nombre:    persona.nombre,
        mail:      persona.mail,
        area:      persona.area,
        puesto:    persona.puesto || persona.area || '',
        idPersona: persona.idPersona || ''
      },
      competencia: {
        idCompetencia: comp.idCompetencia || '',
        dominio:       comp.dominio       || ctx.nombresDocumentos,
        categoria:     comp.categoria     || '',
        criticidad:    comp.criticidad    || ''
      },
      evaluacion: {
        idEvaluacion: ctx.idEvaluacion,
        documento:    ctx.nombresDocumentos,
        driveId:      (ctx.docs || []).map(function(d){ return d.id; }).join('|'),
        modo:         'Presencial',
        nroPreguntas: (ctx.todasLasPreguntas || []).length,
        evaluador:    Session.getActiveUser().getEmail(),
        formUrl:      '',
        formEditUrl:  ''
      }
    });
  });

  // Borrador informativo (sin link de Form)
  _crearBorradorGmail(mailsEvaluados, nombresEvaluados, ctx.nombresDocumentos,
                      (ctx.todasLasPreguntas || []).length, '', 'presencial');

  Logger.log('[Codigo] Evaluación presencial registrada — ID: ' + ctx.idEvaluacion +
             (comp.idCompetencia ? ' | Comp: ' + comp.idCompetencia : ''));

  // Abrir diálogo de corrección presencial pre-cargado
  Utilities.sleep(500);
  abrirDialogoPresencialParaEval(ctx.idEvaluacion);

  return {
    ok:           true,
    idEvaluacion: ctx.idEvaluacion,
    modo:         'presencial'
  };
}

/**
 * Agrega una pregunta de opción múltiple al Form.
 * @param {Form} form
 * @param {Object} p — { enunciado, opciones[], correcta, justificacion }
 */
function _agregarPreguntaAlForm(form, p) {
  var item = form.addMultipleChoiceItem();
  item.setTitle(p.enunciado);
  item.setRequired(true);
  item.setChoices(p.opciones.map(function(op) { return item.createChoice(op); }));
}

/**
 * Crea un borrador de Gmail con el link a la evaluación.
 * No envía — el evaluador revisa y envía manualmente.
 */
function _crearBorradorGmail(mails, nombres, documentos, nroPreguntas, formUrl, modo) {
  try {
    const asunto = 'Evaluación GeCO — ' + nombres + ' (' + nroPreguntas + ' preguntas)';

    var cuerpoHtml =
      '<html><head><meta charset="UTF-8"></head><body>' +
      '<p>Hola,</p>';

    if (modo === 'presencial' || !formUrl) {
      cuerpoHtml +=
        '<p>Se ha generado una <strong>evaluación presencial</strong> sobre: ' +
        '<strong>' + documentos + '</strong>.</p>' +
        '<p>El evaluador coordinará la fecha y modalidad de la evaluación.</p>';
    } else {
      cuerpoHtml +=
        '<p>Tu evaluación está lista. Hacé clic en el link para completarla:</p>' +
        '<p><a href="' + formUrl + '" style="color:#1a73e8;font-weight:600;">' +
        'Acceder a la evaluación →</a></p>';
    }

    cuerpoHtml +=
      '<p style="color:#888;font-size:12px;margin-top:16px;">' +
      'Documento(s): ' + documentos + ' &nbsp;|&nbsp; ' +
      'Preguntas: ' + nroPreguntas + ' &nbsp;|&nbsp; ' +
      'Modo: ' + (modo === 'presencial' ? 'Presencial' : 'Autónomo') +
      '</p>' +
      '</body></html>';

    GmailApp.createDraft(mails, asunto, '', { htmlBody: cuerpoHtml, charset: 'UTF-8' });

  } catch (e) {
    // No bloquear el flujo por un error de borrador
    Logger.log('[Codigo] Error creando borrador Gmail: ' + e.message);
  }
}
