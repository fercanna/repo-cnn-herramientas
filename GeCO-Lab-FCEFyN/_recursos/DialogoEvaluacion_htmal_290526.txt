<script>
// ── Estado global ─────────────────────────────────────────────────
var personas       = [];
var personaSel     = null;   // objeto persona seleccionada

var compPendientes = [];     // competencias pendientes/vencidas
var compTodas      = [];     // todas las competencias del puesto
var compSel        = null;   // competencia seleccionada

var modoSel        = null;   // 'autonomo' | 'presencial'
var docsDrive      = [];     // para cuando no hay driveId en la competencia

var pasoActual     = 1;

// ── Init ──────────────────────────────────────────────────────────
window.onload = function() {
  cargarPersonas();
};

// ── PASO 1 — Persona ──────────────────────────────────────────────
function cargarPersonas() {
  google.script.run
    .withSuccessHandler(function(data) {
      personas = data || [];
      var sel = document.getElementById('selectPersona');
      personas.forEach(function(p) {
        var o = document.createElement('option');
        o.value = p.idPersona;
        o.text  = p.nombre + (p.puesto ? ' — ' + p.puesto : '') + (p.area ? ' (' + p.area + ')' : '');
        sel.appendChild(o);
      });
    })
    .withFailureHandler(function(err) { mostrarAlerta('alertaPaso1', 'error', 'Error cargando personas: ' + err.message); })
    .getPersonas();
}

function onPersonaChange() {
  var id = document.getElementById('selectPersona').value;
  personaSel = id ? personas.find(function(p) { return p.idPersona === id; }) : null;

  var card = document.getElementById('personaCard');
  if (personaSel) {
    card.innerHTML =
      '<strong>' + esc(personaSel.nombre) + '</strong><br>' +
      (personaSel.puesto ? '📌 ' + esc(personaSel.puesto) + '<br>' : '') +
      (personaSel.area   ? '🏢 ' + esc(personaSel.area) + '<br>'   : '') +
      (personaSel.mail   ? '✉️ ' + esc(personaSel.mail)             : '');
    card.classList.add('visible');
    ocultarAlerta('alertaPaso1');
  } else {
    card.classList.remove('visible');
  }

  document.getElementById('btnSiguiente').disabled = !personaSel;
}

// ── PASO 2 — Competencia ──────────────────────────────────────────
function cargarCompetencias() {
  document.getElementById('emptyPendientes').textContent = 'Cargando…';
  document.getElementById('listaPendientes').innerHTML = '<div class="empty-hint" id="emptyPendientes">Cargando…</div>';
  document.getElementById('listaTodas').innerHTML = '';
  document.getElementById('listaTodas').classList.remove('visible');
  document.getElementById('btnToggleTodas').textContent = '▸ Ver todas las competencias del puesto';

  var cargados = 0;
  function check() { cargados++; if (cargados === 2) document.getElementById('btnSiguiente').disabled = !compSel; }

  // Pendientes/vencidas
  google.script.run
    .withSuccessHandler(function(data) {
      compPendientes = data || [];
      renderCompLista('listaPendientes', compPendientes, true);
      document.getElementById('cntPendientes').textContent = compPendientes.length;
      if (compPendientes.length === 0) {
        document.getElementById('listaPendientes').innerHTML =
          '<div class="empty-hint">✓ No hay competencias vencidas o pendientes para este puesto.</div>';
      }
      check();
    })
    .withFailureHandler(function(err) {
      document.getElementById('listaPendientes').innerHTML =
        '<div class="empty-hint" style="color:#c5221f">Error: ' + esc(err.message) + '</div>';
      check();
    })
    .getCompetenciasPendientesPorPersona(personaSel.idPersona, personaSel.puesto);

  // Todas del puesto
  google.script.run
    .withSuccessHandler(function(data) {
      compTodas = data || [];
      renderCompLista('listaTodas', compTodas, false);
      if (compTodas.length === 0) {
        document.getElementById('listaTodas').innerHTML =
          '<div class="empty-hint">No hay competencias definidas para el puesto "' + esc(personaSel.puesto) + '".</div>';
      }
      check();
    })
    .withFailureHandler(function() { check(); })
    .getCompetencias(personaSel.puesto);
}

function renderCompLista(contenedorId, lista, esPendiente) {
  var cont = document.getElementById(contenedorId);
  if (!lista || lista.length === 0) return;

  cont.innerHTML = lista.map(function(c) {
    var claseTipo = '';
    var badgePendiente = '';

    if (esPendiente) {
      if (c.diasVencida !== null && c.diasVencida !== undefined) {
        claseTipo = 'vencido';
        badgePendiente = '<span class="badge badge-vencido">Vencida ' + c.diasVencida + 'd</span>';
      } else {
        claseTipo = 'nunca-eval';
        badgePendiente = '<span class="badge badge-nunca">Nunca evaluada</span>';
      }
    }

    var badgeCrit = 'badge-' + (c.criticidad || 'media').toLowerCase();
    var badgeNat  = (c.naturaleza || '').toLowerCase() === 'tácito' ? 'badge-tacito' : 'badge-explicito';
    var badgeDoc  = c.driveId ? '<span class="badge badge-doc">📄 ' + esc(c.driveNombre || c.dominio) + '</span>'
                              : '<span class="badge badge-sin-doc">Sin doc</span>';
    var ultEval   = c.ultimaEvaluacion ? '<span style="font-size:11px;color:#6b7280">Última: ' + esc(c.ultimaEvaluacion) + '</span>' : '';

    var idComp = c.idCompetencia || (c.dominio + '|' + (c.puesto || ''));

    return '<div class="comp-item ' + claseTipo + '" id="ci-' + esc(idComp) + '" onclick="seleccionarComp(' + esc(JSON.stringify(c)) + ')">' +
      '<div class="comp-nombre">' + esc(c.dominio) + '</div>' +
      '<div class="comp-meta">' +
        '<span class="badge ' + badgeCrit + '">' + esc(c.criticidad) + '</span>' +
        '<span class="badge ' + badgeNat  + '">' + esc(c.naturaleza) + '</span>' +
        badgeDoc + badgePendiente + ultEval +
      '</div>' +
      '<div class="comp-seleccionada-icon">✓</div>' +
    '</div>';
  }).join('');
}

function seleccionarComp(comp) {
  compSel = comp;
  // Quitar selección de todos
  document.querySelectorAll('.comp-item').forEach(function(el) { el.classList.remove('seleccionado'); });
  // Marcar el seleccionado
  var idComp = comp.idCompetencia || (comp.dominio + '|' + (comp.puesto || ''));
  var el = document.getElementById('ci-' + idComp);
  if (el) el.classList.add('seleccionado');

  document.getElementById('btnSiguiente').disabled = false;
  ocultarAlerta('alertaPaso2');
}

function toggleTodas() {
  var lista = document.getElementById('listaTodas');
  var btn   = document.getElementById('btnToggleTodas');
  var visible = lista.classList.toggle('visible');
  btn.textContent = visible ? '▴ Ocultar todas' : '▸ Ver todas las competencias del puesto';
}

// ── PASO 3 — Configuración ────────────────────────────────────────
function prepararPaso3() {
  // Resumen
  document.getElementById('resPersona').textContent    = personaSel.nombre + (personaSel.puesto ? ' — ' + personaSel.puesto : '');
  document.getElementById('resCompetencia').textContent = compSel.dominio + ' (' + compSel.criticidad + ')';
  document.getElementById('resDocumento').textContent  = compSel.driveNombre || compSel.driveId || '— Sin documento vinculado';

  // Si no hay driveId, mostrar selector de Drive
  var grupoDoc = document.getElementById('grupoDocManual');
  if (!compSel.driveId) {
    grupoDoc.style.display = 'flex';
    cargarDocsDrive();
  } else {
    grupoDoc.style.display = 'none';
  }

  // Sugerir modo según naturaleza
  var naturaleza = (compSel.naturaleza || '').toLowerCase();
  if (naturaleza === 'tácito' || naturaleza === 'tacito') {
    seleccionarModo('presencial');
    document.getElementById('modoSugerido').classList.add('visible');
    document.getElementById('modoSugerido').textContent = '💡 Sugerido: Presencial (competencia tácita)';
  } else {
    seleccionarModo('autonomo');
    document.getElementById('modoSugerido').classList.add('visible');
    document.getElementById('modoSugerido').textContent = '💡 Sugerido: Autónomo (competencia explícita)';
  }

  document.getElementById('btnGenerar').disabled = false;
}

function cargarDocsDrive() {
  var sel = document.getElementById('selectDocDrive');
  while (sel.options.length > 1) sel.remove(1);

  google.script.run
    .withSuccessHandler(function(docs) {
      docsDrive = docs || [];
      docs.forEach(function(d) {
        var o = document.createElement('option');
        o.value = d.id;
        o.text  = d.nombre;
        sel.appendChild(o);
      });
      // Si la competencia tiene driveNombre, intentar pre-seleccionar
      if (compSel.driveNombre) {
        for (var i = 0; i < sel.options.length; i++) {
          if (sel.options[i].text.toLowerCase().includes(compSel.driveNombre.toLowerCase())) {
            sel.selectedIndex = i; break;
          }
        }
      }
    })
    .getDocumentosDrive();
}

function seleccionarModo(modo) {
  modoSel = modo;
  document.getElementById('modoAutonomo').classList.toggle('seleccionado',  modo === 'autonomo');
  document.getElementById('modoPresencial').classList.toggle('seleccionado', modo === 'presencial');
}

function onSlider() {
  document.getElementById('valorCantidad').textContent = document.getElementById('sliderCantidad').value;
}

// ── Navegación entre pasos ────────────────────────────────────────
function siguiente() {
  if (pasoActual === 1) {
    if (!personaSel) { mostrarAlerta('alertaPaso1', 'error', 'Seleccioná una persona primero.'); return; }
    irPaso(2);
    cargarCompetencias();
  } else if (pasoActual === 2) {
    if (!compSel) { mostrarAlerta('alertaPaso2', 'error', 'Seleccioná una competencia para continuar.'); return; }
    irPaso(3);
    prepararPaso3();
  }
}

function irPaso(n) {
  if (n < 1 || n > 3) return;
  document.getElementById('paso' + pasoActual).classList.remove('visible');
  document.getElementById('paso' + n).classList.add('visible');
  pasoActual = n;

  // Tabs
  [1, 2, 3].forEach(function(i) {
    var tab = document.getElementById('tab' + i);
    tab.classList.remove('activo', 'completado');
    if (i < pasoActual)      tab.classList.add('completado');
    else if (i === pasoActual) tab.classList.add('activo');
  });

  // Footer buttons
  document.getElementById('btnVolver').style.display = pasoActual > 1 ? 'inline-block' : 'none';

  if (pasoActual === 3) {
    document.getElementById('btnSiguiente').style.display = 'none';
    document.getElementById('btnGenerar').style.display   = 'inline-block';
  } else {
    document.getElementById('btnSiguiente').style.display = 'inline-block';
    document.getElementById('btnGenerar').style.display   = 'none';
    document.getElementById('btnSiguiente').disabled = (pasoActual === 2 && !compSel);
  }
}

// ── Generar evaluación ────────────────────────────────────────────
function generar() {
  if (!personaSel || !compSel || !modoSel) {
    mostrarAlerta('alertaPaso3', 'error', 'Faltan datos obligatorios.');
    return;
  }

  // Determinar documento final
  var driveId     = compSel.driveId || '';
  var driveNombre = compSel.driveNombre || compSel.dominio;

  if (!driveId) {
    var selDoc = document.getElementById('selectDocDrive');
    if (selDoc && selDoc.value) {
      driveId     = selDoc.value;
      driveNombre = selDoc.options[selDoc.selectedIndex].text;
    }
  }

  // Modo autónomo sin documento → error
  if (modoSel === 'autonomo' && !driveId) {
    mostrarAlerta('alertaPaso3', 'error',
      'Para el modo autónomo es necesario un documento Drive para generar las preguntas. ' +
      'Vinculá uno en la gestión de competencias o seleccioná el modo presencial.');
    return;
  }

  var cantidad = parseInt(document.getElementById('sliderCantidad').value) || 5;

  var params = {
    persona: {
      idPersona: personaSel.idPersona,
      nombre:    personaSel.nombre,
      mail:      personaSel.mail,
      puesto:    personaSel.puesto,
      area:      personaSel.area
    },
    competencia: {
      idCompetencia:   compSel.idCompetencia   || '',
      dominio:         compSel.dominio,
      categoria:       compSel.categoria        || '',
      naturaleza:      compSel.naturaleza        || '',
      criticidad:      compSel.criticidad        || '',
      driveId:         driveId,
      driveNombre:     driveNombre,
      frecuenciaMeses: compSel.frecuenciaMeses  || 12
    },
    cantidad: cantidad,
    modo:     modoSel
  };

  // Bloquear UI
  document.getElementById('btnGenerar').disabled   = true;
  document.getElementById('btnVolver').disabled    = true;
  spin('spinGen', true);
  mostrarAlerta('alertaPaso3', 'info',
    modoSel === 'autonomo'
      ? '⏳ Generando preguntas con IA y creando el formulario… puede tardar unos segundos.'
      : '⏳ Registrando evaluación presencial…');

  google.script.run
    .withSuccessHandler(function(res) {
      spin('spinGen', false);
      if (res.ok) {
        var msg = modoSel === 'autonomo'
          ? '✅ Evaluación generada. Se creó el borrador de Gmail con el link.'
          : '✅ Evaluación presencial registrada. Se abrirá el diálogo de corrección.';
        mostrarAlerta('alertaPaso3', 'success', msg);
        if (modoSel === 'autonomo' && res.formUrl) {
          setTimeout(function() { google.script.host.close(); }, 2500);
        } else {
          setTimeout(function() { google.script.host.close(); }, 1800);
        }
      } else {
        document.getElementById('btnGenerar').disabled = false;
        document.getElementById('btnVolver').disabled  = false;
        mostrarAlerta('alertaPaso3', 'error', '❌ ' + (res.error || 'Error desconocido.'));
      }
    })
    .withFailureHandler(function(err) {
      spin('spinGen', false);
      document.getElementById('btnGenerar').disabled = false;
      document.getElementById('btnVolver').disabled  = false;
      mostrarAlerta('alertaPaso3', 'error', '❌ Error: ' + err.message);
    })
    .generarEvaluacion(params);
}

// ── Utilidades ────────────────────────────────────────────────────
function esc(s) {
  return String(s || '').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}

function spin(id, on) {
  var el = document.getElementById(id);
  if (el) el.classList.toggle('visible', on);
}

function mostrarAlerta(id, tipo, msg) {
  var el = document.getElementById(id);
  if (!el) return;
  el.className = 'alert ' + tipo;
  el.textContent = msg;
}

function ocultarAlerta(id) {
  var el = document.getElementById(id);
  if (el) { el.className = 'alert'; el.textContent = ''; }
}
